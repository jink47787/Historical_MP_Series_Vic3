namespace = wedding

wedding.1 = {
	type = country_event
	placement = ROOT

	title = wedding.1.t
	desc = wedding.1.d
	flavor = wedding.1.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:royal_marriage_character

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	trigger = {
		# triggered by seek_royal_marriage
		exists = scope:royal_marriage_character
		exists = scope:marriage_country
	}

	cancellation_trigger = {
		NOR = {
			exists = scope:royal_marriage_character
			exists = scope:marriage_country
			exists = scope:foreign_ruler
		}
	}

	immediate = {
		capital = {
			save_scope_as = capital_scope
		}
		random_scope_character = {
			limit = {
				OR = {
					is_heir_of_own_country = yes
					is_ruler_of_own_country = yes
				}
				age > define:NCharacters|ADULT_AGE
				NOT = { has_variable = is_married }
			}
			set_variable = getting_married # for clothing script
		}
		scope:marriage_country = {
			ruler = {
				save_scope_as = foreign_ruler
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
	}

	option = { # private wedding for international nobility
		name = wedding.1.a
		default_option = yes
		add_modifier = { # influence mult
			name = royal_marriage
			days = normal_modifier_time
		}
	}

	option = { # public wedding for the masses
		name = wedding.1.b
		scope:royal_marriage_character = {
			add_modifier = { # popularity
				name = marriage_popularity
				days = very_long_modifier_time
			}
		}
	}

	after = {
		scope:royal_marriage_character ?= {
			set_variable = is_married
			remove_variable = getting_married
		}
		scope:foreign_ruler ?= {
			remove_variable = marriage_target
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:foreign_ruler }
	}
}

wedding.3 = {
	type = country_event
	placement = ROOT

	title = wedding.3.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = wedding.3.d1
				trigger = {
					ruler.interest_group ?= {
						OR = {
							ig_approval <= angry
							ig_clout < define:NPolitics|MARGINAL_IG_THRESHOLD
						}
					}
				}
			}
			triggered_desc = {
				desc = wedding.3.d2
				trigger = {
					ruler = {
						popularity <= -50
					}
				}
			}
			triggered_desc = {
				desc = wedding.3.d3
				trigger = {
					ruler = {
						OR = {
							has_trait = psychological_affliction
							age >= 75
						}
					}
				}
			}
		}
	}
	flavor = wedding.3.f

	gui_window = event_window_2char

	left_icon = scope:abdicating_monarch
	right_icon = scope:heir_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/2Characters"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	duration = 3

	dlc = dlc004

	trigger = {
		# triggered by abdicate_monarch interaction
	}

	immediate = {
		ruler = {
			save_scope_as = abdicating_monarch
			set_temporary_event_immortality = { DURATION = 3 }
			interest_group ?= {
				save_scope_as = monarch_ig
			}
		}
		heir ?= {
			save_scope_as = heir_scope
			set_temporary_event_immortality = { DURATION = 3 }
		}
		law_type:law_monarchy = {
			save_scope_as = monarchy_law
		}
	}

	option = { # reassure the aristocracy
		name = wedding.3.a
		default_option = yes
		add_loyalists = {
			value = 0.1
			pop_type = aristocrats
		}
		scope:heir_scope ?= {
			set_character_as_ruler = yes
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:abdicating_monarch }
		scope:abdicating_monarch ?= {
			resign_and_retire = yes
		}
	}

	option = { # reassure the people
		name = wedding.3.b
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
		add_loyalists = {
			value = 0.01
		}
		scope:heir_scope ?= {
			set_character_as_ruler = yes
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:abdicating_monarch }
		scope:abdicating_monarch ?= {
			resign_and_retire = yes
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:abdicating_monarch }
		revoke_temporary_event_immortality = { CHARACTER = scope:heir_scope }
	}
}

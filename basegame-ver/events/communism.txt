# Completion event 1
communism.1 = {
	type = country_event
	placement = root
	title = communism.1.t
	desc = communism.1.desc
	flavor = communism.1.f

	duration = 3

	event_image = {
		video = "europenorthamerica_political_extremism"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_election.dds"

	trigger = {
		# triggered by communsim JE 1
	}

	immediate = {
		if = {
			limit = {
				any_interest_group = {
					law_stance = {
						law = law_type:law_council_republic
						value > neutral
					}
				}
			}
			random_interest_group = {
				limit = {
					law_stance = {
						law = law_type:law_council_republic
						value > neutral
					}
				}
				set_variable = communist_ig_var
				save_scope_as = unions
				leader = {
					save_scope_as = union_leader
				}
			}
		}
		else = {
			ig:ig_trade_unions ?= {
				set_variable = communist_ig_var
				save_scope_as = unions
				leader = {
					save_scope_as = union_leader
				}
			}
		}
		add_journal_entry = {
			type = je_communism_2
		}
	}

	option = { # wage controls
		name = communism.1.a
		trigger = {
			NOT = { has_law_or_variant = law_type:law_wage_subsidies }
			NOT = { has_law_or_variant = law_type:law_old_age_pension }
		}
		activate_law = law_type:law_wage_subsidies
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
		ai_chance = {
			base = 40
		}
	}
	option = { # ban slavery
		name = communism.1.b
		trigger = {
			NOT = { has_law = law_type:law_slavery_banned }
		}
		activate_law = law_type:law_slavery_banned
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
		ai_chance = {
			base = 50
		}
	}
	option = { # Regulatory bodies
		name = communism.1.c
		trigger = {
			NOT = { has_law_or_variant = law_type:law_regulatory_bodies }
			NOT = { has_law_or_variant = law_type:law_worker_protections }
		}
		activate_law = law_type:law_regulatory_bodies
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
		ai_chance = {
			base = 40
		}
	}
	option = {
		name = communism.1.d
		default_option = yes
		if = {
			limit = {
				any_interest_group = {
					has_variable = communist_ig_var
				}
			}
			custom_tooltip = communist_unions.tt
		}
		every_interest_group = {
			limit = {
				leader = {
					OR = {
						has_ideology = ideology:ideology_vanguardist
						has_ideology = ideology:ideology_communist_leader
						has_ideology = ideology:ideology_anarchist
					}
				}
			}
			add_modifier = {
				name = communist_government_modifier
				days = normal_modifier_time
			}
		}
		add_loyalists = {
			value = medium_radicals
			strata = upper
		}
		ai_chance = {
			base = 20
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
	}
}

# Completion event 2, choose ideology
communism.2 = {
	type = country_event
	placement = root
	title = communism.2.t
	desc = communism.2.d
	flavor = communism.2.f

	duration = 3

	event_image = {
		video = "ip5_soviet_deliberation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	trigger = {
		# triggered by communsim JE 2
	}

	fire_only_once = yes

	immediate = {
		ig:ig_trade_unions ?= {
			save_scope_as = unions
			leader = {
				save_scope_as = union_leader
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
	}

	option = { # communism
		name = communism.2.b
		default_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_communism
		}
		custom_tooltip = chose_communism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_communist_leader }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_communist_leader
			}
		}
		ai_chance = {
			base = 50
		}
	}
	option = { # vanguardism
		name = communism.2.a
		trigger = {
			has_technology_researched = political_agitation
			custom_tooltip = {
				text = vanguardism_viable_tt
				vanguardism_coefficient > 1
			}
		}
		show_as_unavailable = {
			has_technology_researched = political_agitation
		}
		highlighted_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_vanguardism
		}
		custom_tooltip = chose_vanguardism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_vanguardist }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_vanguardist
			}
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = {
					vanguardism_coefficient >= 2
				}
				add = 40
			}
		}
	}
	option = { # anarchism
		name = communism.2.c
		trigger = {
			any_political_movement = {
				is_political_movement_type = movement_anarchist
			}
		}
		show_as_unavailable = {
			has_technology_researched = anarchism
		}
		highlighted_option = yes
		ig:ig_trade_unions ?= {
			set_variable = chose_anarchism
		}
		custom_tooltip = chose_anarchism.tt
		if = {
			limit = {
				scope:union_leader = {
					NOT = { has_ideology = ideology:ideology_anarchist }
				}
			}
			scope:union_leader = {
				set_ideology = ideology:ideology_anarchist
			}
		}
		ai_chance = {
			base = 0
			modifier = {
				trigger = {
					any_political_movement = {
						is_political_movement_type = movement_anarchist
						political_movement_support >= 0.1
					}
				}
				add = 25
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:union_leader }
		if = {
			limit = {
				NOT = { has_global_variable = socialist_world_republic }
			}
			add_journal_entry = {
				type = je_state_and_revolution
			}
		}
		add_journal_entry = {
			type = je_social_revolution
		}
		custom_tooltip = {
			text = form_white_movement_tt
			if = {
				limit = {
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_anti_communist
						}
					}
				}
				create_political_movement = {
					type = movement_anti_communist
				}
			}
		}
		je_communism_2_cleanup = yes
	}
}

# The Red Army
communism.8 = {
	type = country_event
	placement = root
	title = communism.8.t
	desc = communism.8.d
	flavor = communism.8.f

	duration = 3

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	fire_only_once = yes

	trigger = {
		# triggered by on_action - on_monthly_pulse_country
		has_law_or_variant = law_type:law_council_republic
		has_law_or_variant = law_type:law_revolutionary_guard
		any_scope_character = {
			OR = {
				has_variable = red_army_candidate
				AND = {
					OR = {
						is_interest_group_type = ig_trade_unions
						has_socialist_ideology = yes
					}
					OR = {
						has_role_of_type = general
						has_role_of_type = admiral
					}
				}
			}
		}
	}

	immediate = {
		ordered_scope_character = {
			limit = {
				OR = {
					has_variable = red_army_candidate
					AND = {
						OR = {
							is_interest_group_type = ig_trade_unions
							has_socialist_ideology = yes
						}
						OR = {
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
				}
			}
			order_by = prominence
			position = 0
			save_scope_as = red_army_general
			set_temporary_event_immortality = { DURATION = 3 }
		}
		ig:ig_armed_forces ?= {
			save_scope_as = armed_forces_ig
		}
	}

	option = {
		name = communism.8.a
		ig:ig_armed_forces ?= {
			custom_tooltip = red_army_tt
			hidden_effect = {
				set_interest_group_name = ig_red_army
				if = {
					limit = {
						has_ideology = ideology:ideology_tsarist
					}
					remove_ideology = ideology_tsarist
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_jingoist
					}
					remove_ideology = ideology_jingoist
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_militarist
					}
					remove_ideology = ideology_militarist
					add_ideology = ideology_militant
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_republican
					}
					remove_ideology = ideology_republican
					add_ideology = ideology_militant
				}
				custom_tooltip = {
					text = loyalist_change_stances_tt
					if = {
						limit = {
							has_ideology = ideology:ideology_loyalist
						}
						remove_ideology = ideology_loyalist
						add_ideology = ideology_disciplinarian
					}
				}
				add_ideology = ideology_internationalist
				owner = {
					set_variable = ig_red_army_active
				}
			}
		}
		if = {
			limit = {
				NOT = {
					scope:red_army_general ?= {
						is_interest_group_type = ig_armed_forces
					}
				}
			}
			scope:red_army_general ?= {
				set_interest_group = scope:armed_forces_ig
				custom_tooltip = {
					text = leader_of_red_army_tt
					set_as_interest_group_leader = yes
				}
			}
		}
		every_scope_character = {
			limit = {
				OR = {
					has_variable = red_army_candidate
					AND = {
						OR = {
							is_interest_group_type = ig_trade_unions
							has_socialist_ideology = yes
						}
						OR = {
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
				}
				NOT = {
					scope:red_army_general ?= this
				}
			}
			if = {
				limit = {
					has_variable = red_army_candidate
				}
				set_interest_group = scope:armed_forces_ig
			}
			else = {
				random = {
					chance = {
						value = 50
					}
					set_interest_group = scope:armed_forces_ig
				}
			}
		}
	}

	after = {
		hidden_effect = {
			scope:red_army_general ?= {
				set_as_interest_group_leader = yes
			}
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:red_army_general }
	}
}

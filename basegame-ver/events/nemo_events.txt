namespace = nemo_events

nemo_events.2 = {
	type = country_event
	placement = scope:nemo_coast_state_scope

	title = nemo_events.2.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = nemo_events.2.d_repeat
				trigger = {
					var:nemo_raid_count_var > 0
				}
			}
			triggered_desc = {
				desc = nemo_events.2.d
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = nemo_events.2.f

	gui_window = event_window_1char_tabloid_halloween_nemo
	left_icon = scope:nemo_scope

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	duration = 3

	trigger = {
		game_date >= 1860.1.25
		has_global_variable = nemo_chain_started_global_var
		has_variable = nemo_pending_raid_var
		any_character_in_void = {
			has_template = nemo_prince
		}
	}

	immediate = {
		remove_variable = nemo_pending_raid_var
		if = {
			limit = {
				any_scope_state = {
					is_coastal = yes
				}
			}
			random_scope_state = {
				limit = {
					is_coastal = yes
				}
				save_scope_as = nemo_coast_state_scope
			}
		}
		else = {
			capital = {
				save_scope_as = nemo_coast_state_scope
			}
		}
		random_character_in_void = {
			limit = {
				has_template = nemo_prince
			}
			save_scope_as = nemo_scope
			set_temporary_event_immortality = { DURATION = 3 }
		}
	}

	cancellation_trigger = {
		scope:nemo_scope ?= {
			is_character_alive = no
		}
	}

	option = {
		name = nemo_events.2.a
		default_option = yes
		add_treasury = {
			value = nemo_cargo_loss_small
		}
		every_interest_group = {
			limit = {
				nemo_ig_pro_exclusion_or_professional_forces = yes
			}
			add_modifier = {
				name = nemo_shipping_losses
				days = normal_modifier_time
			}
		}
		change_variable = {
			name = nemo_raid_count_var
			add = 1
		}
		hidden_effect = {
			if = {
				limit = {
					var:nemo_raid_count_var >= var:nemo_raid_limit_var
				}
				set_variable = nemo_pending_finale_var
			}
			else = {
				set_variable = nemo_pending_raid_var
			}
		}
	}

	option = {
		name = nemo_events.2.b
		custom_tooltip = {
			text = nemo_patrol_promise_tt_a
			set_variable = nemo_patrol_promise_var
		}
		custom_tooltip = {
			text = nemo_patrol_promise_tt_b
		}
		every_interest_group = {
			limit = {
				nemo_ig_pro_exclusion_or_professional_forces = yes
			}
			add_modifier = {
				name = nemo_promised_protection
				days = normal_modifier_time
			}
		}
		hidden_effect = {
			random_list = {
				1 = {
					trigger_event = { id = nemo_events.10 months = 1 }
				}
				1 = {
					trigger_event = { id = nemo_events.10 months = 2 }
				}
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:nemo_scope }
	}
}

nemo_events.10 = {
	type = country_event
	hidden = yes

	trigger = {
		has_variable = nemo_patrol_promise_var
	}

	immediate = {
		if = {
			limit = {
				nemo_fleet_patrolling_indian_ocean = yes
			}
			random_scope_fleet = {
				limit = {
					nemo_fleet_patrols_indian_ocean = yes
				}
				save_scope_as = nemo_engaged_fleet_scope
			}
			set_variable = nemo_pending_finale_var
		}
		else = {
			trigger_event = { id = nemo_events.3 days = 3 }
		}
		remove_variable = nemo_patrol_promise_var
	}
}

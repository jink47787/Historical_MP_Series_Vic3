debug.11 = { # Coup button
	type = country_event
	hidden = yes
	orphan = yes
	immediate = {
		random_scope_character = {
			limit = {
				OR = {
					has_role_of_type = general
					has_role_of_type = admiral
				}
			}
			save_scope_as = temp_golpista
			set_temporary_event_immortality = { DURATION = 3 }
		}
		if = {
			limit = {
				exists = scope:temp_golpista
			}
			create_coup_from_character = {
				CHARACTER = scope:temp_golpista
			}
		}
	}
	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:temp_golpista }
	}
}

debug.16 = { # Gunther Approved
	type = country_event
	hidden = yes
	orphan = yes
	immediate = {
		while = {
			limit = {
				any_scope_state = {
					is_capital = no
				}
			}
			random_scope_state = {
				limit = {
					is_capital = no
				}
				save_scope_as = relevant_state
				set_variable = warlord_state
			}
			create_dynamic_country = {
				origin = scope:relevant_state.owner
				country_type = recognized
				tier = principality
				capital = scope:relevant_state
				cede_state_trigger = {
					has_variable = warlord_state
				}
				on_created = {
					every_scope_state = {
						remove_variable = warlord_state
					}
				}
			}
		}
	}
}

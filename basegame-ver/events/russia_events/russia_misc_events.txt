namespace = russia_misc_events

# Baltic Mass Conversions to Orthodoxy
russia_misc_events.1 = {
	type = country_event
	placement = scope:riga_scope

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	event_image = {
		video = "ip5_countryside_procession"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	title = russia_misc_events.1.t
	desc = russia_misc_events.1.d
	flavor = russia_misc_events.1.f

	duration = 3

	fire_only_once = yes

	dlc = dlc019

	trigger = {
		game_date > 1840.1.1
		c:RUS ?= this
		any_subject_or_below = {
			c:UBD ?= this
			is_subject_of = ROOT
		}
		has_law_or_variant = law_type:law_state_religion
		NOT = {
			c:UBD ?= {
				active_law:lawgroup_church_and_state ?= {
					has_amendment = amendment_type:amendment_orthodox_conversions
				}
			}
		}
	}

	immediate = {
		c:UBD ?= {
			save_scope_as = baltic_governorates_scope
			capital = {
				save_scope_as = riga_scope
			}
		}
		ig:ig_devout ?= {
			save_scope_as = orthodox_church_scope
		}
	}

	option = {
		name = russia_misc_events.1.a
		trigger = {
			authority >= 200
		}
		highlighted_option = yes
		show_as_unavailable = {
			always = yes
		}
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					has_journal_entry = je_tp_tsar_gendarme_of_europe
				}
				add = 100
			}
		}
		add_modifier = {
			name = modifier_sponsoring_baltic_conversions
			days = long_modifier_time
		}
		ig:ig_devout ?= {
			add_modifier = {
				name = modifier_sponsoring_baltic_conversions_ig
				days = long_modifier_time
			}
		}
		c:UBD ?= {
			add_radicals = {
				value = small_radicals
				culture = cu:east_german
			}
			custom_tooltip = {
				text = baltic_governorate_gets_conversions_10_tt
				active_law:lawgroup_church_and_state ?= {
					add_amendment = {
						type = amendment_orthodox_conversions
						sponsor = ROOT.ig:ig_rural_folk
						cooldown = 60
						timeout = 120
					}
				}
			}
		}
	}

	option = {
		name = russia_misc_events.1.b
		default_option = yes
		c:UBD ?= {
			custom_tooltip = {
				text = baltic_governorate_gets_conversions_5_tt
				active_law:lawgroup_church_and_state ?= {
					add_amendment = {
						type = amendment_orthodox_conversions
						sponsor = ROOT.ig:ig_rural_folk
						cooldown = 30
						timeout = 60
					}
				}
			}
		}
	}
}

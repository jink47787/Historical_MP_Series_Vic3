# Letters from Beneath the Gallows
cultural_renaissances_events.7 = {
	type = country_event
	placement = root

	duration = 3

	title = cultural_renaissances_events.7.t
	desc = cultural_renaissances_events.7.d
	flavor = cultural_renaissances_events.7.f

    event_image = {
		video = "votp_prison"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	dlc = dlc019

	fire_only_once = yes

	trigger = {
		# triggered by executing Kalinowski in cultural_renaissances_events.6.c
	}

	immediate = {
		# Gets published in a prominent city in Europe with lenient press laws (historially published in Paris)
		if = {
			limit = {
				any_country_in_europe = {
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
					NOT = { this = root }
				}
			}
			ordered_country_in_europe = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
					NOT = { this = root }
				}
				order_by = country_rank
				position = 0
				save_scope_as = letters_publishing_country
			}
		}
		# If no country in Europe has lenient press laws, go outside of Europe
		else_if = {
			limit = {
				any_country = {
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
					NOT = { this = root }
				}
			}
			ordered_country = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_right_of_assembly
						has_law_or_variant = law_type:law_protected_speech
					}
					NOT = { this = root }
				}
				order_by = country_rank
				position = 0
				save_scope_as = letters_publishing_country
			}
		}
		# If no country in the world has lenient press laws, just pick one that hates root
		else = {
			ordered_country_in_europe = {
				limit = {
					NOT = { this = root }
				}
				order_by = {
					value = relations:root
					multiply = -1
				}
				position = 0
				save_scope_as = letters_publishing_country
			}
		}
	}

	# Intercept publications from Paris as suspect
	option = {
		name = cultural_renaissances_events.7.a
		change_relations = {
			country = scope:letters_publishing_country
			value = -10
		}
		ai_chance = {
			base = 10
			modifier = {
				trigger = { has_variable = ai_belarus_renaissance_repress }
				add = 100
			}
			modifier = {
				trigger = { has_variable = ai_belarus_renaissance_moderate }
				add = 100
			}
		}
	}
	# Arrest those in possession of the letters
	option = {
		name = cultural_renaissances_events.7.b
		default_option = yes
		add_radicals = {
			value = small_radicals
			culture = cu:byelorussian
		}
		ai_chance = {
			base = 10
		}
	}
	# Ignore them
	option = {
		name = cultural_renaissances_events.7.c
		add_modifier = {
			name = letters_from_gallows
			days = long_modifier_time
		}
		cu:byelorussian = { add_fervor = 10 }
		ai_chance = {
			base = 10
			modifier = {
				trigger = { has_variable = ai_belarus_renaissance_accelerate }
				add = 100
			}
		}
	}
}

# Bolshevik or SR coup event
bolsheviks.2 = {
	type = country_event
	placement = root

	duration = 3

	dlc = dlc019

	title = {
		first_valid = {
			triggered_desc = {
				desc = bolsheviks.2.t_1
				trigger = {
					capital ?= {
						state_region = s:STATE_INGRIA
					}
				}
			}
			triggered_desc = {
				desc = bolsheviks.2.t
			}
		}
	}
	desc = bolsheviks.2.d
	flavor = bolsheviks.2.f

    event_image = {
		video = "europenorthamerica_springtime_of_nations"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		# Triggered by je_bolsheviks
	}

	cancellation_trigger = {
	}

	immediate = {
		save_scope_as = russia_scope
		# Bolshevik outcome
		if = {
			limit = {
				has_variable = bolshevik_coup_var
			}
			if = {
				limit = {
					var:bolshevik_leader ?= {
						is_character_alive = yes
					}
				}
				var:bolshevik_leader = {
					set_interest_group = ig:ig_trade_unions
					set_as_interest_group_leader = yes
				}
			}
			# make sure Trade Unions have a vanguardist leader
			# pick an existing one if possible
			# otherwise create a new one
			else_if = {
				limit = {
					ig:ig_trade_unions = {
						leader = {
							NOT = {
								character_counts_as_bolshevik = yes
							}
						}
					}
					any_scope_character = {
						character_counts_as_bolshevik = yes
						is_interest_group_type = ig_trade_unions
					}
				}
				ordered_scope_character = {
					limit = {
						character_counts_as_bolshevik = yes
						is_interest_group_type = ig_trade_unions
					}
					order_by = popularity
					position = 0
					set_as_interest_group_leader = yes
				}
			}
			else_if = {
				limit = {
					ig:ig_trade_unions = {
						leader = {
							NOT = {
								character_counts_as_bolshevik = yes
							}
						}
					}
					NOT = {
						any_scope_character = {
							character_counts_as_bolshevik = yes
							is_interest_group_type = ig_trade_unions
						}
					}
				}
				create_character = {
					role = politician
					interest_group = ig_trade_unions
					ideology = ideology_vanguardist
					on_created = {
						set_as_interest_group_leader = yes
					}
				}
			}
			ig:ig_trade_unions = {
				leader = {
					save_scope_as = rev_leader
					set_character_as_ruler = yes
				}
			}
		}
		else_if = {
			limit = {
				has_variable = sr_coup_var
			}
			if = {
				limit = {
					var:left_sr_leader ?= {
						is_character_alive = yes
					}
				}
				var:left_sr_leader = {
					set_interest_group = ig:ig_rural_folk
					set_as_interest_group_leader = yes
				}
			}
			# make sure Rural Folk have an anarchist leader
			# pick an existing one if possible
			# otherwise create a new one
			else_if = {
				limit = {
					ig:ig_rural_folk = {
						leader = {
							NOT = {
								character_counts_as_left_sr = yes
							}
						}
					}
					any_scope_character = {
						character_counts_as_left_sr = yes
						is_interest_group_type = ig_rural_folk
					}
				}
				ordered_scope_character = {
					limit = {
						character_counts_as_left_sr = yes
						is_interest_group_type = ig_rural_folk
					}
					order_by = popularity
					position = 0
					set_as_interest_group_leader = yes
				}
			}
			else_if = {
				limit = {
					ig:ig_rural_folk = {
						leader = {
							NOT = {
								character_counts_as_left_sr = yes
							}
						}
					}
					NOT = {
						any_scope_character = {
							character_counts_as_left_sr = yes
							is_interest_group_type = ig_rural_folk
						}
					}
				}
				create_character = {
					role = politician
					interest_group = ig_rural_folk
					ideology = ideology_anarchist
					on_created = {
						set_as_interest_group_leader = yes
					}
				}
			}
			ig:ig_rural_folk = {
				leader = {
					save_scope_as = rev_leader
					set_character_as_ruler = yes
				}
			}
		}
	}

	after = {
	}

	option = {
		name = bolsheviks.2.a
		if = {
			limit = {
				NOT = {
					has_law = law_type:law_council_republic
				}
			}
			enact_law = law_type:law_council_republic
		}
		if = {
			limit = {
				has_variable = bolshevik_coup_var
				ig:ig_trade_unions = {
					is_in_government = no
				}
			}
			ig:ig_trade_unions = {
				add_ruling_interest_group = yes
			}
		}
		else_if = {
			limit = {
				has_variable = sr_coup_var
				ig:ig_rural_folk = {
					is_in_government = no
				}
			}
			ig:ig_rural_folk = {
				add_ruling_interest_group = yes
			}
		}
		every_interest_group = {
			limit = {
				leader = {
					NOR = {
						character_counts_as_bolshevik = yes
						character_counts_as_menshevik = yes
						character_counts_as_left_sr = yes
					}
				}
			}
			remove_ruling_interest_group = yes
		}
	}
}

# Bolsheviks and Mensheviks fight over the labor movement
bolsheviks.4 = {
	type = country_event
	placement = root

	duration = 4

	dlc = dlc019

	title = bolsheviks.4.t
	desc = bolsheviks.4.d
	flavor = bolsheviks.4.f

    event_image = {
		video = "ip5_soviet_deliberation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		# Triggered by je_bolsheviks
		var:bolshevik_leader ?= {
			is_character_alive = yes
		}
		var:menshevik_leader ?= {
			is_character_alive = yes
		}
		any_political_movement = {
			is_political_movement_type = movement_labor
			NOR = {
				has_modifier = bolshevik_labor_activism
				has_modifier = menshevik_labor_support
				has_modifier = social_faction_labor_failure
			}
		}
	}

	immediate = {
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_labor
			}
			save_scope_as = labor_movement_scope
		}
		var:bolshevik_leader = {
			save_scope_as = bolshevik_leader
			set_temporary_event_immortality = { DURATION = 3 }
		}
		var:menshevik_leader = {
			save_scope_as = menshevik_leader
			set_temporary_event_immortality = { DURATION = 3 }
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = { exists = var:menshevik_leader }
			NOT = { exists = var:bolshevik_leader }
			NOT = { exists = scope:labor_movement_scope }
		}
	}

	after = {
		revoke_temporary_event_immortality = {
			CHARACTER = var:menshevik_leader
			CHARACTER = var:bolshevik_leader
		}
	}

	option = { # bolsheviks
		name = bolsheviks.4.a
		change_variable = {
			name = bolshevik_labor
			add = 25
		}
		change_variable = {
			name = menshevik_labor
			add = -25
		}
		clamp_variable = {
			name = bolshevik_labor
			min = 0
			max = 100
		}
		clamp_variable = {
			name = menshevik_labor
			min = 0
			max = 100
		}
		custom_tooltip = bol_labor_add
		custom_tooltip = men_labor_subtract
		# labor movement gets more activism
		scope:labor_movement_scope = {
			add_modifier = {
				name = bolshevik_labor_activism
				days = short_modifier_time
				is_decaying = yes
			}
		}
	}

	option = { # mensheviks
		name = bolsheviks.4.b
		change_variable = {
			name = menshevik_labor
			add = 25
		}
		change_variable = {
			name = bolshevik_labor
			add = -25
		}
		clamp_variable = {
			name = bolshevik_labor
			min = 0
			max = 100
		}
		clamp_variable = {
			name = menshevik_labor
			min = 0
			max = 100
		}
		custom_tooltip = men_labor_add
		custom_tooltip = bol_labor_subtract
		# labor movement gets more support
		scope:labor_movement_scope = {
			add_modifier = {
				name = menshevik_labor_support
				days = short_modifier_time
				is_decaying = yes
			}
		}
	}

	option = { # neither
		name = bolsheviks.4.c
		default_option = yes
		change_variable = {
			name = bolshevik_labor
			add = -25
		}
		change_variable = {
			name = menshevik_labor
			add = -25
		}
		clamp_variable = {
			name = bolshevik_labor
			min = 0
			max = 100
		}
		clamp_variable = {
			name = menshevik_labor
			min = 0
			max = 100
		}
		custom_tooltip = bol_labor_subtract
		custom_tooltip = men_labor_subtract
		# labor movement gets less support and radicalism
		scope:labor_movement_scope = {
			add_modifier = {
				name = social_faction_labor_failure
				days = short_modifier_time
				is_decaying = yes
			}
		}
	}

# Big socialist conference
# Everyone threatens to split with everyone else
bolsheviks.5 = {
	type = country_event
	placement = root

	duration = 4

	dlc = dlc019

	title = bolsheviks.5.t
	desc = bolsheviks.5.d
	flavor = bolsheviks.5.f

    event_image = {
		video = "ip5_soviet_deliberation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	trigger = {
		# Triggered by je_bolsheviks
		any_scope_character = {
			OR = {
				character_counts_as_menshevik = yes
				character_counts_as_bolshevik = yes
				character_counts_as_left_sr = yes
			}
			NOR = {
				has_modifier = most_popular_leftist_modifier
				has_modifier = discredited_leftist_modifier
			}
		}
		NOT = {
			any_political_movement = {
				OR = {
					has_modifier = popular_leftist_movement_support
					has_modifier = popular_leftist_movement_setback
				}
			}
		}
	}

	immediate = {
		ordered_scope_character = {
			limit = {
				OR = {
					character_counts_as_menshevik = yes
					character_counts_as_bolshevik = yes
					character_counts_as_left_sr = yes
				}
				NOR = {
					has_modifier = most_popular_leftist_modifier
					has_modifier = discredited_leftist_modifier
				}
			}
			order_by = popularity
			position = 0
			save_scope_as = most_popular_leftist
			set_temporary_event_immortality = { DURATION = 3 }
		}
		if = {
			limit = {
				scope:most_popular_leftist = {
					character_counts_as_menshevik = yes
				}
			}
			set_variable = menshevik_event_target
			if = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_socialist
					}
				}
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_socialist
					}
					save_scope_as = popular_leftists_movement
				}
			}
		}
		else_if = {
			limit = {
				scope:most_popular_leftist = {
					character_counts_as_bolshevik = yes
				}
			}
			set_variable = bolshevik_event_target
			if = {
				limit = {
					any_political_movement = {
						is_political_movement_type = movement_communist
					}
				}
				random_political_movement = {
					limit = {
						is_political_movement_type = movement_communist
					}
					save_scope_as = popular_leftists_movement
				}
			}
		}
		else_if = {
			limit = {
				scope:most_popular_leftist = {
					character_counts_as_left_sr = yes
				}

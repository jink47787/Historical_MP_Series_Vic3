rcw.1 = {
	type = country_event
	placement = root

	duration = 3

	dlc = dlc019

	title = rcw.1.t
	desc = rcw.1.d
	flavor = rcw.1.f

	event_image = {
		video = "europenorthamerica_before_the_battle"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	trigger = {
		# triggered by journal entry
		has_variable = rcw_russian_faction
	}

	cancellation_trigger = {
		OR = {
			NOT = { exists = scope:rcw_opposition_country }
			NOT = { exists = scope:rcw_dp }
			NOT = { exists = scope:scariest_russian_warlord_country }
			NOT = { exists = scope:scariest_russian_warlord }
			NOT = { exists = scope:friendliest_russian_warlord_country }
			NOT = { exists = scope:friendliest_russian_warlord }
		}
	}

	immediate = {
		random_diplomatic_play = {
			limit = {
				is_diplomatic_play_type = dp_revolution
				exists = c:RUS
				target_is = c:RUS
			}
			save_scope_as = rcw_dp
			initiator = {
				save_scope_as = rcw_opposition_country
			}
			ordered_scope_initiator_ally = {
				limit = {
					has_variable = rcw_russian_faction
				}
				order_by = rcw_faction_size
				position = 0
				check_range_bounds = no
				save_scope_as = scariest_russian_warlord_country
				ruler = {
					save_scope_as = scariest_russian_warlord
					set_temporary_event_immortality = { DURATION = 3 }
				}
			}
			ordered_scope_target_ally = {
				limit = {
					has_variable = rcw_russian_faction
				}
				order_by = rcw_faction_size
				position = 0
				check_range_bounds = no
				save_scope_as = friendliest_russian_warlord_country
				ruler = {
					save_scope_as = friendliest_russian_warlord
					set_temporary_event_immortality = { DURATION = 3 }
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:scariest_russian_warlord_country }
				exists = scope:rcw_opposition_country
			}
			scope:rcw_opposition_country = {
				save_scope_as = scariest_russian_warlord_country
				ruler ?= {
					save_scope_as = scariest_russian_warlord
					set_temporary_event_immortality = { DURATION = 3 }
				}
			}
		}
		if = {
			limit = {
				NOT = { exists = scope:friendliest_russian_warlord_country }
			}
			c:RUS ?= {
				save_scope_as = friendliest_russian_warlord_country
				ruler ?= {
					save_scope_as = friendliest_russian_warlord
					set_temporary_event_immortality = { DURATION = 3 }
				}
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = {
			CHARACTER = scope:friendliest_russian_warlord
			CHARACTER = scope:scariest_russian_warlord
		}
	}

	option = {
		name = rcw.1.a
		default_option = yes
	}

	option = {
		name = rcw_switch_country_button_1
		trigger = {
			c:RUS ?= this
			is_player = yes
			has_global_variable = rcw_switch_window
			global_var:rcw_switch_target_1 ?= {
				is_player = no
			}
		}
		play_as = global_var:rcw_switch_target_1
		ai_chance = {
			base = 0
		}
	}

	option = {
		name = rcw_switch_country_button_2
		trigger = {

			is_player = yes
			has_global_variable = rcw_switch_window
			global_var:rcw_switch_target_2 ?= {

			limit = { has_variable = rcw_pre_war_russian_state }

			value = 0.1
		}

			name = rcw_reward_warfare

			is_decaying = yes

# The Red/White Terror
	# Kill a bunch of civilians to reduce radicals/turmoil.obstinence in a state
		# state must have turmoil/radicals/obstinence
	# Or only kill some civilians and get nothing
rcw.4 = {
	type = country_event
	placement = scope:rcw_terror_state

	duration = 3

	dlc = dlc019

	title = {
		first_valid = {
			triggered_desc = {
				desc = rcw.4.t1
				trigger = {
					has_law_or_variant = law_type:law_council_republic
				}
			}
			triggered_desc = {
				desc = rcw.4.t2
				trigger = {
					NOT = { has_law_or_variant = law_type:law_council_republic }
					any_country = {
						country_has_primary_culture = cu:russian
						has_law_or_variant = law_type:law_council_republic
					}
				}
			}
			triggered_desc = {
				desc = rcw.4.t3
				trigger = {
					always = yes
				}
			}
		}
	}
	desc = rcw.4.d
	flavor = rcw.4.f

	event_image = {
		video = "unspecific_devastation"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	trigger = {
		# triggered by journal entry
		is_at_war = yes
		any_scope_state = {
			state_region = { is_homeland = cu:russian }
			turmoil > 0.15
			NOT = {
				has_modifier = rcw_terror
			}
		}
		any_scope_character = {
			has_role_of_type = general
		}
	}

	cancellation_trigger = {
		NOT = { exists = scope:rcw_terror_general }
		NOT = { exists = scope:rcw_terror_state }
	}

	cooldown = { days = very_short_modifier_time }

	immediate = {
		ordered_scope_state = {
			limit = {
				#turmoil > 0.15
				state_region = { is_homeland = cu:russian }
				NOT = {
					has_modifier = rcw_terror
				}
			}
			order_by = turmoil
			position = 0
			save_scope_as = rcw_terror_state
		}
		random_scope_character = {
			limit = {
				has_role_of_type = general
			}
			save_scope_as = rcw_terror_general
			set_temporary_event_immortality = { DURATION = 3 }
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:rcw_terror_general }
	}

	option = {
		name = rcw.4.a
		default_option = yes
		scope:rcw_terror_state = {
			add_modifier = {
				name = rcw_terror
				days = normal_modifier_time
				is_decaying = yes
			}
			if = {
				limit = {
					scope:rcw_terror_general = {
						OR = {
							has_trait = ruthless
							has_trait = imperious
							has_trait = wrathful
							has_trait = cruel
						}
					}
				}
				kill_population_percent_in_state = {
					percent = 0.1
				}
			}
			else = {
				kill_population_percent_in_state = {
					percent = 0.06
				}
			}
		}
	}

	option = {
		name = rcw.4.b
		scope:rcw_terror_state = {
			add_modifier = {
				name = rcw_terror
				days = normal_modifier_time
				is_decaying = yes
				multiplier = 0.5
			}
			if = {
				limit = {
					scope:rcw_terror_general = {
						OR = {
							has_trait = cautious
							has_trait = tactful
							has_trait = reserved
							has_trait = honorable
						}
					}
				}
				kill_population_percent_in_state = {
					percent = 0.02
				}
			}
			else = {
				kill_population_percent_in_state = {
					percent = 0.04
				}
			}
		}
	}
}

rcw.5 = {
	type = country_event
	placement = scope:kazan_state

	duration = 3

	dlc = dlc019

	fire_only_once = yes

	title = rcw.5.t
	desc = rcw.5.d
	flavor = rcw.5.f

	event_image = {
		video = "unspecific_armored_train"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	trigger = {
		# triggered by journal entry
		c:RUS ?= THIS
		NOT = { owns_entire_state_region = STATE_KAZAN }
		s:STATE_KAZAN = {
			any_scope_state = {
				owner = {
					OR = {
						has_war_with = ROOT
						is_diplomatic_play_enemy_of = ROOT
					}
				}
			}
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = {	exists = scope:kazan_state }
			NOT = {	exists = scope:kazan_owner }
			NOT = {	exists = scope:kazan_owner_capital }
			NOT = {	exists = scope:kazan_ruler }
		}
	}

	immediate = {
		s:STATE_KAZAN = {
			random_scope_state = {
				save_scope_as = kazan_state
				owner = {
					save_scope_as = kazan_owner
					capital = {
						save_scope_as = kazan_owner_capital
						state_region = {
							set_variable = russian_gold_state_region
						}
					}
					ruler = {
						save_scope_as = kazan_ruler
						set_temporary_event_immortality = { DURATION = 3 }
					}
				}
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:kazan_ruler }
	}

	option = {
		name = rcw.5.a
		default_option = yes
		custom_tooltip = {
			text = capture_gold_reserves_capital_tt
		}
		if = {
			limit = {
				gold_reserves >= 5
			}
			set_variable = {
				name = rus_gold_var
				value = {
					add = root.gold_reserves
					divide = 2
				}
			}
			add_treasury = {
				value = {
					add = root.gold_reserves
					divide = 2
					multiply = -1
				}
			}
		}

		add_modifier = {
			name = russian_gold_reserves_lost
			days = long_modifier_time
			is_decaying = yes
		}
	}
}

# Implosion of Russia event, formerly used the ID test.122
# check common/scripted_effects/05_russian_civil_war_scripted_effects.txt for specifics of how Russia is partitioned.
rcw.100 = {
	type = country_event
	hidden = yes
	dlc = dlc019
	immediate = {
		every_subject_or_below = {
			limit = {
				NOT = { any_overlord_or_above = { NOT = { ROOT = this } } }
				NOT = { is_subject_type = subject_type_personal_union }
			}
			make_independent = yes
			if = {
				limit = {
					capital ?= { rcw_is_in_any_theatre = yes }
				}
				set_variable = rcw_freed_russian_subject
			}
		}

		# We need to make new leaders for countries that were previously in personal unions (i.e. Finland)
		every_subject_or_below = {
			limit = {
				is_subject_type = subject_type_personal_union
			}
			make_independent = yes
			if = {
				limit = {
					capital ?= { rcw_is_in_any_theatre = yes }
				}
				set_variable = rcw_freed_russian_subject
			}

			random_primary_culture = {
				save_temporary_scope_as = independent_culture
			}

			# Pick the most powerful politician that isn't Russian
			ordered_interest_group = {
				order_by = ig_clout
				position = 0
				check_range_bounds = no
				if = {
					limit = {
						leader = { culture != cu:russian }
					}
					leader = {
						set_character_as_ruler = yes
					}

# je_russification complete event
russification.2 = {
	type = country_event
	placement = root

	duration = 3

	title = russification.2.t
	desc = russification.2.d
	flavor = russification.2.f

    event_image = {
		video = "ip5_countryside_procession"
	}

	dlc = dlc019

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	fire_only_once = yes

	trigger = {
		# triggered by je_russification
	}

	cancellation_trigger = {

	}

	immediate = {
		if = {
			limit = {
				any_scope_character = {
					has_role = character_role_favorite
				}
			}
			random_scope_character = {
				limit = {
					has_role = character_role_favorite
				}
				save_scope_as = relevant_character
				set_temporary_event_immortality = { DURATION = 3 } # scripted effect to ensure character doesn't die during the event
			}
		}
		else = {
			random_interest_group = {
				limit = {
					is_in_government = yes
				}
				leader = {
					save_scope_as = relevant_character
					set_temporary_event_immortality = { DURATION = 3 } # scripted effect to ensure character doesn't die during the event
				}
			}
		}
		set_variable = russification_complete
	}

	option = { # if there are any changes to the effects in this option, event_outcome_completed_effect_desc in je_russification must be updated
		name = russification.2.a
		default_option = yes
		if = {
			limit = {
				has_journal_entry = je_tsarpoints_master
			}
			je_tsarpoints_add_med = yes
		}
	    add_modifier = {
            name = modifier_je_russification_secured_ext
            days = very_long_modifier_time
        }
        every_interest_group = {
			limit ={
				is_in_government = yes
			}
			add_modifier = {
				name = modifier_je_russification_ig_str
				days = moderately_long_modifier_time
			}
		}
	}

	option = { # if there are any changes to the effects in this option, event_outcome_completed_effect_desc in je_russification must be updated
		name = russification.2.b
		if = {
			limit = {
				has_journal_entry = je_tsarpoints_master
			}
			je_tsarpoints_add_med = yes
		}
	    add_modifier = {
            name = modifier_je_russification_secured_int
            days = moderately_long_modifier_time
        }
        scope:relevant_character ?= {
			add_modifier = {
				name = modifier_je_russification_loyal_prominence_gain # update russification_2b_character_modifier_tt if there are changes to modifier_je_russification_loyal_prominence_gain
				days = long_modifier_time
				is_decaying = yes
			}
		}
	}

	after = {
		if = {
			limit = {
				exists = scope:relevant_character
			}
			revoke_temporary_event_immortality = { CHARACTER = scope:relevant_character } # scripted effect to remove any temporary immortality granted
		}
	}
}

# based on Edvard Isto's Hyökkäys (aka The Attack)
russification.4 = {
	type = country_event
	placement = scope:finland_scope.capital

	duration = 3

	title = russification.4.t
	desc = russification.4.d
	flavor = russification.4.f

    event_image = {
		video = "europenorthamerica_art_gallery"
	}

	dlc = dlc019

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	fire_only_once = yes

	trigger = {
		# triggered by je_russification
		ROOT = {
			NOR = {
				has_law_or_variant = law_type:law_protected_speech
				has_law_or_variant = law_type:law_guaranteed_liberties
			}
		}
		any_subject_or_below = {
			country_has_primary_culture = cu:finnish
		}
	}

	immediate = {
		random_subject_or_below = {
			limit = {
				country_has_primary_culture = cu:finnish
			}
			save_scope_as = finland_scope
		}
	}

	cancellation_trigger = {
		scope:finland_scope ?= {
			is_country_alive = no
		}
	}

	option = {
		name = russification.4.a
		default_option = yes
		if ={
			limit = {
				scope:finland_scope ?= {
					relations:root >= relations_threshold:cordial
				}
			}
			scope:finland_scope ?= {
				add_liberty_desire = 15
			}
		}
		else = {
			scope:finland_scope ?= {
				add_liberty_desire = 10
			}
		}
	}

	option = {
		name = russification.4.b
		trigger = {
			any_scope_pop = {
				culture = cu:finnish
			}
			NOT = {
				any_active_law = {
					has_amendment = amendment_type:amendment_degenerate_art
				}
			}
		}
		cu:finnish ?= {
			add_fervor = 10
		}
	}
}

# russification of an unincorporated state
russification.5 = {
	type = country_event
	placement = scope:relevant_state

	duration = 3

	title = russification.5.t
	desc = russification.5.d
	flavor = russification.5.f

	gui_window = event_window_1char_tabloid

	left_icon = scope:relevant_character

	dlc = dlc019

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	on_opened_soundeffect = "event:/SFX/Events/misc/1Character_Banner"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	cooldown = { days = very_long_modifier_time }

	trigger = {
		# triggered by je_russification
        any_scope_state = {
        	is_incorporated = no
        	NOR = {
        		has_modifier = modifier_regional_russification_efforts
        		has_variable = russify_this_state_var
        	}
			state_region = {
				OR = {
					is_homeland = cu:finnish
					is_homeland = cu:lithuanian
					is_homeland = cu:latvian
					is_homeland = cu:estonian
					is_homeland = cu:polish
				}
			}
		}
		any_scope_character = {
			AND = {
				is_ruler_of_own_country = no
				is_heir_of_own_country = no
				prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
				OR = {
					is_interest_group_type = ig_landowners
					is_interest_group_type = ig_armed_forces
				}
			}
			NOT = {
				has_variable = russify_this_state_leader_var
			}
		}
	}

	immediate = {
		random_scope_state = {
			limit = {
	        	is_incorporated = no
	        	NOR = {
	        		has_modifier = modifier_regional_russification_efforts
	        		has_variable = russify_this_state_var
	        	}
				state_region = {
					OR = {
						is_homeland = cu:finnish
						is_homeland = cu:lithuanian
						is_homeland = cu:latvian
						is_homeland = cu:estonian
						is_homeland = cu:polish
					}
				}
			}
			save_scope_as = relevant_state
			set_variable = russify_this_state_var
		}
		random_scope_character = {
		 	limit ={
				AND = {
					is_ruler_of_own_country = no
					is_heir_of_own_country = no
					prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
					OR = {
						is_interest_group_type = ig_landowners
						is_interest_group_type = ig_armed_forces
					}
				}
				NOT = {
					has_variable = russify_this_state_leader_var
				}
		 	}
		 	save_scope_as = relevant_character
		 	set_variable = russify_this_state_leader_var
		 	set_temporary_event_immortality = { DURATION = 3 } # scripted effect to ensure character doesn't die during the event
		}
		scope:relevant_state = {
			set_variable = { name = russify_linked_leader value = scope:relevant_character }
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = {
				has_journal_entry = je_russification
			}
			NOT = {
				exists = scope:relevant_character
			}
			scope:relevant_state ?= {
				NOT = {
					owner = root
				}
			}
		}
	}

	option = {
		name = russification.5.a
		default_option = yes
		scope:relevant_state = {
			if = {
				limit = {
					state_region = {
						is_homeland = cu:finnish
					}
				}
				add_radicals_in_state = {
					culture = cu:finnish
					value = small_radicals
				}
			}
			if = {
				limit = {
					state_region = {
						is_homeland = cu:lithuanian
					}
				}
				add_radicals_in_state = {
					culture = cu:lithuanian
					value = small_radicals
				}
			}
			if = {
				limit = {
					state_region = {
						is_homeland = cu:latvian
					}
				}
				add_radicals_in_state = {
					culture = cu:latvian
					value = small_radicals
				}
			}
			if = {
				limit = {
					state_region = {
						is_homeland = cu:estonian
					}
				}
				add_radicals_in_state = {
					culture = cu:estonian
					value = small_radicals
				}
			}
			if = {
				limit = {
					state_region = {
						is_homeland = cu:polish
					}
				}
				add_radicals_in_state = {
					culture = cu:polish
					value = small_radicals
				}
			}
		}
		add_modifier = {
			name = modifier_regional_russification_payments
			days = moderately_long_modifier_time
			multiplier = money_amount_multiplier_small
		}
		scope:relevant_state = {
			add_modifier = {
				name = modifier_regional_russification_efforts
				days = moderately_long_modifier_time
			}
		}
	}

	option = {
		name = russification.5.b
		scope:relevant_state = {
			add_modifier = {
				name = modifier_local_administration_autonomy
				days = moderately_long_modifier_time
			}

			remove_variable = russify_linked_leader

				days = normal_modifier_time
			}
		}

			remove_variable = russify_this_state_leader_var
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:relevant_character } # scripted effect to remove any temporary immortality granted
	}
}

# follow-up to russification.5.a
russification.6 = {
	type = country_event
	placement = scope:relevant_state

	duration = 3

	title = russification.6.t
	desc = russification.6.d
	flavor = russification.6.f

    event_image = {
		video = "ip5_countryside_procession"
	}

	dlc = dlc019

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	cooldown = { days = very_long_modifier_time }

	trigger = {
		# triggered by je_russification
		any_scope_state = {
			has_variable = russify_this_state_var
			is_incorporated = yes
			exists = var:russify_linked_leader
		}
	}

	immediate = {
		random_scope_state = {
			limit = {
				has_variable = russify_this_state_var
				is_incorporated = yes
				exists = var:russify_linked_leader
			}
			save_scope_as = relevant_state
			var:russify_linked_leader ?= {
				save_scope_as = relevant_character
				set_temporary_event_immortality = { DURATION = 3 } # scripted effect to ensure character doesn't die during the event
				interest_group ?= {
					save_scope_as = relevant_ig
				}
			}
		}
	}

	cancellation_trigger = {
		OR = {
			NOT = {
				exists = scope:relevant_character
			}
			scope:relevant_state ?= {
				NOT = {
					owner = root
				}
			}
		}
	}

	option = {
		name = russification.6.a
		default_option = yes
		scope:relevant_character.interest_group ?= {
			add_modifier = {
				name = modifier_russify_state_success
				days = moderately_long_modifier_time
			}
		}
	}

	option = {
		name = russification.6.b
		trigger = {
			scope:relevant_character = {
				prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
			}
		}
		scope:relevant_state = {
			add_modifier = {
				name = modifier_western_territory_incorporated
				days = moderately_long_modifier_time
			}
		}
	}

	after = {
		scope:relevant_state = {
			remove_variable = russify_this_state_var
			remove_variable = russify_linked_leader
		}
		scope:relevant_character = {
			remove_variable = russify_this_state_leader_var
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:relevant_character } # scripted effect to remove any temporary immortality granted
	}
}

# root = country
# scope:target_state = state
# scope:target_country = state owner
ai_state_ambition_claim = {
	scope:target_state = { has_claim_by = root }
}

ai_state_ambition_homeland = {
	scope:target_state = { is_homeland_of_country_cultures = root }
}

ai_state_ambition_capital_region = {
	scope:target_state.region ?= capital.region # Always desire states in home region
	can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
}

ai_state_ambition_split_state = { # The AI wants to unify split states
	scope:target_state = {
		is_split_state = yes
		state_region = {
			any_scope_state = {
				owner = root
				is_treaty_port = no
			}
		}
	}
}

ai_state_ambition_border = {
	scope:target_state = {
		any_neighbouring_state = {
			owner = {
				OR = {
					this = root
					is_subject_of = root
				}
			}
		}
	}
}

ai_state_ambition_colonial = {
	scope:target_country = { ai_country_is_valid_colonization_target = yes }
	can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
}

ai_state_ambition_overlord_gift = { # The AI wants nearby states if gifted them by its overlord
	has_strategic_adjacency = scope:target_state
	any_overlord_or_above = { this = scope:target_country }
}

ai_state_ambition_sphere = {
	scope:target_state.region ?= {
		root = {
			OR = {
				has_strategic_region_interest_tier = {
					strategic_region = prev
					value >= interest_tier_effect_can_start_diplomatic_plays_min_tier
				}
				has_region_stance = {
					region = prev
					stance = stance_conquer_region
				}
				has_region_stance = {
					region = prev
					stance = stance_colonize_region
				}
			}
		}
	}
	can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
}

ai_state_ambition_regional_objective = {
	OR = {
		is_regional_objective_target_state = {
			type = aro:ai_regional_objective_acquire_states
			target = scope:target_state
		}
		is_regional_objective_target_state = {
			type = aro:ai_regional_objective_recover_claims
			target = scope:target_state
		}
	}
	can_reach_target_state_to_conquer = { TARGET_STATE = scope:target_state }
}

# A state the AI treats as a core national interest rather than an opportunity
ai_state_has_core_ambition = {
	OR = {
		ai_state_ambition_claim = yes
		ai_state_ambition_homeland = yes
		ai_state_ambition_capital_region = yes
		ai_state_ambition_split_state = yes
		ai_state_ambition_regional_objective = yes
	}
}

ai_has_reasons_to_not_oppose_law = {
	OR = {
		has_law_commitment = scope:law.type
		any_political_movement = {
			law_enactment_stance = {
				law = scope:law.type
				value > neutral
			}
			OR = {
				political_movement_radicalism >= 0.75
				AND = {
					political_movement_radicalism >= 0.25
					any_influenced_interest_group = {
						is_in_government = yes
					}
					relevant_law_is_critical_to_modernization = yes
				}
			}
		}
		AND = {
			has_journal_entry = je_government_petition
			var:desired_law_var ?= scope:law
		}
		AND = {
			OR = {
				has_journal_entry = je_government_petition_negotiation_version
				has_journal_entry = je_government_petition_negotiation_version_b
			}
			any_interest_group = {
				has_variable = desired_law_var
				var:desired_law_var	?= scope:law
			}
		}
		AND = {
			OR = {
				has_journal_entry = je_law_imposition
				has_journal_entry = je_british_dictate_law
			}
			imposed_law ?= scope:law.type
		}
		je_tsarpoints_ai_wants_law = yes
		congress_poland_ai_wants_law = yes
	}
}

ai_has_systemic_conquest_reason = {
	OR = {
		$TARGET$ = { # They hold states we claim
			any_scope_state = { has_claim_by = root }
		}

		$TARGET$ = { # They hold our cultural homelands
			any_scope_state = { is_homeland_of_country_cultures = root }
		}

		has_diplomatic_pact = { # Our Rival
			who = $TARGET$
			type = rivalry
		}

		AND = { # Larger nations competing with eachother
			has_overlapping_interests = $TARGET$
			country_rank >= rank_value:major_power
			$TARGET$ = { country_rank >= rank_value:major_power }
		}

		AND = { # Our glorious right to self-determination..
			is_subject_of = $TARGET$
			liberty_desire >= 50
		}

		AND = { # ..and our ungrateful rebellious subjects!
			$TARGET$ = {
				is_subject_of = root
				liberty_desire >= 50
			}
		}

		AND = { # Global pariah
			country_rank >= rank_value:great_power
			$TARGET$ = { infamy >= infamy_threshold:pariah }
		}

		ai_is_regional_objective_local_country = {
			TYPE = aro:ai_regional_objective_acquire_states
			TARGET = $TARGET$
		}
	}

	can_reach_target_country_to_conquer = { TARGET_COUNTRY = $TARGET$ }
}

ai_has_systemic_subjugation_reason = {
	country_rank > $TARGET$.country_rank
	$TARGET$.country_rank <= rank_value:minor_power

	OR = {
		$TARGET$ = { ai_country_is_valid_colonization_target = yes }
		AND = {

				NOT = { is_subject_of = root }
			}
		}
		AND = { # Rebellious subject
			$TARGET$ = {
				is_subject_of = root
				liberty_desire >= 50
			}
		}

		ai_is_regional_objective_subjugate_target = { TARGET = $TARGET$ }
	}

	can_reach_target_country_to_subjugate = { TARGET_COUNTRY = $TARGET$ }
}

			has_strategic_adjacency = $TARGET$
		}

		ai_is_regional_objective_protect_target = { TARGET = $TARGET$ }
	}

	can_reach_target_country_to_protect = { TARGET_COUNTRY = $TARGET$ }
}

# Treaty Shorthands
has_friendly_diplomatic_pact = {
	OR = {
		any_scope_treaty = {
			binds = $TARGET$
			any_scope_article = {
				has_type = trade_privilege
				source_country = $TARGET$
			}
		}
		any_scope_treaty = {
			binds = $TARGET$
			any_scope_article = {
				has_type = alliance
			}
		}
		any_scope_treaty = {
			binds = $TARGET$
			any_scope_article = {
				has_type = defensive_pact
			}
		}
		any_scope_treaty = {
			binds = $TARGET$
			any_scope_article = {
				has_type = foreign_investment_rights
				source_country = $TARGET$
			}
		}
	}
}

is_at_war_with_overlord = {
	custom_tooltip = {
		text = is_at_war_with_overlord_tt
		overlord ?= { has_war_with = prev }
	}
}

is_china_unification_pair = {
	OR = {
		has_journal_entry = je_reunify_china
		global_var:chinese_central_government ?= THIS
		c:CHI ?= this
	}
	$TARGET$ = {
		OR = {
			has_journal_entry = je_reunify_china
			global_var:chinese_central_government ?= THIS
			country_has_primary_culture = cu:han
			country_has_primary_culture = cu:manchu
		}
	}
}

is_ethiopia_unification_pair = {
	NOT = { exists = c:ETH }
	has_journal_entry = je_age_of_princes
	$TARGET$ = {
		custom_tooltip = {
			text = ETH_tt
			can_form_nation = ETH
		}
	}
}

is_arabia_unification_pair = {
	NOT = { exists = c:ARA }
	can_form_nation = ARA
	$TARGET$ = {
		custom_tooltip = {
			text = ARA_tt
			can_form_nation = ARA
		}
	}
}

is_india_unification_pair = {
	OR = {
		custom_tooltip = {
			text = BHT_tt
			can_form_nation = BHT
		}
		c:BHT ?= this
	}
	$TARGET$ = {
		OR = {
			custom_tooltip = {
				text = BHT_tt
				can_form_nation = BHT
			}
			AND = {
				is_in_geographic_region = geographic_region_india
				custom_tooltip = {
					text = indian_primary_culture_tt
					has_indian_primary_culture = yes
				}
			}
		}
	}
}

is_indonesia_unification_pair = {
	NOT = { exists = c:IDN }
	custom_tooltip = {
		text = IDN_tt
		can_form_nation = IDN
	}
	$TARGET$ = {
		custom_tooltip = {
			text = IDN_tt
			can_form_nation = IDN
		}
	}
}

is_russia_unification_pair = {
	has_variable = rcw_russian_faction
	$TARGET$ = {
		custom_tooltip = {
			text = rcw_russian_faction_tt
			has_variable = rcw_russian_faction
		}
	}
}

is_scripted_unification_conquest = {
	OR = {
		is_china_unification_pair = { TARGET = $TARGET$ }
		is_ethiopia_unification_pair = { TARGET = $TARGET$ }
		is_arabia_unification_pair = { TARGET = $TARGET$ }
		is_india_unification_pair = { TARGET = $TARGET$ }
		is_indonesia_unification_pair = { TARGET = $TARGET$ }
		is_russia_unification_pair = { TARGET = $TARGET$ }
	}
}

has_any_strait_province_in_hierarchy = {
	OR = {
		has_any_strait_province = yes
		any_subject_or_below = {
			has_any_strait_province = yes
		}
	}
}

tsar_reactionary_ideology_trigger = {
	OR = {
		has_ideology = ideology:ideology_traditionalist
		has_ideology = ideology:ideology_jingoist_leader
		has_ideology = ideology:ideology_authoritarian
		has_ideology = ideology:ideology_ethno_nationalist
		has_ideology = ideology:ideology_fascist
	}
}

poland_whites_strong_trigger = {
	custom_tooltip = {
		text = poland_whites_strong
		je:je_congress_poland = {
			scripted_bar_progress ={
				name = je_congress_poland_progress_bar
				value <= congress_poland_whites_strong_threshold
			}
		}
	}
}

ai_russia_poland_is_unfriendly_neighbor = {
	OR = {
		relations:root <= relations_threshold:poor
		has_attitude = {
			who = root
			attitude = antagonistic
		}
		has_attitude = {
			who = root
			attitude = belligerent
		}
		has_attitude = {
			who = root
			attitude = domineering
		}
		any_rival_country = {
			this = root
		}
	}
}

ai_russia_poland_has_intransigent_agenda = {
	OR = {
		has_strategy = ai_strategy_conservative_agenda
		has_strategy = ai_strategy_reactionary_agenda
		has_strategy = ai_strategy_nationalist_agenda
	}
}

ai_russia_should_appease_poland = {
	has_journal_entry = je_congress_poland
	NOT = { has_journal_entry = je_tp_tsar_crown_of_perfection }
	has_variable = ai_russia_poland_threshold_var
	ai_russia_poland_appeasement_margin >= 0
}

ai_russia_is_intransigent_on_poland = {
	has_variable = ai_russia_poland_threshold_var
	var:ai_russia_poland_threshold_var >= 60
}

ai_russia_is_reformist_and_threatened_by_europe = {
	ai_russia_poland_has_intransigent_agenda = no
	NOT = { has_journal_entry = je_tp_tsar_crown_of_perfection }
	ruler ?= {
		tsar_reactionary_ideology_trigger = no
	}
	has_variable = ai_russia_poland_threat_var
	var:ai_russia_poland_threat_var >= 40
}

ai_russia_poland_bar_trending_white = {
	has_journal_entry = je_congress_poland
	OR = {
		congress_poland_bar_value <= -120
		AND = {
			has_variable = ai_russia_poland_bar_delta_var
			var:ai_russia_poland_bar_delta_var < 0
		}
	}
}

ai_russia_holds_the_line_on_poland = {
	ruler ?= {
		OR = {
			has_conservative_ideology = yes
			tsar_reactionary_ideology_trigger = yes
		}
	}
	ai_russia_poland_bar_trending_white = yes
}

congress_poland_ai_has_law_agenda = {
	has_variable = ai_poland_white
	has_journal_entry = je_congress_poland
	NOT = { has_modifier = congress_poland_laws }
	NAND = {
		OR = {
			has_law_or_variant = law_type:law_guaranteed_liberties
			has_law_or_variant = law_type:law_national_guard
		}
		OR = {
			has_law_or_variant = law_type:law_protected_speech
			has_law_or_variant = law_type:law_right_of_assembly
		}
	}
}

congress_poland_ai_wants_law = {
	has_variable = ai_poland_white
	has_journal_entry = je_congress_poland
	NOT = { has_modifier = congress_poland_laws }
	OR = {
		scope:law.type ?= law_type:law_guaranteed_liberties
		scope:law.type ?= law_type:law_national_guard
		scope:law.type ?= law_type:law_protected_speech
		scope:law.type ?= law_type:law_right_of_assembly
	}
}

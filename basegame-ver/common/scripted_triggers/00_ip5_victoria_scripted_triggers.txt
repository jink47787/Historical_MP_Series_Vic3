russian_character_is_marked_for_death_chance = {
    OR = {
        AND = { # Tsar Nicholas I must die
            has_template = RUS_nicholas_i
            game_date >= 1860.1.1
        }
        AND = { # Tsar Alexander III died young due to health issues
            has_template = RUS_alexander_iii
			has_role_of_type = ruler
            game_date >= 1894.1.1
        }
        AND = { # Alexei needs Rasputin or else he will die, unless je_tp_fav_rasputin is completed
        	has_template = RUS_alexei_ii
        	game_date >= 1910.1.1
        	NOR = {
        		has_modifier = modifier_faith_healing
        		has_modifier = modifier_holy_blessing
        	}
			owner = {
			    NOT = { has_variable = je_tp_fav_rasputin_complete_var }
			    OR = {
			        NOT = {
			        	any_scope_character = {
			        		has_template = RUS_grigoriy_rasputin
			        	}
			        }
			        any_scope_character = {
			            has_template = RUS_grigoriy_rasputin
			            NOT = { has_role = character_role_favorite }
			        }
			    }
			}
        }
    }
}

country_is_valid_qing_treaty_partner = {
	OR = {
		country_has_primary_culture = cu:manchu
		country_has_primary_culture = cu:han
	}
	is_country_type = unrecognized
	hidden_trigger = {
		is_player = no
	}
	OR = {
		trigger_if = {
			limit = {
				has_journal_entry = je_warlord_china
			}
			custom_tooltip = {
				text = china_explosion_point_2_tt
				var:china_warlord_explosion <= 3
			}
		}
		country_rank < rank_value:unrecognized_major_power
	}
	OR = {
		is_subject = no
		is_subject_of = ROOT
	}
	trigger_if = {
		limit = {
			country_rank >= rank_value:unrecognized_major_power
		}
		OR = {
			custom_tooltip = {
				text = war_with_china_not_russia_tt
				any_enemy_in_war = {
					NOT = {
						c:RUS ?= this
					}
				}
			}
			custom_tooltip = {
				text = owes_russia_obligation_tt
				c:RUS ?= {
					is_owed_obligation_by = PREV
				}
			}
			army_power_projection <= { # Army is less than half Russia's
				value = c:RUS.army_power_projection
				multiply = 0.5
			}
		}
	}
}

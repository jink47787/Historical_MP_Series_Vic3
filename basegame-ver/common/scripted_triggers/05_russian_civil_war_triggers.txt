		var:rcw_camp ?= scope:rcw_shade_picker.var:rcw_camp

	is_in_void = no
	character_is_valid_for_events = yes
	OR = {
		has_role_of_type = politician
		has_role_of_type = agitator
		has_role_of_type = general
		has_role_of_type = admiral
		is_interest_group_leader = yes
	}
	OR = {
		owner ?= {
			OR = {
				has_variable = rcw_partition_owner

		}
		AND = {
			is_in_exile_pool = yes

		AND = {
			is_historical = yes
			culture = cu:russian
		}
	}

rcw_leader_available = {

	NOT = { has_variable = rcw_leader_assigned }
	NOT = { has_variable = void_tsar }
	NOR = {
		is_ruler_of_own_country = yes
		is_heir_of_own_country = yes
	}
	trigger_if = {
		limit = {
			NOT = { owner ?= scope:rcw_leader_faction }
		}
		NOT = { has_variable = true_russian_revolution_leader }
		NOT = { c:RUS.var:russian_revolution_leader ?= this }
		NOT = {
			owner ?= {
				has_variable = rcw_partition_owner
				NOT = { this = c:RUS }
			}
		}
	}
	trigger_if = {
		limit = {
			is_historical = no
		}

	}
}

rcw_character_fits_camp = {
	trigger_if = {
		limit = {
			scope:rcw_leader_faction.var:rcw_camp ?= flag:rcw_red
		}
		OR = {
			rcw_character_ideology_red = yes
			has_ideology = ideology:ideology_moderate
		}
		OR = {
			is_interest_group_type = ig_trade_unions
			is_interest_group_type = ig_intelligentsia
		}
	}
	trigger_else_if = {
		limit = {
			scope:rcw_leader_faction.var:rcw_camp ?= flag:rcw_black
		}
		OR = {
			rcw_character_ideology_black = yes
			has_ideology = ideology:ideology_moderate
		}
		OR = {
			is_interest_group_type = ig_trade_unions
			is_interest_group_type = ig_rural_folk
		}
	}
	trigger_else_if = {
		limit = {
			scope:rcw_leader_faction.var:rcw_camp ?= flag:rcw_green
		}
		OR = {
			rcw_character_ideology_green = yes
			has_ideology = ideology:ideology_moderate
		}
		is_interest_group_type = ig_rural_folk
	}
	trigger_else_if = {
		limit = {
			scope:rcw_leader_faction.var:rcw_camp ?= flag:rcw_democratic
		}
		OR = {
			rcw_character_ideology_democratic = yes
			has_ideology = ideology:ideology_moderate
		}
		OR = {
			is_interest_group_type = ig_intelligentsia
			is_interest_group_type = ig_rural_folk
			is_interest_group_type = ig_petty_bourgeoisie
			is_interest_group_type = ig_trade_unions
		}
	}
	trigger_else = {
		OR = {
			rcw_character_ideology_white = yes
			has_ideology = ideology:ideology_moderate
		}
		OR = {
			is_interest_group_type = ig_armed_forces
			is_interest_group_type = ig_landowners
			is_interest_group_type = ig_devout
			is_interest_group_type = ig_industrialists
			is_interest_group_type = ig_petty_bourgeoisie
		}
	}
}

rcw_factions_are_hostile = {
	$FACTION$ = { has_variable = rcw_axis }
	$OTHER_FACTION$ = { has_variable = rcw_axis }
	OR = {
		rcw_axis_distance_at_least = { FACTION = $FACTION$ OTHER_FACTION = $OTHER_FACTION$ MIN_DISTANCE = 2 }
		AND = {
			has_global_variable = rcw_exhaustion_level
			global_var:rcw_exhaustion_level >= 3
		}
		AND = {
			has_global_variable = rcw_exhaustion_level

	}

rcw_can_be_reincorporated_by = {
	hidden_trigger = {
		rcw_is_active_russian_faction = yes
		NOT = { this = $ABSORBER$ }
		is_player = no
	}
	is_at_war = no
	is_diplomatic_play_committed_participant = no
	trigger_if = {
		limit = {
			has_global_variable = rcw_exhaustion_level
			global_var:rcw_exhaustion_level >= 3
		}
		gdp <= {
			value = $ABSORBER$.gdp
			multiply = 0.75
		}
	}
	trigger_else = {
		gdp <= {
			value = $ABSORBER$.gdp
			multiply = 0.5
		}
	}
	OR = {
		rcw_same_camp = { FACTION = this OTHER_FACTION = $ABSORBER$ }
		rcw_factions_are_aligned = { FACTION = this OTHER_FACTION = $ABSORBER$ }

achievement_the_could_have_been_king = {
	possible = {
		c:RUS ?= this
	}
	happened = {
		custom_tooltip = {
			text = achievement_the_could_have_been_king_tt
			heir = { has_template = RUS_alexei_ii }
		}
		has_gov_regency = yes
		custom_tooltip = {
			text = achievement_the_could_have_been_king_2_tt
			ruler = { has_ideology = ideology:ideology_fascist }
		}
	}
}

achievement_the_wide_dniepr_roars = {
	possible = {
		always = yes
	}
	happened = {
		country_has_primary_culture = cu:ukrainian
		cu:ukrainian = {
			custom_tooltip = {
				text = owns_all_ukrainian_homeland_states_tt
				any_scope_homeland_state = {
					count = all
					owner = root
				}
			}
			culture_current_fervor >= 40
		}
	}
}

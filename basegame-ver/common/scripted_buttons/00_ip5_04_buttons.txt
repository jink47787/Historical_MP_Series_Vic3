je_state_and_revolution_industrialisation_plan_button = {
	name = "je_state_and_revolution_industrialisation_plan_button_name"
	desc = "je_state_and_revolution_industrialisation_plan_button_desc"

	visible = {
		has_journal_entry = je_state_and_revolution
	}

	possible = {
		custom_tooltip = {
			text = state_and_revolution_economic_bar_above_pivot_tt
			scope:journal_entry ?= {
				"scripted_bar_progress(je_state_and_revolution_economic_bar)" > ROOT.var:state_and_revolution_economic_pivot_var
			}
		}
		OR = {
			has_law_or_variant = law_type:law_command_economy
			AND = {
				any_interest_group = {
					is_in_government = yes
					law_stance = {
						law = law_type:law_command_economy
						value > neutral
					}
				}
				custom_tooltip = {
					text = can_enact_command_economy_tt
					any_law = {
						law_is_available = yes
						this.type = law_type:law_command_economy
					}
				}
			}
		}
		NOT = {
			has_journal_entry = je_five_year_plan
		}
	}

	selected = {
		has_journal_entry = je_five_year_plan
	}

	effect = {
		if = {
			limit = {
				NOT = {
					has_law_or_variant = law_type:law_command_economy
				}
			}
			activate_law = law_type:law_command_economy
		}
		add_journal_entry = {
			type = je_five_year_plan
		}
		custom_tooltip = 5_year_plan_repeats_tt
	}

	ai_chance = {
		value = 15

		if = {
			limit = {
				has_strategy = ai_strategy_consolidation_agenda
			}
			add = 25
		}

		if = {
			limit = {
				has_strategy = ai_strategy_industrial_expansion
			}
			add = 10
		}

		if = {
			limit = {
				is_at_war = yes
			}
			multiply = 0
		}

		if = {
			limit = {
				in_default = yes
			}
			multiply = 0
		}
	}
}

		}

		if = {

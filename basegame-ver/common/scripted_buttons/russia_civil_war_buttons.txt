# Reincorporate Walords Russian Civil War
je_rcw_warlords_button = {
	name = "rcw_warlords_button"
	desc = "rcw_warlords_button_desc"

	visible = {
		has_variable = rcw_russian_faction
	}

	possible = {
		is_at_war = no
		is_diplomatic_play_committed_participant = no
		custom_tooltip = {
			text = rcw_warlords_button_target_tt
			any_in_global_list = {
				variable = rcw_russian_factions
				rcw_can_be_reincorporated_by = { ABSORBER = root }
			}
		}
	}

	cooldown = { days = very_short_modifier_time }

	ai_chance = {
		value = 15 # Try occasionally when possible
		if = {
			limit = {
				has_global_variable = rcw_ai_reincorporation_pause
			}
			value = 0
		}
	}

	effect = {
		custom_tooltip = {
			text = annex_a_russian_warlord
			ordered_in_global_list = {
				variable = rcw_russian_factions
				limit = {
					rcw_can_be_reincorporated_by = { ABSORBER = root }
				}
				order_by = gdp
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = rus_warlord_annexed_scope
			}
		}
		if = {
			limit = {
				exists = scope:rus_warlord_annexed_scope
			}
			annex_with_incorporation = scope:rus_warlord_annexed_scope
			hidden_effect = {
				if = {
					limit = {
						is_player = no
					}
					set_global_variable = {
						name = rcw_ai_reincorporation_pause
						days = very_short_modifier_time
					}
				}
			}
		}
	}
}

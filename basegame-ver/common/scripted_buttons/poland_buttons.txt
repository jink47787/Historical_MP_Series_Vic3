# Reopen Warsaw University
je_congress_poland_university_button = {
	name = "warsaw_uni_button"
	desc = "warsaw_uni_button_desc"

	visible = {
		OR = {
			scope:congress_poland_scope ?= THIS
			scope:russia_poland_scope ?= THIS
		}
		exists = scope:warsaw_state_scope
	}

	possible = {
		hidden_trigger = {
			scope:congress_poland_scope ?= {
				has_variable = poland_universities_closed_var
			}
		}
		custom_tooltip = {
			text = poland_owns_warsaw
			scope:warsaw_state_scope ?= {
				c:POL ?= owner
			}
		}
		OR = {
			custom_tooltip = {
				text = poland_has_conciliatory_viceroy
				scope:congress_poland_scope ?= {
					ruler = { has_variable = poland_conciliatory }
				}
			}
			scope:congress_poland_scope ?= {
				is_subject_type = subject_type_personal_union
			}
		}
		trigger_if = {
			limit = {
				scope:russia_poland_scope ?= THIS
			}
			NOT = { has_modifier = appeased_poland_modifier }
		}
	}

	# Poland does this unless it backs the Reds, Russia only to appease Poland
	ai_chance = {
		value = 0
		if = {
			limit = {
				scope:congress_poland_scope ?= this
				NOT = { has_variable = ai_poland_red }
			}
			add = 100
		}
		if = {
			limit = {
				scope:russia_poland_scope ?= this
				ai_russia_should_appease_poland = yes
				ai_russia_poland_bar_trending_white = no
			}
			add = 100
		}
	}

	effect = {
		custom_tooltip = {
			text = poland_can_build_unis_tt
			scope:congress_poland_scope = {
				remove_variable = poland_universities_closed_var
			}
		}
		scope:warsaw_state_scope = {
			create_building = {
 				building="building_university"
 				level=2
 				reserves=1
 			}
		}
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
			}
			add_modifier = {
				name = appeased_poland_modifier
				days = normal_modifier_time
			}
		}
		scope:congress_poland_scope = {
			post_notification = poland_uni
		}
		scope:russia_poland_scope = {
			post_notification = poland_uni
		}
	}

	selected = {
		scope:congress_poland_scope ?= {
			NOT = { has_variable = poland_universities_closed_var }
		}
	}
}

# Appoint Viceroy
je_congress_poland_viceroy_button = {
	name = "poland_viceroy_button"
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					scope:russia_poland_scope = {
						is_player = no
						ruler = {
							tsar_reactionary_ideology_trigger = yes
						}
					}
				}
				desc = pol_viceroy_unlikely
			}
			triggered_desc = {
				trigger = {
					scope:russia_poland_scope = {
						is_player = no
						ruler = {
							tsar_reactionary_ideology_trigger = no
						}
					}
				}
				desc = pol_viceroy_likely
			}
			triggered_desc = {
				trigger = {
					always = yes
				}
				desc = pol_viceroy_for_russia
			}
		}
	}

	cooldown = { days = normal_modifier_time }

	visible = {
		scope:congress_poland_scope ?= {
			NOT = { is_subject_type = subject_type_personal_union }
		}
		OR = {
			scope:congress_poland_scope ?= THIS
			scope:russia_poland_scope ?= THIS
		}
	}

	possible = {
		custom_tooltip = {
			text = "poland_viceroy_cooldown_tt"
			NOT = { has_global_variable = poland_viceroy_cooldown }
		}
		trigger_if = {
			limit = {
				scope:congress_poland_scope ?= THIS
			}
			custom_tooltip = {
				text = poland_already_conciliatory
				ruler = {
					NOT = { has_variable = poland_conciliatory }
				}
			}
			custom_tooltip = {
				text = poland_viceroy_bar_trigger
				je:je_congress_poland = {
					scripted_bar_progress ={
						name = je_congress_poland_progress_bar
						value <= congress_poland_viceroy_request_threshold
					}
				}
			}
		}
	}

	ai_chance = {
		value = 0
		# Poland does this if the tsar might be amenable, unless it backs the Reds
		if = {
			limit = {
				scope:congress_poland_scope ?= this
				NOT = { has_variable = ai_poland_red }
				scope:russia_poland_scope = {
					ruler = {
						tsar_reactionary_ideology_trigger = no
					}
				}
			}
			add = 100
		}
		# Russia appoints a nicer governor once the Reds grow too strong
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
				ai_russia_should_appease_poland = yes
				ai_russia_poland_bar_trending_white = no
				scope:congress_poland_scope = {
					ruler = {
						NOT = { has_variable = poland_conciliatory }
					}
				}
			}
			add = 100
		}
		# An intransigent Russia brings back a harsher governor once the Reds are contained
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
				ai_russia_should_appease_poland = no
				ai_russia_is_intransigent_on_poland = yes
				congress_poland_bar_value <= 0
				scope:congress_poland_scope = {
					ruler = {
						has_variable = poland_conciliatory
					}
				}
			}
			add = 100
		}
	}

	effect = {
		set_global_variable = {
			name = poland_viceroy_cooldown
			days = normal_modifier_time
		}
		scope:russia_poland_scope = {
			trigger_event = {
				id = congress_poland.1
				days = 0
				popup = yes
			}
		}
	}
}

# Expand Congress Poland
je_congress_poland_expand_button = {
	name = "expand_poland_button"
	desc = "expand_poland_button_desc"

	cooldown = { days = normal_modifier_time }

	visible = {
		scope:congress_poland_scope ?= THIS
	}

	possible = {
		custom_tooltip = {
			text = "poland_territory_cooldown_tt"
			NOT = { has_global_variable = poland_territory_cooldown }
		}
		scope:russia_poland_scope ?= {
			any_scope_state = {
				state_region = {
					is_homeland = cu:polish
				}
				NOT = { state_region = s:STATE_MAZOVIA }
			}
		}
		poland_whites_strong_trigger = yes
	}

	# A White Poland only asks when Russia is in a position to accept, since a refusal emboldens the Reds
	ai_chance = {
		value = 0
		if = {
			limit = {
				OR = {
					NOT = { has_variable = ai_poland_white }
					scope:russia_poland_scope ?= {
						ai_russia_is_reformist_and_threatened_by_europe = yes
					}
				}
			}
			add = 100
		}
	}

	effect = {
		set_global_variable = {
			name = poland_territory_cooldown
			days = normal_modifier_time
		}
		if = {
			limit = {
				scope:congress_poland_scope ?= this
			}
			custom_tooltip = {
				text = poland_red_advance
				je:je_congress_poland = {
					add_progress = {
						value = congress_poland_red_shove
						name = je_congress_poland_progress_bar
					}
				}
			}
		}
		custom_tooltip = russia_decides_expand_poland
		scope:russia_poland_scope = {
			trigger_event = {
				id = congress_poland.2
				days = 0
				popup = yes
			}
		}
	}
}

# Restore the Polish Army
je_congress_poland_army_button = {
	name = "poland_army_button"
	desc = "poland_army_button_desc"

	visible = {
		OR = {
			scope:congress_poland_scope ?= THIS
			scope:russia_poland_scope ?= THIS
		}
	}

	possible = {
		scope:congress_poland_scope ?= {
			has_modifier = congress_poland_restrictions
			OR = {
				is_subject_type = subject_type_personal_union
				custom_tooltip = {
					text = poland_has_conciliatory_viceroy
					ruler ?= {
						has_variable = poland_conciliatory
					}
				}
			}
		}
		trigger_if = {
			limit = {
				scope:congress_poland_scope ?= this
			}
			poland_whites_strong_trigger = yes
		}
		trigger_if = {
			limit = {
				scope:russia_poland_scope ?= THIS
			}
			NOT = { has_modifier = appeased_poland_modifier }
		}
	}

	ai_chance = {
		value = 0
		# Poland always does this if it has the white influence, a White Poland only once the Sejm is restored
		# Russia will never initiate this
		if = {
			limit = {
				scope:congress_poland_scope ?= this
				OR = {
					NOT = { has_variable = ai_poland_white }
					NOT = { has_modifier = congress_poland_laws }
				}
			}
			add = 100
		}
	}

	effect = {
		scope:congress_poland_scope ?= {
			if = {
				limit = {
					has_modifier = congress_poland_restrictions
				}
				remove_modifier = congress_poland_restrictions
			}
		}
		if = {
			limit = {
				scope:russia_poland_scope ?= THIS
			}
			add_modifier = {
				name = appeased_poland_modifier
				days = normal_modifier_time
			}
		}
		custom_tooltip = {
			text = poland_red_advance_large
			je:je_congress_poland = {
				add_progress = {
					value = congress_poland_red_shove_large
					name = je_congress_poland_progress_bar
				}
			}
		}
		scope:congress_poland_scope = {
			post_notification = poland_army_toast
		}
		scope:russia_poland_scope = {
			post_notification = poland_army_toast
		}
	}

	selected = {
		scope:congress_poland_scope ?= {
			NOT = { has_modifier = congress_poland_restrictions }
		}
	}
}

amendment_factory_councils = {

	parent = law_factory_councils

	allowed_laws = {
		law_combination_acts
		law_guild_system
		law_anti_strike_laws
		law_corporatized_unions
		law_right_to_associate
	}

	modifier = {
		interest_group_ig_trade_unions_pol_str_mult = 0.25
		state_pop_support_movement_labor_mult = -0.10
		state_pop_support_movement_communist_mult = 0.25
		state_pop_support_movement_socialist_mult = 0.25
		state_pop_support_movement_anarchist_mult = 0.25
		building_group_bg_manufacturing_allowed_collectivization_add = 0.1
		building_group_bg_extraction_allowed_collectivization_add = 0.1
		building_group_bg_power_allowed_collectivization_add = 0.1
		building_group_bg_private_infrastructure_allowed_collectivization_add = 0.1
		building_group_bg_arts_allowed_collectivization_add = 0.1
	}

	possible = {
		has_technology_researched = socialism
	}

	would_sponsor = {
		is_interest_group_type = ig_trade_unions
	}

	on_activate = {
		ig:ig_trade_unions ?= {
			remove_ideology = ideology_populist
			add_ideology = ideology_communist
			add_ideology = ideology_internationalist
		}
	}

	on_deactivate = {
		every_political_movement = {
			limit = {
				law_stance = {
					law = law_type:law_factory_councils
					value > neutral
				}
			}
			add_modifier = {
				name = modifier_factory_councils_amendment_repealed
				days = very_short_modifier_time
			}
		}
	}

	ai_will_revoke = {
		any_interest_group = {
			is_in_government = yes
			law_stance = {
				law = law_type:law_factory_councils
				value < neutral
			}
		}
	}

	can_repeal = {
		legitimacy >= legitimacy_to_repeal_amendment
	}

	amendment_activism_multiplier = 1
}

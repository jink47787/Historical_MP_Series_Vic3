je_bolshevik_bar = {
	name = "je_bolshevik_bar"
	desc = "je_bolshevik_bar_desc"

	default_bad = yes

	monthly_progress = {
		progress_condition = {
			limit = {
				owner = {
					has_variable = bolshevik_pivot_var
					has_variable = bolshevik_current_var
				}
			}
			add = {
				desc = "bolshevik_bar_moderating_factor_tt"
				value = owner.var:bolshevik_pivot_var
				subtract = owner.var:bolshevik_current_var
				multiply = socialist_faction_bar_drift_factor
			}
		}
	}

	start_value = 0
	min_value = 0
	max_value = 200
}

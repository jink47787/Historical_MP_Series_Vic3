building_government_administration = {
	building_group = bg_bureaucracy
	city_type = city

	levels_per_mesh = 50

	unlocking_technologies = {
		tech_bureaucracy
	}

	should_auto_expand = {
		occupancy >= 0.8
		OR = {
			state.tax_capacity < state.tax_capacity_usage
			owner = {
				bureaucracy < -10
			}
		}
	}

	production_method_groups = {
		pmg_base_building_government_administration
		pmg_government_administration_bureaucrat_professionalism
		pmg_government_administration_religious_administration
	}

	icon = "gfx/interface/icons/building_icons/building_government_administration.dds"

	required_construction = construction_cost_very_low

	ai_value = {
		value = 1000

		if = { # Bureaucratic Reform agenda - only the agenda's target states, while they still lack tax capacity
			limit = {
				owner ?= {
					has_journal_entry = je_tp_pool_bureaucratic_reform
					is_target_in_variable_list = {
						name = tp_bureaucratic_reform_states
						target = prev
					}
				}
				tax_capacity < tax_capacity_usage
			}
			add = 1000
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}

building_university = {

	building_group = bg_technology

	city_type = city

	levels_per_mesh = 50

	unlocking_technologies = {
		academia
	}

	production_method_groups = {
		pmg_base_building_university
		pmg_university_academia
		pmg_principle_freedom_of_movement_3
	}

	icon = "gfx/interface/icons/building_icons/building_university.dds"

	required_construction = construction_cost_medium

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"

	can_build_government = {
		custom_tooltip = {
			text = pol_cannot_build_unis
			owner = {
				NOT = { has_variable = poland_universities_closed_var }
			}
		}
	}

	ai_value = {
		value = 2000 # Higher base value

		if = { # British dictates
			limit = {
				owner = {
					has_journal_entry = je_british_dictate_universities
				}
				OR = {
					state_region = s:STATE_WEST_BENGAL
					state_region = s:STATE_BOMBAY
					state_region = s:STATE_MADRAS
				}
				NOR = {
					has_building = building_university
					any_scope_building = {
						is_building_type = building_university
						level > 3
					}
				}
			}
			add = 2000
		}
		
		if = { # Taming the North
			limit = {
				owner = {
					has_journal_entry = je_taming_the_north
				}
				state_region = s:STATE_HOKKAIDO
				NOT = {
					has_building = building_university
				}
			}
			add = 10000
		}

		if = { # Educational Reform agenda - only while innovation is below cap
			limit = {
				owner = {
					has_journal_entry = je_tp_pool_educational_reform
					country_innovation < max_innovation
				}
			}
			add = 2000
		}

		if = {
			limit = {
				OR = {
					AND = {
						owner ?= { has_variable = ai_ukraine_renaissance_accelerate }
						state_region = { is_homeland = cu:ukrainian }
					}
					AND = {
						owner ?= { has_variable = ai_belarus_renaissance_accelerate }
						state_region = { is_homeland = cu:byelorussian }
					}
				}
			}
			add = 1000
		}

		if = {
			limit = {
				has_building = building_university
				owner ?= {
					has_variable = ai_poland_white
					has_journal_entry = je_congress_poland
					NOT = { has_variable = poland_university_requirement_met_var }
				}
			}
			add = 1000
		}
	}
}

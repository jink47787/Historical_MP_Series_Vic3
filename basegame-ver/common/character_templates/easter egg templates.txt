    first_name = E_A
    last_name = Nasir
    dna = dna_ea_nasir
    historical = yes
    age = 3621
    culture = cu:mashriqi
    religion = random_accepted
    interest_group = ig_industrialists
    ideology = ideology_market_liberal
    home_region = STATE_BASRA # Ur
    traits = {
        arrogant # "Who am I that you are treating me with such contempt?"
        direct # "If you will take it, take it - If you will not take it, go away!"
        experienced_entrepreneur # Or so I have been told by the merchants guild of Dilmun
    }
    executive_usage = {
        country_trigger = {
            has_game_rule = allow_fantastical_content
            owns_entire_state_region = STATE_BASRA
            OR = {
                c:IRQ ?= this
                AND = {
                    "goods_production_rank(g:copper)" = 1
                    market = {
                        mg:copper ?= {
                            market_goods_production >= 1000
                        }
                    }
                }
            }
        }
		company_trigger = {
			is_company_type = company_type:company_basic_metal_mining
		}
		earliest_usage_date = 1836.1.1
		latest_usage_date = 1935.12.30
		chance = 1
	}

    on_created = {
        set_character_immortal = yes
    }
}

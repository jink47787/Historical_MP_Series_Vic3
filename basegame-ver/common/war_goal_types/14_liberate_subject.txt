liberate_subject = {
	icon = "gfx/interface/icons/war_goals/liberate_subject.dds"

	kind = liberate_subject

	settings = {
		overlord_is_stakeholder
		requires_interest
		targets_enemy_subject

		# Validation
		has_other_stakeholder
		validate_conflicts_war_goals_all
		validate_conflicts_existing_subject
	}

	execution_priority = 58

	contestion_type = control_half_target_country_states

	side_switch = always

	target_type = country

	possible = {
		# trigger to determine if a goal with its target data is listed when selecting a war goal in the diplo play panel
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
	}

	valid = {
		# trigger in addition to some basic validation code-side
		# scopes: root = holder, creator_country, diplomatic_play, target_country, target_state, stakeholder, target_region, article_options
		scope:target_country = {
			overlord = scope:stakeholder
		}
	}

	maneuvers = {
		value = 0

		add = {
			desc = "MANEUVERS_BASE_VALUE"
			value = 20
		}
	}

	infamy_severity = {
		value = 0.2

		# No infamy if target has high liberty desire
		if = {
			limit = {
				scope:target_country = {
					liberty_desire >= define:NDiplomacy|WAR_GOAL_INCREASE_AUTONOMY_MIN_LIBERTY_DESIRE
				}
			}
			multiply = 0
		}
	}

	infamy = {
		value = 0

		scope:target_country = {
			add = country_infamy_value
		}
	}

	on_enforced = {
		# script effect on top of the predefined code effect
	}

	mirrored_wargoal = {
		method = restore_subject
	}

	ai = {
		is_significant_demand = no
	}
}

je_gran_colombia = {
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_foreign_affairs
	dlc = dlc008

	activation_description = {
		always = yes
	}

    is_shown_in_lobby = {
        has_dlc_feature = rp1_content #Colossus of the South
        country_has_primary_culture = cu:north_andean
        is_in_geographic_region = geographic_region_north_andes
        NOT = {
            OR = {
                country_tier = empire
                country_tier = hegemony
            }
        }
    }

    is_shown_when_inactive = {
        country_has_primary_culture = cu:north_andean
        is_in_geographic_region = geographic_region_north_andes
		has_dlc_feature = rp1_content
        NOT = {
            OR = {
                country_tier = empire
                country_tier = hegemony
            }
        }
    }
    possible = {
        country_rank >= rank_value:minor_power
        army_power_projection >= 500
        country_has_primary_culture = cu:north_andean
    }
    immediate = {
        trigger_event = { id = gran_colombia.1 popup = yes }
    }
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = formed_gran_colombia_tt
                c:GCO ?= this
            }
        }
        effect = {
            trigger_event = { id = gran_colombia.2 popup = yes }
            if = {
                limit = {
                    has_modifier = modifier_seeking_gran_colombia
                }
                remove_modifier = modifier_seeking_gran_colombia
            }
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            OR = {
                NOT = {
                    country_has_primary_culture = cu:north_andean
                }
                custom_tooltip = {
                    text = has_rejected_gran_colombia_tt
                    has_variable = rejected_gran_colombia
                }
            }
        }
        effect = {
            if = {
                limit = {
                    has_modifier = modifier_seeking_gran_colombia
                }
                remove_modifier = modifier_seeking_gran_colombia
            }
            je_gran_colombia_cleanup = yes
        }
    }

    weight = 1000


	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_andean_federation = {
    icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_foreign_affairs
	dlc = dlc008

	activation_description = {
		always = yes
	}

    is_shown_in_lobby = {
        has_dlc_feature = rp1_content #Colossus of the South
        OR = {
            country_has_primary_culture = cu:north_andean
            country_has_primary_culture = cu:south_andean
        }
        is_in_geographic_region = geographic_region_south_america
    }

    is_shown_when_inactive = {
        OR = {
            country_has_primary_culture = cu:north_andean
            country_has_primary_culture = cu:south_andean
        }
        is_in_geographic_region = geographic_region_south_america
        NOR = {
            has_variable = rejected_gran_colombia
            has_variable = rejected_la_plata
        }
		has_dlc_feature = rp1_content
    }
    possible = {
        OR = {
            country_has_primary_culture = cu:north_andean
            country_has_primary_culture = cu:south_andean
        }
        country_rank >= rank_value:major_power
        OR = {
            custom_tooltip = {
                text = andean_federation_reqs_tt
                OR = {
                    c:GCO ?= this
                    c:PBC ?= this
                    c:PLT ?= this
                }
            }
            any_scope_state = {
                is_in_geographic_region = geographic_region_south_america
                is_incorporated = yes
                count >= 10
            }
        }
    }
    immediate = {
        trigger_event = { id = andean_federation.1 popup = yes }
    }
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            custom_tooltip = {
                text = formed_andean_federation_tt
                c:FND ?= this
            }
        }
        effect = {
            trigger_event = { id = andean_federation.2 popup = yes }
            show_as_tooltip = {
                if = {
                    limit = {
                        NOT = {
                            country_has_primary_culture = cu:south_andean
                        }
                    }
                    add_primary_culture = cu:south_andean
                }
                if = {
                    limit = {
                        NOT = {
                            country_has_primary_culture = cu:north_andean
                        }
                    }
                    add_primary_culture = cu:north_andean
                }
            }
        }
    }

    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            OR = {
                NOT = {
                    country_has_primary_culture = cu:north_andean
                    country_has_primary_culture = cu:south_andean
                }
                custom_tooltip = {
                    text = has_rejected_andean_federation_tt
                    has_variable = rejected_andean_federation
                }
            }
        }
        effect = {
            je_andean_federation_cleanup = yes
        }
    }

    weight = 1000

    	
	should_be_pinned_by_default_uninvolved_or_context = yes
}

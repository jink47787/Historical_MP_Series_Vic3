je_conference_of_london = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_global_international_situations

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		OR = {
			c:BEL ?= THIS
			c:NET ?= THIS
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = no
	should_be_pinned_by_default_involved = yes

	can_revolution_inherit = yes

	should_be_involved = {
		OR = {
			c:BEL ?= this
			c:NET ?= this
		}
	}

	should_show_when_not_involved = {
		OR = {
			has_strategic_region_interest_tier = {
				strategic_region = sr:region_western_europe
				value >= 3
			}
			capital ?= {
				is_in_geographic_region = geographic_region_western_europe
			}
		}
	}

	scripted_button = favour_belgium_button
    scripted_button = favour_netherlands_button
    scripted_button = partition_compromise_button
    scripted_button = luxembourg_compromise_button
    scripted_button = london_conference_approve_button
    scripted_button = london_conference_disapprove_button

	immediate = {
		c:BEL ?= {
			save_scope_as = belgium_scope
		}
		c:NET ?= {
			save_scope_as = netherlands_scope
		}

		scope:journal_entry ?= {
			if = {
				limit = {
					NOT = {
						exists = var:je_conference_of_london_proposal_1
					}
				}
				set_variable = {
					name = je_conference_of_london_proposal_1
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						exists = var:je_conference_of_london_proposal_2
					}
				}
				set_variable = {
					name = je_conference_of_london_proposal_2
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						exists = var:je_conference_of_london_proposal_3
					}
				}
				set_variable = {
					name = je_conference_of_london_proposal_3
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						exists = var:je_conference_of_london_proposal_4
					}
				}
				set_variable = {
					name = je_conference_of_london_proposal_4
					value = 0
				}
			}
			if = {
				limit = {
					NOT = {
						exists = var:je_conference_of_london_timer
					}
				}
				set_variable = {
					name = je_conference_of_london_timer
					value = 6
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_conference_of_london_tt_1
				has_global_variable = london_treaty_has_concluded_global_var

			}
			custom_tooltip = {
				text = je_conference_of_london_tt_2
				has_global_variable = london_treaty_signed_bel_global_var
			}
			custom_tooltip = {
				text = je_conference_of_london_tt_4
				has_global_variable = london_treaty_signed_net_global_var
			}
		}
		effect_all_involved = {
			trigger_event = { id = treaty_of_london_events.4 popup = yes }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_conference_of_london_tt_3
					OR = {
						has_global_variable = london_treaty_not_signed_net_global_var
						has_global_variable = london_treaty_not_signed_bel_global_var
					}
				}
				custom_tooltip = {
					text = je_conference_of_london_tt_5
					c:BEL ?= {
						has_war_with = c:NET
					}
				}
			}
		}
		effect_all_involved = {
			trigger_event = { id = treaty_of_london_events.5 popup = yes }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOR = {
				exists = c:NET
				exists = c:BEL
			}
		}
		effect = {
			je_conference_of_london_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					scope:journal_entry.var:je_conference_of_london_timer >= 1
				}
				scope:journal_entry ?= {
					change_variable = {
						name = je_conference_of_london_timer
						add = -1
					}
				}
			}
			if = {
				limit = {
					exists = global_var:je_conference_of_london_acceptance_timer
					global_var:je_conference_of_london_acceptance_timer >= 1
				}
				change_global_variable = {
					name = je_conference_of_london_acceptance_timer
					add = -1
				}
			}
			if = {
				limit = {
					scope:journal_entry.var:je_conference_of_london_timer <= 0
				}
				if = { # If a great power has not chosen a proposal by now, a random one will be chosen for it.
					limit = {
						NOT = {
							exists = scope:journal_entry.var:je_conference_of_london_random_votes_assigned
						}
						any_country = {
							is_involved_in_journal_entry = je_conference_of_london
							NOR = {
								c:NET ?= this
								c:BEL ?= this
							}
							NOR = {
								has_variable = favour_belgium_button_var # proposal 1
								has_variable = favour_netherlands_button_var #proposal 2
								has_variable = partition_compromise_button_var # proposal 3
								has_variable = luxembourg_compromise_button_var # proposal 4
							}
						}
					}
					scope:journal_entry ?= {
						set_variable = je_conference_of_london_random_votes_assigned # this effect should only run once for performance reasons
					}
					every_country = {
						limit = {
							is_involved_in_journal_entry = je_conference_of_london
							NOR = {
								c:NET ?= this
								c:BEL ?= this
							}
							NOR = {
								has_variable = favour_belgium_button_var # proposal 1
								has_variable = favour_netherlands_button_var #proposal 2
								has_variable = partition_compromise_button_var # proposal 3
								has_variable = luxembourg_compromise_button_var # proposal 4
							}
						}
						save_temporary_scope_as = voting_country
						random_list = {
							1 = {
								set_variable = favour_belgium_button_var
								scope:journal_entry ?= {
									change_variable = {
										name = je_conference_of_london_proposal_1
										add = scope:voting_country.prestige
									}
								}
								scope:belgium_scope = {
									change_relations = { country = scope:voting_country value = 20 }
									create_diplomatic_catalyst = {
										type = catalyst_event_positive
										target = scope:voting_country
									}
								}
								scope:netherlands_scope = {
									change_relations = { country = scope:voting_country value = -20 }
									create_diplomatic_catalyst = {
										type = catalyst_event_negative
										target = scope:voting_country
									}
								}
							}
							1 = {
								set_variable = favour_netherlands_button_var
								scope:journal_entry ?= {
									change_variable = {
										name = je_conference_of_london_proposal_2
										add = scope:voting_country.prestige
									}
								}
								scope:netherlands_scope = {
									change_relations = { country = scope:voting_country value = 20 }
									create_diplomatic_catalyst = {
										type = catalyst_event_positive
										target = scope:voting_country
									}
								}
								scope:belgium_scope = {
									change_relations = { country = scope:voting_country value = -20 }
									create_diplomatic_catalyst = {
										type = catalyst_event_negative
										target = scope:voting_country
									}
								}
							}
							1 = {
								set_variable = partition_compromise_button_var
								scope:journal_entry ?= {
									change_variable = {
										name = je_conference_of_london_proposal_3
										add = scope:voting_country.prestige
									}
								}
								every_country = {
									limit = {
										any_primary_culture = {
											has_discrimination_trait = heritage_germanic
										}
										capital = {
											OR = {
												is_in_geographic_region = geographic_region_north_germany_old
												is_in_geographic_region = geographic_region_south_germany_old
												is_in_geographic_region = geographic_region_rhine_old
											}
										}
										NOT = {
											this ?= scope:voting_country
										}
									}
									change_relations = { country = scope:voting_country value = 10 }
								}
							}
							1 = {
								set_variable = luxembourg_compromise_button_var
								scope:journal_entry ?= {
									change_variable = {
										name = je_conference_of_london_proposal_4
										add = scope:voting_country.prestige
									}
								}
							}
						}
					}
				}
				if = {
					limit = {
						NOR = {
							has_global_variable = favour_belgium_global_var
							has_global_variable = favour_netherlands_global_var
							has_global_variable = partition_compromise_global_var
							has_global_variable = luxembourg_compromise_global_var
						}
					}
					# Clear winners
					if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_4
						}
						set_global_variable = favour_belgium_global_var
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_1
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_4
						}
						set_global_variable = favour_netherlands_global_var
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_3 > scope:journal_entry.var:je_conference_of_london_proposal_1
							scope:journal_entry.var:je_conference_of_london_proposal_3 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_3 > scope:journal_entry.var:je_conference_of_london_proposal_4
						}
						set_global_variable = partition_compromise_global_var
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_4 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_4 > scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_4 > scope:journal_entry.var:je_conference_of_london_proposal_1
						}
						set_global_variable = luxembourg_compromise_global_var
					}
					else_if = { # Tiebreakers
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_1 = scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_4
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = favour_belgium_button_var # proposal 1
							    	has_variable = favour_netherlands_button_var #proposal 2
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_belgium_button_var
						        }
						        set_global_variable = favour_belgium_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = favour_netherlands_button_var
						        }
						        set_global_variable = favour_netherlands_global_var
						    }
						}
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_1 = scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_4
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = favour_belgium_button_var # proposal 1
							    	has_variable = partition_compromise_button_var # proposal 3
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_belgium_button_var
						        }
						        set_global_variable = favour_belgium_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = partition_compromise_button_var
						        }
						        set_global_variable = partition_compromise_global_var
						    }
						}
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_1 = scope:journal_entry.var:je_conference_of_london_proposal_4
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_1 > scope:journal_entry.var:je_conference_of_london_proposal_3
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = favour_belgium_button_var # proposal 1
							    	has_variable = luxembourg_compromise_button_var # proposal 4
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_belgium_button_var
						        }
						        set_global_variable = favour_belgium_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = luxembourg_compromise_button_var
						        }
						        set_global_variable = luxembourg_compromise_global_var
						    }
						}
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_2 = scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_4
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_1
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = favour_netherlands_button_var # proposal 2
							    	has_variable = partition_compromise_button_var # proposal 3
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_netherlands_button_var
						        }
						        set_global_variable = favour_netherlands_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = partition_compromise_button_var
						        }
						        set_global_variable = partition_compromise_global_var
						    }
						}
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_2 = scope:journal_entry.var:je_conference_of_london_proposal_4
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_3
							scope:journal_entry.var:je_conference_of_london_proposal_2 > scope:journal_entry.var:je_conference_of_london_proposal_1
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = favour_netherlands_button_var # proposal 2
							    	has_variable = luxembourg_compromise_button_var # proposal 4
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_netherlands_button_var
						        }
						        set_global_variable = favour_netherlands_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = luxembourg_compromise_button_var
						        }
						        set_global_variable = luxembourg_compromise_global_var
						    }
						}
					}
					else_if = {
						limit = {
							scope:journal_entry.var:je_conference_of_london_proposal_3 = scope:journal_entry.var:je_conference_of_london_proposal_4
							scope:journal_entry.var:je_conference_of_london_proposal_3 > scope:journal_entry.var:je_conference_of_london_proposal_2
							scope:journal_entry.var:je_conference_of_london_proposal_3 > scope:journal_entry.var:je_conference_of_london_proposal_1
						}
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							    OR = {
							    	has_variable = partition_compromise_button_var # proposal 3
							    	has_variable = luxembourg_compromise_button_var # proposal 4
							    }
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = partition_compromise_button_var
						        }
						        set_global_variable = partition_compromise_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = luxembourg_compromise_button_var
						        }
						        set_global_variable = luxembourg_compromise_global_var
						    }
						}
					}
					else = {
						ordered_country = {
							limit = {
							    is_involved_in_journal_entry = je_conference_of_london
							}
						    order_by = country_rank
							position = 0
							check_range_bounds = no
						    if = {
						        limit = {
						            has_variable = favour_belgium_button_var
						        }
						        set_global_variable = favour_belgium_global_var
						    }
						    if = {
						        limit = {
						            has_variable = favour_netherlands_button_var
						        }
						        set_global_variable = favour_netherlands_global_var
						    }
						    if = {
						        limit = {
						            has_variable = partition_compromise_button_var
						        }
						        set_global_variable = partition_compromise_global_var
						    }
						    else_if = {
						        limit = {
						            has_variable = luxembourg_compromise_button_var
						        }
						        set_global_variable = luxembourg_compromise_global_var
						    }
						}
					}
					set_global_variable = london_treaty_has_concluded_global_var
				}
			}
			if = {
				limit = {
					has_global_variable = london_treaty_has_concluded_global_var
					NOT = {
						exists = global_var:je_conference_of_london_acceptance_timer
					}
				}
				set_global_variable = {
					name = je_conference_of_london_acceptance_timer
					value = 6
				}
			}
			if = {
				limit = {
					exists = global_var:je_conference_of_london_acceptance_timer
					global_var:je_conference_of_london_acceptance_timer <= 0
				}
				c:NET ?= {
					if = {
						limit = {
							NOT = {
								has_variable = decided_on_london_treaty_var
							}
						}
						set_variable = decided_on_london_treaty_var
						set_global_variable = london_treaty_signed_net_global_var
					}
				}
				c:BEL ?= {
					if = {
						limit = {
							NOT = {
								has_variable = decided_on_london_treaty_var
							}
						}
						set_variable = decided_on_london_treaty_var
						set_global_variable = london_treaty_signed_bel_global_var
					}
				}
			}
		}
	}
	on_yearly_pulse = {
	}

	weight = 100

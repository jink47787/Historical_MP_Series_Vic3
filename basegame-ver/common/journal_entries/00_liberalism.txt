# Path to Liberalism
je_liberalism_1 = {
	icon = "gfx/interface/icons/event_icons/event_election.dds"

   group = je_group_internal_affairs

   activation_description = {
      always = yes
   }

	is_shown_when_inactive = {
		NOT = { has_variable = liberalism_1_done }
		has_technology_researched = democracy
		has_technology_researched = international_trade
	}

	possible = {
		any_interest_group = {
			is_in_government = yes
			OR = {
				trigger_if = {
					limit = {
						owner = {
							has_technology_researched = feminism
						}
					}
					OR = {
						has_ideology = ideology:ideology_liberal_modern
						has_ideology = ideology:ideology_universalist_modern
					}
				}
				trigger_else = {
					OR = {
						has_ideology = ideology:ideology_liberal
						has_ideology = ideology:ideology_universalist
					}
				}
				any_scope_character = {
					OR = {
						AND = {
							prominence >= define:NCharacters|PROMINENCE_IMPORTANT_THRESHOLD
							has_role_of_type = politician
						}
						has_role_of_type = agitator
					}
					OR = {
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_market_liberal
						has_ideology = ideology:ideology_humanitarian
						has_ideology = ideology:ideology_humanitarian_royalist
						has_ideology = ideology:ideology_feminist
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_census_voting
				has_law_or_variant = law_type:law_universal_suffrage
			}

			government_legitimacy >= 50
			any_interest_group = {
				is_in_government = yes
				OR = {
					custom_tooltip = {
						text = has_ideology_liberal_tt
						OR = {
							has_ideology = ideology:ideology_liberal
							has_ideology = ideology:ideology_liberal_modern
							has_ideology = ideology:ideology_universalist
							has_ideology = ideology:ideology_universalist_modern
						}
					}
					leader = { has_ideology = ideology:ideology_radical }
					leader = { has_ideology = ideology:ideology_market_liberal }
				}
			}
			is_at_war = no
			NOT = {
				any_interest_group = {
					is_insurrectionary = yes
				}
			}
		}
		effect = {
			set_variable = liberalism_1_done
			trigger_event = { id = liberalism.1 }
		}
	}

	weight = 10000
}

je_liberalism_2 = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			has_law_or_variant = law_type:law_free_trade
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		effect = {
			trigger_event = { id = liberalism.2 }
		}
	}

	current_value = {
		value = total_trade_value
	}

	goal_add_value = {
		value = total_trade_value
		multiply = 0.5
	}

	progressbar = yes

	weight = 5000
}

# Freedom of Thought
je_liberalism_3 = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_right_of_assembly
				has_law_or_variant = law_type:law_protected_speech
			}
			OR = {
				has_law_or_variant = law_type:law_freedom_of_conscience
				has_law_or_variant = law_type:law_total_separation
			}
			any_scope_building = {
				is_building_type = building_university
				level >= 3
				occupancy > 0.9
			}
			any_scope_building = {
				is_building_type = building_art_academy
				level >= 3
				occupancy > 0.9
			}
		}
		effect = {
			trigger_event = { id = liberalism.3 }
		}
	}

	weight = 5000
}

je_french_natural_borders = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_france_expansion
	dlc = dlc004

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		exists = c:FRA
		c:FRA ?= THIS
		has_dlc_feature = voice_of_the_people_content
	}

	possible = {
		has_technology_researched = nationalism
		any_interest_group = {
			is_powerful = yes
			is_in_government = yes
			leader = {
				OR = {
					has_ideology = ideology:ideology_jingoist_leader
					has_ideology = ideology:ideology_ethno_nationalist
					has_ideology = ideology:ideology_fascist
					has_ideology = ideology:ideology_integralist
					has_ideology = ideology:ideology_bonapartist
				}
			}
		}
	}

	immediate = {
		trigger_event = { id = natural_borders.1 popup = yes }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_at_war = no
			custom_tooltip = {
				text = own_all_french_homelands
				NOT = {
					any_state_in_france = {
						NOT = { owner = ROOT }
					}
				}
			}
			any_scope_state = {
				state_region = s:STATE_WALLONIA
				is_incorporated = yes
				is_split_state = no
			}
			any_scope_state = {
				state_region = s:STATE_FLANDERS
				is_incorporated = yes
				is_split_state = no
			}
			any_scope_state = {
				state_region = s:STATE_RHINELAND
				is_incorporated = yes
				is_split_state = no
			}
			any_scope_state = {
				state_region = s:STATE_NORTH_RHINE
				is_incorporated = yes
				is_split_state = no
			}
		}
		effect = {
			set_variable = completed_natural_borders_je
			trigger_event = { id = natural_borders.2 popup = yes }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = reject_natural_borders
		}
		effect = {
			je_french_natural_borders_cleanup = yes
		}
	}


	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 100
}

je_confederation_of_the_rhine = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_france_expansion

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		OR = {
			any_subject_or_below = {
				capital = {
					is_in_geographic_region = geographic_region_german_confederation
				}
			}
			any_scope_state = {
				state_region = {
					OR = {
						is_homeland = cu:south_german
						is_homeland = cu:north_german
					}
				}
				is_in_geographic_region = geographic_region_german_confederation
				is_incorporated = no
				NOT = {
					ROOT = {
						has_claim = prev.state_region
					}
				}
			}
		}
		NOR = {
			country_has_primary_culture = cu:north_german
			country_has_primary_culture = cu:south_german
		}
	}

	possible = {
		country_rank >= rank_value:great_power
		any_state_in_german_confederation = {
			OR = {
				owner = {
					is_subject_of = ROOT
					NOT = {
						has_diplomatic_pact = {
							who = ROOT
							type = personal_union
							is_initiator = no
						}
					}
					capital = {
						is_in_geographic_region = geographic_region_german_confederation
					}
					OR = {
						country_has_primary_culture = cu:north_german
						country_has_primary_culture = cu:south_german
					}
				}
				owner = ROOT
			}
			count >= 4
		}
	}

	immediate = {
		save_scope_as = rhine_overlord
		cu:north_german = {
			save_scope_as = north_german_scope
		}
		cu:south_german = {
			save_scope_as = south_german_scope
		}
	}

	scripted_button = je_confederation_of_the_rhine_confederate_button
	scripted_button = je_confederation_of_the_rhine_expand_button

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = no_german_great_powers_tt
				NOT = {
					any_country_in_german_confederation = {
						OR = {
							country_has_primary_culture = cu:north_german
							country_has_primary_culture = cu:south_german
						}
						country_rank = rank_value:great_power
					}
				}
			}
		}
		effect = {
			je_confederation_of_the_rhine_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			exists = c:RHN
			NOT = { c:RHN ?= { is_at_war_with_overlord = yes } }
			c:RHN ?= {
				NOT = {
					is_subject_of = ROOT
				}
			}
		}
		effect = {
			add_modifier = {
				name = modifier_fall_of_germany
				days = normal_modifier_time
				is_decaying = yes
			}
			je_confederation_of_the_rhine_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOR = {
				any_country_in_german_confederation = {
					is_subject_of = ROOT
				}
				any_scope_state = {
					state_region = {
						OR = {
							is_homeland = cu:south_german
							is_homeland = cu:north_german
						}
					}
					is_in_geographic_region = geographic_region_german_confederation
					is_incorporated = no
				}
			}
		}
		effect = {
			je_confederation_of_the_rhine_cleanup = yes
		}
	}

	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 100
}

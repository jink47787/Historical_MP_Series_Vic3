je_russian_revolution = {
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	group = je_group_historical_content
	dlc = dlc019

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		has_dlc_feature = ip5_content
		c:RUS ?= this
	}

	is_shown_when_inactive = {
		has_dlc_feature = ip5_content
		c:RUS ?= this
		NOT = { has_variable = russian_revolution_concluded }
	}

	possible = {
		has_technology_researched = nationalism
		has_technology_researched = labor_movement
        any_political_movement ?= {
            political_movement_support >= 0.05 # Above marginal
            political_movement_radicalism >= define:NPolitics|RADICALISM_AGITATING_THRESHOLD # 0.25
        }
	}

	immediate = {
		trigger_event = { id = ip5_revolutions.2 }
		if = {
			limit = {
				NOT = { has_russian_revolution_alignment_variables = yes }
			}
			initialize_russian_revolution_alignments = yes
		}
		calculate_russian_revolution_alignments = yes
		designate_russian_revolution_leader = yes

		if = {
			limit = {
				country_opposition_is_liberal = yes
				any_political_movement = {
					movement_is_liberal = yes
				}
			}
			ordered_political_movement = {
				limit = {
					movement_is_liberal = yes
				}
				order_by = political_movement_support
				position = 0
				ROOT = {
					set_variable = {
						name = russian_government_popular_opposition_var
						value = prev
					}
				}
				add_modifier = {
					name = russian_government_popular_opposition
					multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
				}
			}
		}
		if = {
			limit = {
				country_opposition_is_conservative = yes
				any_political_movement = {
					movement_is_conservative = yes
				}
			}
			ordered_political_movement = {
				limit = {
					movement_is_conservative = yes
				}
				order_by = political_movement_support
				position = 0
				ROOT = {
					set_variable = {
						name = russian_government_popular_opposition_var
						value = prev
					}
				}
				add_modifier = {
					name = russian_government_popular_opposition
					multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
				}
			}
		}
		if = {
			limit = {
				country_opposition_is_socialist = yes
				any_political_movement = {
					movement_is_socialist = yes
				}
			}
			ordered_political_movement = {
				limit = {
					movement_is_socialist = yes
				}
				order_by = political_movement_support
				position = 0
				ROOT = {
					set_variable = {
						name = russian_government_popular_opposition_var
						value = prev
					}
				}
				add_modifier = {
					name = russian_government_popular_opposition
					multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			designate_russian_revolution_leader = yes
			if = {
				limit = {
					any_political_movement = {
						has_modifier = russian_government_popular_opposition
					}
				}
				every_political_movement = {
					limit = {
						has_modifier = russian_government_popular_opposition
					}
					remove_modifier = russian_government_popular_opposition
				}
			}
			if = {
				limit = {
					has_variable = russian_government_popular_opposition_var
				}
				remove_variable = russian_government_popular_opposition_var
			}

			if = {
				limit = {
					country_opposition_is_liberal = yes
					any_political_movement = {
						movement_is_liberal = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_liberal = yes
					}
					order_by = political_movement_support
					position = 0
					ROOT = {
						set_variable = {
							name = russian_government_popular_opposition_var
							value = prev
						}
					}
					add_modifier = {
						name = russian_government_popular_opposition
						multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
					}
				}
			}
			else_if = {
				limit = {
					country_opposition_is_conservative = yes
					any_political_movement = {
						movement_is_conservative = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_conservative = yes
					}
					order_by = political_movement_support
					position = 0
					ROOT = {
						set_variable = {
							name = russian_government_popular_opposition_var
							value = prev
						}
					}
					add_modifier = {
						name = russian_government_popular_opposition
						multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
					}
				}
			}
			else_if = {
				limit = {
					country_opposition_is_socialist = yes
					any_political_movement = {
						movement_is_socialist = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_socialist = yes
					}
					order_by = political_movement_support
					position = 0
					ROOT = {
						set_variable = {
							name = russian_government_popular_opposition_var
							value = prev
						}
					}
					add_modifier = {
						name = russian_government_popular_opposition
						multiplier = { value = 1 multiply = ROOT.russian_revolution_radicalism_value }
					}
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = russia_is_conservative
				trigger = {
					country_is_conservative = yes
				}
			}
			triggered_desc = {
				desc = russia_is_liberal
				trigger = {
					country_is_liberal = yes
				}
			}
			triggered_desc = {
				desc = russia_is_socialist
				trigger = {
					country_is_socialist = yes
				}
			}
			triggered_desc = {
				desc = russia_is_conservative_and_liberal
				trigger = {
					country_is_conservative_and_liberal = yes
				}
			}
			triggered_desc = {
				desc = russia_is_conservative_and_socialist
				trigger = {
					country_is_conservative_and_socialist = yes
				}
			}
			triggered_desc = {
				desc = russia_is_liberal_and_socialist
				trigger = {
					country_is_liberal_and_socialist = yes
				}
			}
			triggered_desc = {
				desc = russia_is_a_land_of_contrasts
				trigger = {
					country_is_a_land_of_contrasts = yes
				}
			}
		}
		first_valid = {
			triggered_desc = {
				desc = russian_revolutionary_leader_is
				trigger = {
					has_variable = russian_revolution_leader
					var:russian_revolution_leader ?= { is_character_alive = yes }
				}
			}
			triggered_desc = {
				desc = russian_revolutionary_leaderless
				trigger = {
					NOT = {
						AND = {
							has_variable = russian_revolution_leader
							var:russian_revolution_leader ?= { is_character_alive = yes }
						}
					}
				}
			}
		}
		first_valid = {
			triggered_desc = {
				desc = russia_lib_opposition
				trigger = {
					ROOT = {
						country_opposition_is_liberal = yes
					}
				}
			}
			triggered_desc = {
				desc = russia_soc_opposition
				trigger = {
					ROOT = {
						country_opposition_is_socialist = yes
					}
				}
			}
			triggered_desc = {
				desc = russia_con_opposition
				trigger = {
					ROOT = {
						country_opposition_is_conservative = yes
					}
				}
			}

			triggered_desc = {
				desc = russia_lib_soc_opposition
				trigger = {
					country_opposition_is_liberal_socalist = yes
				}
			}
			triggered_desc = {
				desc = russia_lib_con_opposition
				trigger = {
					country_opposition_is_liberal_conservative = yes
				}
			}
			triggered_desc = {
				desc = russia_con_soc_opposition
				trigger = {
					country_opposition_is_conservative_socialist = yes
				}
			}
		}
	}

	outcome = {
		header = je_russian_revolution_revolution_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"

		conclusion_desc = je_russian_revolution_revolution_conclusion_desc

		trigger = {
			custom_tooltip = {
				text = russian_revolutionary_leader_must_be
				has_variable = russian_revolution_leader
				var:russian_revolution_leader ?= { is_character_alive = yes }
			}
			OR = {
				capital = {
					is_in_revolt = yes
				}
				any_scope_state = {
					is_incorporated = yes
					is_in_revolt = yes
					percent >= 0.5
				}
			}
			any_civil_war = {
				is_civil_war_type = revolution
				civil_war_progress >= 0.75
				OR = {
					AND = {
						custom_tooltip = {
							text = je_russian_revolution_liberal_complete
							ROOT = {
								OR = {
									country_opposition_is_liberal = yes
									country_opposition_is_liberal_socalist = yes
									country_opposition_is_liberal_conservative = yes
								}
							}
						}
						custom_tooltip = {
							text = caused_by_liberal_movement_tt
							political_movement ?= {
								movement_is_liberal = yes
							}
						}
					}
					AND = {
						custom_tooltip = {
							text = je_russian_revolution_socialist_complete
							ROOT = {
								OR = {
									country_opposition_is_socialist = yes
									country_opposition_is_liberal_socalist = yes
									country_opposition_is_conservative_socialist = yes
								}
							}
						}
						custom_tooltip = {
							text = caused_by_socialist_movement_tt
							political_movement ?= {
								movement_is_socialist = yes
							}
						}
					}
					AND = {
						custom_tooltip = {
							text = je_russian_revolution_conservative_complete
							ROOT = {
								OR = {
									country_opposition_is_conservative = yes
									country_opposition_is_liberal_conservative = yes
									country_opposition_is_conservative_socialist = yes
								}
							}
						}
						custom_tooltip = {
							text = caused_by_conservative_movement_tt
							political_movement ?= {
								movement_is_conservative = yes
							}
						}
					}
				}
			}
		}

		effect = {
			var:russian_revolution_leader ?= {
				set_variable = true_russian_revolution_leader
			}

			if = {
				limit = {
					any_civil_war = {
						is_civil_war_type = revolution
						political_movement ?= { movement_is_liberal = yes }
					}
				}
				set_variable = { name = provisional_government_identity value = flag:liberal }
				if = {
					limit = { country_is_conservative = yes }
					set_variable = { name = provisional_government_opposition_ideology value = flag:socialist }
				}
				else = {
					set_variable = { name = provisional_government_opposition_ideology value = flag:conservative }
				}
			}
			else_if = {
				limit = {
					any_civil_war = {
						is_civil_war_type = revolution
						political_movement ?= { movement_is_socialist = yes }
					}
				}
				set_variable = { name = provisional_government_identity value = flag:socialist }
				if = {
					limit = { country_is_conservative = yes }
					set_variable = { name = provisional_government_opposition_ideology value = flag:liberal }
				}
				else = {
					set_variable = { name = provisional_government_opposition_ideology value = flag:conservative }
				}
			}
			else_if = {
				limit = {
					any_civil_war = {
						is_civil_war_type = revolution
						political_movement ?= { movement_is_conservative = yes }
					}
				}
				set_variable = { name = provisional_government_identity value = flag:conservative }
				if = {
					limit = {
						any_civil_war = {
							is_civil_war_type = revolution
							political_movement ?= {
								OR = {
							        is_political_movement_type = movement_royalist_constitutional
							        is_political_movement_type = movement_royalist_absolutist
							        is_political_movement_type = movement_orleanist
							        is_political_movement_type = movement_legitimist
							        is_political_movement_type = movement_bonapartist
							        is_political_movement_type = movement_carlist
							        is_political_movement_type = movement_miguelist
						       	}
					    	}
						}
					}
				    set_variable = provisional_government_is_monarchist
				}
				if = {
					limit = { country_is_liberal = yes }
					set_variable = { name = provisional_government_opposition_ideology value = flag:socialist }
				}
				else = {
					set_variable = { name = provisional_government_opposition_ideology value = flag:liberal }
				}
			}
			else = {
				set_variable = { name = provisional_government_identity value = flag:liberal }
				set_variable = { name = provisional_government_opposition_ideology value = flag:socialist }
			}

			if = {
				limit = {
					has_variable_list = russian_revolution_igs
				}
				clear_variable_list = russian_revolution_igs
			}
			every_interest_group = {
				limit = {
					is_insurrectionary = yes
				}
				ROOT = {
					add_to_variable_list = {
						name = russian_revolution_igs
						target = prev
					}
				}
				abandon_revolution = yes
			}
			every_civil_war = {
				add_civil_war_progress = revolution_retreat_exceptionally_large
			}
			every_interest_group = {
				limit = {
					is_in_government = yes
				}
				remove_ruling_interest_group = yes
			}
			ordered_in_list = {
				variable = russian_revolution_igs
				order_by = ig_clout
				max = 2
				add_ruling_interest_group = yes
			}

			# For achievement_dont_trust_the_liberals
			if = {
				limit = { var:provisional_government_identity ?= flag:liberal }
				set_variable = achievement_dont_trust_the_liberals_1_var
			}

			set_variable = russian_revolution_concluded
			if = {
				limit = {
					has_variable = russian_government_popular_opposition_var
				}
				remove_variable = russian_government_popular_opposition_var
			}
			trigger_event = { id = ip5_revolutions.1 popup = yes }

			hidden_effect = {
				if = {
					limit = {
						any_political_movement = {
							has_modifier = russian_government_popular_opposition
						}
					}
					every_political_movement = {
						limit = {
							has_modifier = russian_government_popular_opposition
						}
						remove_modifier = russian_government_popular_opposition
					}
				}
			}
		}
	}

	weight = 5000


	should_be_pinned_by_default_uninvolved_or_context = yes
}

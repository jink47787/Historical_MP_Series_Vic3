je_texas_tex = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_subject = no
			is_at_war = no
			exists = c:USA
			c:USA ?= {
				relations:root >= relations_threshold:cordial
			}
		}
		effect = {
			trigger_event = { id = texan_war_of_independence.1 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { exists = c:TEX }
		}
		effect = {
			je_texas_tex_cleanup = yes
		}
	}

	weight = 100

	transferable = no
}

	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			exists = c:TEX
			c:TEX ?= {
				relations:root >= relations_threshold:cordial
				is_subject = no

		effect = {
			custom_tooltip = texas_je_tt

		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc

		}
		effect = {
			je_texas_usa_cleanup = yes
		}
	}

	weight = 100

	transferable = no
	should_be_pinned_by_default_uninvolved_or_context = yes

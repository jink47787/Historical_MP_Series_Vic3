je_consolidate_colonial_rule = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_british_india

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:BIC = THIS
	}

	immediate = {
		ig:ig_industrialists ?= {
			save_scope_as = industrialists_ig
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_state_in_indian_frontier = {
				OR = {
					owner = ROOT
					owner = {
						is_subject_of = ROOT
					}
					trigger_if = {
						limit = {
							exists = ROOT.top_overlord
						}
						OR = {
							owner = ROOT.top_overlord
							owner = {
								is_subject_of = ROOT.top_overlord
							}
						}
					}
				}
				percent >= 0.80
				list_scopes = yes
				show_trigger_per_scope = yes
			}
			scope:journal_entry = {
				is_goal_complete = yes
			}
			highest_secession_progress < 0.5
		}
		effect = {
			trigger_event = { id = east_indies.1 }
			set_variable = consolidated_colonial_rule
		}
		event_outcome_desc = {
			first_valid = {
				triggered_desc = {
					desc = event_effects_east_indies.1_tt
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					exists = c:GBR
				}
				trigger_if = {
					limit = {
						exists = c:GBR
					}
					NOT = {
						is_subject_of = c:GBR
					}
				}
				is_subject_type = subject_type_colony
			}
		}
		effect = {
			je_consolidate_colonial_rule_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					highest_secession_progress < 0.3
				}
				change_variable = {
					name = east_indies_revolt_var
					add = 1
				}
				clamp_variable = { name = east_indies_revolt_var max = 120 min = 0 }
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = bic_collapse_var
		}
		effect = {
			je_consolidate_colonial_rule_cleanup = yes
		}
	}

	current_value = {
		value = root.var:east_indies_revolt_var
	}

	goal_add_value = {
		add = 120
	}

		
	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 100
}

je_reform_indies = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		AND = {
			exists = c:DEI
			c:DEI = THIS
		}
	}

	possible = {
		OR = {
			has_law_or_variant = law_type:law_free_trade
			has_law_or_variant = law_type:law_mercantilism
			NOT = { has_law_or_variant = law_type:law_land_based_taxation }
			NOT = { has_law_or_variant = law_type:law_serfdom }
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			OR = {
				has_law_or_variant = law_type:law_free_trade
				has_law_or_variant = law_type:law_mercantilism
			}
			NOT = { has_law_or_variant = law_type:law_land_based_taxation }
			NOT = { has_law_or_variant = law_type:law_serfdom }
			has_law_or_variant = law_type:law_slavery_banned
		}
		effect = {
			trigger_event = { id = east_indies.2 }
			show_as_tooltip = {
				add_journal_entry = {
					type = je_ethical_policy
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_variable = bic_collapse_var
		}
		effect = {
			je_reform_indies_cleanup = yes
		}
	}

	current_value = {
		value = root.gdp
	}

	goal_add_value = {
		value = root.gdp
		multiply = 0.5
	}

	progressbar = yes

	weight = 100
}

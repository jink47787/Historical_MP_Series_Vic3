je_russia_provisional_government = {
	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	group = je_group_historical_content
	dlc = dlc019

	activation_description = {
		always = yes
	}

	# triggered by je_russian_revolution outcome

	scripted_progress_bar = je_rus_prov_gov_bar

	immediate = {
		c:RUS = {
			save_scope_as = russia_scope
			every_scope_state = {
				set_variable = rcw_pre_war_russian_state
			}
			every_subject_or_below = {
				every_scope_state = {
					set_variable = rcw_pre_war_russian_state
				}
			}
		}
		trigger_event = {
			id = rus_prov_gov.2
			days = 3
			popup = yes
		}
		# This is the hidden event that triggers nationalities autonomy
		trigger_event = {
			id = rcw.101
			days = 1
			popup = yes
		}
	}

	on_monthly_pulse = {
		events = {
			rus_prov_gov.3
		}
		random_events = {
			5 = 0
			1 = rcw.101
		}
		effect = {
			if = {
				limit = {
					NOT = {
						any_diplomatic_play = {
							is_diplomatic_play_type = dp_revolution
							target_is = ROOT
						}
					}
				}
				rcw_snapshot_effect = yes
			}
			if = {
				limit = {
					any_political_movement = {
						has_modifier = provisional_government_popular_opposition
					}
				}
				every_political_movement = {
					limit = {
						has_modifier = provisional_government_popular_opposition
					}
					remove_modifier = provisional_government_popular_opposition
				}
			}
			if = {
				limit = {
					var:provisional_government_opposition_ideology ?= flag:socialist
					any_political_movement = {
						movement_is_socialist = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_socialist = yes
					}
					order_by = political_movement_support
					position = 0
					add_modifier = {
						name = provisional_government_popular_opposition
					}
				}
			}
			if = {
				limit = {
					var:provisional_government_opposition_ideology ?= flag:liberal
					any_political_movement = {
						movement_is_liberal = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_liberal = yes
					}
					order_by = political_movement_support
					position = 0
					add_modifier = {
						name = provisional_government_popular_opposition
					}
				}
			}
			if = {
				limit = {
					var:provisional_government_opposition_ideology ?= flag:conservative
					any_political_movement = {
						movement_is_conservative = yes
					}
				}
				ordered_political_movement = {
					limit = {
						movement_is_conservative = yes
					}
					order_by = political_movement_popular_support
					position = 0
					add_modifier = {
						name = provisional_government_popular_opposition
					}
				}
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = rus_prov_gov_legitimate_status
				trigger = {
					government_legitimacy >= 75
				}
			}
			triggered_desc = {
				desc = rus_prov_gov_illegitimate_status
				trigger = {
					government_legitimacy < 75
				}
			}
		}
		first_valid = {
			triggered_desc = {
				desc = rus_prov_gov_civilwar_status
				trigger = {
					any_civil_war = {
						is_civil_war_type = revolution
					}
				}
			}
			triggered_desc = {
				desc = rus_prov_gov_no_civilwar_status
			}
		}
	}

	# Provisional government survives
	outcome = {
		header = je_rus_prov_gov_survive_header
		effect_header = je_rus_prov_gov_survive_effect_header
		color = "gold_positive"
		conclusion_desc = je_rus_prov_gov_survive_conclusion_desc

		trigger = {
			OR = {
				AND = {
					government_legitimacy >= 75
					custom_tooltip = {
						text = rus_prov_gov_no_civil_wars
						NOT = {
							any_civil_war = {
								is_civil_war_type = revolution
							}
						}
					}
					custom_tooltip = {
						text = rus_prov_gov_bar_full_tt
						scope:journal_entry = {
							scripted_bar_progress = {
								name = je_rus_prov_gov_bar
								value >= 60
							}
						}
					}
				}
				custom_tooltip = {
					text = rus_prov_gov_survived_ten_years_tt
					scope:journal_entry = {
						journal_entry_age >= 3650
					}
				}
			}
		}

		effect = {
			rus_prov_gov_stabilized_effect = yes

			trigger_event = {
				id = rcw.102
				days = 0
			}
			hidden_effect = {
				if = {
					limit = {
						any_political_movement = {
							has_modifier = provisional_government_popular_opposition
						}
					}
					every_political_movement = {
						limit = {
							has_modifier = provisional_government_popular_opposition
						}
						remove_modifier = provisional_government_popular_opposition
					}
				}
			}
		}
	}

	# Provisional government fails via civil war
	outcome = {
		header = je_rus_prov_gov_fail_header
		effect_header = je_rus_prov_gov_fail_effect_header
		color = "blue_negative"

		conclusion_desc = je_rus_prov_gov_fail_conclusion_desc

		flags = { historical }

		trigger = {
			any_diplomatic_play = {
				is_diplomatic_play_type = dp_revolution
				target_is = ROOT
				any_scope_play_involved = {
					is_revolutionary = yes
					civil_war_origin_country ?= ROOT
				}
			}
		}

		effect = {
			set_variable = {
				name = keep_current_ruler_var
				days = 7
			}
			every_subject_or_below = {
				limit = {
					OR = {
						is_in_geographic_region = geographic_region_europe
						is_in_geographic_region = geographic_region_asia
					}
				}
				make_independent = yes
				if = {
					limit = {
						capital ?= { rcw_is_in_any_theatre = yes }
					}
					set_variable = rcw_freed_russian_subject
				}
			}
			if = {
				limit = {
					any_civil_war = {
						civil_war_progress_high = yes
					}
				}
				every_civil_war = {
					add_civil_war_progress = 1
				}
			}
			hidden_effect = {
				# Hidden event that creates warlords and starts the civil war
				trigger_event ={
					id = rcw.100
					days = 0
					popup = yes
				}
			}
			hidden_effect = {
				if = {
					limit = {
						any_political_movement = {
							has_modifier = provisional_government_popular_opposition
						}
					}
					every_political_movement = {
						limit = {
							has_modifier = provisional_government_popular_opposition
						}
						remove_modifier = provisional_government_popular_opposition
					}
				}
			}
		}
	}

	weight = 5000


	should_be_pinned_by_default_uninvolved_or_context = yes
	should_be_pinned_by_default_involved = yes
}

je_russia_civil_war = {
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	group = je_group_global_international_situations
	dlc = dlc019

	activation_description = {
		always = yes
	}

	should_be_involved = {
		OR = {
			has_variable = rcw_russian_faction
			has_variable = rcw_separatist
			AND = {
				NOT = { has_global_variable = rcw_v2 }
				country_has_primary_culture = cu:russian
			}
		}
	}

	should_show_when_not_involved = {
		OR = {
			is_in_geographic_region = geographic_region_europe
			is_in_geographic_region = geographic_region_asia
		}
	}

	immediate = {
		c:RUS ?= {
			save_scope_as = russia_scope

			trigger_event = {
				id = rcw.1
				days = 2
				popup = yes
			}
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = rus_cw_pale_horse
				trigger = {
					always = yes
				}
			}
		}
		first_valid = {
			triggered_desc = {
				desc = rus_cw_phase_1_status
				trigger = {
					NOT = { has_global_variable = russia_civil_war_phase_1 }
				}
			}
			triggered_desc = {
				desc = rus_cw_phase_2_status
				trigger = {
					has_global_variable = russia_civil_war_phase_1
				}
			}
		}
	}

	scripted_button = je_rcw_warlords_button
	scripted_button = je_rcw_switch_country_button_1
	scripted_button = je_rcw_switch_country_button_2
	scripted_button = je_rcw_switch_country_button_3
	scripted_button = je_rcw_switch_country_button_4

	map_color = {
		if = {
			limit = {
				owner ?= { has_variable = rcw_separatist }
			}
			value = "rcw_separatist"
		}
		else_if = {
			limit = {
				owner ?= { var:rcw_camp ?= flag:rcw_red }
			}
			value = "rcw_red_1"
		}
		else_if = {
			limit = {
				owner ?= { var:rcw_camp ?= flag:rcw_white }
			}
			value = "rcw_white_1"
		}
		else_if = {
			limit = {
				owner ?= { var:rcw_camp ?= flag:rcw_green }
			}
			value = "rcw_green_1"
		}
		else_if = {
			limit = {
				owner ?= { var:rcw_camp ?= flag:rcw_black }
			}
			value = "rcw_black_1"
		}
		else_if = {
			limit = {
				owner ?= { var:rcw_camp ?= flag:rcw_democratic }
			}
			value = "rcw_democratic_1"
		}
	}

	map_color_second = {
		if = {
			limit = {
				controller ?= {
					var:rcw_camp ?= flag:rcw_red
					NOT = { this = prev.owner }
				}
			}
			value = "rcw_red_1"
		}
		else_if = {
			limit = {
				controller ?= {
					var:rcw_camp ?= flag:rcw_white
					NOT = { this = prev.owner }
				}
			}
			value = "rcw_white_1"
		}
		else_if = {
			limit = {
				controller ?= {
					var:rcw_camp ?= flag:rcw_green
					NOT = { this = prev.owner }
				}
			}
			value = "rcw_green_1"
		}
		else_if = {
			limit = {
				controller ?= {
					var:rcw_camp ?= flag:rcw_black
					NOT = { this = prev.owner }
				}
			}
			value = "rcw_black_1"
		}
		else_if = {
			limit = {
				controller ?= {
					var:rcw_camp ?= flag:rcw_democratic
					NOT = { this = prev.owner }
				}
			}
			value = "rcw_democratic_1"
		}
	}

	map_tooltip = {
		owner ?= {
			save_scope_as = rcw_map_owner
			if = {
				limit = {
					has_variable = rcw_separatist
				}
				custom_tooltip_no_bullet = rcw_map_separatist
			}
			else_if = {
				limit = {
					var:rcw_camp ?= flag:rcw_red
				}
				custom_tooltip_no_bullet = rcw_map_red
			}
			else_if = {
				limit = {
					var:rcw_camp ?= flag:rcw_white
				}
				custom_tooltip_no_bullet = rcw_map_white
			}
			else_if = {
				limit = {
					var:rcw_camp ?= flag:rcw_green
				}
				custom_tooltip_no_bullet = rcw_map_green
			}
			else_if = {
				limit = {
					var:rcw_camp ?= flag:rcw_black
				}
				custom_tooltip_no_bullet = rcw_map_black
			}
			else_if = {
				limit = {
					var:rcw_camp ?= flag:rcw_democratic
				}
				custom_tooltip_no_bullet = rcw_map_democratic
			}
		}
	}

	modifiers_while_active = {
		rcw_modifier
	}

	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_global_variable = rcw_v2
				}
				rcw_annul_faction_truces = yes
				rcw_rederive_war_leader_camps = yes
			}
			else = {
				if = {
					limit = {
						c:RUS ?= {
							is_at_war = no
							is_diplomatic_play_committed_participant = no
							NOT = { has_global_variable = russia_civil_war_phase_1 }
						}
						any_country = {
							country_has_primary_culture = cu:russian
							is_subject = no
							OR = {
								is_in_geographic_region = geographic_region_europe
								is_in_geographic_region = geographic_region_asia
							}
							count > 1
						}
					}
				}
				set_global_variable = russia_civil_war_phase_1
				if = {
					limit = {
						any_country = {
							country_has_primary_culture = cu:russian
							has_truce_with = c:RUS
						}
					}
					every_country = {
						limit = {
							country_has_primary_culture = cu:russian
							has_truce_with = c:RUS
						}
						end_truce = c:RUS
					}
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_global_variable = rcw_v2
				}
				rcw_update_exhaustion_level = yes
				rcw_grant_separatist_claims = yes
				rcw_collapse_check = yes
				if = {
					limit = {
						scope:journal_entry = { journal_entry_age >= 180 }
					}
					rcw_fallback_declarations = yes
				}
			}
			scope:journal_entry = {
				every_scope_je_involved = {
					limit = {
						OR = {
							NOT = { has_global_variable = rcw_v2 }
							has_variable = rcw_russian_faction
						}
					}
					trigger_event = rcw.5 # gold intro
					trigger_event = rcw.6 # gold recovery
					random_list = {
						10 = { }
						1 = { trigger_event = rcw.4 } # red/white terror
						1 = {
							if = {
								limit = {
									OR = {
										NOT = { has_global_variable = rcw_v2 }
										c:RUS ?= this
									}
								}
								trigger_event = rcw.7 # fate of the tsar
							}
						}
					}
				}
			}
		}
	}

	outcome = {
		header = je_rus_civil_war_header
		effect_header = je_rus_civil_war_effect_header
		color = "blue_negative"

		conclusion_desc = je_rus_civil_war_conclusion_desc

		trigger = {
			hidden_trigger = {
				NOT = { has_global_variable = rcw_v2 }
				any_country = {
					country_has_primary_culture = cu:russian
					is_subject = no
					# This makes sure that the civil war doesn't last forever due to possible colonial governments
					OR = {
						is_in_geographic_region = geographic_region_europe
						is_in_geographic_region = geographic_region_asia
					}
					count = 1
				}
			}
		}

		event_outcome_effect_desc = {
			header = rcw.3.a
			effect = {
				add_modifier = {
					name = rcw_reward_construction
					days = very_long_modifier_time
					is_decaying = yes
				}
				add_loyalists = {
					value = 0.1
				}
			}
		}
		event_outcome_effect_desc = {
			header = rcw.3.b
			effect = {
				add_modifier = {
					name = rcw_reward_warfare
					days = very_long_modifier_time
					is_decaying = yes
				}
			}
		}
		event_outcome_effect_desc = {
			header = rcw.3.c
			effect = {
				add_modifier = {
					name = rcw_reward_enactment
					days = very_long_modifier_time
					is_decaying = yes
				}
			}
		}

		effect_all_involved = {
			# If the Provisional Government wins the war
			# For achievement achievement_dont_trust_the_liberals
			if = {
				limit = {
					c:RUS ?= this
					var:provisional_government_identity ?= flag:liberal
				}
				set_variable = achievement_dont_trust_the_liberals_3_var
			}
			trigger_event = {
				id = rcw.3
				popup = yes
			}
			add_loyalists = {
				value = 0.1
			}
		}
	}

	outcome = {
		header = je_rus_civil_war_header
		effect_header = je_rus_civil_war_effect_header
		color = "blue_negative"

		conclusion_desc = je_rus_civil_war_conclusion_desc

		trigger = {
			hidden_trigger = {
				has_global_variable = rcw_v2
			}
			custom_tooltip = {
				text = rcw_one_russian_faction_left_tt
				any_country = {
					rcw_is_active_russian_faction = yes
					count = 1
				}
			}
		}

		effect = {
			hidden_effect = {
				rcw_resolve_victory = yes
				rcw_global_end_cleanup = yes
			}
		}

		event_outcome_effect_desc = {
			header = rcw.3.a
			effect = {
				add_modifier = {
					name = rcw_reward_construction
					days = very_long_modifier_time
					is_decaying = yes
				}
				add_loyalists = {
					value = 0.1
				}
			}
		}
		event_outcome_effect_desc = {
			header = rcw.3.b
			effect = {
				add_modifier = {
					name = rcw_reward_warfare
					days = very_long_modifier_time
					is_decaying = yes
				}
			}
		}
		event_outcome_effect_desc = {
			header = rcw.3.c
			effect = {
				add_modifier = {
					name = rcw_reward_enactment
					days = very_long_modifier_time
					is_decaying = yes
				}
			}
		}

		effect_all_involved = {
			custom_tooltip = rcw_victor_tt
			if = {
				limit = {
					has_variable = rcw_civil_war_victor
				}
				custom_tooltip = rcw_victor_tt
				# If the Provisional Government wins the war
				# For achievement achievement_dont_trust_the_liberals
				if = {
					limit = {
						c:RUS ?= this
						var:provisional_government_identity ?= flag:liberal
					}
					set_variable = achievement_dont_trust_the_liberals_3_var
				}
				trigger_event = {
					id = rcw.3
					popup = yes
				}
				add_loyalists = {
					value = 0.1
				}
			}
			else_if = {
				limit = {
					rcw_separatist_has_survived = yes
				}
				custom_tooltip = rcw_separatist_survived_tt
				add_modifier = {
					name = rcw_independence_secured
					days = long_modifier_time
					is_decaying = yes
				}
			}
			hidden_effect = {
				rcw_country_end_cleanup = yes
			}
		}
	}

	outcome = {
		header = je_rus_civil_war_partitioned_header
		effect_header = je_rus_civil_war_partitioned_effect_header
		color = "blue_negative"

		conclusion_desc = je_rus_civil_war_partitioned_conclusion_desc

		trigger = {
			hidden_trigger = {
				has_global_variable = rcw_v2
			}
			custom_tooltip = {
				text = rcw_no_russian_faction_left_tt
				NOT = {
					any_country = {
						rcw_is_active_russian_faction = yes
					}
				}
			}
		}

		effect = {
			rcw_global_end_cleanup = yes
		}

		effect_all_involved = {
			custom_tooltip = rcw_separatist_survived_tt
			if = {
				limit = {
					rcw_separatist_has_survived = yes
				}
				custom_tooltip = rcw_separatist_survived_tt
				add_modifier = {
					name = rcw_independence_secured
					days = long_modifier_time
					is_decaying = yes
				}
			}
			hidden_effect = {
				rcw_country_end_cleanup = yes
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = je_rus_civil_war_frozen_conclusion_desc

		effect = {
			rcw_global_end_cleanup = yes
		}

		effect_all_involved = {
			custom_tooltip = rcw_separatist_survived_tt
			if = {
				limit = {
					rcw_separatist_has_survived = yes
				}
				custom_tooltip = rcw_separatist_survived_tt
				add_modifier = {
					name = rcw_independence_secured
					days = long_modifier_time
					is_decaying = yes
				}
			}
			hidden_effect = {
				rcw_country_end_cleanup = yes
			}
		}
	}

	timeout = 2190

	weight = 5000

	should_be_pinned_by_default_uninvolved_or_context = yes
	should_be_pinned_by_default_involved = yes
}

je_portuguese_colonialism = {

	group = je_group_historical_content
	dlc = dlc016

	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_military.dds"

		
	should_be_pinned_by_default_uninvolved_or_context = yes
	transferable = no
	weight = 10000

	immediate = {
		sr:region_equatorial_africa = {
			save_scope_as = congo_scope
		}
		sr:region_east_africa = {
			save_scope_as = zanj_scope
		}
		trigger_event = portuguese_colonialism.1
	}

	is_shown_when_inactive = {
        has_dlc_feature = ip4_content
		country_has_primary_culture = cu:portuguese
		is_in_geographic_region = geographic_region_iberia_old
		NOT = {
			has_law_or_variant = law_type:law_no_colonial_affairs
		}
		OR = { # If has any state in Angola or Mozambique at all
			any_state_in_angola = {
				owner = {
					OR = {
						this = ROOT
						AND = {
							country_is_colonial_or_company = yes
							is_subject_of = ROOT
						}
					}
				}
			}
			any_state_in_mozambique = {
				owner = {
					OR = {
						this = ROOT
						AND = {
							country_is_colonial_or_company = yes
							is_subject_of = ROOT
						}
					}
				}
			}
		}
	}

	possible = {
		NOT = {
			has_law_or_variant = law_type:law_no_colonial_affairs
		}
		OR = {
			any_state_in_angola = {
				is_split_state = no
				custom_tooltip = {
					text = owned_by_root_or_a_colonial_subject_tt
					owner = {
						OR = {
							this = ROOT
							AND = {
								country_is_colonial_or_company = yes
								is_subject_of = ROOT
							}
						}
					}
				}
			}
			any_state_in_mozambique = {
				is_split_state = no
				custom_tooltip = {
					text = owned_by_root_or_a_colonial_subject_tt
					owner = {
						OR = {
							this = ROOT
							AND = {
								country_is_colonial_or_company = yes
								is_subject_of = ROOT
							}
						}
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_state_in_angola = {
				is_split_state = no
				custom_tooltip = {
					text = owned_by_root_or_a_colonial_subject_tt
					owner = {
						OR = {
							this = ROOT
							AND = {
								country_is_colonial_or_company = yes
								is_subject_of = ROOT
							}
							has_war_with = ROOT
						}
					}
				}
				count >= 3
			}
			any_state_in_mozambique = {
				is_split_state = no
				custom_tooltip = {
					text = owned_by_root_or_a_colonial_subject_tt
					owner = {
						OR = {
							this = ROOT
							AND = {
								country_is_colonial_or_company = yes
								is_subject_of = ROOT
							}
							has_war_with = ROOT
						}
					}
				}
				count >= 3
			}
		}
		effect = {
			trigger_event = portuguese_colonialism.2
			set_variable = portuguese_colonialism_je_completed
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = does_not_own_any_state_in_angola_or_mozambique_tt
				NAND = {
					any_state_in_angola = {
						owner = {
							OR = {
								this = ROOT
								AND = {
									country_is_colonial_or_company = yes
									is_subject_of = ROOT
								}
							}
						}
					}
					any_state_in_mozambique = {
						owner = {
							OR = {
								this = ROOT
								AND = {
									country_is_colonial_or_company = yes
									is_subject_of = ROOT
								}
							}
						}
					}
				}
			}
		}
		effect = {
			trigger_event = portuguese_colonialism.3
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				hidden_trigger = {
					has_variable = rejected_colonialism_ambitions_var
				}
				AND = {
					is_subject = yes
					NOT = { is_at_war_with_overlord = yes }
				}
			}
		}
		effect = {
			je_portuguese_colonialism_cleanup = yes
		}
	}
}

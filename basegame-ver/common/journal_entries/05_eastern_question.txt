je_eastern_question_russia = {
	group = je_group_foreign_affairs
	dlc = dlc014

	activation_description = {
		always = yes
	}

	icon = "gfx/interface/icons/event_icons/event_skull.dds"

	scripted_progress_bar = control_eastern_crisis_bar

	is_shown_when_inactive = {
		has_dlc_feature = ip3_content
		OR = {
			c:RUS ?= this
			c:UKR ?= this
		}
		NOT = {
			exists = c:BUL
		}
		c:TUR ?= {
			has_strategic_adjacency = ROOT
		}
	}

	possible = {
		c:TUR ?= {
			has_journal_entry = je_great_eastern_crisis
		}
	}

	modifiers_while_active = {
		positive_crisis_influence
	}

	immediate = {
		trigger_event = { id = eastern_crisis.1 popup = yes }
		c:TUR ?= {
			save_scope_as = ottoman_scope
		}
		ROOT = {
			save_scope_as = root_scope
		}
		if = {
			limit = {
				c:TUR ?= {
					any_scope_state = {
						state_region = s:STATE_KARS
					}
				}
			}
			set_variable = ottomans_own_kars_at_start
		}
		if = {
			limit = {
				c:TUR ?= {
					any_scope_state = {
						state_region = s:STATE_DOBRUDJA
					}
				}
			}
			set_variable = ottomans_own_dobrudja_at_start
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = bulgaria_is_independent_tt
				any_country_in_balkans = {
					country_has_primary_culture = cu:bulgarian
					NOR = {
						is_diplomatic_play_enemy_of = c:TUR
						has_war_with = c:TUR
					}
				}
			}
			trigger_if = {
				limit = {
					has_variable = ottomans_own_kars_at_start
				}
				owns_entire_state_region_uncontested = STATE_KARS
			}
			trigger_if = {
				limit = {
					has_variable = ottomans_own_dobrudja_at_start
				}
				custom_tooltip = {
					text = ottomans_not_owns_dobruja_tt
					s:STATE_DOBRUDJA = {
						NOT = {
							any_scope_state = {
								c:TUR ?= owner
							}
						}
					}
				}
			}
		}
		effect = {
			set_variable = russia_eastern_question_victory
			trigger_event = { id = eastern_question_russia.1 popup = yes }
		}
		event_outcome_desc = {
			first_valid = {
				triggered_desc = {
					desc = event_effects_eastern_question_russia.1_tt
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = bulgaria_is_not_free_tt
				NOT = {
					exists = c:BUL
				}
			}
			c:TUR ?= {
				NOT = {
					has_journal_entry = je_great_eastern_crisis
				}
			}
		}
		effect = {
			je_eastern_question_russia_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				exists = c:TUR
			}
		}
		effect = {
			je_eastern_question_russia_cleanup = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_eastern_question_russia_cleanup = yes
		}
	}

	timeout = 3650 # 10 years


	should_be_pinned_by_default_uninvolved_or_context = yes
	weight = 100
	transferable = no

}

je_eastern_question_austria = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/relations_icons/hostile.dds"

	scripted_progress_bar = control_eastern_crisis_bar

	group = je_group_foreign_affairs
	dlc = dlc014

	activation_description = {
		always = yes
	}

	timeout = 3650

	modifiers_while_active = {
		positive_crisis_influence
	}

	immediate = {
		if = {
			limit = {
				c:TUR ?= {
					has_journal_entry = je_great_eastern_crisis
				}
			}
			trigger_event = { id = eastern_crisis.1 }
		}
		random_political_lobby = {
			limit = {
				target = {
					OR = {
						c:RUS ?= this
						c:UKR ?= this
					}
				}
				is_political_lobby_type = lobby_anti_country
			}
			save_scope_as = relevant_lobby
			target = {
				save_scope_as = relevant_country
			}
		}
		s:STATE_BOSNIA ?= {
			random_scope_state = {
				owner ?= {
					if = {
						limit = {
							is_subject = yes
						}
						top_overlord = {
							save_scope_as = bosnia_owner_scope
						}
					}
					else = {
						save_scope_as = bosnia_owner_scope
					}
				}
			}
		}
		ROOT = {
			set_variable = { name = anti_lobby_demand_target value = scope:relevant_country }
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				owns_entire_state_region_uncontested = STATE_BOSNIA
				any_subject_or_below = {
					owns_entire_state_region_uncontested = STATE_BOSNIA
				}
				trigger_if = {
					limit = {
						is_power_bloc_leader = yes
					}
					scope:power_bloc ?= {
						any_power_bloc_member = {
							owns_entire_state_region_uncontested = STATE_BOSNIA
						}
					}
				}
			}
		}
		effect = {
			scope:relevant_lobby = {
				change_appeasement = {
					amount = 3
					factor = appeasement_lobby_request_accepted
				}
			}
			trigger_event = { id = eastern_question_austria.2 popup = yes }
			clear_anti_lobby_vars = yes
		}
		event_outcome_desc = {
			first_valid = {
				triggered_desc = {
					desc = event_effects_eastern_question_austria.2_tt
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			c:TUR ?= {
				NOT = {
					has_journal_entry = je_great_eastern_crisis
				}
			}
		}
		effect = {
			je_eastern_question_austria_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOR = {
				owns_entire_state_region_uncontested = STATE_BOSNIA
				any_subject_or_below = {
					owns_entire_state_region_uncontested = STATE_BOSNIA
				}
				trigger_if = {
					limit = {
						is_power_bloc_leader = yes
					}
					scope:power_bloc ?= {
						any_power_bloc_member = {
							owns_entire_state_region_uncontested = STATE_BOSNIA
						}
					}
				}
				scope:bosnia_owner_scope ?= {
					owns_entire_state_region_uncontested = STATE_BOSNIA
				}
			}
		}
		effect = {
			je_eastern_question_austria_cleanup = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			scope:relevant_lobby = {
				change_appeasement = {
					amount = -5
					factor = appeasement_lobby_request_ignored
				}
			}
			clear_anti_lobby_vars = yes
			je_eastern_question_austria_cleanup = yes
		}
	}

	weight = 1000
}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = pebrine_cure_tt
					has_variable = pebrine_cure
				}
			}
			scope:journal_entry = {
				is_goal_complete = yes

		}
		effect = {
			trigger_event = { id = silkworm_diseases.9 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {

			custom_tooltip = {
				text = pebrine_je_fail_tt
			}

		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {
					is_incorporated = yes
					has_modifier = modifier_pebrine_state_modifier
				}
			}
		}
		effect = {
			je_silkworm_diseases_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = pebrine_cure
					any_scope_state = {
						filter = {
							has_modifier = modifier_pebrine_state_modifier
							owner = ROOT
						}
						state_has_decree_variant = { DECREE = decree_encourage_agricultural_industry }

					}
				}
				change_variable = {
					name = pebrine_progress_var

				}
			}
			random_list = {
				80 = {

				}
				20 = {
					random_country = {
						limit = {
							any_scope_state = {
								is_incorporated = yes
								any_scope_building = {
						            is_building_type = building_silk_plantation
						            level >= 2
						        }
							}
							OR = {
								any_country = {
									NOT = { THIS = PREV }
									market = ROOT.market
									any_scope_state = {
										has_modifier = modifier_pebrine_state_modifier
									}
								}
								any_neighbouring_state = { #Why? Africa.
									is_incorporated = yes
									owner = {
										any_scope_state = {
											is_incorporated = yes
											has_modifier = modifier_pebrine_state_modifier
										}
									}
								}
							}
							OR = {
								is_in_geographic_region = geographic_region_europe
								is_in_geographic_region = geographic_region_north_america
							}
							NOT = {
								has_variable = pebrine_outbreak_event
							}
						}
						trigger_event = { id = silkworm_diseases.3 weeks = 1 }
					}
				}
			}
		}
		random_events = {
			150 = 0
			40 = silkworm_diseases.2
			20 = silkworm_diseases.4
			20 = silkworm_diseases.5
			20 = silkworm_diseases.6

	}

		set_variable = {
			name = pebrine_progress_var

		ROOT = {
			save_scope_as = pebrine_country_scope
		}
		every_country = {
			limit = {
				OR = {
					any_scope_state = {
						any_neighbouring_state = {
							owner = ROOT
						}
					}
					any_scope_state = {
						has_modifier = modifier_pebrine_state_modifier

			}

		}
	}

		value = root.var:pebrine_progress_var
	}

	goal_add_value = {
		add = 12
	}

	should_be_pinned_by_default_uninvolved_or_context = yes

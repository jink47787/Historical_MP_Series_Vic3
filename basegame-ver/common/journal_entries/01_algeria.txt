je_conquest_of_algeria = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_france_expansion
	dlc = dlc004

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		has_dlc_feature = voice_of_the_people_content #France DLC
		c:FRA = THIS
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			country_or_subject_owns_entire_state_region = STATE_ORAN
			country_or_subject_owns_entire_state_region = STATE_ALGIERS
			country_or_subject_owns_entire_state_region = STATE_CONSTANTINE
			trigger_if = {
				limit = {
					exists = c:TUG
				}
				c:TUG ?= {
					OR = {
						is_country_alive = no
						is_subject_of = root
					}
				}
			}
		}
		effect = {
			custom_tooltip = je_conquest_of_algeria_tt
			trigger_event = algeria_events.1
			show_as_tooltip = {
				every_scope_state = {
					limit = {
						OR = {
							state_region = s:STATE_ORAN
							state_region = s:STATE_ALGIERS
							state_region = s:STATE_CONSTANTINE
						}
					}
					add_modifier = {
						name = crushed_algerian_resistance
						days = very_long_modifier_time
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ORAN }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ALGIERS }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_CONSTANTINE }
		}
		effect = {
			trigger_event = algeria_events.3
		}
	}

	on_monthly_pulse = {
		random_events = {
			90 = 0
			10 = algeria_events.5 # Raid Event
		}
	}

	transferable = no


	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 5000
}

je_the_algerian_departements = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_france_expansion
	dlc = dlc004

	activation_description = {
		always = yes
	}

	immediate = {
		random_scope_state = {
			limit = {
				state_region = s:STATE_ALGIERS
			}
			save_scope_as = je_algiers_state_scope
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_ORAN
			}
			save_scope_as = je_oran_state_scope
		}
		random_scope_state = {
			limit = {
				state_region = s:STATE_CONSTANTINE
			}
			save_scope_as = je_constantine_state_scope
		}
		cu:berber = {
			save_scope_as = berber_culture_scope
		}
		cu:maghrebi = {
			save_scope_as = maghrebi_culture_scope
		}
		cu:french = {
			save_scope_as = french_culture_scope
		}
		every_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_ORAN
					state_region = s:STATE_ALGIERS
					state_region = s:STATE_CONSTANTINE
				}
			}
			add_modifier = {
				name = crushed_algerian_resistance
				days = very_long_modifier_time
			}
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_the_algerian_departements_tt_1
				any_scope_state = {
					state_region = s:STATE_ALGIERS
					is_incorporated = yes
					is_split_state = no
					turmoil < 0.1
				}
				any_scope_state = {
					state_region = s:STATE_ORAN
					is_incorporated = yes
					is_split_state = no
					turmoil < 0.1
				}
				any_scope_state = {
					state_region = s:STATE_CONSTANTINE
					is_incorporated = yes
					is_split_state = no
					turmoil < 0.1
				}
			}
			OR = {
				custom_tooltip = {
					text = je_the_algerian_departements_tt_3
					any_scope_state = {
						state_region = s:STATE_ALGIERS
						culture_percent_state = {
							target = cu:french
						    value >= 0.2
						}
					}
					any_scope_state = {
						state_region = s:STATE_ORAN
						culture_percent_state = {
							target = cu:french
						    value >= 0.2
						}
					}
					any_scope_state = {
						state_region = s:STATE_CONSTANTINE
						culture_percent_state = {
							target = cu:french
						    value >= 0.2
						}
					}
				}
				AND = {
					country_average_religious_acceptance = { target = rel:sunni value >= religious_acceptance_medium }
					country_average_culture_pop_acceptance = { culture = cu:maghrebi value >= acceptance_status_3 }
					country_average_culture_pop_acceptance = { culture = cu:berber value >= acceptance_status_3 }
				}
			}
		}
		effect = {
			trigger_event = algeria_events.4
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ORAN }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_ALGIERS }
			NOT = { has_or_is_contesting_state_in_state_region = STATE_CONSTANTINE }
		}
		effect = {
			trigger_event = algeria_events.3
		}
	}

	on_monthly_pulse = {
		random_events = {
			90 = 0
			10 = algeria_events.6
		}
	}

	transferable = no


	should_be_pinned_by_default_uninvolved_or_context = yes

	weight = 5000
}

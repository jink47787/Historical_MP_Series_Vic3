je_victoria = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			trigger_if = {
				limit = {
					c:GBR ?= {
						NOT = {
							has_variable = ruler_selector_var
						}
					}
					c:HAN ?= {
						NOT = {
							has_variable = ruler_selector_var
						}
					}
				}
				custom_tooltip = {
					text = billy_died
					ruler = {
						NOT = {
							has_variable = is_billy
						}
					}
				}
			}
			trigger_else = {
				hidden_trigger = {
					always = no
				}
			}
		}
		effect = {
			custom_tooltip = hanover_union_ends
			if = {
				limit = {
					exists = c:GBR
					c:GBR = ROOT
				}
				trigger_event = { id = victoria.1 }
			}
			je_victoria_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					exists = c:GBR
					c:GBR = ROOT
					ruler = {
						has_variable = is_billy
					}
				}
				random_list = {
					11 = { }
					1 = {
						ruler = {
							kill_character = yes 
						}
					}
				}
			}
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					exists = c:GBR
				}
				NOT = {
					exists = c:HAN
				}
				c:HAN ?= {
					NOT = {
						is_subject_of = c:GBR
					}
				}
				c:GBR ?= {
					has_variable = ruler_selector_var
				}
				c:GBR ?= {
					custom_tooltip = {
						text = "DEBUG_NO_CHARACTER_IN_GBR_IS_VICTORIA"
						NOT = {
							any_scope_character = {
								has_variable = is_victoria
							}
						}
					}
				}
			}
		}
		effect = {
			je_victoria_cleanup = yes
		}
	}

	weight = 10

	should_be_pinned_by_default_uninvolved_or_context = yes
}

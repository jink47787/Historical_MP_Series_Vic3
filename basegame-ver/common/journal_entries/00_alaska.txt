je_sale_of_alaska = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		c:ALK ?= {
			is_player = no
		}
		OR = {
			is_in_geographic_region = geographic_region_east_asia
			is_in_geographic_region = geographic_region_europe
		}
		OR = {
			has_state_in_state_region = STATE_ALASKA
			any_subject_or_below = {
				has_state_in_state_region = STATE_ALASKA
			}
		}
	}

	scripted_button = je_sell_alaska_button

	possible = {
		is_subject = no
		OR = {
			any_scope_state = {
				state_region = s:STATE_ALASKA
			}
			any_subject_or_below = {
				any_scope_state = {
					state_region = s:STATE_ALASKA
				}
			}
		}
		custom_tooltip = {
			text = unfriendly_nation_with_interest_in_alaska_tt
			sr:region_pacific_coast ?= {
				any_scope_interest_marker = {
					owner ?= {
						country_rank >= root.country_rank
						relations:root < 0
					}
				}
			}
		}
		custom_tooltip = {
			text = friendly_nation_with_interest_in_alaska_tt
			any_country_in_north_america = {
				NOT = {
					is_subject = yes
				}
				relations:root >= 0
				country_rank >= rank_value:great_power
				any_scope_state = {
					region = sr:region_pacific_coast
					is_incorporated = yes
				}
			}
		}
	}

	immediate = {
		s:STATE_ALASKA ?= {
			random_scope_state = {
				limit = {
					OR = {
						owner = ROOT
						owner = {
							is_subject_of = ROOT
						}
					}
				}
				save_scope_as = alaska_scope
				owner = {
					save_scope_as = alaska_owner_scope
				}
			}
		}
		ruler = {
			save_scope_as = ruler_scope
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = sold_alaska_tt
				has_variable = successful_sale_of_alaska
			}
		}
		effect = {
			custom_tooltip = {
				text = selling_alaska_tt
			}
			je_sale_of_alaska_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				OR = {
					has_state_in_state_region = STATE_ALASKA
					any_subject_or_below = {
						has_state_in_state_region = STATE_ALASKA
					}
				}
				custom_tooltip = {
					text = not_sold_alaska_tt
					has_variable = successful_sale_of_alaska
				}
			}
		}
		effect = {
			je_sale_of_alaska_cleanup = yes
		}
	}

	weight = 200

	transferable = no


	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_alaska = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		has_interest_marker_in_region = sr:region_pacific_coast
		has_technology_researched = nationalism
		hidden_trigger = {
			NOT = { has_global_variable = alaska_sold }
		}
	}

	scripted_button = je_buy_alaska_button

	possible = {
		is_subject = no
		custom_tooltip = {
			text = je_alaska_pacific_homeland_tt
			any_scope_state = {
				is_incorporated = yes
				region = sr:region_pacific_coast
				is_homeland_of_country_cultures = ROOT
			}
		}
		NOR = {
			owns_entire_state_region = STATE_ALASKA
			any_subject_or_below = {
				owns_entire_state_region = STATE_ALASKA
			}
		}
		country_rank >= rank_value:great_power
	}

	immediate = {
		s:STATE_ALASKA ?= {
			random_scope_state = {
				limit = {
					owner = {
						NOT = {
							is_country_type = decentralized
						}
					}
				}
				save_scope_as = alaska_state
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			s:STATE_ALASKA = {
				any_scope_state = {
					OR = {
						owner = root
						owner = {
							AND = {
								is_subject_of = root
								capital = {
									state_region = s:STATE_ALASKA
								}
							}
						}
					}
				}
				any_scope_state = {
					OR = {
						owner = root
						owner = {
							AND = {
								is_subject_of = root
								capital = {
									state_region = s:STATE_ALASKA
								}
							}
						}
						owner = {
							is_country_type = decentralized
						}
					}
					count = all
				}
			}
		}
		effect = {
			set_global_variable = alaska_sold
			add_modifier = {
				name = conquest_of_alaska
				days = normal_modifier_time
			}
			je_alaska_cleanup = yes
		}
	}

	timeout = 7300

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_alaska_cleanup = yes
		}
	}

	weight = 200

	transferable = no

		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_suez_survey = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = survey_suez_taken
				add = 1
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
		}
		effect = {
			remove_modifier = modifier_surveying_suez
			trigger_event = { id = canal_events.100 }
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { has_variable = survey_suez_taken }
				s:STATE_SINAI = {
					any_scope_state = {
						NOT = { owner = ROOT }
						has_building = building_suez_canal
						owner = { NOT = { has_war_with = root } }
					}
				}
			}
		}
		effect = {
			remove_variable = survey_suez_taken
			remove_modifier = modifier_surveying_suez
			je_suez_survey_cleanup = yes
		}
	}

	current_value = {
		value = root.var:survey_suez_taken
	}

	goal_add_value = {
		value = 18
	}

	progressbar = yes

	weight = 10
		
	should_be_pinned_by_default_uninvolved_or_context = no
}

		}

		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				any_scope_state = {
					region = sr:region_northeast_asia
					is_treaty_port = yes
				}
				AND = {
					country_or_subject_owns_entire_state_region = STATE_SHENGJING
					country_or_subject_owns_entire_state_region = STATE_SOUTHERN_MANCHURIA
					country_or_subject_owns_entire_state_region = STATE_NORTHERN_MANCHURIA
				}
			}
	        any_state_in_manchuria_old = {
	            filter = {
					custom_tooltip = {
						text = owned_by_russia_or_chinese_country
						OR = {
							owner = c:RUS
							owner = {
								OR = {
									country_has_primary_culture = cu:manchu
									country_has_primary_culture = cu:han
								}
							}
						}
					}
	            }

					is_building_type = building_railway
					levels_owned_by_country = {
						target = c:RUS
						value  >= 1
					}
				}

	        }
		}

			set_global_variable = manchurian_railway_variable
			great_game_russia_progress_15 = yes
			je_chinese_concessions_cleanup = yes

		flags = { hidden invalid_outcome }

		trigger = {
			NOT = {
				is_involved_in_journal_entry = je_great_game_control

			je_chinese_concessions_cleanup = yes
		}
	}

	weight = 1000

je_portugal_and_brazil = {
	icon = "gfx/interface/icons/invention_icons/navigation.dds"

	group = je_group_historical_content
	dlc = dlc016

	activation_description = {
		always = yes
	}

	scripted_button = je_portugal_and_brazil_pedro_button
	scripted_button = je_portugal_and_brazil_pursue_united_kingdom_button

	is_shown_when_inactive = {
        has_dlc_feature = ip4_content
		OR = {
			c:POR ?= this
			AND = {
				c:IBE ?= this
				was_formed_from = POR
			}
		}
		exists = c:BRZ
	}

	immediate = {
		c:BRZ ?= {
			save_scope_as = brazil_scope
		}
	}

	outcome = {
		header = je_portugal_and_brazil_complete_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				trigger_if = {
					limit = {
						has_variable = je_portugal_and_brazil_pursue_united_kingdom_button_var
					}
					custom_tooltip = {
						text = has_completed_uk_je_tt
						has_variable = completed_uk_var
					}
				}
				trigger_if = {
					limit = {
						exists = scope:brazil_scope
					}
					OR = {
						ROOT = {
							has_diplomats_expelled = scope:brazil_scope
						}
						scope:brazil_scope ?= {
							has_diplomats_expelled = ROOT
						}
					}
				}
				trigger_else = {
					custom_tooltip = {
						text = has_expelled_brazilian_diplomats_tt
						OR = {
							ROOT = {
								has_diplomats_expelled = c:BRZ
							}
							c:BRZ ?= {
								has_diplomats_expelled = ROOT
							}
						}
					}
				}
			}
		}
		effect = {
			je_portugal_and_brazil_cleanup = yes
		}
	}

	possible = {
		has_interest_marker_in_region = sr:region_brazil
		"relations(c:BRZ)" > 10
	}

	on_monthly_pulse = {

	}

	modifiers_while_active = {

	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					exists = scope:brazil_scope
				}
				has_variable = failed_uk_var
				AND = {
					custom_tooltip = {
						text = portugal_formation_uk_je_tt
						NOT = {
							c:POR ?= this
						}
					}
					custom_tooltip = {
						text = iberia_formation_uk_je_tt
						AND = {
							NOT = {
								c:IBE ?= this
							}
							was_formed_from = POR
						}
					}
				}
			}
		}
		effect = {
			je_portugal_and_brazil_cleanup = yes
		}
	}

	weight = 5000


	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_portugal_united_kingdom = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_historical_content
	dlc = dlc016

	activation_description = {
		always = yes
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = portugal_ruler_is_ruler_tt
				trigger = {
					c:BRZ.ruler ?= ROOT.ruler
				}
			}
			triggered_desc = {
				desc = portugal_ruler_in_line_nominal_tt
				trigger = {
					has_dlc_feature = rp1_content
					exists = c:BRZ.heir
					c:BRZ.heir ?= ROOT.ruler
					c:BRZ.var:pedropoints <= 5
				}
			}
			triggered_desc = {
				desc = portugal_ruler_in_line_securely_tt
				trigger = {
					has_dlc_feature = rp1_content
					exists = c:BRZ.heir
					c:BRZ.heir ?= ROOT.ruler
					c:BRZ.var:pedropoints > 5
				}
			}
			triggered_desc = {
				desc = portugal_ruler_in_line_tt
				trigger = {
					NOT = { has_dlc_feature = rp1_content }
					exists = c:BRZ.heir
					c:BRZ.heir ?= ROOT.ruler
				}
			}
			triggered_desc = {
				desc = portugal_ruler_not_in_line_tt
				trigger = {
					NOT = {
						c:BRZ.heir ?= ROOT.ruler
					}
				}
			}
		}
	}

	immediate = {
		c:BRZ ?= {
			save_scope_as = brazil_scope
		}
	}

	on_monthly_pulse = {

	}

	modifiers_while_active = {

	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			c:BRZ ?= {
				has_law = law_type:law_monarchy
				has_diplomatic_pact = {
					who = ROOT
					type = personal_union
					is_initiator = no
				}
				trigger_if = {
					limit = {
						has_dlc_feature = rp1_content
					}
					NOT = {
						has_journal_entry = je_pedro_brazil
					}
				}
			}
		}
		effect = {
			set_variable = { name = completed_uk_var days = short_modifier_time }
			trigger_event = {
				id = lusofonia.3
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { has_law = law_type:law_monarchy }
				NOT = { has_journal_entry = je_portugal_and_brazil }
				AND = {
					c:BRZ ?= { NOT = { has_law = law_type:law_monarchy } }
					custom_tooltip = {
						text = por_not_restoring_union_tt
						NOT = {
							has_variable = por_seeking_restore_union_var
						}
					}
				}
			}
		}
		effect = {
			set_variable = { name = failed_uk_var days = short_modifier_time }
			je_portugal_united_kingdom_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				exists = c:BRZ
			}
			NOT = { war_may_restore_country = BRZ }
		}
		effect = {
			je_portugal_united_kingdom_cleanup = yes
		}
	}

	weight = 5000

    	
	should_be_pinned_by_default_uninvolved_or_context = yes
}

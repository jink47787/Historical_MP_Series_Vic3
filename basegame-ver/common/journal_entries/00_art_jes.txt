je_romanticism = {

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		has_technology_researched = rationalism
		NOT = { has_technology_researched = realism }
		NOT = { has_variable = completed_romanticism_je }
		NOT = { is_country_type = unrecognized }
	}

	possible = {
		has_technology_researched = romanticism
		NOT = { has_technology_researched = realism }
		any_scope_state = {
			has_building = building_urban_center
		}
	}

	immediate = {
		scope:journal_entry ?= {
			set_variable = { name = romanticism_level_target value = 2 }
		}
		ordered_scope_building = {
			limit = {
				has_active_production_method = pm_traditional_art
				occupancy >= 0.9
			}
			order_by = this.level
			position = 0
			scope:journal_entry ?= {
				change_variable = { name = romanticism_level_target add = prev.level }
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			trigger_if = {
				limit = {
					scope:journal_entry ?= {
						has_variable = romanticism_level_target
					}
				}
				any_scope_building = {
					is_building_type = building_art_academy
					has_active_production_method = pm_traditional_art
					occupancy >= 0.9
					level >= scope:journal_entry.var:romanticism_level_target
				}
			}
			trigger_else = {
				any_scope_building = {
					is_building_type = building_art_academy
					has_active_production_method = pm_traditional_art
					occupancy >= 0.9
					level >= 2
				}
			}
		}
		effect = {
			trigger_event = { id = society_tech_events.302 }
			set_variable = {
				name = completed_romanticism_je
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_technology_researched = realism
		}
		effect = {
			je_romanticism_cleanup = yes
		}
	}

	timeout = 3650

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_romanticism_cleanup = yes
		}
	}
}

		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_building = {
				is_building_type = building_art_academy
				level >= 5
				has_active_production_method = pm_photographic_art
			}
			any_scope_building = {
				filter = {
					is_building_type = building_art_academy
				}
				percent > 0.75
				has_active_production_method = pm_photographic_art
				occupancy > 0.75
			}
		}
		effect = {
			set_variable = camera_je_done

		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_technology_researched = film
		}
		effect = {
			je_camera_cleanup = yes

	}

	timeout = 3650

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC

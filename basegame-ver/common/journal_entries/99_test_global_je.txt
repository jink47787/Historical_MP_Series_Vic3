# Sample JE for global journal entries

je_global_test = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_global_test

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		always = no
		#always = yes
	}
	
	modifiers_while_active = {
		lippe_crisis_ongoing
	}

	scripted_button = je_global_test_button
	scripted_button = je_global_test_button_2
	
	should_be_involved = {
		OR = {
			country_rank >= rank_value:great_power
			country_definition = cd:LIP
			country_definition = cd:SCM
		}		
	}
	
	should_show_when_not_involved = {
		OR = {
			country_rank >= rank_value:major_power
			country_definition = cd:LIP
			country_definition = cd:SCM
		}	
	}	

	possible = {
		always = no
		#any_country = {
		#	count >= 2
		#	OR = {
		#		country_definition = cd:LIP
		#		country_definition = cd:SCM
		#	}
		#}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_country = {
				count < 1
				OR = {
					country_definition = cd:LIP
					country_definition = cd:SCM
				}
			}
		}
		effect = {
			je_global_test_cleanup = yes
		}
		effect_all_involved = {
			add_modifier = { name = lippe_crisis_averted years = 10 }
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			any_country = {
				count > 2
				OR = {
					country_definition = cd:LIP
					country_definition = cd:SCM
				}
			}
		}
		effect = {
			je_global_test_cleanup = yes
		}
		effect_all_involved = {
			add_modifier = { name = lippe_crisis years = 10 }
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_global_test_cleanup = yes
		}
		effect_all_involved = {
			add_modifier = { name = lippe_crisis years = 10 }
		}
	}

	immediate_all_involved = {
		if = {
			limit = { exists = c:LIP }
			change_relations = { country = c:LIP value = -10 }
		}
		if = {
			limit = { exists = c:SCM }
			change_relations = { country = c:SCM value = -10 }
		}
	}	

	on_become_involved_after_activation = {
		if = {
			limit = { exists = c:LIP }
			change_relations = { country = c:LIP value = -10 }
		}
		if = {
			limit = { exists = c:SCM }
			change_relations = { country = c:SCM value = -10 }
		}
	}

	on_no_longer_involved = {
		if = {
			limit = { exists = c:LIP }
			change_relations = { country = c:LIP value = 10 }
		}
		if = {
			limit = { exists = c:SCM }
			change_relations = { country = c:SCM value = 10 }
		}
	}

	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				every_scope_je_involved = {
					limit = {
						NOR = {
							country_rank >= rank_value:great_power
							country_definition = cd:LIP
							country_definition = cd:SCM
						}					
					}
					
					scope:journal_entry = { remove_involved_country = prev }
				}		
			}

			scope:journal_entry = { remove_involved_country = c:FRA }
			scope:journal_entry = { add_involved_country = c:ARG }
		}
	}

	weight = 1000

	timeout = 1825

	should_be_pinned_by_default_uninvolved_or_context = yes
}

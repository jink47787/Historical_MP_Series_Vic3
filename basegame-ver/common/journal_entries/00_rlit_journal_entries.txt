	immediate = {

				1 = {
					set_variable = {
						name = rus_rlit_path_var
						value = flag:autocratic
					}

				1 = {
					set_variable = {

						value = flag:radical

						name = rus_rlit_path_var

				}
			}
		}

		effect_header = je_standard_on_completion_header

		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = rlit_je1_complete_tt
				has_variable = rlit_cluster1_concluded
			}
		}
		effect = {
			add_modifier = {
				name = golden_age_of_russian_literature
				days = long_modifier_time
			}
			if = {
				limit = {
					scope:journal_entry = {
						"scripted_bar_progress(rlit_prestige_bar)" >= 100
					}
				}
				add_modifier = {
					name = golden_age_literary_triumph
					days = long_modifier_time
				}
			}
			trigger_event = {
				id = rlit.015
				popup = yes
			}
		}
	}
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			hidden_trigger = {
				NOT = {
					has_variable = rlit_cluster1_concluded
				}
			}
			OR = {
				AND = {
					custom_tooltip = {
						text = rlit_fail_free_word_tt

							"scripted_bar_progress(rlit_censorship_bar)" <= 0
						}
					}
					rlit_speech_laws_restrictive = yes
				}

					custom_tooltip = {

						}
					}

			}

			add_modifier = {
				name = literature_suppressed_legacy
				days = normal_modifier_time
			}
			rlit_abandon_arc = yes
		}
	}

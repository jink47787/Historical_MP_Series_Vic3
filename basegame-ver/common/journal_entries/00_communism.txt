je_communism_1 = {
	icon = "gfx/interface/icons/event_icons/event_revolutionary_flag.dds"

	group = je_group_socialism

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOT = { has_variable = communism_1_done }
		has_technology_researched = socialism
	}

	scripted_progress_bar = communism_1_progress_bar

	possible = {
		ig:ig_trade_unions ?= {
			is_marginal = no
		}
		OR = {
			ig:ig_trade_unions ?= {
				ig_approval <= define:NPolitics|APPROVAL_THRESHOLD_UNHAPPY
				law_stance = {
					law = law_type:law_factory_councils
					value > neutral
				}
			}
			any_political_movement = {
				OR = {
					AND = {
						is_political_movement_type = movement_labor
						this.political_movement_radicalism >= define:NPolitics|RADICALISM_AGITATING_THRESHOLD
					}
					is_political_movement_type = movement_socialist
				}
			}
		}
	}

	immediate = {
		save_scope_as = root_country
		set_communism_vars = yes
		show_as_tooltip = {
			if = {
				limit = {
					country_has_laws_in_need_of_critical_modernization = no
					NOT = {
						any_political_movement = {
							is_political_movement_type = movement_socialist
						}
					}
				}
				create_political_movement = { type = movement_socialist }
			}
		}
		trigger_event = { id = communism.10 days = 1 popup = yes }
	}

	on_weekly_pulse = {
		effect = {
			calculate_communism_progress = yes
		}
	}

	on_monthly_pulse = {
		random_events = {
			200 = 0
			5 = communism.7
		}
		effect = {
			calculate_communism_progress = yes
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = communism_progress_bar_at_full_tt
					scope:journal_entry = {
						"scripted_bar_progress(communism_1_progress_bar)" >= 100
					}
				}
				has_law_or_variant = law_type:law_council_republic
			}
		}
		effect = {
			set_variable = communism_1_done
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_council_republic
					}
				}
				trigger_event = { id = communism.1 popup = yes }
			}
			else = { # Mirrors effects of communism.1, skips je_communism_2
				if = {
					limit = {
						any_interest_group = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
					}
					random_interest_group = {
						limit = {
							law_stance = {
								law = law_type:law_council_republic
								value > neutral
							}
						}
						set_variable = communist_ig_var
					}
				}
				else = {
					ig:ig_trade_unions ?= {
						set_variable = communist_ig_var
					}
				}
				if = {
					limit = {
						NOT = {
							has_journal_entry = je_state_and_revolution
						}
					}
					trigger_event = { id = communism.2 popup = yes }
				}
			}
			custom_tooltip = communism_je_1_fail_tt
			je_communism_1_cleanup = yes

			effect = {
				add_journal_entry = { type = je_communism_2 }
			}

			header = communism.1.a

				activate_law = law_type:law_wage_subsidies
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.b
			effect = {
				activate_law = law_type:law_slavery_banned
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.c
			effect = {
				activate_law = law_type:law_regulatory_bodies
			}
		}
		event_outcome_effect_desc = {
			header = communism.1.d
			effect = {
				every_interest_group = {
					limit = {
						leader = {
							OR = {
								has_ideology = ideology:ideology_vanguardist
								has_ideology = ideology:ideology_communist_leader
								has_ideology = ideology:ideology_anarchist
							}
						}
					}

						name = communist_government_modifier

					}

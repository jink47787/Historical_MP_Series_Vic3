central_america_falls_apart = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_uninvolved_or_context = yes


	is_shown_in_lobby = {
		c:UCA ?= THIS
	}

	on_monthly_pulse = {
		random_events = {
			10 = 0
			1 = central_america.3
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			any_scope_state = {
				state_region = s:STATE_GUATEMALA
				turmoil < 0.2
			}
			any_scope_state = {
				state_region = s:STATE_HONDURAS
				turmoil < 0.2
			}
			any_scope_state = {
				state_region = s:STATE_COSTA_RICA
				turmoil < 0.2
			}
			any_scope_state = {
				state_region = s:STATE_SAN_SALVADOR
				turmoil < 0.2
			}
			any_scope_state = {
				state_region = s:STATE_NICARAGUA
				turmoil < 0.2
			}
			hidden_trigger = {
				 game_date > "1836.1.1"
			}
		}
		effect = {
			central_america_falls_apart_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = central_america_fail_tt
				any_scope_state = {
					OR = {
						state_region = s:STATE_GUATEMALA
						state_region = s:STATE_HONDURAS
						state_region = s:STATE_COSTA_RICA
						state_region = s:STATE_SAN_SALVADOR
						state_region = s:STATE_NICARAGUA
					}
					count < 3
				}
			}
		}
		effect = {
			trigger_event = { id = central_america.2 popup = yes }
			custom_tooltip = central_america_capital_tt
		}
	}

	status_desc = {
		first_valid = {
			triggered_desc = {
				desc = central_america_breakup_status
			}
		}
	}
}



ragamuffin_war = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_brazil

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_at_war = no
			OR = {
				NOT = { exists = c:PNI }
				c:PNI ?= {
					is_country_alive = no
				}
			}
			OR = {
				NOT = { exists = c:PRA }
				c:PRA ?= {
					is_country_alive = no
				}
			}
		}
		effect = {
			hidden_effect = { set_variable = ragamuffin_war_completed }
			ragamuffin_war_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			is_at_war = no
			OR = {
				c:PNI ?= {
					is_country_alive = yes
				}
				c:PRA ?= {
					is_country_alive = yes
				}
			}
		}
		effect = {
			trigger_event = { id = south_america.5 }
			hidden_effect = { set_variable = ragamuffin_war_completed }
			if = {
				limit = {
					has_journal_entry = je_pedro_brazil
				}
				remove_pedropoint_effect = yes
			}
		}
	}

		
	should_be_pinned_by_default_uninvolved_or_context = yes

}

ragamuffin_war_minors = {
	icon = "gfx/interface/icons/event_icons/bicorne_hat.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		OR = {
			c:PNI ?= THIS
			c:PRA ?= THIS
		}
	}


	immediate = {
		c:BRZ ?= {
			save_scope_as = brz_scope
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			is_at_war = no
			is_country_alive = yes
		}
		effect = {
			if = {
				limit = {
					c:PNI ?= THIS
				}
				custom_tooltip = {
					text = ragamuffin_war_minors_tt
				}
			}
			else = {
				trigger_event = { id = brazilian_minors.4 popup = yes }
			}
			ragamuffin_war_minors_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			is_country_alive = no
		}
		effect = {
			ragamuffin_war_minors_cleanup = yes
		}
	}


	should_be_pinned_by_default_uninvolved_or_context = yes

}

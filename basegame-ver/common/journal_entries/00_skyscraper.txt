je_skyscraper_site = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = skyscraper_var
				add = 1
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = { is_goal_complete = yes }
			any_scope_state = {
				is_split_state = no
				is_incorporated = yes
				any_scope_building = {
					is_building_type = building_urban_center
					level >= 20
				}
			}
		}
		effect = {
			remove_modifier = skyscraper_survey
			trigger_event = { id = skyscraper.1 } # site found
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {
					is_split_state = no
					is_incorporated = yes
					any_scope_building = {
						is_building_type = building_urban_center
						level >= 20
					}
				}
			}
		}
		effect = {
			remove_modifier = skyscraper_survey
			remove_variable = skyscraper_var
			je_skyscraper_site_cleanup = yes
		}
	}

	current_value = {
		value = root.var:skyscraper_var
	}

	goal_add_value = {
		value = 24
	}

	progressbar = yes

	weight = 10
		
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_skyscraper_construction = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:target = {
				any_scope_building = {
					is_building_type = building_skyscraper
					is_under_construction = no
				}
			}
		}
		effect = {
			trigger_event = { id = skyscraper.2 }
			remove_variable = skyscraper_var
			scope:target = {
				remove_modifier = skyscraper_site
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			NOT = {
				any_scope_state = {
					has_modifier = skyscraper_site
				}
			}
		}
		effect = {
			remove_variable = skyscraper_var
			je_skyscraper_construction_cleanup = yes
		}
	}
		
	should_be_pinned_by_default_uninvolved_or_context = no
}

je_splitting_oman = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		c:OMA = THIS
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = said_bin_sultan_al_said_dies
				ruler = {
					exists = yes
					NOT = { has_template = oma_said_bin_sultan }
				}
				country_has_monarchy_law = yes
			}
		}
		effect = {
			custom_tooltip = omani_realm_splits
			if = {
				limit = {
					exists = c:OMA
					c:OMA = ROOT
				}
				trigger_event = { id = zanzibar.1 }
			}
			je_splitting_oman_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { exists = c:OMA }
				NOT = { country_has_monarchy_law = yes }
				exists = c:ZAN
				c:OMA ?= {
					has_variable = ruler_selector_var
				}
			}
		}
		effect = {
			je_splitting_oman_cleanup = yes
		}
	}

	# no monthly pulse to kill the sultan.

	weight = 10

		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

# Simba wa Zanzibar
je_lion_of_the_zanj = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	immediate = {
		s:STATE_RIFT_VALLEY = {
			save_scope_as = rift_valley_region
		}
		s:STATE_TANGANYIKA = {
			save_scope_as = tanganyika_region
		}
		s:STATE_KENYA = {
			save_scope_as = kenya_region
		}
		s:STATE_ZANZIBAR = {
			save_scope_as = zanzibar_region
		}
		s:STATE_LINDI = {
			save_scope_as = lindi_region
		}

		cu:swahili = {
			save_scope_as = swahili_scope
		}

		cu:bedouin = {
			save_scope_as = omani_scope
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				is_country_type = recognized
				AND = {
					owns_entire_state_region_uncontested = STATE_RIFT_VALLEY
					owns_entire_state_region_uncontested = STATE_TANGANYIKA
					owns_entire_state_region_uncontested = STATE_KENYA
					owns_entire_state_region_uncontested = STATE_ZANZIBAR
					owns_entire_state_region_uncontested = STATE_LINDI
				}
			}
		}
		effect = {
			trigger_event = { id = zanzibar.4 }

			custom_tooltip = {
				text = je_lion_of_the_zanj_claims

				if = {
					limit = { NOT = { owns_entire_state_region_uncontested = STATE_RIFT_VALLEY } }
					s:STATE_RIFT_VALLEY = {
						add_claim = ROOT
					}
				}
				if = {
					limit = { NOT = { owns_entire_state_region_uncontested = STATE_TANGANYIKA } }
					s:STATE_TANGANYIKA = {
						add_claim = ROOT
					}
				}
				if = {
					limit = { NOT = { owns_entire_state_region_uncontested = STATE_KENYA } }
					s:STATE_KENYA = {
						add_claim = ROOT
					}
				}
				if = {
					limit = { NOT = { owns_entire_state_region_uncontested = STATE_ZANZIBAR } }
					s:STATE_ZANZIBAR = {
						add_claim = ROOT
					}
				}
				if = {
					limit = { NOT = { owns_entire_state_region_uncontested = STATE_LINDI } }
					s:STATE_LINDI = {
						add_claim = ROOT
					}
				}
			}


			if = {
				limit = {
					any_primary_culture = { cu:bedouin = this }
				}
				custom_tooltip = {
					text = je_lion_of_the_zanj_homelands_bedouin
					if = {
						limit = {
							owns_entire_state_region_uncontested = STATE_RIFT_VALLEY
							owns_entire_state_region_uncontested = STATE_TANGANYIKA
							owns_entire_state_region_uncontested = STATE_KENYA
							owns_entire_state_region_uncontested = STATE_ZANZIBAR
							owns_entire_state_region_uncontested = STATE_LINDI
						}
						s:STATE_ZANZIBAR = {
							add_homeland = cu:bedouin
						}
					}
				}
			}

			if = {
				limit = {
					any_primary_culture = { cu:swahili = this }
				}
				custom_tooltip = {
					text = je_lion_of_the_zanj_homelands_swahili
					if = {
						limit = {
							owns_entire_state_region_uncontested = STATE_RIFT_VALLEY
							owns_entire_state_region_uncontested = STATE_TANGANYIKA
							owns_entire_state_region_uncontested = STATE_KENYA
							owns_entire_state_region_uncontested = STATE_ZANZIBAR
							owns_entire_state_region_uncontested = STATE_LINDI
						}
						s:STATE_RIFT_VALLEY = {
							add_homeland = cu:swahili
						}
						s:STATE_TANGANYIKA = {
							add_homeland = cu:swahili
						}
						s:STATE_LINDI = {
							add_homeland = cu:swahili
						}
					}
				}
			}
		}
	}

	# Fails if Zanzibar is a subject, or if a non-African country manages to establish a foothold in Zanzibar's quote-unquote core territories (roughly Kenya & Tanzania)
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
	    	OR = {
	    		is_subject = yes
	    		custom_tooltip = {
	    			text = je_lion_of_the_zanj_fail

			        any_neighbouring_state = {
						OR = {
							state_region = s:STATE_KENYA
							state_region = s:STATE_RIFT_VALLEY
							state_region = s:STATE_ZANZIBAR
							state_region = s:STATE_TANGANYIKA
							state_region = s:STATE_LINDI
						}
						OR = {
			    			owner = {
			    				is_not_in_geographic_region = geographic_region_africa
			    				NOT = { is_diplomatic_play_enemy_of = ROOT }
			    			}
			    			owner = {
			    				is_subject = yes
			    				top_overlord = {
			    					is_not_in_geographic_region = geographic_region_africa
			    					NOT = { is_diplomatic_play_enemy_of = ROOT }
			    				}
			    			}
						}
			    	}
			    }
	        }
		}
		effect = {

			show_as_tooltip = {
				trigger_event = { id = zanzibar.3 } # it's actually triggered by event zanzibar.2
			}

			every_scope_state = {
				limit = {
					is_capital = no
				}
				custom_tooltip = {
					text = je_lion_of_the_zanj_loss_of_territory
				}
			}

			hidden_effect = {

				# Collapse of Zanzibar event triggers for Zanzibar's top liege if subject failstate is triggered
				if = {
					limit = {
						is_subject = yes
					}
					top_overlord = { trigger_event = { id = zanzibar.2 } }
				}

				# Collapse of Zanzibar event triggers for colonizer that successfully intruded on Zanzibar's 'natural sphere' if territorial integrity failstate is triggered
				else_if = {
					limit = {
				        any_neighbouring_state = {
							OR = {
								state_region = s:STATE_KENYA
								state_region = s:STATE_RIFT_VALLEY
								state_region = s:STATE_ZANZIBAR
								state_region = s:STATE_TANGANYIKA
								state_region = s:STATE_LINDI
							}
							OR = {
								owner = {
				    				is_not_in_geographic_region = geographic_region_africa
				    			}
				    			owner = {
				    				is_subject = yes
				    				top_overlord = {
				    					is_not_in_geographic_region = geographic_region_africa
				    				}
				    			}
							}
						}
		    		}

			    	random_neighbouring_state = {
			    		limit = {
							OR = {
								state_region = s:STATE_KENYA
								state_region = s:STATE_RIFT_VALLEY
								state_region = s:STATE_ZANZIBAR
								state_region = s:STATE_TANGANYIKA
								state_region = s:STATE_LINDI
							}
							OR = {
								owner = {
				    				is_not_in_geographic_region = geographic_region_africa
				    			}
				    			owner = {
				    				is_subject = yes
				    				top_overlord = {
				    					is_not_in_geographic_region = geographic_region_africa
				    				}
				    			}
							}
			    		}
	    				if = {
	    					limit = {
	    						owner = { is_not_in_geographic_region = geographic_region_africa }
	    					}
	    					owner = { trigger_event = { id = zanzibar.2 } }
	    				}
	    				else_if = {
	    					limit = {
	    						owner = {
	    							NOT = { is_diplomatic_play_enemy_of = ROOT }
			        				AND = {
				        				is_subject = yes
				        				top_overlord = {
				        					is_not_in_geographic_region = geographic_region_africa
				        					NOT = { is_diplomatic_play_enemy_of = ROOT }
				        				}
			        				}
			        			}
	    					}
	    					owner = {
	    						top_overlord = { trigger_event = { id = zanzibar.2 } }
	    					}
	    				}
			    	}
			    }
		    }
			je_lion_of_the_zanj_cleanup = yes
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			c:ZAN ?= {
				is_not_in_geographic_region = geographic_region_africa
			}
		}
		effect = {
			je_lion_of_the_zanj_cleanup = yes
		}
	}

	on_yearly_pulse = {
		random_events = {
			chance_of_no_event = 80

			20 = zanzibar.5 # tippu tip
		}
	}

	weight = 10

		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

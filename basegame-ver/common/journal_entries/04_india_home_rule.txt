je_india_home_rule = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_british_india
	dlc = dlc011

	activation_description = {
		always = yes
	}

	scripted_progress_bar = je_india_home_rule_progressbar

	scripted_button = je_india_home_rule_request_democracy_permission

	is_shown_when_inactive = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		is_subject = yes
		has_technology_researched = pan-nationalism
	}

	possible = {
		NOT = {
			is_subject_type = subject_type_chartered_company
		}
		any_political_movement = {
			is_political_movement_type = movement_india_pan_national
		}
		is_subject = yes
		custom_tooltip = {
			text = no_south_asian_primary_cultures_tt
			NOT = {
				any_primary_culture = {
					has_discrimination_trait_group = heritage_group_south_asian
				}
			}
		}
	}

	immediate = {
		top_overlord = {
			save_scope_as = overlord_scope
			add_journal_entry = {
				type = je_india_nationalism_britain
			}
		}
		ig:ig_intelligentsia ?= {
			save_scope_as = intelligentsia_scope
		}
		ig:ig_industrialists ?= {
			save_scope_as = government_scope
		}
		ig:ig_armed_forces ?= {
			save_scope_as = armed_forces_scope
		}
		ROOT = {
			save_scope_as = india_scope
		}
		random_political_movement = {
			limit = {
				is_political_movement_type = movement_india_pan_national
			}
			save_scope_as = indian_national_movement
		}
		trigger_event = { id = india_home_rule_events.7 popup = yes }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = is_dominion_or_protectorate_tt
				OR = {
					has_diplomatic_pact = {
						who = overlord
						type = dominion
						is_initiator = no
					}
					has_diplomatic_pact = {
						who = overlord
						type = protectorate
						is_initiator = no
					}
				}
			}
			country_has_voting_franchise = yes
			has_law_or_variant = law_type:law_elected_bureaucrats
			custom_tooltip = {
				text = accepts_south_asian_culture_tt
				any_scope_culture = {
					has_discrimination_trait_group = heritage_group_south_asian
					root = {
						country_average_culture_pop_acceptance = {
							culture = prev
							value >= 60
						}
					}
				}
			}
		}
		effect = {
			trigger_event = { id = india_home_rule_events.1 popup = yes }
			set_global_variable = india_home_rule
		}
	    event_outcome_desc = {
	        first_valid = {
	            triggered_desc = {
	                desc = event_effects_india_home_rule_events.1_tt
	            }
	        }
	    }
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				has_any_secessionists_broken_out = yes
				any_political_movement = {
					is_political_movement_type = movement_india_pan_national
					political_movement_radicalism >= india_independence_radicalism_threshold
				}
				any_civil_war = {
					civil_war_progress >= 0.75
					is_civil_war_type = revolution
				}
				NOT = {
					OR = {
						has_war_with = scope:overlord_scope
						is_at_war_with_overlord = yes
					}
				}
			}
		}
		effect = {
			trigger_event = { id = india_home_rule_events.3 popup = yes}
			set_global_variable = india_home_rule_failed
		}
	    event_outcome_desc = {
	        first_valid = {
	            triggered_desc = {
	                desc = event_effects_india_home_rule_events.3_tt
	            }
	        }
	    }
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					exists = scope:overlord_scope
				}
				scope:india_scope ?= {
					NOT = {
						is_subject_of = scope:overlord_scope
					}
				}
			}
		}
		effect = {
			je_india_home_rule_cleanup = yes
		}
	}

	weight = 1000
}

je_india_nationalism_britain = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_british_india
	dlc = dlc011

	activation_description = {
		always = yes
	}

	is_shown_in_lobby = {
		has_dlc_feature = ip2_content #Pivot of Empire
		c:GBR ?= THIS
	}

	scripted_progress_bar = je_india_home_rule_progressbar

	scripted_button = je_india_nationalism_britain_indian_self_governance
	scripted_button = je_india_nationalism_britain_dispatch_simon_commission

	immediate = { # Activated by this popping up for India.
		c:BIC ?= {
			save_scope_as = india_scope
		}
		ROOT = {
			save_scope_as = overlord_scope
		}
		trigger_event = { id = india_home_rule_events.8 popup = yes }
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			trigger_if = {
				limit = {
					c:BIC ?= {
						has_journal_entry = je_india_home_rule
					}
				}
				custom_tooltip = {
					text = granted_india_home_rule_tt
					has_global_variable = india_home_rule
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = preserve_british_rule_forever_tt
					always = no
				}
			}
		}
		effect = {
			je_india_nationalism_britain_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			c:BIC ?= {
				has_revolution = yes
			}
		}
		effect = {
			je_india_nationalism_britain_cleanup = yes
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = simon_commission_timer_var
				}
				change_variable = { name = simon_commission_timer_var subtract = 1 }
				if = {
					limit = {
						var:simon_commission_timer_var = 0
					}
					trigger_event = { id = india_home_rule_events.10 popup = yes }
					remove_variable = simon_commission_timer_var
				}
			}
		}
		random_events = {
			400 = 0
			10 = india_nationalism_events.6
			10 = india_nationalism_events.7
		}
	}

	outcome = {
		flags = { hidden invalid_outcome }
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				AND = {
					NOT = { exists = scope:india_scope }
					NOT = { war_may_restore_country = BIC }
				}
				scope:india_scope ?= {
					NOT = {
						is_subject_of = ROOT
					}
				}
				NOT = {
					OR = {
						has_war_with = ROOT
						is_at_war_with_overlord = yes
					}
				}
			}
		}
		effect = {
			je_india_nationalism_britain_cleanup = yes
		}
	}

	weight = 1000
}

je_india_nationalism = {
		
	should_be_pinned_by_default_uninvolved_or_context = yes
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_british_india
	dlc = dlc011

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		has_dlc_feature = ip2_content
		c:BIC ?= this
		is_subject = yes
		has_technology_researched = pan-nationalism
	}

	possible = {
		any_political_movement = {
			is_political_movement_type = movement_india_pan_national
			political_movement_radicalism >= india_independence_radicalism_threshold
		}
		is_subject = yes
		custom_tooltip = {
			text = no_south_asian_primary_cultures_tt
			NOT = {
				any_primary_culture = {
					has_discrimination_trait_group = heritage_group_south_asian
				}
			}
		}
	}

	immediate = {
		top_overlord = {
			save_scope_as = overlord_scope
			if = {
				limit = {
					NOT = {
						has_journal_entry = je_india_nationalism_britain
					}
				}
				add_journal_entry = {
					type = je_india_nationalism_britain
				}
			}
		}
		ROOT = {
			save_scope_as = india_scope
		}
		ig:ig_industrialists ?= {
			save_scope_as = government_scope
		}
	}

	on_monthly_pulse = {
		random_events = {
			100 = 0
			5 = india_nationalism_events.3
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				custom_tooltip = {
					text = india_independence_movement_defeated_tt
					OR = {
						NOT = {
							any_political_movement = {
								is_political_movement_type = movement_india_pan_national
							}
						}
						any_political_movement = {
							is_political_movement_type = movement_india_pan_national
							political_movement_support <= 0.05
							political_movement_radicalism <= 0.1
						}
					}
				}
				custom_tooltip = {
					text = india_independent_tt
					is_independent_and_not_being_subjugated = yes
				}
			}
		}
		effect = {
			je_india_nationalism_cleanup = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			has_revolution = yes
		}
		effect = {
			custom_tooltip = britain_might_abandon_india_tt
			top_overlord = {
				trigger_event = { id = india_nationalism_events.1 popup = yes }
			}
		}
		event_outcome_desc = {
			first_valid = {
				triggered_desc = {
					desc = event_effects_india_nationalism_events.1_tt
				}
			}
		}
	}

	weight = 1000
}

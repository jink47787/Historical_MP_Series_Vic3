je_shinbutsu_bunri = {
	icon = "gfx/interface/icons/religion_icons/shinto.dds"

	group = je_group_historical_content
	dlc = dlc018

	activation_description = {
		always = yes
	}

	possible = {
		# triggered by shinbutsu_bunri_decision
	}

	scripted_button = je_taikyo_proclamation_button

	immediate = {
		every_scope_homeland_state = {
			convert_population = {
				target = rel:shinto
				value = 0.10
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			random_scope_state = {
				limit = {
					state_region = { is_homeland = cu:japanese }
					religion_percent_state = {  
						target = rel:shinto
						value <= 0.5
					}
				}
				convert_population = {
					target = rel:shinto
					value = 0.02
				}
			}
		}
		random_events = {
			100 = 0
			10 = japan_religion.1
			10 = japan_religion.2
			10 = japan_religion.3
			10 = japan_religion.4
			10 = japan_religion.5
			10 = japan_religion.6
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			# All starting Japanese Homelands
			# Trigger targets states individually for clearer tooltip
			s:STATE_KANTO.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_TOKAI.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_KYUSHU.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_HOKUSHINETSU.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_CHUGOKU.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_KANSAI.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_TOHOKU.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_SHIKOKU.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_KYOTO.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
			s:STATE_HOKKAIDO.region_state:JAP ?= {
				religion_percent_state = {
					target = rel:shinto
					value >= 0.6
				}
			}
		}
		effect = {
			trigger_event = {
				id = japan_religion.11
				popup = yes
			}
		}
		event_outcome_effect_desc = {
			effect = {
				add_modifier = {
					name = arahitogami
				}
				every_scope_homeland_state = {
					convert_population = {
						target = rel:shinto
						value = 0.1
					}
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				has_law_or_variant = law_type:law_total_separation
				has_law_or_variant = law_type:law_state_atheism
				AND = {
					NOT = { country_has_state_religion = rel:shinto }
					NOT = { has_law_or_variant = law_type:law_freedom_of_conscience }
				}
			}
		}
		effect = {
			je_shinbutsu_bunri_cleanup = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_shinbutsu_bunri_cleanup = yes
		}
	}

	weight = 5000

	timeout = 9125

	should_be_pinned_by_default_uninvolved_or_context = yes
}

je_elevate_buddhism = {
	icon = "gfx/interface/icons/religion_icons/mahayana.dds"

	group = je_group_historical_content
	dlc = dlc018

	activation_description = {
		always = yes
	}

	possible = {
		# triggered by elevate_buddhism_decision
	}

	immediate = {
		set_variable = {
			name = je_elevate_buddhism_var
			value = 0
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					ig:ig_devout = { is_powerful = yes }
				}
				change_variable = {
					name = je_elevate_buddhism_var
					add = 1
				}
			}
		}
		random_events = {
			100 = 0
			10 = japan_religion.7
			10 = japan_religion.8
			10 = japan_religion.9
			10 = japan_religion.6
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			ig:ig_devout = { is_powerful = yes }
		}
		effect = {
			trigger_event = {
				id = japan_religion.12
				popup = yes
			}
		}
		event_outcome_effect_desc = {
			effect = {
				add_modifier = {
					name = extended_danka_system
				}
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				ig:ig_devout = { is_marginal = yes }
				has_law_or_variant = law_type:law_total_separation
				has_law_or_variant = law_type:law_state_atheism
			}
		}
		effect = {
			je_elevate_buddhism_cleanup = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			je_elevate_buddhism_cleanup = yes
		}
	}

	current_value = {
		value = root.var:je_elevate_buddhism_var
	}

	goal_add_value = {
		add = 60
	}

	progressbar = yes

	weight = 5000

	timeout = 7300

	should_be_pinned_by_default_uninvolved_or_context = yes
}

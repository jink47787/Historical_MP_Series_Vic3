je_meiji_restoration_get_faction_sgui = {
    scope = state
    saved_scopes = { frame }

    is_shown = {
        owner ?= {
            has_journal_entry = je_meiji_restoration
        }
    }

    is_valid = {
        owner ?= {
            has_journal_entry = je_meiji_restoration
        }
        trigger_if = {
            limit = {
                scope:frame = 0 # Tokugawa
            }
            OR = {
                root.state_region = s:STATE_KANTO
                root.state_region = s:STATE_TOKAI
                root.state_region = s:STATE_KANSAI
            }
        }
        trigger_else_if = {
            limit = {
                scope:frame = 1 # Date
            }
            root.state_region = s:STATE_TOHOKU
        }
        trigger_else_if = {
            limit = {
                scope:frame = 2 # Hachisuka
            }
            root.state_region = s:STATE_SHIKOKU
        }
        trigger_else_if = {
            limit = {
                scope:frame = 3 # ii
            }
            root.state_region = s:STATE_KYOTO
        }
        trigger_else_if = {
            limit = {
                scope:frame = 4 # Maeda
            }
            root.state_region = s:STATE_HOKUSHINETSU
        }
        trigger_else_if = {
            limit = {
                scope:frame = 5 # Matsumae
            }
            root.state_region = s:STATE_HOKKAIDO
        }
        trigger_else_if = {
            limit = {
                scope:frame = 6 # Mori
            }
            root.state_region = s:STATE_CHUGOKU
        }
        trigger_else_if = {
            limit = {
                scope:frame = 7 # Shimazu
            }
            root.state_region = s:STATE_KYUSHU
        }
        trigger_else = {
            always = no
        }
    }
}

je_colonize_korea_states_sgui = {
    scope = country

    effect = {
        if = {
            limit = {
                has_variable_list = colonize_korea_states_states
            }
            every_in_list = {
                variable = colonize_korea_states_states

                if = {
                    limit = {
                        NAND = { # this is based on a section within complete in je_colonize_korea
                            c:JAP ?= owner
                            is_incorporated = yes
                            turmoil < je_colonize_korea_state_turmoil_max # set in event_values
                            culture_percent_state = {
                                target = cu:japanese
                                value >= je_colonize_korea_state_japanese_culture_min # set in event_values
                            }
                        }
                    }
                    # for each state print a line:
                    custom_tooltip_no_bullet = "JE_COLONIZE_KOREA_STATES_LIST_ENTRY_OWNED_STATE"
                    custom_tooltip_no_bullet = "JE_COLONIZE_KOREA_STATES_LIST_ENTRY_INCORPORATED_STATE"
                    custom_tooltip_no_bullet = "JE_COLONIZE_KOREA_STATES_LIST_ENTRY_TURMOIL"
                    custom_tooltip_no_bullet = "JE_COLONIZE_KOREA_STATES_LIST_ENTRY_CULTURE"
                }
                else = {
                    custom_tooltip_no_bullet = "JE_COLONIZE_KOREA_STATES_LIST_ENTRY_COMPLETE"
                }
            }
        }
    }
}

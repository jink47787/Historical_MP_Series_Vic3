	value = root.var:rus_gov_soc

russian_total_alignment_calc = {
	value = russian_conservative_monthly_change
	add = russian_liberal_monthly_change
	add = russian_socialist_monthly_change
}

russian_conservative_share = {
	value = 0
	if = {
		limit = {
			russian_total_alignment_calc > 0
		}
		add = russian_conservative_monthly_change
		divide = russian_total_alignment_calc
	}
}

russian_conservative_monthly_change = {
	value = 0
	if = {
		limit = {
			ruler ?= {
				OR = {
					has_reactionary_ideology = yes
					has_conservative_ideology = yes
					has_monarchist_ideology = yes
				}
			}
		}
		add = {
			desc = "russian_alignment_change_ruler"
			add = 1
		}
	}
	if = {
		limit = {
			any_political_movement = {
				OR = {
					movement_is_nationalist = yes
					movement_is_conservative = yes
				}
				political_movement_support >= 0.05
			}
		}
		add = {
			desc = "russian_alignment_change_movements"
			every_political_movement = {
				limit = {
				OR = {
					movement_is_nationalist = yes
					movement_is_conservative = yes
				}
					political_movement_support >= 0.05
				}
				add = this.political_movement_support
			}
		}
	}
	if = {
		limit = {
			any_interest_group = {
				OR = {
					ig_is_reactionary = yes
					interest_group_has_reactionary_ideology = yes
					interest_group_has_conservative_ideology = yes
				}
				is_marginal = no
			}
		}
		add = {
			desc = "russian_alignment_change_interest_groups"
			every_interest_group = {
				limit = {
				OR = {
					ig_is_reactionary = yes
					interest_group_has_reactionary_ideology = yes
					interest_group_has_conservative_ideology = yes
				}
					is_marginal = no
				}
				add = this.ig_clout
			}
		}
	}
	if = {
		limit = {
			russian_liberal_laws_value <= 0
			russian_socialist_laws_value <= 0
		}
		add = {
			desc = "russian_alignment_change_laws"
			add = 1
		}
	}
	if = {
		limit = {
			NOT = { has_technology_researched = socialism }
		}
		divide = 2
	}
	if = {
		limit = {
			NOT = { has_technology_researched = egalitarianism }
		}
		divide = 2
	}
}

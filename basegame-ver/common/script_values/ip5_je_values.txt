# Ceiling on a country's Innovation, scaling with literacy. Read by the Tsarpoints JE, its buttons and its scripted effects.
max_innovation = {
	value = 150
	multiply = literacy_rate
	add = 50
}

# The literacy rate a country is expected to have reached by the current year, ramping from 20% in 1836. Tsarpoints uses it to judge whether literacy is on track.
expected_literacy_rate_year = {
	value = year
	subtract = 1836
	multiply = 0.01
	min = 0.20
	max = 1
}

# Plain GDP per capita, guarded against division by zero on countries with no population.
# Base for five_year_plan_gdp_per_capita_value and read directly by 00_communism.txt.
country_gdp_per_capita = {
	value = gdp
	if = {
		limit = {
			total_population > 0
		}
		divide = total_population
	}
	else = {
		divide = 1
	}
}

# Socialism strength - shared by both journal entries #
#######################################################

# Country. How socialist a country is overall: 3x socialist movement support (saturating near 33% support) plus 2x socialist party clout (saturating near 50% clout).
# The root input of the whole RoR weighting system - feeds normalized_socialism_coefficient, the je_reform_or_revolution map colour, and the Council Republic Confrontation drift factor.
socialism_coefficient = {
	value = 0
	if = { # Max - 33% support
		limit = {
			any_political_movement = {
				is_political_movement_type = movement_socialist
			}
		}
		add = {
			desc = "socialism_coefficient_movement_support"
			every_political_movement = {
				limit = {
					is_political_movement_type = movement_socialist
				}
				value = {
					value = this.political_movement_support
					multiply = 3
				}
			}
		}
	}
	if = { # Max - 50% clout
		limit = {
			any_active_party = {
				party_counts_as_socialist_party = yes
				party_has_members = yes
			}
		}
		add = {
			desc = "socialism_coefficient_party_clout"
			every_active_party = {
				limit = {
					party_counts_as_socialist_party = yes
					party_has_members = yes
				}
				every_member = {
					add = this.ig_clout
				}
			}
			multiply = 2
		}
	}
}

# Party. Total clout of every interest group that is a member of this party.
# Read by the reform_or_revolution.1 event options and by loc; not part of the drift maths.
party_total_clout = { # Party
	value = 0
	every_member = {
		add = this.ig_clout
	}
}

# Country. This country's socialism_coefficient relative to the average across involved countries, centred on 1.
# Above 1 = more socialist than the involved average, below 1 = less. Reads the average_socialism_coefficient variable cached on the JE.
normalized_socialism_coefficient = {
	value = 1
	add = {
		value = this.socialism_coefficient
		subtract = je:je_reform_or_revolution.var:average_socialism_coefficient
	}
}

# Squared so that countries far from the average dominate the sum rather than merely leading it.
# This is the term that makes involved_country_percentage_of_total_involved_population_modified socialism-weighted.
normalized_socialism_coefficient_squared = {
	value = normalized_socialism_coefficient
	multiply = normalized_socialism_coefficient
}

# JE. Count of countries involved in je_reform_or_revolution.
# Cached monthly into the num_involved_countries_var JE variable, which everything else reads instead of recomputing this.
num_involved_countries = {
	value = 0
	every_scope_je_involved = {
		add = 1
	}
}

# JE. Summed incorporated population of every involved country - the denominator for all population weighting.
# Cached monthly into the total_incorporated_population_of_involved_countries_var JE variable.
total_incorporated_population_of_involved_countries = {
	value = 0
	if = {
		limit = {
			var:num_involved_countries_var > 0
		}
		every_scope_je_involved = {
			add = cached_ai_incorporated_population
		}
	}
}

# Country. What fraction of this country's own population is incorporated.
# Used by the per-country map contribution to scale factors that only apply to incorporated pops.
percent_incorporated_population = { # country scope
	value = cached_ai_incorporated_population
	divide = total_population
}

# Country. This country's share of the total involved population. The plain population weight.
# Used by the coverage-style drift factors (institutions, factory councils), which are not socialism-weighted.
involved_country_percentage_of_total_involved_population = {
	if = {
		limit = {
			je:je_reform_or_revolution.var:total_incorporated_population_of_involved_countries_var > 0
		}
		value = total_population
		divide = je:je_reform_or_revolution.var:total_incorporated_population_of_involved_countries_var
	}
}

# Country. The same population share, multiplied by normalized_socialism_coefficient_squared.
# The socialism-weighted population weight, used by the political drift factors (parties in government, movements, trade union stances, Council Republic Confrontation, scripted bonuses).
involved_country_percentage_of_total_involved_population_modified = {
	value = involved_country_percentage_of_total_involved_population
	multiply = normalized_socialism_coefficient_squared
}

# Country. One country's net push on the RoR bar: positive is Revolutionary, negative is Reformist.
# Every factor here mirrors a reform_or_revolution_*_var block in journal_entry_update_reform_or_revolution_noninit_variables - keep the two in step when adding a factor.
# Cached per country into reform_or_revolution_contribution_cache, which drives the map mode tooltip, the communist Pop Attraction modifier and the reform_or_revolution event options.
reform_or_revolution_map_country_contribution_value = { # country scope
	if = {
		limit = {
			NOT = { has_factory_councils_law_or_amendment = yes }
			any_interest_group = {
				is_in_government = yes
				party ?= {
					is_party_type = socialist_party
				}
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_SOCIALIST_PARTIES_IN_GOVERNMENT_TT"
			add = {
				value = -10
				multiply = involved_country_percentage_of_total_involved_population_modified
			}
		}
	}
	if = {
		limit = {
			has_law_or_variant = law_type:law_council_republic
			has_journal_entry = je_state_and_revolution
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_COUNCIL_REPUBLIC_CONFRONTATION_TT"
			add = {
				value = state_and_revolution_average_confrontation # 0-100
				multiply = {
					value = 0.1
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
		}
	}
	if = {
		limit = {
			any_political_movement = {
				OR = {
					is_political_movement_type = movement_socialist
					is_political_movement_type = movement_labor
				}
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_INSURRECTIONARY_LABOR_OR_SOCIALIST_MOVEMENTS_TT"
			if = {
				limit = {
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_socialist
							is_political_movement_type = movement_labor
						}
						is_insurrectionary = yes
					}
				}
				add = {
					value = 10
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_ACTIVISM_OF_LABOR_OR_SOCIALIST_MOVEMENTS_TT"
			if = {
				limit = {
					NOT = {
						has_law_or_variant = law_type:law_council_republic
					}
					any_political_movement = {
						OR = {
							is_political_movement_type = movement_socialist
							is_political_movement_type = movement_labor
						}
						this.political_movement_support >= 0.05
					}
				}
				every_political_movement = {
					limit = {
						OR = {
							is_political_movement_type = movement_socialist
							is_political_movement_type = movement_labor
						}
						this.political_movement_support >= 0.05
					}
					add = {
						value = -0.25
						add = this.political_movement_radicalism
						multiply = 10
					}
				}
				multiply = involved_country_percentage_of_total_involved_population_modified
			}
		}
	}
	if = {
		limit = {
			has_factory_councils_law_or_amendment = yes
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_FACTORY_COUNCILS_TT"
			add = {
				value = percent_incorporated_population
				multiply = involved_country_percentage_of_total_involved_population
			}
			multiply = 10
		}
	}
	if = {
		limit = {
			NOT = { has_factory_councils_law_or_amendment = yes }
			has_institution = institution_workplace_safety
		}
		subtract = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_WORKPLACE_SAFETY_TT"
			add = {
				value = percent_incorporated_population
				multiply = involved_country_percentage_of_total_involved_population
			}
		}
	}
	if = {
		limit = {
			NOR = {
				has_factory_councils_law_or_amendment = yes
				has_law_or_variant = law_type:law_poor_laws
			}
			has_institution = institution_social_security
		}
		subtract = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_SOCIAL_SECURITY_TT"
			add = {
				value = percent_incorporated_population
				multiply = involved_country_percentage_of_total_involved_population
			}
		}
	}
	if = {
		limit = {
			NOT = { has_factory_councils_law_or_amendment = yes }
			has_law = law_type:law_public_health_insurance
			has_institution = institution_health_system
		}
		subtract = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_HEALTH_TT"
			add = {
				value = percent_incorporated_population
				multiply = involved_country_percentage_of_total_involved_population
			}
		}
	}
	if = {
		limit = {
			ig:ig_trade_unions ?= {
				law_stance = {
					law = law_type:law_factory_councils
					value > neutral
				}
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_TRADE_UNIONS_FACTORY_COUNCIL_STANCE_PRO_TT"
			add = {
				value = ig:ig_trade_unions.ig_clout
				multiply = {
					value = 10
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
		}
	}
	if = {
		limit = {
			ig:ig_trade_unions ?= {
				law_stance = {
					law = law_type:law_factory_councils
					value < neutral
				}
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_TRADE_UNIONS_FACTORY_COUNCIL_STANCE_ANTI_TT"
			subtract = {
				value = ig:ig_trade_unions.ig_clout
				multiply = {
					value = 10
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
		}
	}
	if = {
		limit = {
			OR = {
				this.modifier:country_reform_or_revolution_revolution_bonus_add != 0
				this.modifier:country_reform_or_revolution_reform_bonus_add != 0
			}
		}
		add = {
			desc = "REFORM_OR_REVOLUTION_COUNTRY_BONUS_VALUES_TT"
			if = {
				limit = {
					this.modifier:country_reform_or_revolution_revolution_bonus_add != 0
				}
				add = {
					value = this.modifier:country_reform_or_revolution_revolution_bonus_add
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
			if = {
				limit = {
					this.modifier:country_reform_or_revolution_reform_bonus_add != 0
				}
				subtract = {
					value = this.modifier:country_reform_or_revolution_reform_bonus_add
					multiply = involved_country_percentage_of_total_involved_population_modified
				}
			}
		}
	}
}

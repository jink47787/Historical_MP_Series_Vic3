favorite_appointment_controversy = {
	value = 0
	if = {
		limit = {
			exists = owner.ruler
		}
		value = average_progressiveness
		subtract = owner.ruler.average_progressiveness
		abs = yes
		if = {
			limit = {
				exists = interest_group
				exists = owner.ruler.interest_group
			}
			multiply = 0.5
			add = {
				value = interest_group.average_progressiveness
				subtract = owner.ruler.interest_group.average_progressiveness
				abs = yes
				multiply = 0.5
			}
		}
		subtract = 15
		min = 0
		divide = 10
		multiply = {
			value = 150
			subtract = prominence
			divide = 100
		}
		multiply = {
			value = 100
			subtract = owner.ruler.popularity
			divide = 100
		}
		min = 0
		max = 10
	}
}

favorite_appointment_ig_penalty = {
	value = 0
	if = {
		limit = {
			is_in_government = yes
			NOT = { root.interest_group ?= this }
			root.favorite_appointment_controversy > 0
		}
		value = root.favorite_appointment_controversy
		multiply = 0.5
		min = 1
	}
	else_if = {
		limit = {
			NOT = { root.interest_group ?= this }
			root.favorite_appointment_controversy > 0
			exists = root.owner.ruler
			0 < {
				add = {
					value = average_progressiveness
					subtract = root.average_progressiveness
					abs = yes
				}
				subtract = {
					value = average_progressiveness
					subtract = root.owner.ruler.average_progressiveness
					abs = yes
				}
			}
		}
		value = root.favorite_appointment_controversy
		multiply = 0.25
		min = 1
	}
}

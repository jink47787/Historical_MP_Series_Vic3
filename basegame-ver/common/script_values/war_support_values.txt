# root = country
# scope:war = the war we're evaluating war support change for
# No randomness is allowed in this script value

military_size = {
	add = army_size_including_conscripts
	add = navy_size
}

war_support_from_casualties = {
	add = "scope:war.num_country_casualties(root)"
	if = {
		limit = { military_size > 0 }
		divide = military_size
	}
	divide = 1000
	multiply = define:NWar|WAR_SUPPORT_CHANGE_KIA_FACTOR
}

war_support_from_battles = {
	add = "size_weighted_won_battles_fraction(scope:war)"
	subtract = "size_weighted_lost_battles_fraction(scope:war)"
	multiply = define:NWar|WAR_SUPPORT_CHANGE_BATTLES_FACTOR
	multiply = {
		value = "scope:war.num_significant_battles"
		divide = define:NWar|WAR_SUPPORT_BATTLES_RAMP_THRESHOLD
		max = 1
	}
}

# The mirror of the duration drain: a new war carries its own enthusiasm, decaying to nothing as the war stops being new

	min = 0

war_support_from_gdp = {
	value = "scope:war.gdp_change_since_war_start(root)"
	save_temporary_value_as = temp_gdp_change

	value = 0
	if = {
		limit = { scope:temp_gdp_change > 0 }
		add = {
			value = scope:temp_gdp_change
			max = define:NWar|WAR_SUPPORT_CHANGE_GDP_GROWTH_MAX_AT
			subtract = define:NWar|WAR_SUPPORT_CHANGE_GDP_TOLERANCE
			min = 0
			divide = {
				value = define:NWar|WAR_SUPPORT_CHANGE_GDP_GROWTH_MAX_AT
				subtract = define:NWar|WAR_SUPPORT_CHANGE_GDP_TOLERANCE
			}
			multiply = define:NWar|WAR_SUPPORT_CHANGE_GDP_GROWTH_FACTOR
		}
	}
	else = {
		add = {
			# The collapse side needs no cap of its own: GDP cannot fall by more than everything the country had
			value = scope:temp_gdp_change
			multiply = -1
			subtract = define:NWar|WAR_SUPPORT_CHANGE_GDP_TOLERANCE
			min = 0
			divide = {
				value = 1
				subtract = define:NWar|WAR_SUPPORT_CHANGE_GDP_TOLERANCE
			}
			multiply = define:NWar|WAR_SUPPORT_CHANGE_GDP_DECLINE_FACTOR
		}
	}
}

# Only says anything when the subject is actually fighting this war alongside its overlord: not when the overlord stayed out of it, and not when the subject is fighting the overlord itself
war_support_from_subject_attitude = {
	if = {
		limit = {
			is_ai = yes
			is_subject = yes
			scope:war = {
				is_war_participant = root
				is_war_participant = root.overlord
				NOT = { is_fighting_overlord_in_war = root }
			}
		}
		if = {
			limit = { attitude_towards_overlord = rebellious }
			add = define:NWar|WAR_SUPPORT_CHANGE_REBELLIOUS_SUBJECT_EFFECT
		}
		else_if = {
			limit = { attitude_towards_overlord = defiant }
			add = define:NWar|WAR_SUPPORT_CHANGE_DEFIANT_SUBJECT_EFFECT
		}
		else_if = {
			limit = { attitude_towards_overlord = aloof }
			add = define:NWar|WAR_SUPPORT_CHANGE_ALOOF_SUBJECT_EFFECT
		}
		else_if = {
			limit = { attitude_towards_overlord = loyal }
			add = define:NWar|WAR_SUPPORT_CHANGE_LOYAL_SUBJECT_EFFECT
		}
	}
}

# Wars where a goal cannot actually be taken have nothing pulling the two sides together, so a stalled demand converges war support on the neutral point.
# Split in two so the breakdown can say which half is acting; they are mutually exclusive, one applying only below the neutral point and the other only above it.
# Both are time-gated exactly like war goal progress.
war_support_from_stalled_goal_against_us = {
	if = {
		limit = {
			scope:war_support < define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
			scope:war = { has_stalled_wargoal_against = root }
		}
		value = define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
		subtract = scope:war_support
		divide = define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
		min = 0
		max = 1
		if = {
			limit = { scope:war = { has_stalled_state_wargoal_against = root } }
			multiply = define:NWar|WAR_SUPPORT_CHANGE_STALLED_STATE_GOAL_CONVERGENCE
		}
		else = {
			multiply = define:NWar|WAR_SUPPORT_CHANGE_STALLED_GOAL_CONVERGENCE
		}
		multiply = "scope:war.war_goal_time_ramp"
	}
}

war_support_from_stalled_goal_held_by_us = {
	if = {
		limit = {
			scope:war_support > define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
			scope:war = { has_stalled_wargoal_held_by = root }
		}
		value = scope:war_support
		subtract = define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
		divide = define:NDiplomacy|WAR_SUPPORT_DRIFT_TARGET
		min = 0
		max = 1
		multiply = define:NWar|WAR_SUPPORT_CHANGE_STALLED_GOAL_CONVERGENCE
		multiply = "scope:war.war_goal_time_ramp"
		multiply = -1
	}
}

war_support_from_journal_entries = {
	if = {
		limit = {
			has_journal_entry = je_the_first_carlist_war
			exists = c:SPA
			exists = c:SPC
			scope:war = {
				is_war_participant = c:SPA
				is_war_participant = c:SPC
			}
			c:SPA = { is_diplomatic_play_enemy_of = c:SPC }
		}
		add = {
			value = 10.0
			desc = "je_the_first_carlist_war"
		}
	}
	if = {
		limit = {
			has_variable = rcw_russian_faction
			has_global_variable = rcw_exhaustion_level
			global_var:rcw_exhaustion_level >= 1
			scope:war.diplomatic_play = {
				OR = {
					is_diplomatic_play_type = dp_unify_russia
					is_diplomatic_play_type = dp_revolution
				}
			}
		}
		add = {
			value = enemy_occupation
			if = {
				limit = { global_var:rcw_exhaustion_level >= 4 }
				multiply = -2.0
			}
			else_if = {
				limit = { global_var:rcw_exhaustion_level >= 3 }
				multiply = -1.0
			}
			else_if = {
				limit = { global_var:rcw_exhaustion_level >= 2 }
				multiply = -0.5
			}
			else = {
				multiply = -0.25
			}
			desc = "WAR_SUPPORT_CHANGE_FROM_RCW_EXHAUSTION"
		}
	}
}

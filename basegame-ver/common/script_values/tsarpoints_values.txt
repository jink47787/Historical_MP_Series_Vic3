tp_bootlicker_ruler_law_preference = {
	value = 0
	if = {
		limit = {
			owner.ruler ?= {
				law_stance = {
					law = prev.type
					value >= strongly_approve
				}
			}
		}
		add = 2
	}
	else_if = {
		limit = {
			owner.ruler ?= {
				law_stance = {
					law = prev.type
					value >= approve
				}
			}
		}
		add = 1
	}
}

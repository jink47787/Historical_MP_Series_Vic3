# Monthly AI helper maps for decree targeting (root = country).
# Keys are states, values are the state's rank (0 = most levels) in the matching building group; only the top 5 (ranks 0-4) are stored.

ai_assemble_decree_state_maps = {
	if = {
		limit = {
			has_variable_map = industrial_states_map
		}
		clear_variable_map = industrial_states_map
	}
	if = {
		limit = {
			has_variable_map = agricultural_states_map
		}
		clear_variable_map = agricultural_states_map
	}
	if = {
		limit = {
			has_variable_map = resource_states_map
		}
		clear_variable_map = resource_states_map
	}
	if = {
		limit = {
			has_variable_map = army_states_map
		}
		clear_variable_map = army_states_map
	}
	set_local_variable = {
		name = ai_decree_map_rank
		value = 0
	}
	ordered_scope_state = {
		limit = {
			"state_has_building_group_levels(bg_manufacturing)" >= 10
			any_scope_building = {
				is_building_group = bg_manufacturing
			}
		}
		order_by = "state_has_building_group_levels(bg_manufacturing)"
		min = 0
		max = 5
		check_range_bounds = no
		ROOT = {
			add_to_variable_map = {
				name = industrial_states_map
				key = prev
				value = local_var:ai_decree_map_rank
			}
			change_local_variable = {
				name = ai_decree_map_rank
				add = 1
			}
		}
	}
	set_local_variable = {
		name = ai_decree_map_rank
		value = 0
	}
	ordered_scope_state = {
		limit = {
			ai_decree_agricultural_levels_script_value >= 10
			any_scope_building = {
				OR = {
					is_building_group = bg_agriculture
					is_building_group = bg_ranching
					is_building_group = bg_plantations
				}
			}
		}
		order_by = ai_decree_agricultural_levels_script_value
		min = 0
		max = 5
		check_range_bounds = no
		ROOT = {
			add_to_variable_map = {
				name = agricultural_states_map
				key = prev
				value = local_var:ai_decree_map_rank
			}
			change_local_variable = {
				name = ai_decree_map_rank
				add = 1
			}
		}
	}
	set_local_variable = {
		name = ai_decree_map_rank
		value = 0
	}
	ordered_scope_state = {
		limit = {
			"state_has_building_group_levels(bg_extraction)" >= 10
			any_scope_building = {
				is_building_group = bg_extraction
			}
		}
		order_by = "state_has_building_group_levels(bg_extraction)"
		min = 0
		max = 5
		check_range_bounds = no
		ROOT = {
			add_to_variable_map = {
				name = resource_states_map
				key = prev
				value = local_var:ai_decree_map_rank
			}
			change_local_variable = {
				name = ai_decree_map_rank
				add = 1
			}
		}
	}
	set_local_variable = {
		name = ai_decree_map_rank
		value = 0
	}
	ordered_scope_state = {
		limit = {
			"state_has_building_group_levels(bg_army)" >= 10
			any_scope_building = {
				is_building_group = bg_army
			}
		}
		order_by = "state_has_building_group_levels(bg_army)"
		min = 0
		max = 5
		check_range_bounds = no
		ROOT = {
			add_to_variable_map = {
				name = army_states_map
				key = prev
				value = local_var:ai_decree_map_rank
			}
			change_local_variable = {
				name = ai_decree_map_rank
				add = 1
			}
		}
	}
}

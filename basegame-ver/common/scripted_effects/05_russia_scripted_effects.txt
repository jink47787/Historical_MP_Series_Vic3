russian_separatist_country_setup = {
	# Poland
	if = {
		limit = {
			any_primary_culture = { cu:polish = this }
			NOT = { exists = c:POL }
		}
		change_tag = POL

		russian_separatist_claim_homeland_states = { CULTURE = cu:polish TAG = c:POL }

		# MORE
		russian_separatist_claim_homeland_states = { CULTURE = cu:polish TAG = c:POL }

		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_GREATER_POLAND

				}
			}
			set_capital = STATE_GREATER_POLAND
		}

		limit = {
			any_primary_culture = { cu:ukrainian = this }
			NOT = { exists = c:UKR }
		}

		russian_separatist_claim_homeland_states = { CULTURE = cu:ukrainian TAG = c:UKR }
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_KIEV
					is_capital = no
				}
			}
			set_capital = STATE_KIEV
		}
		if = {
			limit = {
				any_scope_state = {
					state_region = s:STATE_TAURIDA
				}
				s:STATE_CRIMEA.region_state:RUS ?= {
					NOT = { has_variable = rcw_reserved_black }
				}
			}
			s:STATE_CRIMEA.region_state:RUS ?= {
				set_state_owner = c:UKR
			}
		}
	}

	# Belarus
	else_if = {
		limit = {
			any_primary_culture = { cu:byelorussian = this }
			NOT = { exists = c:BYE }
		}
		change_tag = BYE

		russian_separatist_claim_homeland_states = { CULTURE = cu:byelorussian TAG = c:BYE }
	}

	# Finland
	else_if = {
		limit = {
			any_primary_culture = { cu:finnish = this }
			NOT = { exists = c:FIN }
		}
		change_tag = FIN

		russian_separatist_claim_homeland_states = { CULTURE = cu:finnish TAG = c:FIN }
	}

	# Chechnya
	else_if = {
		limit = {
			any_primary_culture = { cu:chechen = this }
			NOT = { exists = c:CHC }
		}

		russian_separatist_claim_homeland_states = { CULTURE = cu:chechen TAG = c:CHC }
	}

	# Kalmykia
	else_if = {
		limit = {
			any_primary_culture = { cu:kalmyk = this }
			NOT = { exists = c:KLM }
		}
		change_tag = KLM

		russian_separatist_claim_homeland_states = { CULTURE = cu:kalmyk TAG = c:KLM }
	}

	# Bukhara
	else_if = {
		limit = {
			capital = {
				state_region = s:STATE_UZBEKIA
			}
			NOT = { exists = c:BUK }
		}
		change_tag = BUK

		if = {
			limit = {
				NOT = { country_has_primary_culture = cu:uzbek }
			}
			add_primary_culture = cu:uzbek
		}

		every_neighbouring_state = {
			limit = {
				owner ?= { has_variable = rcw_partition_owner }
				is_capital = no
				NOT = { has_variable = rcw_reserved_black }
				state_region = {
					is_homeland = cu:uzbek
					NOT = { s:STATE_KHIVA = this }
				}
			}
			set_state_owner = c:BUK
		}
	}

	# Khiva

russian_separatist_claim_homeland_states = {
	every_neighbouring_state = {
		limit = {
			owner ?= { has_variable = rcw_partition_owner }
			is_capital = no
			NOT = { has_variable = rcw_reserved_black }
			state_region = { is_homeland = $CULTURE$ }
		}
		if = {
			limit = {
				state_region = { is_homeland = cu:russian }
			}
			random = {
				chance = {
					value = 20
					if = {
						limit = {
							culture_percent_state = {
								target = $CULTURE$
								value >= 0.25
							}
						}
						add = 30
					}
					if = {
						limit = {
							culture_percent_state = {
								target = $CULTURE$
								value >= 0.5
							}
						}
						add = 30
					}
				}
				set_state_owner = $TAG$
			}
		}
		else = {
			set_state_owner = $TAG$
		}
	}
}

# Riga Vicariate - monthly conversion of Estonian and Latvian peasants to Orthodoxy
baltic_riga_vicariate_conversion_effect = {
	every_scope_state = {
		limit = {
			is_incorporated = yes
		}
		every_scope_pop = {
			limit = {
				OR = {
					has_pop_culture = estonian
					has_pop_culture = latvian
				}
				strata = lower
				NOT = { has_pop_religion = orthodox }
			}
			change_pop_religion = {
				target = rel:orthodox
				value = state.modifier:state_orthodox_lower_strata_conversion_rate_add
			}
		}
	}
}

congress_poland_ai_refresh_russia_stance_effect = {

		limit = { has_variable = ai_russia_poland_bar_last_var }
		set_variable = {
			name = ai_russia_poland_bar_delta_var
			value = {
				value = congress_poland_bar_value
				subtract = var:ai_russia_poland_bar_last_var
			}
		}
	}
	set_variable = {
		name = ai_russia_poland_bar_last_var
		value = congress_poland_bar_value
	}
	set_variable = {
		name = ai_russia_poland_threat_var
		value = ai_russia_poland_threat_score
	}
	set_variable = {
		name = ai_russia_poland_threshold_var
		value = ai_russia_poland_appeasement_threshold
	}
}

socialist_faction_pick_leader_effect = {
	if = {
		limit = {
			any_scope_character = {
				$LEADER_TRIGGER$ = yes
			}
		}
		ordered_scope_character = {
			limit = {
				$LEADER_TRIGGER$ = yes
			}
			order_by = prominence
			position = 0
			save_scope_as = socialist_faction_candidate
		}
		if = {
			limit = {
				NOT = {
					var:$LEADER_VAR$ ?= {
						is_character_alive = yes
						owner = ROOT
						$LEADER_TRIGGER$ = yes
						prominence >= {
							value = scope:socialist_faction_candidate.prominence
							subtract = define:NCharacters|PROMINENCE_NOTEWORTHY_THRESHOLD
						}
					}
				}
			}
			set_variable = {
				name = $LEADER_VAR$
				value = scope:socialist_faction_candidate
			}
		}
	}
	else = {
		remove_variable = $LEADER_VAR$
	}
}

create_russian_national_state_for_culture = {
	save_scope_as = national_state_origin
	if = {
		limit = {
			exists = scope:independent_national_state
		}
		clear_saved_scope = independent_national_state
	}
	if = {
		limit = {
			exists = scope:independent_national_state_culture
		}
		clear_saved_scope = independent_national_state_culture
	}
	every_scope_state = {
		limit = {
			has_variable = independent_national_state_cede
		}
		remove_variable = independent_national_state_cede
	}
	$CULTURE$ = {
		save_scope_as = independent_national_state_culture
		ordered_scope_homeland_state = {
			limit = {
				owner = scope:national_state_origin
				is_capital = no
				national_state_culture_is_largest_homeland_culture = yes
			}
			order_by = gdp
			position = 0
			check_range_bounds = no
			save_scope_as = independent_national_state
		}
	}
	if = {
		limit = {
			exists = scope:independent_national_state
		}
		scope:independent_national_state_culture = {
			every_scope_homeland_state = {
				limit = {
					owner = scope:national_state_origin
					is_capital = no
					OR = {
						is_in_same_market_area = scope:independent_national_state
						AND = {
							scope:independent_national_state_culture = cu:ukrainian
							rcw_ukraine_renaissance_elevated = yes
						}
					}
					national_state_culture_is_largest_homeland_culture = yes
				}
				set_variable = independent_national_state_cede
			}
		}
		create_independent_national_state_in_russia = yes
	}
}

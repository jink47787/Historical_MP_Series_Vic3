je_tsarpoints_add_partial = {
	hidden_effect = {
		scope:journal_entry ?= {
			set_variable = tp_je_partial_completed_var
			set_variable = {
				name = tp_partial_progress_var
				value = 0
			}
			while = {
				limit = {
					scripted_bar_progress = {
						name = $PROGRESS_BAR$
						value > var:tp_partial_progress_var
					}
				}
				change_variable = {
					name = tp_partial_progress_var
					add = 1
				}
			}
			set_variable = {
				name = tp_partial_progress_percent_var
				value = {
					value = var:tp_partial_progress_var
					divide = $PROGRESS_BAR_MAX$
					max = 1
				}
			}
			set_variable = {
				name = tp_partial_reward_var
				value = {
					value = $REWARD$
					multiply = var:tp_partial_progress_percent_var
					round = yes
				}
			}
		}
	}
	custom_tooltip = {
		text = je_tsarpoints_add_partial_tt
		change_variable = {
			name = tsarpoints_var
			add = scope:journal_entry.var:tp_partial_reward_var
		}
		set_variable = {
			name = tp_last_mission_reward
			value = scope:journal_entry.var:tp_partial_reward_var
		}
	}
}

je_tsarpoints_upgrade_favourite = {
	if = {
		limit = {
			has_trait = expert_political_operator
		}
		random_list = {
			10 = {
				trigger = {
					NOT = {
						has_trait = ambitious
					}
				}
				add_trait = ambitious
			}
			10 = {
				trigger = {
					NOT = {
						has_trait = persistent
					}
				}
				add_trait = persistent
			}
			10 = {
				trigger = {
					NOT = {
						has_trait = charismatic
					}
				}
				add_trait = charismatic
			}
		}
	}
	else_if = {
		limit = {
			has_trait = experienced_political_operator
		}
		remove_trait = experienced_political_operator
		add_trait = expert_political_operator
	}
	else_if = {
		limit = {
			has_trait = basic_political_operator
		}
		remove_trait = basic_political_operator
		add_trait = experienced_political_operator
	}
	else = {
		add_trait = basic_political_operator
	}
}

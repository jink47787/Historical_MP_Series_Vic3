designate_russian_revolution_leader = {
	if = {
		limit = {
			NOT = {
				AND = {
					has_variable = russian_revolution_leader
					var:russian_revolution_leader ?= { is_character_alive = yes }
				}
			}
		}
		ordered_scope_character = {
			limit = {
				character_is_valid_for_events = yes
				NOR = {
					is_ruler_of_own_country = yes
					is_heir_of_own_country = yes
				}
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
				}
				trigger_if = {
					limit = {
						owner = { country_opposition_is_liberal = yes }
					}
					OR = {
						has_liberal_ideology = yes
						has_progressive_ideology = yes
					}
				}
				trigger_else_if = {
					limit = {
						owner = { country_opposition_is_socialist = yes }
					}
					has_socialist_ideology = yes
				}
				trigger_else_if = {
					limit = {
						owner = { country_opposition_is_conservative = yes }
					}
					OR = {
						has_reactionary_ideology = yes
						has_conservative_ideology = yes
						has_monarchist_ideology = yes
					}
				}
				trigger_else_if = {
					limit = {
						owner = { country_opposition_is_liberal_socalist = yes }
					}
					OR = {
						has_liberal_ideology = yes
						has_progressive_ideology = yes
						has_socialist_ideology = yes
					}
				}
				trigger_else_if = {
					limit = {
						owner = { country_opposition_is_liberal_conservative = yes }
					}
					OR = {
						has_liberal_ideology = yes
						has_progressive_ideology = yes
						has_reactionary_ideology = yes
						has_conservative_ideology = yes
						has_monarchist_ideology = yes
					}
				}
				trigger_else_if = {
					limit = {
						owner = { country_opposition_is_conservative_socialist = yes }
					}
					OR = {
						has_reactionary_ideology = yes
						has_conservative_ideology = yes
						has_monarchist_ideology = yes
						has_socialist_ideology = yes
					}
				}
			}
			order_by = prominence
			position = 0
			ROOT = {
				set_variable = {
					name = russian_revolution_leader
					value = prev
				}
			}
		}
	}
}

circassian_expulsion_pick_destination_effect = { # for je_circassian_expulsions and caucasus_expulsions.1, sets the destination country and state
	random_country = {
		limit = {
			any_scope_state = {
				OR = {
					state_region = s:STATE_EASTERN_THRACE
					state_region = s:STATE_KASTAMONU
					state_region = s:STATE_TRABZON
				}
				NOT = { owner = ROOT }
			}
		}
		save_scope_as = expulsion_destination_country
		random_scope_state = {
			limit = {
				OR = {
					state_region = s:STATE_EASTERN_THRACE
					state_region = s:STATE_KASTAMONU
					state_region = s:STATE_TRABZON
				}
				NOT = { owner = ROOT }
			}
			save_scope_as = expulsion_destination_state
		}
	}
}

nihilist_leaders_successor_effect = { # for dissolving nihilism
	if = {
		limit = {
			nihilist_successor_socialist = yes
		}
		every_scope_character = {
			limit = {
				has_ideology = ideology:ideology_nihilist
			}
			set_ideology = ideology:ideology_communist_leader
		}
	}
	else_if = {
		limit = {
			nihilist_successor_anarchist = yes
		}
		every_scope_character = {
			limit = {
				has_ideology = ideology:ideology_nihilist
			}
			set_ideology = ideology:ideology_anarchist
		}
	}
	else = {
		every_scope_character = {
			limit = {
				has_ideology = ideology:ideology_nihilist
			}
			set_ideology = ideology:ideology_radical
		}
	}
}

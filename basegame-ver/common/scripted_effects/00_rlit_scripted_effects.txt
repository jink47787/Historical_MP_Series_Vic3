rlit_classify_cluster1_route = {
	if = {
		limit = {
			rlit_route_official_leads = yes
		}
		set_variable = {
			name = rlit_cluster1_official
			value = yes
		}
	}
	else_if = {
		limit = {
			rlit_route_radical_leads = yes
		}
		set_variable = {
			name = rlit_cluster1_radical
			value = yes
		}
	}
	else_if = {
		limit = {
			rlit_route_spiritual_leads = yes
		}
		set_variable = {
			name = rlit_cluster1_spiritual
			value = yes
		}
	}
	else = {
		set_variable = {
			name = rlit_cluster1_universal
			value = yes
		}
	}
}

rlit_drop_prestige_tier_effect = {
	if = {
		limit = {
			has_modifier = $TIER$
		}
		remove_modifier = $TIER$
	}
}

rlit_take_prestige_tier_effect = {
	if = {
		limit = {
			NOT = {
				has_modifier = $TIER$
			}
		}
		add_modifier = {
			name = $TIER$
		}
	}
}

rlit_update_prestige_tiers = {
	if = {
		limit = {
			has_variable = rlit_prestige_score
			var:rlit_prestige_score >= 180
		}
		rlit_take_prestige_tier_effect = { TIER = rlit_prestige_tier_4 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_3 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_2 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_1 }
	}
	else_if = {
		limit = {
			has_variable = rlit_prestige_score
			var:rlit_prestige_score >= 100
		}
		rlit_take_prestige_tier_effect = { TIER = rlit_prestige_tier_3 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_4 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_2 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_1 }
	}
	else_if = {
		limit = {
			has_variable = rlit_prestige_score
			var:rlit_prestige_score >= 50
		}
		rlit_take_prestige_tier_effect = { TIER = rlit_prestige_tier_2 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_4 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_3 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_1 }
	}
	else_if = {
		limit = {
			has_variable = rlit_prestige_score
			var:rlit_prestige_score >= 20
		}
		rlit_take_prestige_tier_effect = { TIER = rlit_prestige_tier_1 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_4 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_3 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_2 }
	}
	else = {
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_4 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_3 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_2 }
		rlit_drop_prestige_tier_effect = { TIER = rlit_prestige_tier_1 }
	}
}

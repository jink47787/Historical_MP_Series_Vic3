rcw_snapshot_effect = {
	set_global_variable = {
		name = rcw_labor_red_ratio
		value = rcw_labor_red_ratio_value
	}
	set_global_variable = {
		name = rcw_weight_red
		value = rcw_camp_weight_red
	}
	set_global_variable = {
		name = rcw_weight_black
		value = rcw_camp_weight_black
	}
	set_global_variable = {
		name = rcw_weight_green
		value = rcw_camp_weight_green
	}
	set_global_variable = {
		name = rcw_weight_democratic
		value = rcw_camp_weight_democratic
	}
	set_global_variable = {
		name = rcw_weight_white
		value = rcw_camp_weight_white
	}
	set_global_variable = {
		name = rcw_weight_total
		value = {
			value = global_var:rcw_weight_red
			add = global_var:rcw_weight_black
			add = global_var:rcw_weight_green
			add = global_var:rcw_weight_democratic
			add = global_var:rcw_weight_white
		}
	}
	set_global_variable = {
		name = rcw_share_red
		value = {
			value = global_var:rcw_weight_red
			divide = global_var:rcw_weight_total
		}
	}
	set_global_variable = {
		name = rcw_share_black
		value = {
			value = global_var:rcw_weight_black
			divide = global_var:rcw_weight_total
		}
	}
	set_global_variable = {
		name = rcw_share_green
		value = {
			value = global_var:rcw_weight_green
			divide = global_var:rcw_weight_total
		}
	}
	set_global_variable = {
		name = rcw_share_democratic
		value = {
			value = global_var:rcw_weight_democratic
			divide = global_var:rcw_weight_total
		}
	}
	set_global_variable = {
		name = rcw_share_white
		value = {
			value = global_var:rcw_weight_white
			divide = global_var:rcw_weight_total
		}
	}
	if = {
		limit = {
			any_political_movement = {
				OR = {
					is_political_movement_type = movement_royalist_absolutist
					is_political_movement_type = movement_royalist_constitutional
				}
				has_core_ideology = ideology:ideology_tsarist_movement
				political_movement_support >= rcw_monarchist_movement_support
			}
		}
		set_global_variable = rcw_monarchist_pending
	}
	else_if = {
		limit = {
			has_global_variable = rcw_monarchist_pending
		}
		remove_global_variable = rcw_monarchist_pending
	}
}

rcw_set_camp = {
	if = {
		limit = {
			NOT = { var:rcw_camp ?= flag:rcw_$CAMP$ }
		}
		set_variable = {
			name = rcw_camp
			value = flag:rcw_$CAMP$
		}
		rcw_pick_shade = yes
	}
	set_variable = {
		name = rcw_axis
		value = $AXIS$
	}
}

rcw_pick_shade = {
	save_temporary_scope_as = rcw_shade_picker
	if = {
		limit = {
			NOT = { rcw_shade_taken_in_camp = { SHADE = 1 } }
		}
		set_variable = {
			name = rcw_shade
			value = 1
		}
	}
	else_if = {
		limit = {
			NOT = { rcw_shade_taken_in_camp = { SHADE = 2 } }
		}
		set_variable = {
			name = rcw_shade
			value = 2
		}
	}
	else_if = {
		limit = {
			NOT = { rcw_shade_taken_in_camp = { SHADE = 3 } }
		}
		set_variable = {
			name = rcw_shade
			value = 3
		}
	}
	else_if = {
		limit = {
			NOT = { rcw_shade_taken_in_camp = { SHADE = 4 } }
		}
		set_variable = {
			name = rcw_shade
			value = 4
		}
	}
	else_if = {
		limit = {
			NOT = { rcw_shade_taken_in_camp = { SHADE = 5 } }
		}
		set_variable = {
			name = rcw_shade
			value = 5
		}
	}
	else = {
		set_variable = {
			name = rcw_shade
			value = {
				value = 0
				every_in_global_list = {
					variable = rcw_russian_factions
					limit = {
						NOT = { this = scope:rcw_shade_picker }
						has_variable = russian_warlord_country
						is_country_alive = yes
						var:rcw_camp ?= scope:rcw_shade_picker.var:rcw_camp
					}
					add = 1
				}
				modulo = 5
				add = 1
			}
		}
	}
}

rcw_sync_camp_to_axis = {
	if = {
		limit = {
			var:rcw_axis <= 0
		}
		rcw_set_camp = { CAMP = black AXIS = 0 }
	}
	else_if = {
		limit = {
			var:rcw_axis <= 1
		}
		rcw_set_camp = { CAMP = red AXIS = 1 }
	}
	else_if = {
		limit = {
			var:rcw_axis <= 2
		}
		rcw_set_camp = { CAMP = green AXIS = 2 }
	}
	else_if = {
		limit = {
			var:rcw_axis <= 3
		}
		rcw_set_camp = { CAMP = democratic AXIS = 3 }
	}
	else = {
		rcw_set_camp = { CAMP = white AXIS = 4 }
	}
}

		limit = {

		}
		rcw_set_camp = { CAMP = democratic AXIS = 3 }
	}

		limit = {

		}
		rcw_set_camp = { CAMP = white AXIS = 4 }
	}
}

rcw_assign_russia_camp = {
	if = {
		limit = {
			has_law = law_type:law_council_republic
		}
		rcw_set_camp = { CAMP = red AXIS = 1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_anarchy
		}
		rcw_set_camp = { CAMP = black AXIS = 0 }
	}
	else_if = {
		limit = {
			var:provisional_government_identity ?= flag:conservative
		}
		rcw_set_camp = { CAMP = white AXIS = 4 }
	}
	else_if = {
		limit = {
			OR = {
				var:provisional_government_identity ?= flag:liberal
				var:provisional_government_identity ?= flag:socialist
			}
		}
		rcw_set_camp = { CAMP = democratic AXIS = 3 }
	}
	else_if = {
		limit = {
			exists = ruler
		}
		rcw_set_camp_from_character_ideology = { CHARACTER = ruler }
	}
	if = {
		limit = {
			NOT = { has_variable = rcw_camp }
		}
		rcw_set_camp = { CAMP = white AXIS = 4 }
	}
}

rcw_assign_revolution_camp = {
	save_scope_as = rcw_camp_target
	if = {
		limit = {
			exists = scope:rcw_revolution_movement
		}
		clear_saved_scope = rcw_revolution_movement
	}
	ROOT = {
		random_civil_war = {
			limit = {
				is_civil_war_type = revolution
			}
			political_movement ?= {
				save_scope_as = rcw_revolution_movement
			}
		}
	}
	scope:rcw_revolution_movement ?= {
		if = {
			limit = {
				rcw_movement_is_black = yes
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = black AXIS = 0 } }
		}
		else_if = {
			limit = {
				rcw_movement_is_red = yes
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = red AXIS = 1 } }
		}
		else_if = {
			limit = {
				rcw_movement_is_green = yes
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = green AXIS = 2 } }
		}
		else_if = {
			limit = {
				OR = {
					is_political_movement_type = movement_liberal
					is_political_movement_type = movement_positivist
				}
				ROOT.var:rcw_axis = 3
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = white AXIS = 4 } }
		}
		else_if = {
			limit = {
				rcw_movement_is_democratic = yes
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = democratic AXIS = 3 } }
		}
		else_if = {
			limit = {
				rcw_movement_is_white = yes
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = white AXIS = 4 } }
		}
		else_if = {
			limit = {
				is_political_movement_type = movement_socialist
				ROOT.var:rcw_axis >= 2
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = red AXIS = 1 } }
		}
		else_if = {
			limit = {
				is_political_movement_type = movement_socialist
			}
			scope:rcw_camp_target = { rcw_set_camp = { CAMP = democratic AXIS = 3 } }
		}
	}
	if = {
		limit = {
			NOT = { has_variable = rcw_camp }
			exists = ruler
		}
		rcw_set_camp_from_character_ideology = { CHARACTER = ruler }
	}
	if = {
		limit = {
			NOT = { has_variable = rcw_camp }
		}
		rcw_set_camp = { CAMP = red AXIS = 1 }
	}
	if = {
		limit = {
			var:rcw_axis = ROOT.var:rcw_axis
		}
		if = {
			limit = {
				var:rcw_axis >= 2
			}
			change_variable = {
				name = rcw_axis
				subtract = 1
			}
		}
		else = {
			change_variable = {
				name = rcw_axis
				add = 1
			}
		}
		rcw_sync_camp_to_axis = yes
	}
}

rcw_rederive_war_leader_camp = {
	if = {
		limit = {
			has_law = law_type:law_anarchy
			NOT = { var:rcw_camp ?= flag:rcw_black }
		}
		rcw_set_camp = { CAMP = black AXIS = 0 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_council_republic
			NOT = { has_law = law_type:law_anarchy }
			NOT = { var:rcw_camp ?= flag:rcw_red }
		}
		rcw_set_camp = { CAMP = red AXIS = 1 }
	}
}

rcw_mark_partition_owners = {
	set_variable = rcw_partition_owner
	set_variable = rcw_russian_faction
	set_variable = rcw_war_leader
	add_to_global_variable_list = {
		name = rcw_russian_factions
		target = this
	}
	every_diplomatic_play = {
		limit = {
			is_diplomatic_play_type = dp_revolution
			target_is = ROOT
		}
		initiator = {
			if = {
				limit = {
					NOT = { has_variable = rcw_partition_owner }
				}
				set_variable = rcw_partition_owner
				set_variable = rcw_russian_faction
				add_to_global_variable_list = {
					name = rcw_russian_factions
					target = this
				}
			}
		}
	}
	ordered_country = {
		limit = {
			has_variable = rcw_partition_owner
			NOT = { this = ROOT }
		}
		order_by = rcw_faction_size
		position = 0
		check_range_bounds = no
		save_scope_as = rcw_revolution_country
		set_variable = rcw_war_leader
		if = {
			limit = {
				NOT = { exists = c:RSR }
			}
			change_tag = RSR
		}
	}
	rcw_assign_russia_camp = yes
	scope:rcw_revolution_country ?= {
		rcw_assign_revolution_camp = yes
		set_global_variable = {
			name = rcw_revolution_axis
			value = var:rcw_axis
		}
	}
	every_country = {
		limit = {
			has_variable = rcw_partition_owner
			NOT = { has_variable = rcw_camp }
		}
		rcw_assign_revolution_camp = yes
	}
	if = {
		limit = {
			NOT = { has_global_variable = rcw_revolution_axis }
		}
		set_global_variable = {
			name = rcw_revolution_axis
			value = 1
		}
	}
}

rcw_reserve_black_heartland = {
	if = {
		limit = {
			global_var:rcw_share_black >= rcw_black_reservation_share
		}
		every_state = {
			limit = {
				owner ?= { has_variable = rcw_partition_owner }
				is_capital = no
				OR = {
					state_region = s:STATE_CHERSON
					state_region = s:STATE_TAURIDA
				}
			}
			add_to_temporary_list = rcw_black_heartland_states
		}
		if = {
			limit = {
				list_size = {
					name = rcw_black_heartland_states
					value >= rcw_min_faction_states
				}
			}
			every_in_list = {
				list = rcw_black_heartland_states
				set_variable = rcw_reserved_black
			}
		}
	}
}

rcw_drop_unviable_black_reservation = {
	if = {
		limit = {
			NOT = {
				any_state = {
					has_variable = rcw_reserved_black
					NOT = { rcw_state_theatre_retained = yes }
					count >= rcw_min_faction_states
				}
			}
		}
		every_state = {
			limit = {
				has_variable = rcw_reserved_black
			}
			remove_variable = rcw_reserved_black
		}
	}
}

rcw_pick_switch_targets = {
	c:RSR ?= {
		if = {
			limit = {
				is_player = no
			}
			set_global_variable = {
				name = rcw_switch_target_1
				value = this
				days = rcw_switch_window_days
			}
		}
	}
	c:POL ?= {
		if = {
			limit = {
				rcw_is_switch_candidate = yes
			}
			rcw_assign_switch_target = { TARGET = this }
		}
	}
	c:UKR ?= {
		if = {
			limit = {
				rcw_is_switch_candidate = yes
			}
			rcw_assign_switch_target = { TARGET = this }
		}
	}
	random_country = {
		limit = {
			var:rcw_theatre_key ?= flag:rcw_free_territory
			rcw_is_switch_candidate = yes
		}
		rcw_assign_switch_target = { TARGET = this }
	}
	while = {
		limit = {
			NOT = { has_global_variable = rcw_switch_target_4 }
			any_country = {
				rcw_is_switch_candidate = yes
			}
		}
		ordered_country = {
			limit = {
				rcw_is_switch_candidate = yes
			}
			order_by = rcw_interesting_value
			position = 0
			check_range_bounds = no
			rcw_assign_switch_target = { TARGET = this }
		}
	}
	every_country = {
		limit = {
			has_variable = rcw_switch_picked
		}
		remove_variable = rcw_switch_picked
	}
	set_global_variable = {
		name = rcw_switch_window
		days = rcw_switch_window_days
	}
}

rcw_release_elevated_ukraine = {
	if = {
		limit = {
			rcw_ukraine_renaissance_elevated = yes
			NOT = { exists = c:UKR }
			any_state = {
				rcw_is_elevated_ukraine_state = yes
			}
		}
		ordered_state = {
			limit = {
				rcw_is_elevated_ukraine_state = yes
			}
			order_by = state_population
			position = 0
			check_range_bounds = no
			set_variable = rcw_sep_seed_done
			save_temporary_scope_as = russian_warlord_state
		}
		cu:ukrainian = {
			save_temporary_scope_as = russian_warlord_state_culture
		}
		rcw_release_separatist = yes
		scope:russian_warlord_state.owner ?= {
			if = {
				limit = {
					NOT = { has_variable = rcw_partition_owner }
					country_has_primary_culture = cu:ukrainian
				}
				save_temporary_scope_as = rcw_elevated_ukraine
				every_state = {
					limit = {
						rcw_is_elevated_ukraine_state = yes
					}
					set_state_owner = scope:rcw_elevated_ukraine
				}
			}
		}
	}
}

rcw_release_separatists = {
	every_country = {
		limit = {
			has_variable = rcw_partition_owner
		}
		while = {
			limit = {
				any_scope_state = {
					rcw_is_separatist_seed = yes
				}
			}
			ordered_scope_state = {
				limit = {
					rcw_is_separatist_seed = yes
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				set_variable = rcw_sep_seed_done

		capital = scope:russian_warlord_state
		color = list "rcw_breakaway_colors"

		}

				has_variable = rcw_cede

				limit = {
					NOT = { has_variable = rcw_partition_owner }
					NOT = { has_variable = rcw_separatist }

rcw_spin_off_aligned_separatists = {
	every_country = {
		limit = {
			has_variable = rcw_separatist
			NOT = { has_variable = rcw_camp }
			any_scope_state = {
				has_variable = rcw_revolution_exclave
			}
		}
		save_temporary_scope_as = rcw_parent_separatist
		while = {
			limit = {
				any_state = {
					rcw_is_separatist_spin_off_state = yes
					any_neighbouring_state = {
						rcw_is_separatist_spin_off_state = yes
					}
				}
			}
			ordered_state = {
				limit = {
					rcw_is_separatist_spin_off_state = yes
					any_neighbouring_state = {
						rcw_is_separatist_spin_off_state = yes
					}
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = rcw_pocket_capital
				set_variable = rcw_pocket_member
			}
			while = {
				limit = {
					any_state = {
						rcw_is_separatist_spin_off_state = yes
						NOT = { has_variable = rcw_pocket_member }
						any_neighbouring_state = {
							has_variable = rcw_pocket_member
						}
					}
				}
				every_state = {
					limit = {
						rcw_is_separatist_spin_off_state = yes
						NOT = { has_variable = rcw_pocket_member }
						any_neighbouring_state = {
							has_variable = rcw_pocket_member
						}
					}
					set_variable = rcw_pocket_member
				}
			}
			scope:rcw_parent_separatist = {
				random_primary_culture = {
					save_temporary_scope_as = rcw_spin_off_culture
				}
			}
			scope:rcw_pocket_capital = {
				ordered_scope_culture = {
					limit = {
						scope:rcw_parent_separatist = {
							country_has_primary_culture = prev
						}
					}
					order_by = "scope:rcw_pocket_capital.culture_percent_state(this)"
					position = 0
					check_range_bounds = no
					save_temporary_scope_as = rcw_spin_off_culture
				}
			}
			create_dynamic_country = {
				origin = scope:rcw_parent_separatist
				tier = kingdom
				culture = scope:rcw_spin_off_culture
				religion = scope:rcw_spin_off_culture.religion
				capital = scope:rcw_pocket_capital
				color = list "rcw_breakaway_colors"
				cede_state_trigger = {
					has_variable = rcw_pocket_member
				}
			}
			if = {
				limit = {
					scope:rcw_pocket_capital ?= {
						owner ?= {
							NOT = { this = scope:rcw_parent_separatist }
							NOT = { has_variable = rcw_partition_owner }
							NOT = { has_variable = rcw_separatist }
						}
					}
				}
				scope:rcw_pocket_capital.owner = {
					rcw_register_separatist = yes
					set_variable = rcw_aligned_separatist
					set_variable = {
						name = rcw_camp
						value = scope:rcw_revolution_country.var:rcw_camp
					}
					set_variable = {
						name = rcw_axis
						value = scope:rcw_revolution_country.var:rcw_axis
					}
					rcw_apply_camp_government = yes
					rcw_aligned_separatist_pick_leader = yes
				}
			}
			every_state = {
				limit = {
					has_variable = rcw_pocket_member
				}
				remove_variable = rcw_pocket_member
				set_variable = rcw_pocket_done
			}
		}
	}
	every_state = {
		limit = {
			OR = {
				has_variable = rcw_revolution_exclave
				has_variable = rcw_pocket_done
			}
		}
		if = {
			limit = {
				has_variable = rcw_revolution_exclave
			}
			remove_variable = rcw_revolution_exclave
		}
		if = {
			limit = {
				has_variable = rcw_pocket_done
			}
			remove_variable = rcw_pocket_done
		}
	}
}

rcw_aligned_separatist_pick_leader = {
	save_scope_as = rcw_leader_faction
	if = {
		limit = {
			exists = scope:rcw_chosen_leader

		limit = {

	}
}

rcw_create_theatre_faction = {
	every_state = {
		limit = {
			var:rcw_faction_key ?= flag:rcw_$THEATRE$
			owner ?= { has_variable = rcw_partition_owner }
		}
		add_to_temporary_list = rcw_faction_states_$THEATRE$
	}
	if = {
		limit = {
			list_size = {
				name = rcw_faction_states_$THEATRE$
				value >= rcw_min_faction_states
			}
		}
		random_list = {
			50 = {
				set_global_variable = {
					name = rcw_capital_rank
					value = 0
				}
			}
			30 = {
				trigger = {
					list_size = {
						name = rcw_faction_states_$THEATRE$
						value >= 2
					}
				}
				set_global_variable = {
					name = rcw_capital_rank
					value = 1
				}
			}
			20 = {
				trigger = {
					list_size = {
						name = rcw_faction_states_$THEATRE$
						value >= 3
					}
				}
				set_global_variable = {
					name = rcw_capital_rank
					value = 2
				}
			}
		}
		if = {
			limit = {
				global_var:rcw_capital_rank = 2
			}
			ordered_in_list = {
				list = rcw_faction_states_$THEATRE$
				order_by = {
					value = state_population
					multiply = russian_culture_fraction
				}
				position = 2
				save_temporary_scope_as = rcw_faction_capital_$THEATRE$
			}
		}
		else_if = {
			limit = {
				global_var:rcw_capital_rank = 1
			}
			ordered_in_list = {
				list = rcw_faction_states_$THEATRE$
				order_by = {
					value = state_population
					multiply = russian_culture_fraction
				}
				position = 1
				save_temporary_scope_as = rcw_faction_capital_$THEATRE$
			}
		}
		else = {
			ordered_in_list = {
				list = rcw_faction_states_$THEATRE$
				order_by = {
					value = state_population
					multiply = russian_culture_fraction
				}
				position = 0
				save_temporary_scope_as = rcw_faction_capital_$THEATRE$
			}
		}
		scope:rcw_faction_capital_$THEATRE$.owner = {
			save_temporary_scope_as = rcw_faction_origin_$THEATRE$
		}
		every_in_list = {
			list = rcw_faction_states_$THEATRE$
			limit = {
				owner = scope:rcw_faction_origin_$THEATRE$
			}
			set_variable = rcw_cede
		}
		create_dynamic_country = {
			origin = scope:rcw_faction_origin_$THEATRE$
			tier = kingdom
			culture = cu:russian
			religion = rel:orthodox
			capital = scope:rcw_faction_capital_$THEATRE$
			cede_state_trigger = {
				has_variable = rcw_cede
			}
			on_created = {
				every_scope_state = {
					limit = {
						has_variable = rcw_cede
					}
					remove_variable = rcw_cede
				}
			}
		}
		if = {
			limit = {
				scope:rcw_faction_capital_$THEATRE$ ?= {
					owner ?= {
						NOT = { has_variable = rcw_partition_owner }
						NOT = { has_variable = rcw_russian_faction }
					}
				}
			}
			scope:rcw_faction_capital_$THEATRE$.owner = {
				save_temporary_scope_as = rcw_new_faction_$THEATRE$
				set_variable = rcw_russian_faction
				set_variable = russian_warlord_country
				set_variable = {
					name = rcw_theatre_key
					value = flag:rcw_$THEATRE$
				}
				if = {
					limit = {
						NOT = { scope:rcw_faction_origin_$THEATRE$ = ROOT }
					}
					set_variable = rcw_from_revolution
				}
				add_to_global_variable_list = {
					name = rcw_russian_factions
					target = this
				}
			}
			every_in_list = {
				list = rcw_faction_states_$THEATRE$
				limit = {
					owner ?= { has_variable = rcw_partition_owner }
				}
				set_state_owner = scope:rcw_new_faction_$THEATRE$
			}
		}
		every_state = {
			limit = {
				has_variable = rcw_cede
			}
			remove_variable = rcw_cede
		}
	}
}

rcw_mark_revolution_connected = {
	capital = {
		set_variable = rcw_revolution_connected
	}
	while = {
		limit = {
			any_state = {
				owner ?= scope:rcw_revolution_country
				NOT = { has_variable = rcw_revolution_connected }
				any_neighbouring_state = {
					has_variable = rcw_revolution_connected
				}
			}
		}
		every_state = {
			limit = {
				owner ?= scope:rcw_revolution_country
				NOT = { has_variable = rcw_revolution_connected }
				any_neighbouring_state = {
					has_variable = rcw_revolution_connected
				}
			}
			set_variable = rcw_revolution_connected
		}
	}
}

rcw_mark_revolution_exclaves = {
	scope:rcw_revolution_country ?= {
		if = {
			limit = {
				exists = capital
			}
			rcw_mark_revolution_connected = yes
			every_state = {
				limit = {
					rcw_is_revolution_pocket_state = yes
					any_neighbouring_state = {
						rcw_is_revolution_pocket_state = yes
					}
				}
				set_variable = rcw_revolution_exclave
			}
		}
	}
	every_state = {
		limit = {
			has_variable = rcw_revolution_connected
		}
		remove_variable = rcw_revolution_connected
	}
}

rcw_release_revolution_pockets = {
	scope:rcw_revolution_country ?= {
		if = {
			limit = {
				exists = capital
			}
			rcw_mark_revolution_connected = yes
			while = {
				limit = {
					any_state = {
						rcw_is_revolution_pocket_state = yes
						any_neighbouring_state = {
							rcw_is_revolution_pocket_state = yes
						}
					}
				}
				ordered_state = {
					limit = {
						rcw_is_revolution_pocket_state = yes
						any_neighbouring_state = {
							rcw_is_revolution_pocket_state = yes
						}
					}
					order_by = state_population
					position = 0
					check_range_bounds = no
					save_temporary_scope_as = rcw_pocket_capital
					set_variable = rcw_pocket_member
				}
				while = {
					limit = {
						any_state = {
							rcw_is_revolution_pocket_state = yes
							NOT = { has_variable = rcw_pocket_member }
							any_neighbouring_state = {
								has_variable = rcw_pocket_member
							}
						}
					}
					every_state = {
						limit = {
							rcw_is_revolution_pocket_state = yes
							NOT = { has_variable = rcw_pocket_member }
							any_neighbouring_state = {
								has_variable = rcw_pocket_member
							}
						}
						set_variable = rcw_pocket_member
					}
				}
				scope:rcw_pocket_capital = {
					ordered_scope_culture = {
						order_by = "scope:rcw_pocket_capital.culture_percent_state(this)"
						position = 0
						check_range_bounds = no
						save_temporary_scope_as = rcw_pocket_culture
					}
				}
				create_dynamic_country = {
					origin = scope:rcw_revolution_country
					tier = kingdom
					culture = scope:rcw_pocket_culture
					religion = scope:rcw_pocket_culture.religion
					capital = scope:rcw_pocket_capital
					color = list "rcw_breakaway_colors"
					cede_state_trigger = {
						has_variable = rcw_pocket_member
					}
				}
				if = {
					limit = {
						scope:rcw_pocket_capital ?= {
							owner ?= {
								NOT = { has_variable = rcw_partition_owner }
								NOT = { has_variable = rcw_russian_faction }
							}
						}
					}
					scope:rcw_pocket_capital.owner = {
						set_variable = rcw_russian_faction
						set_variable = russian_warlord_country
						if = {
							limit = {
								scope:rcw_pocket_capital = {
									has_variable = rcw_theatre
								}
							}
							set_variable = {
								name = rcw_theatre_key
								value = scope:rcw_pocket_capital.var:rcw_theatre
							}
						}
						set_variable = rcw_from_revolution
						set_variable = {
							name = rcw_camp
							value = scope:rcw_revolution_country.var:rcw_camp
						}
						set_variable = {
							name = rcw_axis
							value = scope:rcw_revolution_country.var:rcw_axis
						}
						rcw_pick_shade = yes
						add_to_global_variable_list = {
							name = rcw_russian_factions
							target = this
						}
					}
				}
				every_state = {
					limit = {
						has_variable = rcw_pocket_member
					}
					remove_variable = rcw_pocket_member
					set_variable = rcw_pocket_done
				}
			}
		}
	}
	every_state = {
		limit = {
			OR = {
				has_variable = rcw_revolution_connected
				has_variable = rcw_pocket_done
			}
		}
		if = {
			limit = {
				has_variable = rcw_revolution_connected
			}
			remove_variable = rcw_revolution_connected
		}
		if = {
			limit = {
				has_variable = rcw_pocket_done
			}
			remove_variable = rcw_pocket_done
		}
	}
}

rcw_resolve_exclave_state = {
	if = {
		limit = {
			has_variable = rcw_faction_key
			any_neighbouring_state = {
				has_variable = rcw_faction_connected
				owner ?= {
					has_variable = russian_warlord_country
					var:rcw_theatre_key ?= scope:rcw_exclave_state.var:rcw_faction_key
				}
			}
		}
		random_neighbouring_state = {
			limit = {
				has_variable = rcw_faction_connected
				owner ?= {
					has_variable = russian_warlord_country
					var:rcw_theatre_key ?= scope:rcw_exclave_state.var:rcw_faction_key
				}
			}
			owner = {
				save_temporary_scope_as = rcw_exclave_taker
			}
		}
		set_state_owner = scope:rcw_exclave_taker

					is_subject = no

						has_variable = rcw_freed_russian_subject
					}

			}
		}
	}
}

rcw_exclave_pass = {
	rcw_mark_factions_connected = yes
	while = {
		limit = {
			any_state = {
				rcw_is_faction_exclave_state = { SPARE_PLAYER = $SPARE_PLAYER$ }
			}
		}
		if = {
			limit = {
				any_state = {
					rcw_is_faction_exclave_state = { SPARE_PLAYER = $SPARE_PLAYER$ }
					any_neighbouring_state = {
						has_variable = rcw_faction_connected
					}
				}
			}
			ordered_state = {
				limit = {
					rcw_is_faction_exclave_state = { SPARE_PLAYER = $SPARE_PLAYER$ }
					any_neighbouring_state = {
						has_variable = rcw_faction_connected
					}
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = rcw_exclave_state
			}
		}
		else = {
			ordered_state = {
				limit = {
					rcw_is_faction_exclave_state = { SPARE_PLAYER = $SPARE_PLAYER$ }
				}
				order_by = state_population
				position = 0
				check_range_bounds = no
				save_temporary_scope_as = rcw_exclave_state
			}
		}
		scope:rcw_exclave_state = {
			rcw_resolve_exclave_state = { SPARE_PLAYER = $SPARE_PLAYER$ }
		}
		rcw_mark_factions_connected = yes
	}
	every_state = {
		limit = {
			OR = {
				has_variable = rcw_faction_connected
				has_variable = rcw_exclave_moved
				has_variable = rcw_exclave_skipped
			}
		}
		if = {
			limit = {
				has_variable = rcw_faction_connected
			}
			remove_variable = rcw_faction_connected
		}
		if = {
			limit = {
				has_variable = rcw_exclave_moved
			}
			remove_variable = rcw_exclave_moved
		}
		if = {
			limit = {
				has_variable = rcw_exclave_skipped
			}
			remove_variable = rcw_exclave_skipped
		}
	}
}

rcw_faction_pick_leader = {
	save_scope_as = rcw_leader_faction
	if = {
		limit = {
			exists = scope:rcw_chosen_leader
		}
		clear_saved_scope = rcw_chosen_leader
	}
	ordered_in_global_list = {
		variable = rcw_leader_candidates
		limit = {
			rcw_leader_available = yes
			is_historical = yes
			rcw_candidate_fits_faction = yes
		}
		alternative_limit = {
			rcw_leader_available = yes
			rcw_candidate_fits_faction = yes
			prominence >= define:NCharacters|PROMINENCE_NOTEWORTHY_THRESHOLD
		}
		order_by = rcw_leader_score
		position = 0
		check_range_bounds = no
		save_scope_as = rcw_chosen_leader
	}
	if = {
		limit = {
			NOT = { exists = scope:rcw_chosen_leader }
			global_var:rcw_setup_index <= rcw_template_spawn_faction_limit
		}
		rcw_try_spawn_camp_template = yes
	}
	if = {
		limit = {
			NOT = { exists = scope:rcw_chosen_leader }
		}
		ordered_in_global_list = {
			variable = rcw_leader_candidates
			limit = {
				rcw_leader_available = yes
				rcw_candidate_fits_faction = yes
			}
			order_by = rcw_leader_score
			position = 0
			check_range_bounds = no
			save_scope_as = rcw_chosen_leader
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:rcw_chosen_leader }
		}
		rcw_create_camp_leader = yes
	}
	scope:rcw_chosen_leader ?= {
		rcw_install_leader = yes
	}
}

rcw_install_leader = {
	scope:rcw_leader_faction = {
		set_variable = {
			name = keep_current_ruler_var
			days = 7
		}
	}
	if = {
		limit = {
			NOT = { owner ?= scope:rcw_leader_faction }
		}
		transfer_character = scope:rcw_leader_faction
	}
	if = {
		limit = {
			has_modifier = in_exile_modifier
		}
		hidden_effect = {
			free_from_exile_effect = yes
		}
	}
	siberian_exile_var_cleanup = yes
	if = {
		limit = {
			scope:rcw_leader_faction = {
				government_transfer_of_power = parliamentary_elective
			}
		}
		if = {
			limit = {
				is_interest_group_leader = no
			}
			set_as_interest_group_leader = yes
		}
	}
	if = {
		limit = {
			scope:rcw_leader_faction = {
				OR = {
					government_transfer_of_power = parliamentary_elective
					has_variable = rcw_camp
				}
			}
		}
		interest_group ?= {
			if = {
				limit = {
					is_in_government = no
				}
				add_ruling_interest_group = yes
			}
		}
	}
	set_character_as_ruler = yes
	if = {
		limit = {
			scope:rcw_leader_faction = {
				var:rcw_camp ?= flag:rcw_white
			}
			is_interest_group_type = ig_armed_forces
			NOR = {
				has_role_of_type = general
				has_role_of_type = admiral
			}
			scope:rcw_leader_faction = {
				ruler_can_have_command = yes
			}
		}
		add_character_role = general
	}
	if = {
		limit = {
			scope:rcw_leader_faction = {
				OR = {
					var:rcw_camp ?= flag:rcw_democratic
					var:rcw_camp ?= flag:rcw_black
				}
			}
			is_interest_group_leader = no
		}
		set_as_interest_group_leader = yes
	}
	set_variable = rcw_leader_assigned
}

rcw_collapse_check = {
	every_in_global_list = {
		variable = rcw_russian_factions
		limit = {
			rcw_is_active_russian_faction = yes
			is_player = no
			is_at_war = yes
			NOT = {
				AND = {
					has_variable = rcw_war_leader
					any_diplomatic_play = {
						is_diplomatic_play_type = dp_revolution
						OR = {
							target_is = prev
							initiator ?= prev
						}
					}
				}
			}
			enemy_occupation >= 0.4
			capital ?= {
				controller ?= {
					has_variable = rcw_russian_faction
					NOT = { this = prev.owner }
				}
			}
		}
		save_temporary_scope_as = rcw_collapsing_faction
		if = {
			limit = {
				has_variable = rcw_collapsing
			}
			capital = {
				controller = {
					save_temporary_scope_as = rcw_collapse_victor
				}
			}
			every_scope_state = {
				limit = {
					is_capital = no
					controller ?= {
						has_variable = rcw_russian_faction
						NOT = { this = scope:rcw_collapsing_faction }
						NOT = { this = scope:rcw_collapse_victor }
					}
				}
				controller = {
					save_temporary_scope_as = rcw_collapse_state_taker
				}
				set_state_owner = scope:rcw_collapse_state_taker
			}
			scope:rcw_collapse_victor = {
				annex_with_incorporation = scope:rcw_collapsing_faction
			}
		}
		else = {
			set_variable = {
				name = rcw_collapsing
				days = 45
			}
		}
	}
	if = {
		limit = {
			exists = scope:rcw_collapse_victor
		}
		rcw_exclave_pass = { SPARE_PLAYER = yes }
	}
}

			}
		}
		if = {

				}
			}

			}
		}
	}

			limit = {

rcw_distribute_agitators = {
	every_country = {
		limit = {
			has_variable = rcw_partition_owner
		}
		every_scope_character = {
			limit = {
				has_role_of_type = agitator
				is_interest_group_leader = no
				NOR = {
					is_ruler_of_own_country = yes
					is_heir_of_own_country = yes
				}
				NOT = { has_variable = rcw_leader_assigned }
				home_state ?= {
					owner ?= {
						has_variable = russian_warlord_country
						has_variable = rcw_camp
					}
				}
				var:rcw_candidate_camp ?= home_state.owner.var:rcw_camp
			}
			home_state = {
				owner = {
					save_temporary_scope_as = rcw_agitator_destination
				}
			}
			transfer_character = scope:rcw_agitator_destination
		}
	}
}

rcw_grant_separatist_claims = {
	if = {
		limit = {
			global_var:rcw_exhaustion_level >= 2
			NOT = { has_global_variable = rcw_separatist_claims_granted }
		}
		set_global_variable = rcw_separatist_claims_granted
		every_in_global_list = {
			variable = rcw_russian_factions
			limit = {
				rcw_is_active_russian_faction = yes
			}
			save_temporary_scope_as = rcw_claimant
			every_neighbouring_state = {
				limit = {
					owner ?= { has_variable = rcw_separatist }
					state_region = { is_homeland = cu:russian }
				}
				state_region = {
					add_claim = scope:rcw_claimant
				}
			}
		}
	}
}

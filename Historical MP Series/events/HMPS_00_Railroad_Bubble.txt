# Event that tracks variables which could trigger recession
# Railroad bubble 

railroad_bubble.1 = {
	type = country_event
	placement = root
	hidden = yes

	trigger = {
		# triggered by on_actions
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_variable = rr_built }
			}
			# the first expansion counts too
			set_variable = {
				name = rr_built
				value = 1
			}
		}
		else = {
			change_variable = {
				name = rr_built
				add = 1
			}
		}
	}
}

railroad_bubble.2 = {

	hidden = yes

	}

	}
}

#journal entry, set variables for intial recipient

railroad_bubble.3 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.3.t
	desc = railroad_bubble.3.d
	flavor = railroad_bubble.3.f
	
	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	
	trigger = {
		# triggered by je_starting
	}

	# Embrace the mania: full-strength boom and bust. Not under Command Economy or Cooperative Ownership,
	# unless legitimacy is too low to regulate (so the event always has an option)
	option = {
		name = railroad_bubble.3.a
		default_option = yes
		trigger = {
			OR = {
				NOR = {
					has_law = law_type:law_command_economy
					has_law = law_type:law_cooperative_ownership
				}
				legitimacy < 75
			}
		}
		#add modifier to country
		add_modifier = {
			name = railroad_bubble
		}
		custom_tooltip = {
			text = railroad_bubble.3.a.t
			every_scope_building = {
				limit = {
					is_building_type = building_railway
				}
				add_modifier = {
					name = overvalued_railroads_building
				}
			}
			every_scope_state = {
				add_modifier = {
					name = overvalued_railroads_state
				}
			}
		}
		every_primary_culture = {
			add_cultural_obsession = transportation
		}
	}

	# Regulate the exchanges: always costs bureaucracy (until the burst, removed in railroad_bubble.6) and Industrialists -5 for 2 years.
	# 50%: it works - boom and bust at 75%. 50%: speculators evade it - full-strength boom and bust.
	# The two regulator modifiers have the same effect; their names show the player which outcome happened. Not under Laissez-Faire; needs 75 legitimacy
	option = {
		name = railroad_bubble.3.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
			custom_tooltip = {
				text = railroad_bubble_legitimacy_tt
				legitimacy >= 75
			}
		}
		show_as_unavailable = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = rr_bubble_regulated_exchanges
				days = 730
			}
		}
		random_list = {
			50 = {
				custom_tooltip = railroad_bubble.3.b.success_tt
				add_modifier = {
					name = rr_exchange_regulators
				}
				set_variable = {
					name = rr_bubble_scale
					value = 0.75
				}
				add_modifier = {
					name = railroad_bubble
					multiplier = 0.75
				}
				custom_tooltip = {
					text = railroad_bubble.3.b.t
					every_scope_building = {
						limit = {
							is_building_type = building_railway
						}
						add_modifier = {
							name = overvalued_railroads_building
							multiplier = 0.75
						}
					}
					every_scope_state = {
						add_modifier = {
							name = overvalued_railroads_state
							multiplier = 0.75
						}
					}
				}
			}
			50 = {
				custom_tooltip = railroad_bubble.3.b.failure_tt
				add_modifier = {
					name = rr_exchange_regulators_evaded
				}
				add_modifier = {
					name = railroad_bubble
				}
				custom_tooltip = {
					text = railroad_bubble.3.a.t
					every_scope_building = {
						limit = {
							is_building_type = building_railway
						}
						add_modifier = {
							name = overvalued_railroads_building
						}
					}
					every_scope_state = {
						add_modifier = {
							name = overvalued_railroads_state
						}
					}
				}
			}
		}
		every_primary_culture = {
			add_cultural_obsession = transportation
		}
	}
}

# The peak: the response decides when the bubble bursts (railroad_bubble.6) and how hard.
# rr_bust_size = bubble size x strength x response factor

railroad_bubble.5 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.5.t
	desc = railroad_bubble.5.d
	flavor = railroad_bubble.5.f
	
	duration = 2

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	
	trigger = {
		# triggered by je
	}
	
	immediate = {
		if = {
			limit = { NOT = { has_variable = rr_bubble_scale } }
			set_variable = {
				name = rr_bubble_scale
				value = 1
			}
		}
		set_variable = {
			name = rr_bust_size
			value = var:railroad_bubble_mult
		}
		change_variable = {
			name = rr_bust_size
			multiply = var:rr_bubble_scale
		}
	}

	# Let it run: bursts in 183 days (6 months) at full strength
	option = {
		name = railroad_bubble.5.a
		default_option = yes
		custom_tooltip = {
			text = railroad_bubble.5.a.t
			trigger_event = {
				id = railroad_bubble.6
				days = 183
				popup = yes
			}
		}
	}

	# Cool the market: bursts now at 75% strength, but a flat 12-month credit squeeze (rr_tight_money) and Industrialists -5 approval for 2 years. Not under Laissez-Faire
	option = {
		name = railroad_bubble.5.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		add_modifier = {
			name = rr_tight_money
			days = 365
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = rr_market_cooled
				days = 730
			}
		}
		custom_tooltip = {
			text = railroad_bubble.5.b.t
			change_variable = {
				name = rr_bust_size
				multiply = 0.75
			}
			trigger_event = {
				id = railroad_bubble.6
				popup = yes
			}
		}
	}

	# Prop up the shares: weekly expense for the 365 days (12 months) until the burst, 5% of GDP in total (rr_shares_propped); bursts at 125% strength. Interventionism only, and needs room to borrow
	option = {
		name = railroad_bubble.5.c
		trigger = {
			has_law = law_type:law_interventionism
			custom_tooltip = {
				text = railroad_bubble_fiscal_room_tt
				scaled_debt < 0.5
			}
		}
		show_as_unavailable = {
			has_law = law_type:law_interventionism
		}
		add_modifier = {
			name = rr_shares_propped
			days = 365
			# weekly amount: 5% of GDP spread over 52 weeks
			multiplier = {
				value = root.gdp
				multiply = 0.05
				divide = 52
			}
		}
		custom_tooltip = {
			text = railroad_bubble.5.c.t
			change_variable = {
				name = rr_bust_size
				multiply = 1.25
			}
			trigger_event = {
				id = railroad_bubble.6
				days = 365
				popup = yes
			}
		}
	}

	# Private bank bailout: paid from the investment pool (5% of GDP, needs that much in the pool), bursts in 365 days (12 months) at 125% strength. Laissez-Faire only
	option = {
		name = railroad_bubble.5.e
		trigger = {
			has_law = law_type:law_laissez_faire
			custom_tooltip = {
				text = railroad_bubble_investment_pool_tt
				investment_pool >= {
					value = root.gdp
					multiply = 0.05
				}
			}
		}
		show_as_unavailable = {
			has_law = law_type:law_laissez_faire
		}
		add_investment_pool = {
			value = root.gdp
			multiply = -0.05
		}
		custom_tooltip = {
			text = railroad_bubble.5.c.t
			change_variable = {
				name = rr_bust_size
				multiply = 1.25
			}
			trigger_event = {
				id = railroad_bubble.6
				days = 365
				popup = yes
			}
		}
	}
}

# Railroad bubble burst event - the crash scales with rr_bust_size (set in railroad_bubble.5):
# rr_bubble_crash at full strength for 12 months, then railroad_bubble.8 adds rr_bubble_pop fading over 24 months

railroad_bubble.6 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.6.t
	desc = railroad_bubble.6.d
	flavor = railroad_bubble.6.f
	
	duration = 2

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"	
	
	trigger = {
		# triggered by event
	}

	# bursts scheduled before rr_bust_size existed use the bubble size
	immediate = {
		if = {
			limit = { NOT = { has_variable = rr_bust_size } }
			set_variable = {
				name = rr_bust_size
				value = var:railroad_bubble_mult
			}
		}
	}

	# We can rebuild: the bust as set at the peak
	option = {
		name = railroad_bubble.6.a
		default_option = yes
		# only remove what the country has: a missing timed modifier logs an error (repeatedly, while the event's tooltips are shown)
		if = {
			limit = { has_modifier = railroad_bubble }
			remove_modifier = railroad_bubble
		}
		if = {
			limit = { has_modifier = rr_exchange_regulators }
			remove_modifier = rr_exchange_regulators
		}
		if = {
			limit = { has_modifier = rr_exchange_regulators_evaded }
			remove_modifier = rr_exchange_regulators_evaded
		}
		custom_tooltip = {
			text = railroad_bubble.6.a.t
			every_scope_building = {
				limit = {
					has_modifier = overvalued_railroads_building
				}
				remove_modifier = overvalued_railroads_building
			}
			every_scope_state = {
				limit = { has_modifier = overvalued_railroads_state }
				remove_modifier = overvalued_railroads_state
			}
		}
		add_modifier = {
			name = rr_bubble_crash
			multiplier = root.var:rr_bust_size
			days = 365
		}
		custom_tooltip = {
			text = railroad_bubble.6.recovery_tt
			set_variable = {
				name = rr_crash_size
				value = root.var:rr_bust_size
			}
			trigger_event = {
				id = railroad_bubble.8
				days = 365
			}
		}
		every_primary_culture = {
			remove_cultural_obsession = transportation
		}
		hidden_effect = {
			remove_variable = rr_bust_size
		}
	}

	# Fund relief works: weekly expense of 0.1% of GDP for 12 months (rr_relief_works, about 5% of GDP in total) and a flat +15% dependents' wage for those months (rr_relief_works_wages). Needs room to borrow; not under Laissez-Faire.
	# The crash itself is the same as railroad_bubble.6.a - keep the two in step
	option = {
		name = railroad_bubble.6.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
			custom_tooltip = {
				text = railroad_bubble_fiscal_room_tt
				scaled_debt < 0.5
			}
		}
		show_as_unavailable = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		add_modifier = {
			name = rr_relief_works
			days = 365
			multiplier = money_amount_multiplier_large
		}
		# flat +15% dependents' wage for the 12 full-strength months (the smallest possible crash wage loss, see z_HMPS_flavor_modifiers.txt)
		custom_tooltip = railroad_bubble.6.b.t
		add_modifier = {
			name = rr_relief_works_wages
			days = 365
		}
		# only remove what the country has: a missing timed modifier logs an error (repeatedly, while the event's tooltips are shown)
		if = {
			limit = { has_modifier = railroad_bubble }
			remove_modifier = railroad_bubble
		}
		if = {
			limit = { has_modifier = rr_exchange_regulators }
			remove_modifier = rr_exchange_regulators
		}
		if = {
			limit = { has_modifier = rr_exchange_regulators_evaded }
			remove_modifier = rr_exchange_regulators_evaded
		}
		custom_tooltip = {
			text = railroad_bubble.6.a.t
			every_scope_building = {
				limit = {
					has_modifier = overvalued_railroads_building
				}
				remove_modifier = overvalued_railroads_building
			}
			every_scope_state = {
				limit = { has_modifier = overvalued_railroads_state }
				remove_modifier = overvalued_railroads_state
			}
		}
		add_modifier = {
			name = rr_bubble_crash
			multiplier = root.var:rr_bust_size
			days = 365
		}
		custom_tooltip = {
			text = railroad_bubble.6.recovery_tt
			set_variable = {
				name = rr_crash_size
				value = root.var:rr_bust_size
			}
			trigger_event = {
				id = railroad_bubble.8
				days = 365
			}
		}
		every_primary_culture = {
			remove_cultural_obsession = transportation
		}
		hidden_effect = {
			remove_variable = rr_bust_size
		}
	}
}

# Crash phase 2 - fired by railroad_bubble.6 365 days after the burst: the same penalties, fading over 24 months

railroad_bubble.8 = {
	type = country_event
	placement = ROOT
	hidden = yes

	trigger = {
		has_variable = rr_crash_size
	}

	immediate = {
		add_modifier = {
			name = rr_bubble_pop
			multiplier = var:rr_crash_size
			days = 730
			is_decaying = yes
		}
		remove_variable = rr_crash_size
	}
}

# The Railway King - fired once per bubble by je_railroad_bubble when the bubble reaches 20.
# Bubble size changes show on the progress bar now and reach the modifiers at the next monthly pulse

railroad_bubble.7 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.7.t
	desc = railroad_bubble.7.d
	flavor = railroad_bubble.7.f

	duration = 3

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"

	trigger = {
		# triggered by je_railroad_bubble
	}

	# Grant his charters: bubble +5, Industrialists +5 approval for 2 years
	option = {
		name = railroad_bubble.7.a
		custom_tooltip = {
			text = railroad_bubble.7.a.t
			change_variable = {
				name = railroad_bubble_mult
				add = 5
			}
			clamp_variable = {
				name = railroad_bubble_mult
				max = 40
				min = 0
			}
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = rr_railway_king_charters
				days = 730
			}
		}
	}

	# Investigate his accounts: bubble -5, Industrialists -5 approval for 2 years. Not under Laissez-Faire
	option = {
		name = railroad_bubble.7.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		custom_tooltip = {
			text = railroad_bubble.7.b.t
			change_variable = {
				name = railroad_bubble_mult
				subtract = 5
			}
			clamp_variable = {
				name = railroad_bubble_mult
				max = 40
				min = 0
			}
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = rr_railway_king_investigated
				days = 730
			}
		}
	}

	# Stay out of it
	option = {
		name = railroad_bubble.7.c

	}

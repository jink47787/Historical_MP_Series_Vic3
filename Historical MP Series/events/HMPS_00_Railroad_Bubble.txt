# Event that tracks variables which could trigger recession
# Railroad bubble 

railroad_bubble.1 = {
	type = country_event
	placement = root
	hidden = yes

	trigger = {
		# triggered by on_actions
	}
	
	immediate = {
		if = {
			limit = {
				NOT = { has_variable = rr_built }
			}
			set_variable = {
				name = rr_built
				value = 0
			}
		}
		else = {
			change_variable = {
				name = rr_built
				add = 1
			}
		}
	}
}

railroad_bubble.3 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.3.t
	desc = railroad_bubble.3.d
	flavor = railroad_bubble.3.f
	
	duration = 3

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	
	trigger = {
		# triggered by je_starting
	}

	# Embrace the mania: full-strength boom and bust. Not under Command Economy or Cooperative Ownership
	option = {
		name = railroad_bubble.3.a
		default_option = yes
		trigger = {
			NOR = {
				has_law = law_type:law_command_economy
				has_law = law_type:law_cooperative_ownership
			}
		}
		#add modifier to country
		add_modifier = {
			name = railroad_bubble
		}
		custom_tooltip = {
			text = railroad_bubble.3.a.t
			every_scope_building = {
				limit = {
					is_building_type = building_railway
				}
				add_modifier = {
					name = overvalued_railroads_building
				}
			}
			every_scope_state = {
				add_modifier = {
					name = overvalued_railroads_state
				}
			}
		}
		every_primary_culture = {
			add_cultural_obsession = transportation
		}
	}

	# Regulate the exchanges: boom and bust at 75%. Not under Laissez-Faire
	option = {
		name = railroad_bubble.3.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		set_variable = {
			name = rr_bubble_scale
			value = 0.75
		}
		add_modifier = {
			name = railroad_bubble
			multiplier = 0.75
		}
		custom_tooltip = {
			text = railroad_bubble.3.b.t
			every_scope_building = {
				limit = {
					is_building_type = building_railway
				}
				add_modifier = {
					name = overvalued_railroads_building
					multiplier = 0.75
				}
			}
			every_scope_state = {
				add_modifier = {
					name = overvalued_railroads_state
					multiplier = 0.75
				}
			}
		}
		ig:ig_industrialists = {
			add_modifier = {
				name = rr_bubble_regulated_exchanges
				months = 24
			}
		}
		every_primary_culture = {
			add_cultural_obsession = transportation
		}
	}
}

# The peak: the response decides when the bubble bursts (railroad_bubble.6) and how hard.
# rr_bust_size = bubble size x strength x response factor

railroad_bubble.5 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.5.t
	desc = railroad_bubble.5.d
	flavor = railroad_bubble.5.f
	
	duration = 2

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	
	trigger = {
		# triggered by je
	}
	
	immediate = {
		if = {
			limit = { NOT = { has_variable = rr_bubble_scale } }
			set_variable = {
				name = rr_bubble_scale
				value = 1
			}
		}
		set_variable = {
			name = rr_bust_size
			value = var:railroad_bubble_mult
		}
		change_variable = {
			name = rr_bust_size
			multiply = var:rr_bubble_scale
		}
	}

	# Let it run: bursts in 150 days at full strength
	option = {
		name = railroad_bubble.5.a
		default_option = yes
		custom_tooltip = {
			text = railroad_bubble.5.a.t
			trigger_event = {
				id = railroad_bubble.6
				days = 150
				popup = yes
			}
		}
	}

	# Cool the market: bursts now at half strength. Not under Laissez-Faire
	option = {
		name = railroad_bubble.5.b
		trigger = {
			NOT = { has_law = law_type:law_laissez_faire }
		}
		custom_tooltip = {
			text = railroad_bubble.5.b.t
			change_variable = {
				name = rr_bust_size
				multiply = 0.5
			}
			trigger_event = {
				id = railroad_bubble.6
				popup = yes
			}
		}
	}

	# Prop up the shares: costs 2% of GDP from the treasury, bursts in 330 days at 125% strength. Interventionism only
	option = {
		name = railroad_bubble.5.c
		trigger = {
			has_law = law_type:law_interventionism
		}
		add_treasury = {
			value = root.gdp
			multiply = -0.02
		}
		custom_tooltip = {
			text = railroad_bubble.5.c.t
			change_variable = {
				name = rr_bust_size
				multiply = 1.25
			}
			trigger_event = {
				id = railroad_bubble.6
				days = 330
				popup = yes
			}
		}
	}

	# Private bank bailout: as Prop up the shares, but paid from the investment pool. Laissez-Faire only
	option = {
		name = railroad_bubble.5.e
		trigger = {
			has_law = law_type:law_laissez_faire
			investment_pool >= {
				value = root.gdp
				multiply = 0.02
			}
		}
		show_as_unavailable = {
			has_law = law_type:law_laissez_faire
		}
		add_investment_pool = {
			value = root.gdp
			multiply = -0.02
		}
		custom_tooltip = {
			text = railroad_bubble.5.c.t
			change_variable = {
				name = rr_bust_size
				multiply = 1.25
			}
			trigger_event = {
				id = railroad_bubble.6
				days = 330
				popup = yes
			}
		}
	}
}

# Railroad bubble burst event - the rr_bubble_pop penalties scale with rr_bust_size (set in railroad_bubble.5) and decay over 36 months

railroad_bubble.6 = {
	type = country_event
	placement = ROOT

	title = railroad_bubble.6.t
	desc = railroad_bubble.6.d
	flavor = railroad_bubble.6.f
	
	duration = 2

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"	
	
	trigger = {
		# triggered by event
	}

	# bursts scheduled before rr_bust_size existed use the bubble size
	immediate = {
		if = {
			limit = { NOT = { has_variable = rr_bust_size } }
			set_variable = {
				name = rr_bust_size
				value = var:railroad_bubble_mult
			}
		}
	}

	option = {
		name = railroad_bubble.6.a
		# at full strength the bust is 18-40: 18-25 small, 26-32 medium, 33-40 large
		if = {
			limit = { root.var:rr_bust_size < 26 }
			add_radicals = {
				value = small_radicals
			}
		}
		else_if = {
			limit = { root.var:rr_bust_size < 33 }
			add_radicals = {
				value = medium_radicals
			}	
		}
		else = {
			add_radicals = {
				value = large_radicals
			}	
		}
		remove_modifier = railroad_bubble
		custom_tooltip = {
			text = railroad_bubble.6.a.t
			every_scope_building = {
				limit = {
					has_modifier = overvalued_railroads_building
				}
				remove_modifier = overvalued_railroads_building
				}
			every_scope_state = {
				remove_modifier = overvalued_railroads_state
			}
		}
		add_modifier = {
			name = rr_bubble_pop
			multiplier = root.var:rr_bust_size
			months = 36
			is_decaying = yes
		}
		every_primary_culture = {
			remove_cultural_obsession = transportation
		}
		hidden_effect = {
			remove_variable = rr_bust_size
		}
	}
}

			}

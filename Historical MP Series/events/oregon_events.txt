	placement = ROOT

		video = "europenorthamerica_capitalists_meeting"

	title = oregon.2.t
	desc = oregon.2.d
	flavor = oregon.2.f
	
	duration = 3
	
	trigger = {
	}
	
	immediate = {
		s:STATE_WASHINGTON = {
			random_scope_state = {
				limit = {
					owner = {
						OR = {
							c:GBR = THIS
							is_subject_of = c:GBR
						}
					}
				}
				save_scope_as = washington_scope
			}
		}
		s:STATE_IDAHO = {
			random_scope_state = {
				limit = {
					owner = {
						OR = {
							c:GBR = THIS
							is_subject_of = c:GBR
						}
					}
				}
				save_scope_as = idaho_scope
			}
		}

			random_scope_state = {
				limit = {
					owner = {
						OR = {
							c:GBR = THIS
							is_subject_of = c:GBR
						}
					}
				}
				save_scope_as = oregon_scope
			}
		}
		c:USA ?= {
			save_scope_as = usa_scope
		}
	}
	
	option = {
		name = oregon.2.a
		default_option = yes
		c:USA ?= {
			change_relations = {
				country = c:GBR
				value = 20
			}
			trigger_event = {
				id = oregon.3
				days = 0
			}
		}
		show_as_tooltip = {
			s:STATE_IDAHO = {
				random_scope_state = {
					limit = {
						exists = c:GBR
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
					set_state_owner = c:USA
					set_state_type = unincorporated
				}
				if = {
					limit = {
						c:ORG ?= {
							has_claim = s:STATE_IDAHO
						}
					}
					s:STATE_IDAHO = {
						remove_claim = c:ORG
					}
					s:STATE_IDAHO = {
						add_claim = c:USA
					}
				}
			}
			s:STATE_WASHINGTON = {
				random_scope_state = {
					limit = {
						exists = c:GBR
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
					set_state_owner = c:USA
					set_state_type = unincorporated
				}
				if = {
					limit = {
						c:ORG ?= {
							has_claim = s:STATE_WASHINGTON
						}
					}
					s:STATE_WASHINGTON = {
						remove_claim = c:ORG
					}
					s:STATE_WASHINGTON = {
						add_claim = c:USA
					}
				}
			}
			s:STATE_OREGON = {
				random_scope_state = {
					limit = {
						exists = c:GBR
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
					set_state_owner = c:USA
					set_state_type = unincorporated
				}
				if = {
					limit = {
						c:ORG ?= {
							has_claim = s:STATE_OREGON
						}
					}
					s:STATE_OREGON = {
						remove_claim = c:ORG
					}
					s:STATE_OREGON = {
						add_claim = c:USA
					}
				}
			}
			s:STATE_WYOMING = {
				add_homeland = cu:yankee
				random_scope_state = {
					limit = {
						exists = c:GBR
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
					set_state_owner = c:USA
					set_state_type = unincorporated
				}
			}
			s:STATE_MONTANA = {
				add_homeland = cu:yankee
				random_scope_state = {
					limit = {
						exists = c:GBR
						owner = {
							OR = {
								c:GBR = THIS
								is_subject_of = c:GBR
							}
						}
					}
					set_state_owner = c:USA
					set_state_type = unincorporated
				}
			}
			hidden_effect = {
				if = {
					limit = {
						s:STATE_BRITISH_COLUMBIA = {
							any_scope_state = {
								owner = {
									OR = {
										c:GBR = THIS
										is_subject_of = c:GBR
									}
								}
							}
						}
					}
					s:STATE_BRITISH_COLUMBIA = {
						random_scope_state = {
							set_state_name = STATE_BRITISH_COLUMBIA_british
						}
					}
				}
			}
		}
		ai_chance = { # AI should do this by default
			base = 10
		}
	}
	option = {
		name = oregon.2.b
		c:USA ?= {
			change_relations = {
				country = c:GBR
				value = -30
			}
			trigger_event = {
				id = oregon.4
				days = 0
			}
		}
		ai_chance = {
			base = 0
		}
	}
}

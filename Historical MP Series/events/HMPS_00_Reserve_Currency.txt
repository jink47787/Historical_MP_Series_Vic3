# A new holder
reserve_currency.1 = {
	type = country_event
	placement = root

	title = reserve_currency.1.t
	desc = {
		first_valid = {
			triggered_desc = {
				desc = reserve_currency.1.d_gained
				trigger = { global_var:reserve_currency_holder ?= root }
			}
			triggered_desc = {
				desc = reserve_currency.1.d_lost
				trigger = { global_var:rc_previous_holder ?= root }
			}
			triggered_desc = {
				desc = reserve_currency.1.d
				trigger = { always = yes }
			}
		}
	}
	flavor = reserve_currency.1.f

	duration = 3

	event_image = {
		video = "europenorthamerica_london_center"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		# triggered by rc_notify_new_holder
	}

	option = {
		name = reserve_currency.1.a
		default_option = yes
	}
}

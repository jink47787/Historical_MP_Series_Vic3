# Joining the Scramble for Africa
africa_colonial_events.1 = {
	type = country_event
	placement = scope:african_colony_state

	title = africa_colonial_events.1.t
	desc = africa_colonial_events.1.d
	flavor = africa_colonial_events.1.f

	event_image = {
		video = "africa_desert_expedition"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = no
				is_in_geographic_region = geographic_region_africa
			}
			save_scope_as = african_colony_state
		}
		hmps_save_african_colony_state_fallback = yes #HMPS: countries that only have African subjects have no colony to place the event on
		#HMPS
		if = {
			limit = {
				NOT = {
					has_global_variable = global_africa_scramble
				}
			}
			set_global_variable = {
				name = global_africa_scramble
			}
			set_variable = first_to_scramble
			set_global_variable = { #HMPS: africa_colonial_events.8 is offered to newly eligible powers while this lasts
				name = africa_scramble_offer_window
				years = 10
			}
			trigger_event = {
				id = congo_free_state.1
				days = 45
			}
		}
	} 
	option = {
		name = africa_colonial_events.1.a
		default_option = yes
		
		#HMPS Flavor
		if = {
			limit = {
				has_variable = first_to_scramble
			}
			hidden_effect = { #HMPS: explained by africa_colonial_events.1.a.t; raw script would show in the tooltip
				every_country = {
					limit = {
						hmps_africa_scramble_offer_eligible = yes
					}
					trigger_event = {
						id = africa_colonial_events.8
						days = 0
						popup = yes
					}
				}
			}
			custom_tooltip = {
				text = africa_colonial_events.1.a.t
			}
			add_modifier = {
				name = scrambling_for_africa
				multiplier = 2
				days = 3650 #HMPS: 10 years
			}
			add_modifier = {
				name = first_scrambler
				months = 60
			}
			change_institution_investment_level = {
				institution = institution_colonial_affairs
				investment = 1
			}
		}
		
		if = {
			limit = {
				NOT = { has_variable = first_to_scramble }
				has_variable = has_mp
			}
			add_modifier = {
				name = scrambling_for_africa
				multiplier = 1.5
				days = 3650 #HMPS: 10 years
			}
		}
		if = {
			limit = {
				NOT = { has_variable = first_to_scramble }
				NOT = { has_variable = has_mp }
			}
			add_modifier = {
				name = scrambling_for_africa
				days = 3650 #HMPS: 10 years
			}
		}
		hidden_effect = { #HMPS: last, after the bonus checks above; so a restarted journal entry can't give the first-scrambler bonus twice
			if = {
				limit = { has_variable = first_to_scramble }
				remove_variable = first_to_scramble
			}
		}
	}
}

africa_colonial_events.4 = {
	type = country_event
	placement = scope:african_colony_state

	title = africa_colonial_events.4.t
	desc = africa_colonial_events.4.d
	flavor = africa_colonial_events.4.f

	event_image = {
		video = "southamerica_christianity"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	cooldown = { days = stupidly_long_modifier_time }

	trigger = {
		religion = {
			has_discrimination_trait = heritage_christian
		}
		any_scope_state = {
			is_incorporated = no
			is_in_geographic_region = geographic_region_africa
			NOT = { has_modifier = christian_mission }
			any_neighbouring_state = {
				owner = {
					is_country_type = decentralized
					NOT = {
						religion = {
							has_discrimination_trait = heritage_christian
						}
					}
				}
			}
		}
	}

	immediate = {
		random_scope_state = {
			limit = {
				is_incorporated = no
				is_in_geographic_region = geographic_region_africa
				NOT = { has_modifier = christian_mission }
				any_neighbouring_state = {
					owner = {
						is_country_type = decentralized
						NOT = {
							religion = {
								has_discrimination_trait = heritage_christian
							}
						}
					}
				}
			}
			save_scope_as = african_colony_state
		}
	}

	option = {
		name = africa_colonial_events.4.a
		scope:african_colony_state = {
			add_modifier = {
				name = christian_mission
				days = normal_modifier_time
			}
		}
		ig:ig_devout ?= {
			add_modifier = {
				name = approved_mission
				days = normal_modifier_time
			}
		}
		every_country = {
			limit = {
				is_country_type = decentralized
				NOT = {
					religion = {
						has_discrimination_trait = heritage_christian
					}
				}
				any_neighbouring_state = { this = scope:african_colony_state }
			}
			change_tension = {
				country = root
				value = 3
			}
		}
	}

	option = {
		name = africa_colonial_events.4.b
		default_option = yes

		ig:ig_devout ?= {
			add_modifier = {
				name = rejected_mission
				days = normal_modifier_time
			}
		}
		every_country = {
			limit = {
				is_country_type = decentralized
				NOT = {
					religion = {
						has_discrimination_trait = heritage_christian
					}
				}
				any_neighbouring_state = { this = scope:african_colony_state }
			}
			change_tension = {
				country = root
				value = -3
			}
		}
	}
}

africa_colonial_events.5 = {
	type = country_event

	flavor = africa_colonial_events.5.f

		video = "southamerica_child_labor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	trigger = {
		has_law_or_variant = law_type:law_slavery_banned
		NOT = { has_modifier = mission_to_abolish_slavery }
		any_interest_group = {
			law_stance = {
				law = law_type:law_slave_trade

			}

			is_incorporated = no

					is_country_type = decentralized

			}
		}
	}

	cooldown = { days = stupidly_long_modifier_time }

	immediate = {
		random_interest_group = {
			limit = {
				law_stance = {
					law = law_type:law_slave_trade
					value < neutral
				}
			}
			save_scope_as = anti_slave_trade_ig
			leader = { save_scope_as = anti_slave_trade_ig_leader }
		}
		random_scope_state = {
			limit = {
				is_incorporated = no
				is_in_geographic_region = geographic_region_africa

			add_modifier = {
				name = pledged_to_end_slave_trade
				days = normal_modifier_time
			}
		}
		every_country = {
			limit = {
				is_country_type = decentralized
				NOT = {	has_law_or_variant = law_type:law_slavery_banned }
				any_neighbouring_state = { this = scope:african_colony_state }

				country = root

		default_option = yes

			add_modifier = {
				name = refused_pledge_to_end_slave_trade

#HMPS Flavor: sent by africa_colonial_events.1 to the other colonial powers when the first country joins the Scramble, then monthly (HMPS_on_action.txt) to powers that become eligible within 10 years
africa_colonial_events.8 = {
	type = country_event
	placement = scope:african_colony_state

	title = africa_colonial_events.8.t
	desc = africa_colonial_events.8.d
	flavor = africa_colonial_events.8.f

	event_image = {
		video = "africa_diplomats_negotiating"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	trigger = { # offered once per country, while the window set by africa_colonial_events.1 lasts
		has_global_variable = africa_scramble_offer_window
		NOT = { has_variable = africa_scramble_offered }
		hmps_africa_scramble_offer_eligible = yes
	}

	immediate = {
		set_variable = africa_scramble_offered
		clear_saved_scope = african_colony_state # drop the first scrambler's state passed on by africa_colonial_events.1
		random_scope_state = {
			limit = {
				is_incorporated = no
				is_in_geographic_region = geographic_region_africa
			}
			save_scope_as = african_colony_state
		}
		hmps_save_african_colony_state_fallback = yes
	}

	option = {
		name = africa_colonial_events.8.a
		default_option = yes
		if = {
			limit = {
				NOT = { has_technology_researched = civilizing_mission }
			}
			custom_tooltip = {
				text = africa_colonial_events.8.a.tt
			}
			add_technology_researched = civilizing_mission
			add_technology_researched = malaria_prevention
		}
		else = {
			custom_tooltip = {
				text = africa_colonial_events.8.a.tt_mp
				set_variable = has_mp # gives the 1.5x scrambling_for_africa in africa_colonial_events.1
			}
			add_technology_researched = malaria_prevention
		}
		ai_chance = {
			base = 90
			modifier = { # Great Powers with nothing in Africa: 50/50, as Germany joined but Russia and Austria never did
				trigger = {
					NOT = {
						any_scope_state = {
							is_in_geographic_region = geographic_region_africa
						}
					}
					NOT = {
						any_subject_or_below = {
							is_in_geographic_region = geographic_region_africa
						}
					}
				}
				add = -40
			}
		}
	}
	option = {
		name = africa_colonial_events.8.b
		add_modifier = {
			name = reject_colonialism
			days = 1825 # 5 years
		}
		ai_chance = {
			base = 10
			modifier = { # see option a
				trigger = {
					NOT = {
						any_scope_state = {
							is_in_geographic_region = geographic_region_africa
						}
					}
					NOT = {
						any_subject_or_below = {
							is_in_geographic_region = geographic_region_africa
						}
					}
				}
				add = 40
			}
		}
	}
}

namespace = vatican

vatican.1 = { 	# the rights of the vatican city
	type = country_event
	placement = ROOT
	title = vatican.1.t
	desc = vatican.1.d
	#flavor = vatican.1.f # only show flavour if you are using the random target version.
	
	duration = 3
	
	event_image = {
		video = "southamerica_aristocrats"
	}
	
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		#owns_state = STATE_LAZIO
	}
	
	immediate = {
		ROOT = {
			save_scope_as = vatican_scope
		}
	}
	
	option = { # vatican city
		name = vatican.1.a
		set_variable = vatican_city_var
		create_country = {
			origin = root
			tag = PAP
			province = p:xB040A0
			on_created = {
				activate_law = law_type:law_theocracy
				activate_law = law_type:law_oligarchy
				activate_law = law_type:law_state_religion
				ig:ig_devout ?= {
					set_interest_group_name = ig_roman_curia
					remove_ideology = ideology_moralist
					add_ideology = ideology_papal_moralist
				}
				ig:ig_intelligentsia ?= {
					remove_ideology = ideology_anti_clerical
				}
				ig:ig_landowners ?= {
					set_interest_group_name = ig_roman_landowners
					remove_ideology = ideology_paternalistic
					add_ideology = ideology_papal_paternalistic
				}
				#gov_papacy somehow 
				
			}
		}
		create_diplomatic_pact = {  
			country = c:PAP 
			type = protectorate
		}
		hidden_effect = {
			every_country = {
				create_unidirectional_truce = {
					country = c:PAP
					months = 9999
				}
			}
		}
		
	}

	option = { # vatican lazio
		name = vatican.1.b
		set_variable = vatican_lazio_var
		create_country = {
			origin = root
			tag = PAP
			province = p:xB5726E
			on_created = {
				s:STATE_LAZIO = {
					set_owner_of_provinces = {
						country = c:PAP
						provinces = { xB040A0 x3040A0 xB0C0A0 xB5726E }
					}
				}
				activate_law = law_type:law_theocracy
				activate_law = law_type:law_oligarchy
				activate_law = law_type:law_state_religion
				ig:ig_devout ?= {
					set_interest_group_name = ig_roman_curia
					remove_ideology = ideology_moralist
					add_ideology = ideology_papal_moralist
				}
				ig:ig_intelligentsia ?= {
					remove_ideology = ideology_anti_clerical
				}
				ig:ig_landowners ?= {
					set_interest_group_name = ig_roman_landowners
					remove_ideology = ideology_paternalistic
					add_ideology = ideology_papal_paternalistic
				}
				#gov_papacy somehow 

			}
		}
		create_diplomatic_pact = {  
			country = c:PAP 
			type = protectorate
		}
		hidden_effect = {
			every_country = {
				create_unidirectional_truce = {
					country = c:PAP
					months = 9999
				}
			}
		}
	}

	option = { # on second thought, maybe not
		name = vatican.1.c
	}
}

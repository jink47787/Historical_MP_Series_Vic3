namespace = HMPS_guano

HMPS_guano.1 = { #HMPS TEST EVENT (console only): force guano depletion or discovery in one of your states. Fire with: event HMPS_guano.1
	type = country_event
	placement = root

	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	event_image = {
		video = "unspecific_steam_ship"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	duration = 3

	title = HMPS_guano.1.t
	desc = HMPS_guano.1.d

	immediate = {
		random_scope_state = {
			limit = {
				has_potential_resource = building_guano_mine
			}
			save_scope_as = guano_test_state
		}
	}

	option = { # Deplete
		name = HMPS_guano.1.a
		trigger = {
			exists = scope:guano_test_state
		}
		scope:guano_test_state = {
			force_resource_depletion = building_guano_mine
		}
	}

	option = { # Discover
		name = HMPS_guano.1.b
		trigger = {
			exists = scope:guano_test_state
		}
		scope:guano_test_state = {
			force_resource_discovery = building_guano_mine
		}
	}

	option = { # Do nothing
		name = HMPS_guano.1.c
		default_option = yes
	}
}

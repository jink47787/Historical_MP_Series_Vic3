namespace = hmps_obd

hmps_obd.1 = {
	type = country_event
	hidden = yes

	immediate = {
		every_state = {
			limit = {
				OR = {
					state_region = s:STATE_IDAHO
					state_region = s:STATE_OREGON
					state_region = s:STATE_WASHINGTON
					state_region = s:STATE_BRITISH_COLUMBIA
				}
				owner = c:USA
			}
			state_region = { add_claim = c:GBR }
		}
		trigger_event = {
			id = hmps_obd.2
			days = 4
		}
	}
}

hmps_obd.2 = {
	type = country_event
	hidden = yes

	immediate = {
		# Britain contests the Oregon Country once the USA holds land in both Washington and Oregon. In 1836 Oregon (ORG)
		# holds no part of Idaho, so Idaho and British Columbia only get war goals where the USA (or, for BC, ORG) holds a state.
		if = {
			limit = {
				exists = c:GBR
				exists = s:STATE_WASHINGTON.region_state:USA
				exists = s:STATE_OREGON.region_state:USA
			}
			c:GBR = {
				create_diplomatic_play = {
					name = hmps_oregon_border_dispute_war
					escalation = 0
					target_country = c:USA
					type = dp_return_state
					target_state = s:STATE_OREGON.region_state:USA
					annex_as_civil_war = no
					add_war_goal = {
						holder = c:GBR
						type = return_state
						target_state = s:STATE_WASHINGTON.region_state:USA
						primary_demand = no
					}
					add_war_goal = {
						holder = c:USA
						type = revoke_all_claims
						target_country = c:GBR
						primary_demand = no
					}
				}
				random_diplomatic_play = {
					limit = {
						initiator = c:GBR
						target = c:USA
					}
					if = {
						limit = {
							exists = s:STATE_IDAHO.region_state:USA
						}
						add_war_goal = {
							holder = c:GBR
							type = return_state
							target_state = s:STATE_IDAHO.region_state:USA
							primary_demand = no
						}
					}
					if = {
						limit = {
							exists = s:STATE_BRITISH_COLUMBIA.region_state:USA
						}
						add_war_goal = {
							holder = c:GBR
							type = return_state
							target_state = s:STATE_BRITISH_COLUMBIA.region_state:USA
							primary_demand = no
						}
					}
					else_if = {
						limit = {
							c:USA = {
								has_variable = hmps_oregon_trail_to_british_columbia
							}
							exists = s:STATE_BRITISH_COLUMBIA.region_state:ORG
						}
						add_war_goal = {
							holder = c:USA
							type = conquer_state
							target_state = s:STATE_BRITISH_COLUMBIA.region_state:ORG
							primary_demand = no
						}
					}
				}
			}
		}
	}
}

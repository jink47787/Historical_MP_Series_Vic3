hmps_obd.1 = {
	type = country_event
	hidden = yes

	immediate = {
		every_state = {
			limit = {
				OR = {
					state_region = s:STATE_IDAHO
					state_region = s:STATE_OREGON
					state_region = s:STATE_WASHINGTON
					state_region = s:STATE_BRITISH_COLUMBIA
				}
				owner = c:USA
			}
			add_claim = c:GBR
		}
		trigger_event = {
			id = hmps_obd.2
			days = 5
		}
	}
}

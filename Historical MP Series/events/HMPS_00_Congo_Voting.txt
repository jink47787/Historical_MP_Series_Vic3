# Setup: picks the nominees and gives the JE to the Great Powers, the nominees and the countries holding Congo states
congo_free_state.1 = {
	type = country_event
	placement = ROOT

	hidden = yes

	trigger = {
		#triggered by africa_colonial_events.1, 45 days after the first country joins the Scramble
	}

	immediate = {
		every_state = {
			limit = {
				hmps_is_congo_state = yes
			}
			set_variable = congo_state
		}
		# Nominee slots: 1 = Belgium, 2-6 = the highest-ranked eligible countries
		c:BEL ?= {
			if = {
				limit = {
					is_revolutionary = no
					has_revolution = no
					any_scope_state = {
						is_coastal = yes
					}
				}
				set_variable = {
					name = congo_nominee
					value = 1
				}
				set_global_variable = {
					name = congo_nominee_1
					value = THIS
				}
			}
		}
		hmps_congo_pick_nominee = { SLOT = 2 NOMINEE = congo_nominee_2 }
		hmps_congo_pick_nominee = { SLOT = 3 NOMINEE = congo_nominee_3 }
		hmps_congo_pick_nominee = { SLOT = 4 NOMINEE = congo_nominee_4 }
		hmps_congo_pick_nominee = { SLOT = 5 NOMINEE = congo_nominee_5 }
		hmps_congo_pick_nominee = { SLOT = 6 NOMINEE = congo_nominee_6 }

		if = {
			limit = {
				any_country = {
					has_variable = congo_nominee
				}
			}
			# Land decisions take the first 2 months (60 days); voting runs 6 more months until the JE times out (240 days)
			set_global_variable = {
				name = congo_land_phase
				days = 60
			}
			set_global_variable = { name = congo_votes_0 value = 0 }
			set_global_variable = { name = congo_votes_1 value = 0 }
			set_global_variable = { name = congo_votes_2 value = 0 }
			set_global_variable = { name = congo_votes_3 value = 0 }
			set_global_variable = { name = congo_votes_4 value = 0 }
			set_global_variable = { name = congo_votes_5 value = 0 }
			set_global_variable = { name = congo_votes_6 value = 0 }
			set_global_variable = { name = congo_best_slot value = 0 }
			set_global_variable = { name = congo_best_votes value = 0 }
			# Shown in the JE status; counted down weekly by je_congo_conference
			set_global_variable = { name = congo_days_to_vote value = 60 }
			set_global_variable = { name = congo_days_left value = 240 }
			hmps_congo_update_nominees = yes
			every_country = {
				limit = {
					country_rank = rank_value:great_power
				}
				set_variable = congo_voter
			}
			# Decentralized nations get no JE and always cede
			every_country = {
				limit = {
					NOT = { is_country_type = decentralized }
					any_scope_state = {
						has_variable = congo_state
					}
				}
				set_variable = congo_landholder
			}
			every_country = {
				limit = {
					OR = {
						has_variable = congo_voter
						has_variable = congo_nominee
						has_variable = congo_landholder
					}
				}
				add_journal_entry = {
					type = je_congo_conference
				}
			}
		}
	}
}

# Result: sent by je_congo_conference when the votes are counted or the conference collapses
congo_free_state.2 = {
	type = country_event
	placement = ROOT

	title = {
		first_valid = {
			triggered_desc = {
				desc = congo_free_state.2.t_winner
				trigger = {
					has_variable = congo_winner
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.t_collapsed
				trigger = {
					has_global_variable = congo_conference_collapsed
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.t
				trigger = {
					always = yes
				}
			}
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				desc = congo_free_state.2.d_winner
				trigger = {
					has_variable = congo_winner
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.d_result
				trigger = {
					has_global_variable = congo_winner
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.d_no_winner
				trigger = {
					has_global_variable = congo_no_winner
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.d_we_blocked
				trigger = {
					has_variable = congo_blocker
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.d_blocked
				trigger = {
					has_global_variable = congo_blocker
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.d_no_nominees
				trigger = {
					always = yes
				}
			}
		}
	}
	flavor = {
		first_valid = {
			triggered_desc = {
				desc = congo_free_state.2.f_winner
				trigger = {
					has_variable = congo_winner
				}
			}
			triggered_desc = {
				desc = congo_free_state.2.f
				trigger = {
					always = yes
				}
			}
		}
	}

	event_image = {
		video = "europenorthamerica_capitalists_meeting"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_map.dds"

	duration = 3

	trigger = {
		#triggered by je_congo_conference
	}

	immediate = {
		if = {
			limit = {
				has_global_variable = congo_winner
			}
			global_var:congo_winner = {
				save_scope_as = congo_winner_scope
			}
		}
		if = {
			limit = {
				has_global_variable = congo_blocker
			}
			global_var:congo_blocker = {
				save_scope_as = congo_blocker_scope
			}
		}
	}

	# We won
	option = {
		name = congo_free_state.2.a
		default_option = yes
		trigger = {
			has_variable = congo_winner
		}
		custom_tooltip = {
			text = congo_free_state.2.a.t
		}
	}
	# Someone else won
	option = {
		name = congo_free_state.2.b
		default_option = yes
		trigger = {
			has_global_variable = congo_winner
			NOT = { has_variable = congo_winner }
		}
	}
	# The Great Powers voted to leave the region unorganized
	option = {
		name = congo_free_state.2.c
		default_option = yes
		trigger = {
			has_global_variable = congo_no_winner
		}
	}
	# We refused to cede the Bas-Congo
	option = {
		name = congo_free_state.2.e
		default_option = yes
		trigger = {
			has_variable = congo_blocker
		}
	}
	# Someone else refused to cede the Bas-Congo
	option = {
		name = congo_free_state.2.g
		default_option = yes
		trigger = {
			has_global_variable = congo_blocker
			NOT = { has_variable = congo_blocker }
		}
	}
	# Every nominee withdrew or no longer exists
	option = {
		name = congo_free_state.2.h
		default_option = yes
		trigger = {
			has_global_variable = congo_no_nominees
		}
	}
}

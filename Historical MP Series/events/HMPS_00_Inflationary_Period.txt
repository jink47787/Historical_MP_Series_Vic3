namespace = inflationary_period

# HMPS Inflationary Period. Rolled monthly for market leaders whose market runs a surplus of 4% of GDP or more, at most once every
# 6 months (sf_monthly_check, common/scripted_effects/HMPS_specie_famine_effects.txt). The choice applies to the whole market for 24 months

inflationary_period.1 = {
	type = country_event
	placement = root

	title = inflationary_period.1.t
	desc = inflationary_period.1.d
	flavor = inflationary_period.1.f

	duration = 3

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		# triggered by sf_monthly_check
	}

	# Allow Inflation: +10% throughput, -15% wages
	option = {
		name = inflationary_period.1.a
		default_option = yes
		add_modifier = {
			name = ip_inflation_allowed
			days = 730
		}
		ip_apply_to_members = { MODIFIER = ip_inflation_allowed }
		ai_chance = {
			base = 50
		}
	}

	# Constrict the Money Supply: +25% import advantage, -25% export advantage, -10% throughput
	option = {
		name = inflationary_period.1.b
		add_modifier = {
			name = ip_money_supply_constricted
			days = 730
		}
		ip_apply_to_members = { MODIFIER = ip_money_supply_constricted }
		ai_chance = {
			base = 50
		}
	}
}

# Players elsewhere in the market: what the leader chose

			triggered_desc = {
				desc = inflationary_period.2.d
				trigger = { always = yes }
			}
		}
	}
	flavor = inflationary_period.2.f

	duration = 3

	event_image = {
		video = "europenorthamerica_rich_and_poor"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_scales.dds"

	trigger = {
		# triggered by ip_apply_to_members
	}

	option = {
		name = inflationary_period.2.a
		default_option = yes
	}
}

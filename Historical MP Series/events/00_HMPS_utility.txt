namespace = hmps_bot

# HMPS Balance of Trade: runs hmps_balance_of_trade_update_effect with the country as root.
# add_modifier's multiplier reads var: from root, so calling the effect inside every_country gave every country
# the root's penalty (or none on day 0). Fired for every country on day 0 (HMPS_balance_of_trade_start in
# common/on_actions/HMPS_on_action.txt) and weekly by HMPS_utility_je (common/journal_entries/z_HMPS_utility.txt).
hmps_bot.1 = {
	type = country_event
	hidden = yes

	immediate = {
		hmps_balance_of_trade_update_effect = yes
	}
}

# HMPS Bank Rate and Reserves: runs hmps_fx_rate_modifiers_effect with the country as root (same reason as hmps_bot.1).
# Fired for every country weekly by hmps_fx_weekly_effect, and for a market's countries when its owner changes the rate
# (hmps_fx_rate_change_effect). Both in common/scripted_effects/HMPS_scripted_effects.txt
hmps_bot.2 = {
	type = country_event
	hidden = yes

	immediate = {
		hmps_fx_rate_modifiers_effect = yes
	}
}

# Railroad Bubble: lists the incorporated states with a railway where transportation costs over 10% above base price
# (the states counted by je_railroad_bubble's expensive-transportation growth). Shown in je_railroad_bubble_reason
rr_bubble_expensive_transport_states_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				any_scope_state = {
					is_incorporated = yes
					any_scope_building = {
						is_building_type = building_railway
						level > 0
					}
					sg:transportation = {
						state_goods_pricier > 0.1
					}
				}
			}
			every_scope_state = {
				limit = {
					is_incorporated = yes
					any_scope_building = {
						is_building_type = building_railway
						level > 0
					}
					sg:transportation = {
						state_goods_pricier > 0.1
					}
				}
				custom_tooltip = "RR_BUBBLE_TRANSPORT_STATES_ENTRY" # for each print a line
			}
		}
		else = {
			custom_tooltip = "RR_BUBBLE_TRANSPORT_STATES_EMPTY"
		}
	}
}

# Congo Conference: lists every voting Great Power and its current vote. Shown in je_congo_conference's voting status
hmps_congo_votes_sgui = {
	scope = country

	effect = {
		every_country = {
			limit = {
				has_variable = congo_voter
			}
			if = {
				limit = { has_variable = congo_vote }
				if = {
					limit = { var:congo_vote = 0 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_0"
				}
				else_if = {
					limit = { var:congo_vote = 1 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_1"
				}
				else_if = {
					limit = { var:congo_vote = 2 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_2"
				}
				else_if = {
					limit = { var:congo_vote = 3 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_3"
				}
				else_if = {
					limit = { var:congo_vote = 4 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_4"
				}
				else_if = {
					limit = { var:congo_vote = 5 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_5"
				}
				else_if = {
					limit = { var:congo_vote = 6 }
					custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_6"
				}
			}
			else = {
				custom_tooltip = "HMPS_CONGO_VOTES_ENTRY_NONE"
			}
		}
	}
}

# Congo Conference: lists the powers that voted for ROOT (a nominee). Shown in congo_free_state.3
hmps_congo_our_voters_sgui = {
	scope = country

	effect = {
		if = {
			limit = {
				has_variable = congo_nominee
				any_country = {
					has_variable = congo_vote
					var:congo_vote = root.var:congo_nominee
				}
			}
			every_country = {
				limit = {
					has_variable = congo_vote
				}
				if = {
					limit = { var:congo_vote = root.var:congo_nominee }
					custom_tooltip = "HMPS_CONGO_OUR_VOTERS_ENTRY"
				}
			}
		}
		else = {
			custom_tooltip = "HMPS_CONGO_OUR_VOTERS_EMPTY"
		}
	}
}

# HMPS Bank Rate and Reserves: the market owner's bank rate coins in the Market panel (gui/market_panel.gui,
# gui/00_trade_partners_market_panel.gui), one per rate from 2% to 10%. Only the coins one or two points from the current rate are
# valid (no rate yet counts as 4%); a change locks the rate for 30 days (hmps_fx_rate_change_effect in HMPS_scripted_effects.txt). is_shown is only read by the coin bar

hmps_fx_rate_set_2_sgui = {
	scope = country

	is_shown = {
		NOT = { is_country_type = decentralized }
	}

	is_valid = {
		is_ai = no
		NOT = { is_country_type = decentralized }
		NOT = { has_variable = hmps_fx_rate_lock }
		OR = {
			NOT = { has_variable = hmps_fx_rate }
			var:hmps_fx_rate = 3
			var:hmps_fx_rate = 4
		}
	}

	effect = {
		hmps_fx_rate_change_effect = { RATE = 2 }
	}
}

sv_interest_reduction = {
    add = root.global_country_ranking
    subtract = 1
    divide = 15
    multiply = -0.10
    add = 0.10
}

# HMPS AI: weekly loan interest as a share of weekly income (0.05 = 5%)
# Principal = scaled_debt * credit; annual rate = add * ( 1 + mult ); weekly = annual / 52
hmps_ai_interest_share = {
	value = scaled_debt
	multiply = credit
	multiply = modifier:country_loan_interest_rate_add
	multiply = {
		value = 1
		add = modifier:country_loan_interest_rate_mult
		min = 0
	}
	divide = 52
	divide = {
		value = income
		min = 1
	}
}

# HMPS AI: how hard to push construction, from +1 (no interest) through 0 (interest = 15% of income) to -1 (30% or more)
# Safety net: at 40% debt-to-credit or more it is always -1, whatever the interest share says
hmps_ai_debt_factor = {
	value = 0.15
	subtract = hmps_ai_interest_share
	divide = 0.15
	min = -1
	max = 1
	if = {
		limit = {
			scaled_debt >= 0.4
		}
		value = -1
	}
}

# HMPS AI: extra wanted construction output on top of vanilla's 1 point per 5000 income (cap 200)
# Factor +1 doubles vanilla's income part; factor -1 cancels it exactly, so total wanted output never goes negative
# (a negative total could read as "far below critical" and make the AI build more)
hmps_ai_extra_construction_output = {
	value = fixed_income
	divide = 5000
	min = 0
	max = 200
	multiply = hmps_ai_debt_factor
}

# HMPS AI: +0.5 (borrow to build) at no interest, 0 at the 15% target, -0.5 at 30% or more
hmps_ai_construction_debt_balance = {
	value = hmps_ai_debt_factor
	multiply = 0.5
}

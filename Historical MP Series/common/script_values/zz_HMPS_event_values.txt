# HMPS Bank Rate and Reserves: estimated yearly capital flow (+ in, - out) if the market owner set the rate to N, for the coin tooltips
# (HMPS_FX_RATE_SET_N_TT). Uses last week's reference rate, which moves with the owner's own rate by its share of world GDP:
# flow = (N - rate) x (1 - GDP share) x 1% of GDP + (rate + pull - reference) x 1% of GDP. 0 if isolationist or before the first week
hmps_fx_flow_rate = {
	value = 4
	if = {
		limit = { has_variable = hmps_fx_rate }
		value = var:hmps_fx_rate
	}
}

hmps_fx_flow_per_point = {
	value = 0
	if = {
		limit = {
			has_variable = hmps_fx_gdp
			has_global_variable = hmps_fx_world_gdp
			global_var:hmps_fx_world_gdp > 0
			has_global_variable = hmps_fx_ref_rate
			NOT = { has_law_or_variant = law_type:law_isolationism }
		}
		value = 1
		subtract = {
			value = var:hmps_fx_gdp
			divide = global_var:hmps_fx_world_gdp
		}
		multiply = var:hmps_fx_gdp
		divide = 100
	}
}

hmps_fx_flow_now = {
	value = 0
	if = {
		limit = {
			has_variable = hmps_fx_gdp
			has_global_variable = hmps_fx_world_gdp
			global_var:hmps_fx_world_gdp > 0
			has_global_variable = hmps_fx_ref_rate
			NOT = { has_law_or_variant = law_type:law_isolationism }
		}
		value = hmps_fx_flow_rate
		if = {
			limit = { has_variable = hmps_fx_pull }
			add = var:hmps_fx_pull
		}
		subtract = global_var:hmps_fx_ref_rate
		multiply = var:hmps_fx_gdp
		divide = 100
	}
}

	add = hmps_fx_flow_now

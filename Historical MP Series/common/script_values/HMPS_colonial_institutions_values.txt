# HMPS: Colonial Affairs investment level of a state's owner, used as the multiplier for the Colonial Institutions state modifiers
# Root = state (HMPS_colonial_institutions_state_effects in HMPS_on_action.txt)
# Counts as 1 if the owner somehow has no Colonial Affairs institution (the laws need State-Sponsored Colonization, which provides it)
colonial_affairs_level_sv = {
	value = 1
	if = {
		limit = {
			owner ?= {
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 2
				}
			}
		}
		value = 2
	}
	if = {
		limit = {
			owner ?= {
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 3
				}
			}
		}
		value = 3
	}
	if = {
		limit = {
			owner ?= {
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 4
				}
			}
		}
		value = 4
	}
	if = {
		limit = {
			owner ?= {
				institution_investment_level = {
					institution = institution_colonial_affairs
					value >= 5
				}
			}
		}
		value = 5
	}
}

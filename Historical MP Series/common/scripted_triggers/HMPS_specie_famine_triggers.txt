# HMPS Specie Famine, Inflationary Period and Reserve Currency (common/journal_entries/HMPS_Specie_Famine.txt)

# Country scope. Leads its own market
sf_is_market_leader = {
	market.owner ?= this
}

# Country scope. Recognized: not an unrecognized or decentralized country
rc_is_recognized = {
	NOT = { is_country_type = unrecognized }
	NOT = { is_country_type = decentralized }
}

# Country scope. The Reserve Currency holder stops suffering Specie Famines once it has Mutual Funds
rc_immune_to_specie_famine = {
	has_global_variable = reserve_currency_holder
	this = global_var:reserve_currency_holder
	has_technology_researched = mutual_funds
}

# Country scope. Can a Specie Famine start here this month? (the roll itself is sf_famine_chance)
# Players only, market leaders only, one at a time per market, 5 years after the last one ended, not before 1837
sf_can_start_famine = {
	game_date >= 1837.1.1
	is_ai = no
	sf_is_market_leader = yes
	has_variable = hmps_bot_share
	NOT = { has_variable = sf_cooldown }
	NOT = { has_journal_entry = je_specie_famine }
	NOT = { has_journal_entry = je_specie_famine_member }
	NOT = { rc_immune_to_specie_famine = yes }
}

# Country scope. Can an Inflationary Period event fire here this month? Market leaders only, 6 months apart, never during a Specie Famine, not before 1837
ip_can_fire = {
	game_date >= 1837.1.1
	sf_is_market_leader = yes
	has_variable = hmps_bot_share
	var:hmps_bot_share >= 0.04
	NOT = { has_variable = ip_cooldown }
	# not while its market is in a Specie Famine
	NOT = { has_journal_entry = je_specie_famine }
	NOT = { has_journal_entry = je_specie_famine_member }
}

# Country scope (the market leader). Suspend Convertibility: at war, and this country has suffered at least 100,000 casualties in that war
sf_has_heavy_war_casualties = {
	any_scope_war = {
		num_country_casualties = {
			target = root
			value >= 100000
		}
	}
}

# Country scope. Can keep the Reserve Currency: alive, a Great Power, leads its own market, not bankrupt and not barred after losing it to bankruptcy.
# The holder also loses it after 6 months without a deficit (rc_surplus_months in rc_monthly_update)
rc_can_hold = {
	is_country_alive = yes
	rc_is_recognized = yes
	country_rank >= rank_value:great_power
	sf_is_market_leader = yes
	in_default = no
	NOT = { has_variable = rc_barred }
}

# Country scope. Its market runs a balance of trade deficit (hmps_bot_balance is the whole market's, see HMPS_specie_famine_values.txt)
rc_has_deficit = {
	has_variable = hmps_bot_balance
	var:hmps_bot_balance < 0
}

# Country scope. Its market does not run a deficit. Not the opposite of rc_has_deficit: a country with no figure yet is neither
rc_has_no_deficit = {
	has_variable = hmps_bot_balance
	var:hmps_bot_balance >= 0
}

# Country scope. Can take the Reserve Currency (challenge the holder, or claim it while vacant): could hold it
# (so a Great Power), and its market runs a deficit
rc_can_take = {
	rc_can_hold = yes
	rc_has_deficit = yes
}

# Country scope. Listed among the top importers (rc_rank_top_importers): the holder and every country that can take it
rc_is_listed = {
	OR = {
		rc_can_take = yes
		AND = {
			has_global_variable = reserve_currency_holder
			this = global_var:reserve_currency_holder
		}
	}
}

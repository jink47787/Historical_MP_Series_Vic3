# Scramble for Africa: can this country be offered africa_colonial_events.8 (free Malaria Prevention to join the Scramble)?
# Same colony test as je_scramble_for_africa; quinine is the 1.14 tech before Civilizing Mission
hmps_africa_scramble_offer_eligible = {
	is_country_type = recognized
	OR = {
		any_scope_state = {
			is_in_geographic_region = geographic_region_africa
			is_incorporated = no
			is_treaty_port = no
		}
		any_subject_or_below = {
			is_in_geographic_region = geographic_region_africa
		}
	}
	has_technology_researched = quinine
	NOT = { has_technology_researched = malaria_prevention }
	OR = {
		has_law_or_variant = law_type:law_colonial_resettlement
		has_law_or_variant = law_type:law_colonial_exploitation
	}
}

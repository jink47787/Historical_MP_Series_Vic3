on_monthly_pulse_country = {
	on_actions = {
		HMPS_India_Primary_Nerf
	}
}

HMPS_India_Primary_Nerf = { # Makes it so india only gets Bengali or Hindustani primary
	effect  = {
		root = {
			if = {
				limit = {
					c:BHT ?= this
				}
				
				every_primary_culture = {
					limit = {
						NOT = { this = cu:bengali }
						NOT = { this = cu:hindustani }
					}
					save_temporary_scope_as = primary_culture_to_remove
					PREV ?= {
						remove_primary_culture = scope:primary_culture_to_remove
					}
				}
				
				if = {
					limit = {
						country_has_primary_culture = cu:bengali
						country_has_primary_culture = cu:hindustani
					}
					if = {
						limit = {
							capital.state_region = {
								is_homeland = cu:bengali
							}
						}
						remove_primary_culture = cu:hindustani
					}
					else = {
						remove_primary_culture = cu:bengali
					}
				}
				if = {
					limit = {
						NOT = { country_has_primary_culture = cu:bengali }
						NOT = { country_has_primary_culture = cu:hindustani }
					}
					add_primary_culture = cu:hindustani
				}
			}
		}
	}
}

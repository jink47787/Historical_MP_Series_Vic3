# HMPS Specie Famine, Inflationary Period and Reserve Currency (common/scripted_effects/HMPS_specie_famine_effects.txt).
# The rolls and the Reserve Currency run monthly; a Specie Famine itself runs weekly from its journal entry

on_monthly_pulse_country = {
	on_actions = {
		HMPS_specie_famine_monthly
	}
}

on_monthly_pulse = {
	on_actions = {
		HMPS_reserve_currency_monthly
	}
}

on_game_started_after_lobby = {
	on_actions = {
		HMPS_reserve_currency_start
	}
}

# Deficit months, the Specie Famine roll and the Inflationary Period roll, for market leaders
HMPS_specie_famine_monthly = {
	effect = {
		sf_monthly_check = yes
	}
}

# Reserve Currency: loss, challenges, vacancy and the tiered modifiers
HMPS_reserve_currency_monthly = {
	effect = {
		rc_monthly_update = yes
	}
}

# Britain starts as the Reserve Currency
HMPS_reserve_currency_start = {
	effect = {
		if = {
			limit = { NOT = { has_global_variable = rc_initialized } }
			rc_initialize = yes
		}
	}
}

# This is done because it is not cool when ALSACE has a revolt to join swiss
# also seems justifable larp wise given that baden (linguistically similar) is considered to be south_german

on_game_started_after_lobby = {
	on_actions = {
		HMPS_alsace_becomes_south_german
		HMPS_spanish_pops_become_spanish
	}
}

HMPS_alsace_becomes_south_german = {
	effect = {
		s:STATE_ALSACE_LORRAINE = {
			remove_homeland = cu:alemannic
			every_scope_state = {
				add_acceptance = { culture = cu:south_german value = 112 } ## Will decay to +10 over 34 years, after that it will be too small to have an impact, [for more discussion See: "Language Discrimination" post by "stormbreaker2030" in "hmps-feedback" from August 2026]
				every_scope_pop = {
					limit = {
						culture = cu:alemannic
					}
					change_pop_culture = {
						target = cu:south_german
						value = 0.5
					}
				}
			}
		}
	}
}

# Unlocking ERA "speedbump" techs once x amount of people fully complete an era of a tech tree [x = 7 currently]

#ERA 2
unified_tech_unlocked_production_era_2 = {
	effect = {
		if = {
			limit = {
				NOT = {
					has_global_variable = era_2_production_unlocked
				}
				any_country = {
					count >= 7
					is_country_type = recognized
					has_technology_researched = tech_block_production_military_era_2
					has_technology_researched = tech_block_production_society_era_2
				}
			}
			every_country = {
				limit = {
					is_country_type = recognized
					country_rank >= rank_value:minor_power
				}
				add_technology_researched = tech_block_production_military_era_2
				add_technology_researched = tech_block_production_society_era_2
			}
			set_global_variable = era_2_production_unlocked
		}
	}
}

#For when a nation becomes Recognized and can catch up on Era Unlocks if possible / For save game compatibility / Backup for if the previous on_actions dont give out the techs to some nations
unified_tech_era_unlock_yearly_check = {
	effect = {
		if = {
			limit = {
				is_country_type = recognized
				country_rank >= rank_value:minor_power
			}
			if = { #Unlocks ERA 2 Production
				limit = {
					has_global_variable = era_2_production_unlocked
					NOT = {
						has_technology_researched = tech_block_production_military_era_2
						has_technology_researched = tech_block_production_society_era_2
					}
				}
				add_technology_researched = tech_block_production_military_era_2
				add_technology_researched = tech_block_production_society_era_2
			}
			if = { #Unlocks ERA 3 Production
				limit = {
					has_global_variable = era_3_production_unlocked
					NOT = {
						has_technology_researched = tech_block_production_military_era_3
						has_technology_researched = tech_block_production_society_era_3
					}
				}
				add_technology_researched = tech_block_production_military_era_3
				add_technology_researched = tech_block_production_society_era_3
			}
			if = { #Unlocks ERA 4 Production
				limit = {
					has_global_variable = era_4_production_unlocked
					NOT = {
						has_technology_researched = tech_block_production_military_era_4
						has_technology_researched = tech_block_production_society_era_4
					}
				}
				add_technology_researched = tech_block_production_military_era_4
				add_technology_researched = tech_block_production_society_era_4
			}
			if = { #Unlocks ERA 5 Production
				limit = {
					has_global_variable = era_5_production_unlocked
					NOT = {
						has_technology_researched = tech_block_production_military_era_5
					}
				}
				add_technology_researched = tech_block_production_military_era_5
			}
			if = { #Unlocks ERA 2 Society
				limit = {
					has_global_variable = era_2_society_unlocked
					NOT = {
						has_technology_researched = tech_block_society_military_era_2
						has_technology_researched = tech_block_society_production_era_2
					}
				}
				add_technology_researched = tech_block_society_military_era_2
				add_technology_researched = tech_block_society_production_era_2
			}
			if = { #Unlocks ERA 3 Society
				limit = {

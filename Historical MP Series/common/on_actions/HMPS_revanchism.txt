on_revolution_start = {
	on_actions  = {
		HMPS_revolution_inherit_revanchism_values 
	}
}

### The Journal Entry itself as well as any simple variables related to it will get inherited after a revolution, however the variable lists are not inherited automatically and need to be done manually via this on action
HMPS_revolution_inherit_revanchism_values = {
	effect  = {
		root = {
			### Held States
			every_in_list = {
				variable = revanchism_core_states_which_have_been_owned_at_one_point
				scope:target = {
					add_to_variable_list = {
						name = revanchism_core_states_which_have_been_owned_at_one_point
						target = prev
					}
				}
			}
			
			### Claimed Cultures
			every_in_list = {
				variable = je_hmps_revanchism_extra_cultures
				scope:target = {
					add_to_variable_list = {
						name = je_hmps_revanchism_extra_cultures
						target = prev
					}
				}
			}
			
			every_in_list = {
				variable = je_hmps_revanchism_extra_cultures_50
				scope:target = {
					add_to_variable_list = {
						name = je_hmps_revanchism_extra_cultures_50
						target = prev
					}
				}
			}
			
			every_in_list = {
				variable = je_hmps_revanchism_extra_cultures_25
				scope:target = {
					add_to_variable_list = {
						name = je_hmps_revanchism_extra_cultures_25
						target = prev
					}
				}
			}
			
			every_in_list = {
				variable = je_hmps_revanchism_extra_cultures_10
				scope:target = {
					add_to_variable_list = {
						name = je_hmps_revanchism_extra_cultures_10
						target = prev
					}
				}
			}
		}
	}
}

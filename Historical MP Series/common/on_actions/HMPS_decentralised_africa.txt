on_game_started_after_lobby = { #on_game_started = {
	on_actions = {
		HMPS_decentralise_africa
	}
}


HMPS_decentralise_africa = {
	effect = {
		every_country_in_africa = {
			limit = {
				has_game_rule = HMPS_game_rule_decentralized_africa_enabled
				is_country_type = unrecognized
				any_primary_culture = {
					has_discrimination_trait_group = heritage_group_african
				}
				
				# Excluding a country which a player has already picked
				is_player = no
				
				# Excluding a countries which own any states outside of Africa
				NOT = {
					any_scope_state = {
						NOT = {
							is_in_geographic_region = geographic_region_africa
						}
					}
				}
				
				#Excluding Subjects
				is_subject = no 
				
				# Excluding Ethiopia
				NOT = { 
					any_primary_culture = {
						has_discrimination_trait = heritage_abyssinian
					}
				}
				
				## Excluding Somalia
				NOT = {
					any_primary_culture = { cu:somali = this }
				}
				## Excluding Swahilia (East coast of Africa)
				NOT = {
					any_primary_culture = { cu:swahili = this }
				}
				NOT = { #Excluding a few specific countries
					OR = {
						this ?= c:WBL # Minor Dutch speaking african country
						this ?= c:PHL # Minor Dutch speaking african country
						this ?= c:ZUL # Zulu
						this ?= c:BST # Basutoland (Modern-day Lesotho)
						this ?= c:SWZ # Swaziland(Modern-day Eswatini)
						this ?= c:GZA # Gaza
						this ?= c:MAD # Madagascar
						this ?= c:SOK # Sokoto
						this ?= c:KON # Kongo
					}
				}
			}
			set_country_type = decentralized
			add_modifier = native_conscription_4
		}
	}
}

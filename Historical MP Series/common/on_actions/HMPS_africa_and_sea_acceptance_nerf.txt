on_monthly_pulse_country = {
	on_actions = {
		HMPS_africa_acceptance_nerf
		HMPS_southeast_asia_acceptance_nerf
	}
}

HMPS_africa_acceptance_nerf = {
	effect  = {
		root = {
			if = {
				limit = {
					has_game_rule = HMPS_game_rule_africa_acceptance_nerf_enabled
					any_primary_culture = {
						has_discrimination_trait_group = heritage_group_african
						#NOT = { has_discrimination_trait = heritage_african_diaspora }
					}
					any_scope_state = {
						is_in_geographic_region = geographic_region_subsaharan_africa 
					}
					NOT = { has_modifier = HMPS_modifier_country_africa_acceptance_nerf }
				}
				add_modifier = HMPS_modifier_country_africa_acceptance_nerf
			}
			else_if = {
				limit = {
					has_modifier = HMPS_modifier_country_africa_acceptance_nerf
				}
				remove_modifier = HMPS_modifier_country_africa_acceptance_nerf
			}
		}
	}
}

					any_primary_culture = {
						has_discrimination_trait_group = heritage_group_southeast_asian
					}
					any_scope_state = {
						is_in_geographic_region = geographic_region_south_east_asia 
					}
					NOT = { has_modifier = HMPS_modifier_country_southeast_asia_acceptance_nerf }
				}
				add_modifier = HMPS_modifier_country_southeast_asia_acceptance_nerf
			}
			else_if = {
				limit = {
					has_modifier = HMPS_modifier_country_southeast_asia_acceptance_nerf
				}
				remove_modifier = HMPS_modifier_country_southeast_asia_acceptance_nerf
			}
		}
	}
}

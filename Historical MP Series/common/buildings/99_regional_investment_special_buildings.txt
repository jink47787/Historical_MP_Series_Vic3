REPLACE_OR_CREATE:building_opium_plantation = {
	building_group = bg_plantations
	icon = "gfx/interface/icons/building_icons/opium_plantation.dds"
	required_construction = construction_cost_low

	unlocking_technologies = {
		enclosure
	}

	city_type = farm
	levels_per_mesh = 50

	production_method_groups = {
		pmg_base_building_opium_plantation
		pmg_train_automation_building_opium_plantation
	}

	can_build_government = {
		AND = {
			owner = { NOT = { is_banning_goods = g:opium } }
			regional_investment_building_allowed = yes
		}
	}
	can_build_private = {
		AND = {
			owner = { NOT = { is_banning_goods = g:opium } }
			regional_investment_building_allowed = yes
		}
	}
	ownership_type = self

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_plantations.dds"

	ai_value = {
		value = 1000

		if = { # British dictates
			limit = {
				owner = {
					has_journal_entry = je_british_dictate_plantations
					var:dictate_plantation_type ?= 1
				}
			}
			add = 2000
		}

		if = { # Colonisation of Korea
			limit = {
				owner ?= {
					c:JAP ?= this
					has_journal_entry = je_colonize_korea
				}
				is_in_geographic_region = geographic_region_korea
			}
			add = 2000
		}
	}
}

REPLACE_OR_CREATE:building_power_bloc_statue = {
	building_group = bg_monuments

	city_type = city

	levels_per_mesh = 1

	expandable = no

	statue = yes

	production_method_groups = {
		pmg_base_building_power_bloc_statue
	}

	icon = "gfx/interface/icons/building_icons/building_powerblock_statue.dds"

	required_construction = construction_cost_monument

	can_build_government = {
		AND = {
			always = yes
			regional_investment_building_allowed = yes
		}
	}

	can_build_private = {
		always = no
	}

	potential = {
		owner ?= {
			is_in_power_bloc = yes
		}
		has_dlc_feature = power_bloc_features
	}

	ai_value = {
		value = 0

		if = {
			limit = {
				is_incorporated = yes
				state_population >= 500000
			}

			add = {
				value = state_population
				divide = 2500
				max = 5000
			}
		}

		if = {
			limit = { is_capital = yes }
			add = 2500
		}

		owner = {
			if = {
				limit = { is_power_bloc_leader = yes }
				multiply = 2.0
			}

			if = {
				limit = { country_rank < rank_value:major_power }
				multiply = 0.0
			}
			else_if = {
				limit = { country_rank < rank_value:great_power }
				multiply = 0.5
			}
		}
	}

	background = "gfx/interface/icons/building_icons/backgrounds/building_panel_bg_monuments.dds"
}

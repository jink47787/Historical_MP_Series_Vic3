
### AFRICA
## SOUTHERN AFRICA
REPLACE_OR_CREATE:je_divide_and_rule_sotho_button = {
	name = "je_divide_and_rule_sotho_button"
	desc = "je_divide_and_rule_reason"
	visible = { OR = {
		capital.region = sr:region_southern_africa
		capital.state_region = { is_homeland = cu:sotho } }
	}
	selected = { has_modifier = secondary_culture_sotho }
	possible = {
		NOT = { has_modifier = modifier_has_shown_colonial_favouritism }
		has_technology_researched = civilizing_mission
		average_sol_for_culture = { target = cu:sotho	value >= 10 }
		any_scope_state = { state_region = { is_homeland = cu:sotho } }
	}
	effect = {
		add_modifier = secondary_culture_sotho
		add_modifier = modifier_has_shown_colonial_favouritism
		every_scope_culture = {
			limit = { shares_heritage_trait_group_with_any_primary_culture = prev }
			prev = { add_radicals = { culture = prev		value = 0.2 } }
		}
	}
	ai_chance = { value = 100 }
}

### CONGO CONFERENCE (je_congo_conference, events/HMPS_00_Congo_Voting.txt)
## Land decisions, first 60 days. AI landholders always cede
hmps_congo_hold_bas_congo_button = {
	name = "hmps_congo_hold_bas_congo_button"
	desc = "hmps_congo_hold_bas_congo_button_desc"
	visible = {
		has_variable = congo_landholder
		has_state_in_state_region = STATE_BAS_CONGO
	}
	possible = {
		custom_tooltip = {
			text = hmps_congo_land_phase_tt
			has_global_variable = congo_land_phase
		}
	}
	effect = {
		set_variable = congo_kept_land
		set_variable = congo_blocker
		set_global_variable = {
			name = congo_blocker
			value = ROOT
		}
		set_global_variable = congo_conference_collapsed
		custom_tooltip = hmps_congo_collapse_tt
		add_modifier = {
			name = resisted_congo_state
			months = 120
		}
		change_infamy = 40
		custom_tooltip = {
			text = hmps_congo_relations_30_tt
			every_country = {
				limit = {
					country_rank >= rank_value:major_power
					NOT = { THIS = ROOT }
				}
				change_relations = {
					country = ROOT
					value = -30
				}
			}
		}
	}
	ai_chance = { value = 0 }
}

hmps_congo_keep_states_button = {
	name = "hmps_congo_keep_states_button"
	desc = "hmps_congo_keep_states_button_desc"
	visible = {
		has_variable = congo_landholder
		NOT = { has_state_in_state_region = STATE_BAS_CONGO }
	}
	possible = {
		custom_tooltip = {
			text = hmps_congo_land_phase_tt
			has_global_variable = congo_land_phase
		}
		custom_tooltip = {
			text = hmps_congo_land_kept_tt
			NOT = { has_variable = congo_kept_land }
		}
		any_scope_state = {
			has_variable = congo_state
		}
	}
	selected = {
		has_variable = congo_kept_land
	}
	effect = {
		set_variable = congo_kept_land
		custom_tooltip = {
			text = hmps_congo_keep_states_tt
			every_scope_state = {
				limit = {
					has_variable = congo_state
				}
				set_variable = greedy_power
			}
		}
		add_modifier = {
			name = resisted_congo_state
			months = 120
			multiplier = 0.5
		}
		change_infamy = 20
		custom_tooltip = {
			text = hmps_congo_relations_15_tt
			every_country = {
				limit = {
					country_rank >= rank_value:major_power
					NOT = { THIS = ROOT }
				}
				change_relations = {
					country = ROOT
					value = -15
				}
			}
		}
	}
	ai_chance = { value = 0 }
}

hmps_congo_withdraw_button = {
	name = "hmps_congo_withdraw_button"
	desc = "hmps_congo_withdraw_button_desc"
	visible = {
		OR = {
			has_variable = congo_nominee
			has_variable = congo_withdrew
		}
	}
	possible = {
		custom_tooltip = {
			text = hmps_congo_land_phase_tt
			has_global_variable = congo_land_phase
		}
		custom_tooltip = {
			text = hmps_congo_is_nominee_tt
			has_variable = congo_nominee
		}
	}
	selected = {
		has_variable = congo_withdrew
	}
	effect = {
		remove_variable = congo_nominee
		set_variable = congo_withdrew
		custom_tooltip = hmps_congo_withdraw_tt
		hmps_congo_update_nominees = yes
	}
	ai_chance = { value = 0 }
}

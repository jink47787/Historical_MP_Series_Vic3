hmps_start_oregon_settle_bc = {
	name = "hmps_start_oregon_settle_bc_button"
	desc = "hmps_start_oregon_settle_bc_button_desc"

	visible = { 
		NOT = {
			has_variable = hmps_oregon_trail_to_british_columbia
		}
	}

	possible = {
		custom_tooltip = {
			text = hmps_settle_bc_tt

			NOT = {
				has_variable = hmps_oregon_trail_to_british_columbia
			}
			NOT = {
				has_variable = hmps_usa_decision_cooldown
			}
		}
	}

	effect = {
		ROOT = {
			set_variable = hmps_oregon_trail_to_british_columbia
		}
		set_variable = {
			name = hmps_usa_decision_cooldown
			months = 6
		}
		hidden_effect = {
			s:STATE_BRITISH_COLUMBIA.region_state:ORG = {
				if = {
					limit = {
						NOT = {
							any_scope_pop = {
								has_pop_culture = yankee
							}
						}
					}
					create_pop = {			
						culture = yankee
						size = 100
					}
				}
			}
		}
	}
}

hmps_survey_the_oregon_trail = {
	name = "hmps_survey_the_oregon_trail_button"
	desc = "hmps_survey_the_oregon_trail_button_desc"

	visible = { 
		NOT = {
			has_global_variable = oregon_trail_mapped
		}
		NOT = {  has_variable = on_expedition }
		OR = {
			is_ai = no
			NOT = { has_variable = ai_expedition_cooldown }
		}
	}

	possible = {
		is_at_war = no
		has_technology_researched = quinine
		any_scope_character = {
			OR = {
				has_role_of_type = general
				has_role_of_type = admiral
			}
			count >= 3
		}
	}

	effect = {
		ROOT = {
			set_variable = hmps_surveying_oregon_trail_usa
		}
		if = {
			limit = {
				is_player = yes
			}
			trigger_event = { id = expedition_events.200 popup = yes }
			set_variable = on_expedition
		}
		else = {
			set_variable = AI_EXPEDITION_WAMERICA
			ai_expedition_simulator_start = yes
		}
		hidden_effect = {
			set_variable = {
				name = ai_expedition_cooldown
				value = yes
				days = 1825
			}
		}
	}
}

# Manifest Destiny Buttons
hmps_initiate_manifest_destiny = {
	name = "hmps_initiate_manifest_destiny"
	desc = "hmps_initiate_manifest_destiny_desc"

	visible = {
		always = yes
	}

	possible = {
		custom_tooltip = {
			text = hmps_brd

			has_global_variable = buenaventura_river_disproven
		}
		custom_tooltip = {
			text = hmps_gslm

			has_global_variable = great_salt_lake_mapped
		}
	}

	effect = {
		if = {
			limit = {
				owns_entire_state_region = STATE_CALIFORNIA
				owns_entire_state_region = STATE_NEVADA
				owns_entire_state_region = STATE_UTAH
				owns_entire_state_region = STATE_ARIZONA
				owns_entire_state_region = STATE_NEW_MEXICO
				owns_entire_state_region = STATE_TEXAS
			}
			trigger_event = { id = manifest_destiny.1 }
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
				multiplier = -1
			}
		}
		if = {
			limit = {
				exists = c:MEX
			}
			create_incident = {
				country = c:MEX
				value = 20
			}
			change_relations = {
				country = c:MEX
				value = -50
			}
		}
		s:STATE_CALIFORNIA = {
			if = {
				limit = {
					any_scope_state = {
						 NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_NEVADA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_UTAH = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_ARIZONA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_NEW_MEXICO = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_TEXAS = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		set_variable = manifested_destiny
	}
}

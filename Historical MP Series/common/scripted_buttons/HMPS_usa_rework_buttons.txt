hmps_survey_the_oregon_trail = {
	name = "hmps_survey_the_oregon_trail_button"
	desc = "hmps_survey_the_oregon_trail_button_desc"

	visible = { 
		NOT = {
			has_global_variable = oregon_trail_mapped
		}
		NOT = {  has_variable = on_expedition }
		OR = {
			is_ai = no
			NOT = { has_variable = ai_expedition_cooldown }
		}
	}

	possible = {
		is_at_war = no
		has_technology_researched = quinine
		any_scope_character = {
			OR = {
				has_role_of_type = general
				has_role_of_type = admiral
			}
			count >= 3
		}
	}

	effect = {
		ROOT = {
			set_variable = hmps_surveying_oregon_trail_usa
		}
		if = {
			limit = {
				is_player = yes
			}
			trigger_event = { id = expedition_events.200 popup = yes }
			set_variable = on_expedition
		}
		else = {
			set_variable = AI_EXPEDITION_WAMERICA
			ai_expedition_simulator_start = yes
		}
		hidden_effect = {
			set_variable = {
				name = ai_expedition_cooldown
				value = yes
				days = 1825
			}
		}
	}
}

	}
}

hmps_take_oregon_by_force = {
	name = "hmps_take_oregon_by_force_button"
	desc = "hmps_take_oregon_by_force_button_desc"

	visible = { 
		NOT = {
			any_state = {
				filter = {
					OR = {
						state_region = s:STATE_IDAHO
						state_region = s:STATE_OREGON
						state_region = s:STATE_WASHINGTON
						state_region = s:STATE_BRITISH_COLUMBIA
					}
				}
				count >= 3
				OR = {
					owner = c:USA
					owner = {
						OR = {
							is_subject_of = c:USA
							is_indirect_subject_of = c:USA
						}
					}
				}
			}
		}
	}

	possible = {
		has_variable = hmps_sent_oregon_demand
		NOT = {
			has_variable = hmps_sent_oregon_demand_countdown
			ROOT = {
				has_truce_with = c:GBR
			}
		}
		NOT = {
			any_scope_treaty = {
				binds = c:GBR
				any_scope_article = {
					count >= 1
					OR = {
						has_type = defensive_pact
						has_type = alliance
					}
				}
			}
		}			
	}

	effect = {
		every_state = {
			limit = {
				OR = {
					state_region = s:STATE_IDAHO
					state_region = s:STATE_OREGON
					state_region = s:STATE_WASHINGTON
					state_region = s:STATE_BRITISH_COLUMBIA
				}
				has_claim_by = c:USA
			}
			set_state_owner = c:USA
			add_claim = c:GBR
		}
		c:GBR = {
			create_diplomatic_play = {
				name = hmps_oregon_border_dispute_war
				escalation = 0
				target_country = c:USA
				type = dp_return_state
				annex_as_civil_war = no
				if = {
					limit = {
						s:STATE_IDAHO = {
							owner = c:USA
						}
					}
					add_war_goal = {
						holder = c:GBR
						war_goal = return_state
						target_state = s:STATE_IDAHO
					}
					add_war_goal = {
						holder = c:USA
						war_goal = revoke_claim
						target_state = s:STATE_IDAHO
					}
				}
				if = {
					limit = {
						s:STATE_OREGON = {
							owner = c:USA
						}

					add_war_goal = {
						holder = c:GBR

					}

					}
				}
				if = {

hmps_initiate_manifest_destiny = {
	name = "hmps_initiate_manifest_destiny"
	desc = "hmps_initiate_manifest_destiny_desc"

	visible = {
		always = yes
	}

	possible = {
		has_global_variable = buenaventura_river_disproven
		has_global_variable = great_salt_lake_mapped
	}

	effect = {
		if = {
			limit = {
				owns_entire_state_region = STATE_CALIFORNIA
				owns_entire_state_region = STATE_NEVADA
				owns_entire_state_region = STATE_UTAH
				owns_entire_state_region = STATE_ARIZONA
				owns_entire_state_region = STATE_NEW_MEXICO
				owns_entire_state_region = STATE_TEXAS
			}
			trigger_event = { id = manifest_destiny.1 }
		}
		ig:ig_landowners ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
				multiplier = -1
			}
		}
		if = {
			limit = {
				exists = c:MEX
			}
			create_incident = {
				country = c:MEX
				value = 20
			}
		}
		s:STATE_CALIFORNIA = {
			if = {
				limit = {
					any_scope_state = {
						 NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_NEVADA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_UTAH = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_ARIZONA = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_NEW_MEXICO = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		s:STATE_TEXAS = {
			if = {
				limit = {
					any_scope_state = {
						NOT = {
							owner = ROOT
						}
					}
				}
				add_claim = ROOT
			}
		}
		set_variable = manifested_destiny
	}
}

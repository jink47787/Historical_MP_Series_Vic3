hmps_revanchism_gain_claims = { #Genuinly cant figure out why the make claim buttons wont work
	name = "hmps_revanchism_gain_claims_button"
	desc = "hmps_revanchism_gain_claims_button_desc"

	visible = { 
		is_country_type = recognized
		any_state = {
			filter = {
				is_homeland_of_country_cultures = ROOT
				NOR = {
					owner = ROOT
					owner = {
						OR = {
							is_subject_of = ROOT
							is_indirect_subject_of = ROOT
						}
					}
				}
			}
			count >= 1
			has_modifier = modifier_hmps_revanchism_target
			NOT = {
				has_claim_by = ROOT
			}
		}
	}

	possible = {
		always = yes
	}

	effect = {
		every_state = {
			limit = {
				is_homeland_of_country_cultures = ROOT
				#NOR = {
				#	owner = ROOT
				#	owner = {
				#		OR = {
				#			is_subject_of = ROOT
				#			is_indirect_subject_of = ROOT
				#		}
				#	}
				#}
				has_modifier = modifier_hmps_revanchism_target
				NOT = {
					has_claim_by = ROOT
				}
			}
			add_claim = ROOT
		}	
	}
}

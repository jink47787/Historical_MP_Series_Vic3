REPLACE:manifest_destiny = {
	is_shown = {
		always = no
	}

	possible = {
		has_technology_researched = nationalism
		OR = {
			has_interest_marker_in_region = sr:region_pacific_coast
			sr:region_great_plains = {
				any_scope_state = {
					percent >= 0.4
					owner = {
						OR = {
							this = root
							is_subject_of = root
						}
					}
				}
			}
		}
	}

	when_taken = {
		ig:ig_landowners ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
			}
		}
		ig:ig_intelligentsia ?= {
			add_modifier = {
				name = ig_manifest_destiny
				days = normal_modifier_time
				multiplier = -1
			}
		}
		if = {
			limit = {
				exists = c:MEX
			}
			create_incident = {
				country = c:MEX
				value = 20
			}
		}
		set_variable = manifested_destiny
	}

	ai_chance = {
		value = 10
	}	
}

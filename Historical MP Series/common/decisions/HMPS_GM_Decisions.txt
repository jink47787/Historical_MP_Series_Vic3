gm_player = {
	is_shown = {
		OR = {
			NOT = { has_global_variable = player_gm }
			AND = {
				has_game_rule = HMPS_game_rule_GM_tool_fail_safe_enabled
				global_country_ranking = 1
			}
		}
		is_player = yes
	}

	when_taken = {
		set_global_variable = player_gm
		every_country = {
			limit = {
				has_modifier = game_master
			}
			remove_modifier = game_master
		}
		add_modifier = {
			name = game_master
		}
	}
}

GM_Self_Action = {
	is_shown = {
		has_modifier = game_master
		NOT = { has_global_variable = GM_taking_action }
		NOT = { has_global_variable = single_player }
	}
	
	when_taken = {
		set_global_variable = {
			name = GM_taking_action
			days = 31
		}
		set_variable = { 
			name = actionee
			days = 95
		}
		trigger_event = {
			id = gm_actions.1
			days = 0
			popup = yes
		}
		every_country = {
			limit = {
				is_player = yes
			}
			post_notification = gm_self_action
		}
	}
}

# Fix (2026-10-12): is_shown used to need c:RUS to own all of Alaska. At the start Russia owns none of it: its subject
# ALK (the Russian-American Company) and the native ATB hold it, so the decision never appeared. It now accepts Russia
# or its subject ALK holding Alaska apart from native land; alaska.4 already finds ALK's overlord as the seller.
# Possible likewise needs Washington apart from native land instead of all of it.
# Delete this file if the Alaska Purchase decision is not needed.
usa_purchase_alaska = {
	is_shown = {
		exists = c:USA
		exists = c:RUS
		c:USA = ROOT
		is_subject = no
		NOT = {
			has_variable = alaska_purchase_var
		}
		c:RUS = {
			is_subject = no
			is_player = no
		}
		OR = {
			c:RUS = {
				hmps_holds_region_bar_natives = { REGION = STATE_ALASKA }
			}
			c:ALK ?= {
				is_subject_of = c:RUS
				hmps_holds_region_bar_natives = { REGION = STATE_ALASKA }
			}
		}
	}

	possible = {
		hmps_holds_region_bar_natives = { REGION = STATE_WASHINGTON }
		in_default = no
		c:RUS = {
			relations:root >= 0
		}
	}

	when_taken = {
		set_variable = {
			name = alaska_purchase_var
			value = yes
		}
		trigger_event = {
			id = alaska.4
			days = 0
			popup = yes
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = { gold_reserves > 3000000 }
			add = 10
		}
		if = {
			limit = { 
				gold_reserves > 0
				net_fixed_income > 100000
			}
			add = 10
		}
		if = {
			limit = {
				OR = {
					is_diplomatic_play_committed_participant = yes
					is_at_war = yes
				}
			}
			add = -1000
		}
	}
}

decision_call_election = {
	is_shown = {
		OR = {
			has_law_or_variant = law_type:law_census_voting
			has_law_or_variant = law_type:law_landed_voting
			has_law_or_variant = law_type:law_wealth_voting
			has_law_or_variant = law_type:law_census_voting 
			has_law_or_variant = law_type:law_universal_suffrage
			has_law_or_variant = law_type:law_single_party_state
		}
	}

	possible = {
		authority >= 100
		is_player = yes
		hidden_trigger = {
			NOT = { has_variable = election_cooldown }
		}
	}

	when_taken = {
		set_variable = {
			name = election_cooldown
			days = 40
		}
		add_modifier = {
			name = modifier_called_election_recently
			multiplier = 1
			years = 10
			is_decaying = yes
		}
		call_election = {
			months = 1
		}
		if = {
			limit = {
				NOT = {
					active_law:lawgroup_distribution_of_power ?= {
						has_amendment = amendment_type:amendment_tradition_of_free_elections
					}
				}
			}
			add_electoral_confidence = -0.10
		}
		custom_tooltip = {
			text = decision_call_election_radicals_tt
			every_scope_state = {
				limit = { is_incorporated = yes }
				add_radicals_in_state = {
					value = small_radicals
				}
			}
		}
	}

	ai_chance = {
		value = 100
	}
}

je_hmps_french_baby_bust = {
	icon = "gfx/interface/icons/invention_icons/unused/secularism.dds"
	
	group = je_group_historical_content
	
	modifiers_while_active = {
	}
	is_shown_when_inactive = {
		c:FRA ?= this
	}

	possible = {
		always = yes
	}
	
	immediate = {
		hidden_effect = {
			every_state_in_france = {
				limit = {
					NOT = { state_region = s:STATE_SAVOY }
					NOT = { state_region = s:STATE_ALSACE_LORRAINE }
					NOT = { state_region = s:STATE_BRITTANY }
				}
				add_modifier = modifier_hmps_french_baby_bust
				add_modifier = {
					name = modifier_hmps_french_baby_bust_early_game_fix			
					is_decaying = yes
					years = 30
				}
			}
		}
		if = {
			limit = {
				NOT = { has_global_variable = hmps_french_baby_bust_progress }
			}
			set_global_variable  = {
				name = hmps_french_baby_bust_progress
				value = 0
			}
		}
		set_variable = {
			name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
			value = 1
		}
	}

	complete = {
		scope:journal_entry = {
			is_goal_complete = yes
		}
	}

	on_complete = {
		ig:ig_devout ?= {
			set_ig_trait = ig_trait:ig_trait_be_fruitful_and_multiply
		}
		hidden_effect = {
			every_state_in_france_old = {
				limit = {
					has_modifier = modifier_hmps_french_baby_bust
				}
				remove_modifier = modifier_hmps_french_baby_bust
			}
		}
		remove_global_variable = hmps_french_baby_bust_progress
	}
	
	invalid = {
	}
	
	on_invalid = {
		hidden_effect = {
			every_state_in_france_old = {
				remove_modifier = modifier_hmps_french_baby_bust
			}
		}
		remove_global_variable = hmps_french_baby_bust_progress
	}

	current_value = {
		value = global_var:hmps_french_baby_bust_progress
	}

	goal_add_value = {
		value = 100
	}
	
	progressbar = yes

	on_weekly_pulse = { #We use weekly pulse because i dont want any degenerate stuff where people lower taxes only on the monthly pulse
		effect = {
			set_variable = {
				name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
				value = 1
			}
			
				
			if = { ### If "Confessional State"
				limit = {
					#tax_level <= low
					#has_law_or_variant = law_type:law_theocracy
					has_law_or_variant = law_type:law_state_religion
					has_law_or_variant = law_type:law_charitable_health_system 
					has_law_or_variant = law_type:law_religious_schools 
				}
				
				### SCHOOL AMENDMENT
				if = {
					limit = {
						NOT = {
							any_active_law = {
								has_amendment = amendment_type:hmps_french_baby_bust_religious_schools_amendment
							}
						}
					}
					active_law:lawgroup_education_system ?= { # Wealth Voting - Reform Act
						add_amendment = {
							type = hmps_french_baby_bust_religious_schools_amendment
							sponsor = PREV.ig:ig_devout
						}
					}
				}
				
				### HEALTH AMENDMENT
				if = {
					limit = {
						NOT = {
							any_active_law = {
								has_amendment = amendment_type:hmps_french_baby_bust_religious_health_amendment
							}
						}
					}
					active_law:lawgroup_health_system ?= { # Wealth Voting - Reform Act
						add_amendment = {
							type = hmps_french_baby_bust_religious_health_amendment
							sponsor = PREV.ig:ig_devout
						}
					}
				}
				
				### Tax Slider
				if = {
					limit = {
						tax_level = very_low
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 2
					}
				}
				if = {
					limit = {
						tax_level = low
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 1
					}
				}
				if = {
					limit = {
						tax_level = high
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = -1
					}
				}
				if = {
					limit = {
						tax_level = very_high
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = -2
					}
				}
				
				### General Tax law
				if = {
					limit = {
						OR = {
							has_law_or_variant = law_type:law_consumption_based_taxation
							has_law_or_variant = law_type:law_proportional_taxation
						}
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 1
					}
				}
				if = {
					limit = {
						OR = {
							has_law_or_variant = law_type:law_graduated_taxation
						}
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 2
					}
				}
				
				### Income Tax law
				if = {
					limit = {
						has_law_or_variant = law_type:law_no_income_taxes
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 1.5
					}
				}
				if = {
					limit = {
						has_law_or_variant = law_type:law_low_income_tax
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 0.5
					}
				}
				if = {
					limit = {
						has_law_or_variant = law_type:law_high_income_tax
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = -0.5
					}
				}
				
				### Tax Decree
				every_state_in_france = {
					limit = {
						owner = ROOT
						has_decree = decree_additional_taxes
					}
					prev = {
						change_variable = {
							name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
							add = -0.1
						}
					}
				}
				
				### Consuption Taxes
				if = {
					limit = {
						NOR = {
							has_consumption_tax = g:grain
							has_consumption_tax = g:clothes
							has_consumption_tax = g:wine
						}
					}
					change_variable = {
						name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
						add = 0.5
					}
				}
			}
			else = { ### Not a Confessional State: remove the amendments
				every_active_law = {
					every_scope_amendment = {
						limit = {
							OR = {
								type = amendment_type:hmps_french_baby_bust_religious_schools_amendment
								type = amendment_type:hmps_french_baby_bust_religious_health_amendment
							}
						}
						remove_amendment = yes
					}
				}
			}
			change_variable = {
				name = hmps_french_baby_bust_progress_cached_weekly_progress_rate
				min = 1
			}
			change_global_variable = {
				name = hmps_french_baby_bust_progress
				add = {
					value = var:hmps_french_baby_bust_progress_cached_weekly_progress_rate
					multiply = 0.01916 # Number means it goes down average of 1 per year
				}
			}
			
			every_state_in_france = {
				limit = {
					NOT = { state_region = s:STATE_SAVOY }
					NOT = { state_region = s:STATE_ALSACE_LORRAINE }
					NOT = { state_region = s:STATE_BRITTANY }
				}
				remove_modifier = modifier_hmps_french_baby_bust
				add_modifier = {
					name = modifier_hmps_french_baby_bust
					multiplier = {
						value = 100
						subtract = global_var:hmps_french_baby_bust_progress
						divide = 100
					}
				}
			}
			
			
				
		}
	}

	weight = 10000
	should_be_pinned_by_default_uninvolved_or_context = yes
	
	can_revolution_inherit = yes
}

REPLACE:je_manifest_destiny_mexico = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_usa_manifest_destiny

	scripted_button = hmps_initiate_manifest_destiny
	scripted_button = hmps_send_demand_for_mexican_states

	scripted_progress_bar = hmps_manifest_destiny_time_limit_bar
	progressbar = yes
	
	immediate = {
		ROOT = {
			set_variable = {
				name = hmps_manifest_destiny_completion_var
				value = 0
			}
		}
		c:MEX = {
			save_scope_as = hmps_mex_scope
		}
	}

	current_value = {
		value = root.var:hmps_manifest_destiny_completion_var
	}

	goal_add_value = {
		value = 60
	}

	complete = {
		scope:journal_entry = { is_goal_complete = yes }
	}

	on_complete = {
		trigger_event = { id = manifest_destiny.1 }
	}

	fail = {
		scope:journal_entry = {
			"scripted_bar_progress(hmps_manifest_destiny_time_limit_bar)" <= 0
		}
		NOT = {
			every_state = {
				limit = {
					OR = {
						state_region = s:STATE_CALIFORNIA
						state_region = s:STATE_NEVADA
						state_region = s:STATE_UTAH
						state_region = s:STATE_ARIZONA
						state_region = s:STATE_NEW_MEXICO
						state_region = s:STATE_TEXAS
					}
				}
				owner = c:USA
				OR = {
					is_incorporated = yes
					incorporation_progress >= 0.25
				}
			}
		}
	}
	
	on_fail = {
		every_state = {
			limit = {
				OR = {
					state_region = s:STATE_CALIFORNIA
					state_region = s:STATE_NEVADA
					state_region = s:STATE_UTAH
					state_region = s:STATE_ARIZONA
					state_region = s:STATE_NEW_MEXICO
					state_region = s:STATE_TEXAS
				}
				owner = c:USA
			}
			set_state_type = unincorporated
			remove_claim = c:USA
		}
	}

	on_monthly_pulse = {
		if = {
			limit = {
				every_state = {
					limit = {
						OR = {
							state_region = s:STATE_CALIFORNIA
							state_region = s:STATE_NEVADA
							state_region = s:STATE_UTAH
							state_region = s:STATE_ARIZONA
							state_region = s:STATE_NEW_MEXICO
							state_region = s:STATE_TEXAS
						}
					}
					owner = c:USA
					is_incorporated = yes
				}
			}
			ROOT = {
				change_variable = {
					name = hmps_manifest_destiny_completion_var
					add = 1
				}
			}
		}
	}

	weight = 100
		
	should_be_pinned_by_default_uninvolved_or_context = yes
}

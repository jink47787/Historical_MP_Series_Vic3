je_decembrist_sympathizers = {
	icon = "gfx/interface/icons/event_icons/event_portrait.dds"
	
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	possible = {
		always = no # HMPS: Decembrist Sympathizers paused; remove this block when re-enabling
	}

	immediate = {
		set_variable = {
			name = decembrist_elimination
			value = 0
		}
	}
	
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			NOT = { has_law_or_variant = law_type:law_no_police }
			OR = {
				has_law_or_variant = law_type:law_censorship
				has_law_or_variant = law_type:law_outlawed_dissent
			}
			NOT = {
				any_interest_group = {
					NOT = { has_ideology = ideology:ideology_liberal }
					ig_counts_as_marginal = no
					leader = {
						OR = {
							has_ideology = ideology:ideology_radical
							has_ideology = ideology:ideology_republican_leader
							has_ideology = ideology:ideology_reformer
						}
					}
				}
			}
			NOT = {
				any_scope_character = { 
					OR = {
						has_role = general
						has_role = admiral
						has_role = agitator
					}
					OR = {
						has_ideology = ideology:ideology_radical
						has_ideology = ideology:ideology_republican_leader
						has_ideology = ideology:ideology_reformer
					}
				}
			}
		}
		effect = {
			custom_tooltip = {
				text = liberals_fail_tt
			}
			add_modifier = {
				name = decembrist_suppression_success
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = { has_law_or_variant = law_type:law_monarchy }
				NOT = { has_law_or_variant = law_type:law_autocracy }
			}
		}
		effect = {
			add_modifier = {
				name = decembrist_suppression_failure
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		flags = { timeout }
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		effect = {
			add_modifier = {
				name = decembrist_suppression_failure
				days = very_long_modifier_time
				is_decaying = yes
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_no_police }
					OR = {
						has_law_or_variant = law_type:law_censorship
						has_law_or_variant = law_type:law_outlawed_dissent
					}
				}
				change_variable = {
					name = decembrist_elimination
					add = 1
				}
			}
		}
	}
	
	timeout = 5475

	current_value = {
		value = root.var:decembrist_elimination
	}

	goal_add_value = {
		value = 60
	}
	
	progressbar = yes
	
	can_revolution_inherit = yes

	weight = 5000
	should_be_pinned_by_default_uninvolved_or_context = yes
}

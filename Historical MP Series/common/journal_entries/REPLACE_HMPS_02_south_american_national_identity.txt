# HMPS: slows South American colonization until railways and the telegraph open the interior; rewards on completion
je_hmps_unconquered_interior = {
	icon = "gfx/interface/icons/event_icons/event_map.dds"

	group = je_group_internal_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		capital = {
			is_in_geographic_region = geographic_region_south_america
		}
		NOT = { is_country_type = decentralized }
		NOT = { has_global_variable = hmps_unconquered_interior_opened }
		NAND = {
			has_technology_researched = railways
			has_technology_researched = electric_telegraph
		}
		any_state = {
			is_in_geographic_region = geographic_region_latin_america
			has_claim_by = ROOT
			owner ?= { is_country_type = decentralized }
		}
	}

	possible = {
		capital = {
			is_in_geographic_region = geographic_region_south_america
		}
		NOT = { is_country_type = decentralized }
		NOT = { has_global_variable = hmps_unconquered_interior_opened }
		NAND = {
			has_technology_researched = railways
			has_technology_researched = electric_telegraph
		}
		any_state = {
			is_in_geographic_region = geographic_region_latin_america
			has_claim_by = ROOT
			owner ?= { is_country_type = decentralized }
		}
	}

	modifiers_while_active = {
		hmps_unconquered_interior
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				AND = {
					has_technology_researched = railways
					has_technology_researched = electric_telegraph
				}
				custom_tooltip = {
					text = hmps_unconquered_interior_claims_colonized_tt
					NOT = {
						any_state = {
							is_in_geographic_region = geographic_region_latin_america
							has_claim_by = ROOT
							owner ?= { is_country_type = decentralized }
						}
					}
				}
				custom_tooltip = { # HMPS: once one South American country opens its interior, all do
					text = hmps_unconquered_interior_opened_elsewhere_tt
					has_global_variable = hmps_unconquered_interior_opened
				}
			}
		}
		effect = {
			set_global_variable = hmps_unconquered_interior_opened
			add_modifier = {
				name = hmps_open_frontier
				years = 5
			}
		}
	}

	weight = 100

	should_be_pinned_by_default_uninvolved_or_context = yes
}

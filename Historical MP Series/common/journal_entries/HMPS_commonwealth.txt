je_commonwealth = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_global_international_situations
	#group = je_group_historical_content
	should_be_pinned_by_default_involved = yes
	should_be_pinned_by_default_uninvolved_or_context = yes	
	
	scripted_progress_bar = ast_commonwealth_progress_bar
	scripted_progress_bar = can_commonwealth_progress_bar
	scripted_progress_bar = saf_commonwealth_progress_bar
	
	should_be_involved = {
		OR = {
			c:GBR ?= this
			AND = {
				OR = {
					is_in_same_power_bloc = c:GBR
					is_subject_of = c:GBR
				}
				OR = {
					c:SAF ?= this
					country_has_primary_culture = cu:anglo_canadian
					country_has_primary_culture = cu:franco_canadian
					country_has_primary_culture = cu:boer
					country_has_primary_culture = cu:australian
				}
			}
		}
	}

	should_show_when_not_involved = {
		OR = {
			is_in_same_power_bloc = c:GBR
			is_subject_of = c:GBR
		}
	}
	is_shown_when_inactive = {
		this ?= c:GBR
	}

	possible = {
		this ?= c:GBR
		#has_technology_researched = nationalism
	}
	
	immediate = {
		set_global_variable = { name = commonwealth_progress_ast value = "scope:journal_entry.scripted_bar_progress(ast_commonwealth_progress_bar)" }
		set_global_variable = { name = commonwealth_progress_can value = "scope:journal_entry.scripted_bar_progress(can_commonwealth_progress_bar)" }
		set_global_variable = { name = commonwealth_progress_saf value = "scope:journal_entry.scripted_bar_progress(saf_commonwealth_progress_bar)" }
	}
	
	#custom_completion_header = je_uneasy_raj_complete_header

	complete = {
		custom_tooltip = {
			text = je_commonwealth_complete_tt
			global_var:commonwealth_progress_ast >= 100
			global_var:commonwealth_progress_can >= 100
			global_var:commonwealth_progress_saf >= 100
		}
		#custom_tooltip = {
		#	text = je_avert_sepoy_mutiny_tt_1
		#	has_variable = consolidated_colonial_rule
		#}
		#has_law = law_type:law_multicultural
		#root.average_sol >= 10
		#radical_fraction = {
		#	value <= 0.10
		#}
	}

	#custom_failure_header = je_uneasy_raj_fail_header

	fail = {
		custom_tooltip = {
			text = je_commonwealth_fail_tt
			NOT = {
				any_country = {
					OR = {
						top_overlord = c:GBR
						is_in_same_power_bloc = c:GBR
					}
					OR = {
						c:SAF ?= this
						country_has_primary_culture = cu:boer
						country_has_primary_culture = cu:anglo_canadian
						country_has_primary_culture = cu:franco_canadian
						country_has_primary_culture = cu:australian
					}
				}
			}
		}
	}

	#timeout = 9000

	#on_timeout = {
	#}

	#event_outcome_timeout_desc = {
	#	first_valid = {
	#		triggered_desc = {
	#			desc = event_effects_sepoy_mutiny_events.3_tt
	#		}
	#	}
	#}

	invalid = {
		NOT = {
			exists = c:GBR
		}
	}

	#custom_on_completion_header = je_uneasy_raj_on_complete_header

	on_complete = {
		custom_tooltip = {
			text = je_commonwealth_complete_reward_loyalists_for_involved_tt
			every_country = {
				limit = {
					is_involved_in_journal_entry = je_commonwealth
				}
				add_loyalists = {
					value = 0.10
				}
			}
		}
		
		custom_tooltip = modifier_HMPS_britain_migration_commonwealth_reward_tt_for_more
		hidden_effect = {
			every_state_in_western_europe = {
				limit = {
					OR = {
						state_region = { is_homeland = cu:british }
						state_region = { is_homeland = cu:scottish }
					}
					NOT = {
						state_region = { is_homeland = cu:irish }
					}
				}
				add_modifier = {
					name = modifier_HMPS_britain_migration_commonwealth_reward
				}
			}
		}
		
		remove_global_variable = commonwealth_progress_ast
		remove_global_variable = commonwealth_progress_can
		remove_global_variable = commonwealth_progress_saf
	}

	event_outcome_completed_desc = {
		#first_valid = {
		#	triggered_desc = {
		#		desc = event_effects_sepoy_mutiny_events.11_tt
		#	}
		#}
	}

	#custom_on_failure_header = je_uneasy_raj_on_fail_header

	on_fail = {
		remove_global_variable = commonwealth_progress_ast
		remove_global_variable = commonwealth_progress_can
		remove_global_variable = commonwealth_progress_saf
	}

	event_outcome_failed_desc = {
		#first_valid = {
		#	triggered_desc = {
		#		desc = event_effects_sepoy_mutiny_events.3_tt
		#	}
		#}
	}

	on_weekly_pulse = {
		events = {
		}
		effect = {
			set_global_variable = { name = commonwealth_progress_ast value = "scope:journal_entry.scripted_bar_progress(ast_commonwealth_progress_bar)" }
			set_global_variable = { name = commonwealth_progress_can value = "scope:journal_entry.scripted_bar_progress(can_commonwealth_progress_bar)" }
			set_global_variable = { name = commonwealth_progress_saf value = "scope:journal_entry.scripted_bar_progress(saf_commonwealth_progress_bar)" }	
		}		
	}

	on_monthly_pulse = {
		effect = {
			scope:journal_entry ?= {
				### AUSTRALIA
				if = {
					limit = {
						global_var:commonwealth_progress_ast >= 50
					}
					c:AST ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = colony
									}
								}
							}
							increase_autonomy = yes
						}
					}
				}
				if = {
					limit = {
						global_var:commonwealth_progress_ast >= 100
					}
					c:AST ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = dominion
									}
								}
							}
							this = {
								remove_diplomatic_pact = {
									country = c:GBR
									type = dominion
								}
							}
							join_power_bloc = c:GBR
						}
					}
				}
				### CANADA
				if = {
					limit = {
						global_var:commonwealth_progress_can >= 50
					}
					c:CAN ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = colony
									}
								}
							}
							increase_autonomy = yes
						}
					}
				}
				if = {
					limit = {
						global_var:commonwealth_progress_can >= 100
					}
					c:CAN ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = dominion
									}
								}
							}
							this = {
								remove_diplomatic_pact = {
									country = c:GBR
									type = dominion
								}
							}
							join_power_bloc = c:GBR
						}
					}
				}
				### SAFRICA
				if = {
					limit = {
						global_var:commonwealth_progress_saf >= 50
					}
					c:SAF ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = colony
									}
								}
							}
							increase_autonomy = yes
						}
					}
				}
				if = {
					limit = {
						global_var:commonwealth_progress_saf >= 100
					}
					c:SAF ?= {
						if = {
							limit = {
								this = {
									has_diplomatic_pact = {
										who = c:GBR
										type = dominion
									}
								}
							}
							this = {
								remove_diplomatic_pact = {
									country = c:GBR
									type = dominion
								}
							}
							join_power_bloc = c:GBR
						}
					}
				}
			}
		}
		random_events = {
		}
	}

	weight = 100
	can_revolution_inherit = yes
}

# Vanilla Canada JEs (00_canada_australia.txt); only complete changed so native nations' 1836 land doesn't block them
REPLACE:je_canada_can = {
	icon = "gfx/interface/icons/event_icons/waving_flag.dds"

	group = je_group_historical_content

	is_shown_when_inactive = {
		capital = {
			is_in_geographic_region = geographic_region_greater_canada
		}
		is_subject = yes
	}

	is_shown_in_lobby = {
		OR = {
			country_has_primary_culture = cu:anglo_canadian
			country_has_primary_culture = cu:franco_canadian
		}
		is_in_geographic_region = geographic_region_greater_canada
	}

	possible = {
		has_technology_researched = nationalism
	}

	scripted_button = je_canada_unite_can_button

	complete = { # HMPS: native land no longer blocks it (1836 North America rework)
		hmps_holds_region_bar_natives = { REGION = STATE_ONTARIO }
		hmps_holds_region_bar_natives = { REGION = STATE_QUEBEC }
		hmps_holds_region_bar_natives = { REGION = STATE_NEW_BRUNSWICK }
		hmps_holds_region_bar_natives = { REGION = STATE_MANITOBA }
		hmps_holds_region_bar_natives = { REGION = STATE_SASKATCHEWAN }
		hmps_holds_region_bar_natives = { REGION = STATE_ALBERTA }
		hmps_holds_region_bar_natives = { REGION = STATE_BRITISH_COLUMBIA }
		hmps_holds_region_bar_natives = { REGION = STATE_YUKON_TERRITORY }
		hmps_holds_region_bar_natives = { REGION = STATE_NORTHWEST_TERRITORIES }
		hmps_holds_region_bar_natives = { REGION = STATE_NUNAVUT }
	}

	on_complete = {
		trigger_event = { id = can_aus.5 }
		hidden_effect = {
			if = {
				limit = {
					any_scope_state = {
						state_region = s:STATE_ONTARIO
					}
				}
			}
			s:STATE_ONTARIO = {
				random_scope_state = {
			        set_hub_name = {
			            type = city
			            name = HUB_NAME_STATE_ONTARIO_city_ottawa
			        }
				}
			}
		}
	}

	invalid = {
		is_subject = no
	}

	weight = 100


	should_be_pinned_by_default_uninvolved_or_context = yes
}

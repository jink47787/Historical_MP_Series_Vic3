je_railroad_bubble = {
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	
	group = je_group_crises
	
	is_shown_when_inactive = {
		has_technology_researched = railways
		OR = {
			NOT = { has_global_variable = num_rr_busts }
			global_var:num_rr_busts < 4
		}
	}

	possible = { #just for Tooltip
		country_rank >= rank_value:major_power
		custom_tooltip = { 
			text = je_railroad_bubble_activation_5_rr_tt
			has_variable = rr_built
			root.var:rr_built >= 5
		}
		custom_tooltip = { 
			text = je_railroad_bubble_activation_first_4_tt
			OR = {
				NOT = { has_global_variable = num_rr_busts }
				global_var:num_rr_busts < 4
			}
		}
	}
	

	immediate = {
		change_variable = {
			name = rr_built
			multiply = 0
		}
		# num_rr_busts counts bubbles started, not busts (name kept for save compatibility)
		if = {
			limit = {
				exists = global_var:num_rr_busts
			}
			change_global_variable = {
				name = num_rr_busts
				add = 1
			}
		}
		else = {
			set_global_variable = {
				name = num_rr_busts
				value = 1
			}
		}
		set_variable = {
			name = railroad_bubble_mult
			value = 0
		}
		set_variable = {
			name = railroad_bubble_pops
			value = 40
		}
		set_variable = {
			name = rr_time_limit
			value = 0
		}
		# strength of the boom and bust; railroad_bubble.3 sets 0.75 when the exchanges are regulated
		set_variable = {
			name = rr_bubble_scale
			value = 1
		}
		if = {
			limit = { has_variable = rr_railway_king_fired }
			remove_variable = rr_railway_king_fired
		}
		# last, so the start event's choice can't be overwritten by the setup above
		trigger_event = {
			id = railroad_bubble.3
			popup = yes
		}
	}
	
	# Ends when the bubble reaches size 40 or after 18 months
	complete = {
		OR = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			custom_tooltip = {
				text = je_railroad_bubble.c.t
				var:rr_time_limit = 18
			}
		}
	}
	
	# happens when variable gets too high or enough time passes
	on_complete = {
		trigger_event = {
			id = railroad_bubble.5
			popup = yes
		}
		custom_tooltip = {
			text = je_railroad_bubble.c.tt
		}
	}
	
	# Every month the bubble grows by 1, plus 1 per railway level expanded, plus 1 if any railway is subsidized, plus 1 if transportation sells over 10% above base price in the market. Capped at 40
	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = rr_time_limit
				add = 1
			}
			change_variable = {
				name = railroad_bubble_mult
				add = 1
			}
			change_variable = {
				name = railroad_bubble_mult
				add = root.var:rr_built
			}
			if = {
				limit = {
					any_scope_state = {
						any_scope_building = {
							is_building_type = building_railway
							is_subsidized = yes
						}
					}
				}
				change_variable = {
					name = railroad_bubble_mult
					add = 1
				}
			}
			if = {
				limit = {
					market_capital.market = {
						mg:transportation = {
							market_goods_pricier > 0.1
						}
					}
				}
				change_variable = {
					name = railroad_bubble_mult
					add = 1
				}
			}
			# Cap at the goal size (railroad_bubble_pops = 40); the modifier maximums assume 40
			clamp_variable = {
				name = railroad_bubble_mult
				max = 40
				min = 0
			}
			# Effective modifier size = bubble size x strength (bubbles started before rr_bubble_scale existed count as full strength)
			if = {
				limit = { NOT = { has_variable = rr_bubble_scale } }
				set_variable = {
					name = rr_bubble_scale
					value = 1
				}
			}
			set_variable = {
				name = rr_bubble_effect
				value = var:railroad_bubble_mult
			}
			change_variable = {
				name = rr_bubble_effect
				multiply = var:rr_bubble_scale
			}
			# Re-apply the railway and state modifiers at the new size
			every_scope_building = {
				limit = {
					has_modifier = overvalued_railroads_building
				}
				remove_modifier = overvalued_railroads_building
			}
			every_scope_building = {
				limit = {
					is_building_type = building_railway
				}
				add_modifier = {
					name = overvalued_railroads_building
					multiplier = var:rr_bubble_effect
				}
			}
			every_scope_state = {
				remove_modifier = overvalued_railroads_state
				add_modifier = {
					name = overvalued_railroads_state
					multiplier = var:rr_bubble_effect
				}
			}
			# The Railway King appears once, halfway to the peak, and not in the month the entry ends; no private railway promoters under Command Economy or Cooperative Ownership
			if = {
				limit = {
					var:railroad_bubble_mult >= 20
					var:railroad_bubble_mult < 40
					var:rr_time_limit < 18
					NOT = { has_variable = rr_railway_king_fired }
					NOR = {
						has_law = law_type:law_command_economy
						has_law = law_type:law_cooperative_ownership
					}
				}
				set_variable = rr_railway_king_fired
				trigger_event = {
					id = railroad_bubble.7
					popup = yes
				}
			}
			change_variable = {
				name = rr_built
				multiply = 0
			}
		}
	}
	
		
	current_value = {
		value = root.var:railroad_bubble_mult
	}

	goal_add_value = {
		value = root.var:railroad_bubble_pops
	}

	progressbar = yes
	can_revolution_inherit = yes
    weight = 200
	should_be_pinned_by_default_uninvolved_or_context = yes
	
}

# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law
#
# law_disorganized_planning (Local Power Structures) must stay first in this file: countries with no law in the group start with the first law defined
# law_disorganized_planning uses the vanilla Elder Council icon

law_disorganized_planning = {
	group = lawgroup_economic_centralization

	icon = "gfx/interface/icons/law_icons/elder_council.dds"

	progressiveness = 0

	can_enact = {
		always = yes
	}

	can_impose = { # overlords cannot force Local Power Structures onto subjects
		always = no
	}

	modifier = {
		building_economy_of_scale_level_cap_add = -15
		building_subsistence_farm_throughput_add = 0.3
		building_subsistence_rice_farm_throughput_add = 0.3
		building_subsistence_orchard_throughput_add = 0.3
		building_subsistence_pasture_throughput_add = 0.3
		building_subsistence_fishing_village_throughput_add = 0.3
		interest_group_ig_rural_folk_pol_str_mult = 0.1
	}

	ai_will_do = { # AI never deliberately goes back to Local Power Structures
		always = no
	}
}

law_market_led_development = {
	group = lawgroup_economic_centralization

	icon = "gfx/interface/icons/law_icons/laissez_faire.dds" # placeholder icon

	progressiveness = 25

	unlocking_technologies = {
		political_economy
	}

	can_enact = { # unlike the other modern laws, allowed with Extractive Economy
		NOT = { has_law_or_variant = law_type:law_traditionalism }
	}

	disallowing_laws = { # these laws also switch Market-Led Development away when enacted: Command Economy to Centralized Economy, Cooperative Ownership to Regional Economic Councils
		law_command_economy
		law_cooperative_ownership
	}

	can_impose = {
		can_impose_law_default = yes
	}

	modifier = {
		country_max_companies_add = 1
		state_shopkeepers_investment_pool_efficiency_mult = 0.05
		state_trade_advantage_mult = 0.1
	}

	ai_enact_weight_modifier = { # AI strongly wants to leave Local Power Structures, IG support decides between the three modern laws
		value = 0

		if = {
			limit = { has_law_or_variant = law_type:law_disorganized_planning }
			add = 500
		}
	}
}

# group = this is the law_group a law belongs to
# icon = graphical icon shown in-game
# modifier = {} modifier on country for having adopted this law
#
# HMPS: Colonial Institutions law group. law_colonial_exploitation is also in this group (REPLACE in HMPS_00_colonial_affairs.txt)
# law_ad_hoc_administration must stay first in this file, and this file must load before 01_colonial_affairs.txt: countries with no law in the group start with the first law defined
# The non-default laws' effects are state modifiers (z_HMPS_other_modifiers.txt) applied to unincorporated states by HMPS_colonial_institutions_state_effects (HMPS_on_action.txt),
# multiplied by the Colonial Affairs institution's investment level. The non-default laws need State-Sponsored Colonization, which provides Colonial Affairs
# All icons are vanilla placeholders

law_ad_hoc_administration = {
	group = lawgroup_colonial_institutions

	icon = "gfx/interface/icons/law_icons/no_colonial_affairs.dds" # placeholder icon

	progressiveness = 0

	can_enact = {
		always = yes
	}

	can_impose = {
		can_impose_law_default = yes
	}

	ai_will_do = { # AI only goes back to Ad Hoc Administration once it has no colonies left
		NOT = {
			any_scope_state = {
				is_incorporated = no
			}
		}
	}
}

law_indirect_rule = {
	group = lawgroup_colonial_institutions

	icon = "gfx/interface/icons/law_icons/frontier_colonization.dds" # placeholder icon

	progressiveness = 0

	unlocking_technologies = {
		colonization
	}

	disallowing_laws = { # needs State-Sponsored Colonization: No Colonial Affairs and Frontier Colonization also switch this law back to Ad Hoc Administration when enacted
		law_no_colonial_affairs
		law_frontier_colonization
		law_sakoku
	}

	can_impose = {
		can_impose_law_default = yes
	}

	modifier = {
		country_institution_cost_institution_colonial_affairs_mult = 0.05
	}

	ai_enact_weight_modifier = { # AI with colonies wants to leave Ad Hoc Administration, IG support decides between the colonial laws
		value = 0

		if = {
			limit = {
				has_law = law_type:law_ad_hoc_administration
				any_scope_state = {
					is_incorporated = no
				}
			}
			add = 250
		}
	}
}

		always = yes
	}

		country_legitimacy_base_add = 5
	}

	tax_modifier_very_low = {

	}	
	
	tax_modifier_low = {
		tax_income_add = 0.12
	}
	
	tax_modifier_medium = {
		tax_income_add = 0.15
	}	
	
	tax_modifier_high = {
		tax_income_add = 0.18
	}
	
	tax_modifier_very_high = {
		tax_income_add = 0.195
	}		
	
	ai_enact_weight_modifier = {
		value = 0
		if = { # HMPS: raise taxes when in debt
			limit = { scaled_debt > 0 }

			add = 750
		}
	}

	ai_will_do = {
		has_law = law_type:law_no_income_taxes
	}
}

# script_context = <script context>	# see later for list of valid script contexts
# texture = <path>					# the path is relative to the "game" folder
# valid = <trigger>					# an invalid alert will not be shown
# open_panel = <panel_name>[|<tab>]	# if specified, this panel will open when the alert is clicked. Cannot specify both open_panel and open_popup
									# an optional tab name can be specified after a | character
# open_popup = <popup_name>			# if specified, this popup will open when the alert is clicked. Cannot specify both open_panel and open_popup
# type = alert | important_action | angry_important_action	# if unspecified or "alert", this will be a standard alert. Otherwise, it will be shown inside the important actions alert (angry_important_action alerts are sorted first)
# alert_group = <alert_group_name>	# if specified, multiple alerts in the same group will be grouped in the important actions alert (if they are "important_action"). "important_actions with unspecified alert_group will never be grouped together

# HMPS ALERTS

low_leverage_alert = {
	texture = "gfx/interface/main_hud/powerblocs.dds"
	# texture = "gfx/interface/icons/generic_icons/warning.dds"
	script_context = player_country

	valid = {
		power_bloc ?= {
			any_power_bloc_member = {
				is_power_bloc_leader = no
				is_subject = no
				power_bloc ?= {
					leverage_advantage = { target = prev value < 50 } # 50 is so that players get enough warning to fix issue
				}
			}
		}
	}

	open_panel = power_bloc_panel
	type = alert
	color = red
}

gm_low_leverage_alert = {
	texture = "gfx/interface/main_hud/powerblocs.dds"
	# texture = "gfx/interface/icons/generic_icons/warning.dds"
	script_context = player_country

	valid = {
		has_modifier = game_master
		any_power_bloc ?= {
			any_power_bloc_member = {
				is_player = yes #GM shouldnt get notifications about AI things
				is_power_bloc_leader = no
				is_subject = no
				power_bloc ?= {
					leverage_advantage = { target = prev value < 20 } # 20 is so that GM can issue a warning incase players are dumb
				}
			}
		}
	}

	open_panel = power_bloc_panel
	type = alert
	color = red
}

# Reserve Currency: shown to every member of the market while it is about to gain (green) or lose (red) the Reserve Currency.
# Reads the variables of rc_monthly_update (common/scripted_effects/HMPS_specie_famine_effects.txt)
# Gaining: our market has out-imported the holder's
reserve_currency_gaining_alert = {
	script_context = player_market
	texture = "gfx/interface/icons/alert_icons/reserve_currency_gaining.dds"

	valid = {
		has_global_variable = reserve_currency_holder
		owner ?= { has_variable = rc_challenge_months }
	}

	open_panel = market
	type = alert
	color = green
}

# Gaining: nobody holds it and our market leads in imports
reserve_currency_vacant_alert = {
	script_context = player_market
	texture = "gfx/interface/icons/alert_icons/reserve_currency_gaining.dds"

	valid = {
		has_global_variable = rc_vacant_candidate
		owner ?= global_var:rc_vacant_candidate
	}

	open_panel = market
	type = alert
	color = green
}

# Losing: our market leader holds it and another market has out-imported ours
reserve_currency_losing_alert = {
	script_context = player_market
	texture = "gfx/interface/icons/alert_icons/reserve_currency_losing.dds"

	valid = {
		has_global_variable = rc_top_challenger
		has_global_variable = reserve_currency_holder
		owner ?= global_var:reserve_currency_holder
	}

	open_panel = market
	type = alert
	color = red
}

# Losing: our market leader holds it and our market has stopped running a deficit
reserve_currency_surplus_alert = {
	script_context = player_market
	texture = "gfx/interface/icons/alert_icons/reserve_currency_losing.dds"

	valid = {
		has_global_variable = reserve_currency_holder
		owner ?= global_var:reserve_currency_holder
		owner ?= { has_variable = rc_surplus_months }
	}

	open_panel = market
	type = alert
	color = red
}

regional_investment_assign_state_slot = {
	if = {
		limit = {
			NOT = {
				is_target_in_variable_list = {
					name = regional_investment_rights_investors
					target = $INVESTOR$
				}
			}
		}
		add_to_variable_list = {
			name = regional_investment_rights_investors
			target = $INVESTOR$
		}
	}
}

regional_investment_remove_state_slot = {
	if = {
		limit = {
			is_target_in_variable_list = {
				name = regional_investment_rights_investors
				target = $INVESTOR$
			}
		}
		remove_list_variable = {
			name = regional_investment_rights_investors
			target = $INVESTOR$
		}
	}
}

regional_investment_clear_state_slots = {
	if = {
		limit = {
			has_variable_list = regional_investment_rights_investors
		}
		clear_variable_list = regional_investment_rights_investors
	}
}

force_single_market = {
	if = {
		limit = {
			is_subject = yes
			overlord = {
				market = prev.market
				NOT = {
					has_diplomatic_pact = {
						who = prev
						type = grant_own_market
					}
				}
			}
		}
		overlord = {
			create_diplomatic_pact = {
				country = prev
				type = grant_own_market
			}
		}
	}
}

calculate_tpg_reconstruction_law_progress = {
	# From Laws
	set_variable = { name = tpg_reconstruction_from_laws value = 0 }
	# Cops
	set_variable = { name = temp_policing_var_contribution value = institution:institution_police.investment }
	if = {
		limit = {
			has_law = law_type:law_no_police
		}
		change_variable = { name = temp_policing_var_contribution add = 0 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_local_police
		}
		change_variable = { name = temp_policing_var_contribution multiply = 0.1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_dedicated_police
		}
		change_variable = { name = temp_policing_var_contribution multiply = 0.2 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_militarized_police
		}
		change_variable = { name = temp_policing_var_contribution multiply = 0.3 }
	}
	
	# Internal Affairs
	set_variable = { name = temp_ha_var_contribution value = institution:institution_home_affairs.investment }
	if = {
		limit = {
			has_law = law_type:law_no_home_affairs
		}
		change_variable = { name = temp_ha_var_contribution add = 0 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_guaranteed_liberties
		}
		change_variable = { name = temp_ha_var_contribution multiply = 0.1 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_national_guard
		}
		change_variable = { name = temp_ha_var_contribution multiply = 0.2 }
	}
	else_if = {
		limit = {
			has_law = law_type:law_secret_police
		}
		change_variable = { name = temp_ha_var_contribution multiply = 0.3 }
	}
	change_variable = { name = tpg_reconstruction_from_laws add = var:temp_policing_var_contribution }
	change_variable = { name = tpg_reconstruction_from_laws add = var:temp_ha_var_contribution }
	remove_variable = temp_policing_var_contribution
	remove_variable = temp_ha_var_contribution
}

# HMPS: Replaces transition_off_of_monarchy_effects in the republic/corporate state on_activate blocks.
# The base effect adds Stamp Out Monarchism to every country activating these laws, including every country that starts as a republic.
# country_was_monarchy_var is set by the monarchy/social monarchy on_activate and cleared by every other governance law's on_activate.
hmps_transition_off_of_monarchy_effects = {
	if = {
		limit = {
			has_variable = country_was_monarchy_var
			game_date > 1836.1.1 # Laws activated during game setup (e.g. a default Monarchy before history sets a republic) don't count
		}
		transition_off_of_monarchy_effects = yes
	}
	else = {
		crown_land_republic_effect = yes
		transition_off_of_monarchy_law_effects = yes
	}
	remove_variable = country_was_monarchy_var
}

# HMPS: Called from the immediate of je_law_imposition and je_british_dictate_law (REPLACE_HMPS_00_impose_law.txt, REPLACE_HMPS_04_india_british_dictates.txt). Root = country imposed upon.
# Puppet, Colony and Crown Land pacts give the subject +10% imposed-law enactment success; player subjects don't get it.
# The modifier sits on the journal entry, so it ends when the imposition does.
hmps_player_subject_imposition_effect = {
	if = {
		limit = {
			is_player = yes
			OR = {
				is_subject_type = subject_type_puppet
				is_subject_type = subject_type_colony
				is_subject_type = subject_type_crown_land
			}
		}
		scope:journal_entry = {
			add_modifier = hmps_player_subject_imposition
		}
	}
}

		}
	}
}

# Country scope: SLOT = option number (0 = leave unorganized), VOTES = global tally of that option. A changed vote is taken off the old option first

			has_variable = congo_vote
		}
		if = {
			limit = { var:congo_vote = 0 }
			change_global_variable = { name = congo_votes_0 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 1 }
			change_global_variable = { name = congo_votes_1 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 2 }
			change_global_variable = { name = congo_votes_2 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 3 }
			change_global_variable = { name = congo_votes_3 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 4 }
			change_global_variable = { name = congo_votes_4 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 5 }
			change_global_variable = { name = congo_votes_5 add = -1 }
		}
		else_if = {
			limit = { var:congo_vote = 6 }

		name = congo_vote
		value = $SLOT$

	change_global_variable = {
		name = $VOTES$
		add = 1
	}

# Counts the votes and hands over the Congo. Runs once, from whichever country's JE ends first
hmps_congo_resolve_conference = {
	if = {
		limit = {
			NOT = { has_global_variable = congo_resolved }
		}
		set_global_variable = congo_resolved

		# AI Great Powers that have not voted by the deadline vote now
		every_country = {
			limit = {
				has_variable = congo_voter
				is_player = no
				NOT = { has_variable = congo_vote }
			}
			hmps_congo_ai_vote = yes
		}

		# "Leave unorganized" starts in the lead; votes for nominees that no longer exist are ignored
		set_global_variable = {
			name = congo_best_slot
			value = 0
		}
		set_global_variable = {
			name = congo_best_votes
			value = global_var:congo_votes_0
		}
		hmps_congo_check_slot = { SLOT = 1 VOTES = congo_votes_1 }
		hmps_congo_check_slot = { SLOT = 2 VOTES = congo_votes_2 }
		hmps_congo_check_slot = { SLOT = 3 VOTES = congo_votes_3 }
		hmps_congo_check_slot = { SLOT = 4 VOTES = congo_votes_4 }
		hmps_congo_check_slot = { SLOT = 5 VOTES = congo_votes_5 }
		hmps_congo_check_slot = { SLOT = 6 VOTES = congo_votes_6 }

		if = {
			limit = {
				global_var:congo_best_slot > 0
			}
			random_country = {
				limit = {
					has_variable = congo_nominee
					var:congo_nominee = global_var:congo_best_slot
				}
				save_scope_as = congo_winner_scope
				set_variable = congo_winner
				set_global_variable = {
					name = congo_winner
					value = THIS
				}
				add_modifier = {
					name = received_congo
					months = 241
				}
			}
			# Every Congo state except those kept with "Keep Our Congo States"
			every_state = {
				limit = {
					hmps_is_congo_state = yes
					NOT = { has_variable = greedy_power }
					NOT = { owner = scope:congo_winner_scope }
				}
				set_state_owner = scope:congo_winner_scope
			}
		}
		else = {
			set_global_variable = congo_no_winner
		}

		# Every remaining nominee gains 20 relations with each power whose final vote was for it, and learns who they were
		every_country = {
			limit = {
				has_variable = congo_nominee
			}
			save_scope_as = congo_nominee_scope
			every_country = {
				limit = {
					has_variable = congo_vote
					var:congo_vote = scope:congo_nominee_scope.var:congo_nominee
					NOT = { THIS = scope:congo_nominee_scope }
				}
				change_relations = {
					country = scope:congo_nominee_scope
					value = 20
				}
			}
			trigger_event = {
				id = congo_free_state.3
				days = 1
				popup = yes
			}
		}
	}
}

# HMPS Balance of Trade. Country scope; root must be the country (add_modifier's multiplier reads var: from root), so it is
# only called from the hidden event hmps_bot.1 (events/00_HMPS_utility.txt), fired for every country on day 0
# (HMPS_balance_of_trade_start, common/on_actions/HMPS_on_action.txt) and weekly by HMPS_utility_je (common/journal_entries/z_HMPS_utility.txt).
# Measured per market: net trade = the market's yearly exports - yearly imports (summed over every country whose market = root.market,
# weekly x 52), as a share of the market's summed yearly GDP. Every member computes the same totals and gets the same modifier.
# Within +-1% of GDP: no effect. Beyond: the leading side loses 5% advantage per 1% of GDP (cap -50%) and the other side
# gains 2.5% per 1% of GDP (cap +25%); both caps are reached at 10% of GDP.
# A surplus lowers export and raises import advantage; a deficit lowers import and raises export advantage.
# Any future Balance of Trade flavor (decisions, events) should target the market owner.
# The variables are shown in the Market panel's trade tab (gui/market_panel.gui, gui/00_trade_partners_market_panel.gui).
# Exports are scaled by the world's imports / exports (global_var:hmps_bot_world_ratio, set by
# hmps_balance_of_trade_world_totals_effect) because the game values imports above exports, which left almost every
# market in deficit. After scaling, the world's balances sum to zero.
hmps_balance_of_trade_world_totals_effect = {
	set_global_variable = { name = hmps_bot_world_exports value = 0 }
	set_global_variable = { name = hmps_bot_world_imports value = 0 }
	every_country = {
		change_global_variable = { name = hmps_bot_world_exports add = total_export_value }
		change_global_variable = { name = hmps_bot_world_imports add = total_import_value }
	}
	if = {
		limit = { global_var:hmps_bot_world_exports > 0 }
		set_global_variable = {
			name = hmps_bot_world_ratio
			value = {
				value = global_var:hmps_bot_world_imports
				divide = global_var:hmps_bot_world_exports
			}
		}
	}
	else = {
		set_global_variable = { name = hmps_bot_world_ratio value = 1 }
	}
}

hmps_balance_of_trade_update_effect = {
	set_variable = {
		name = hmps_bot_exports_weekly
		value = {
			every_country = {
				limit = { market = root.market }
				add = total_export_value
			}
		}
	}
	set_variable = {
		name = hmps_bot_exports
		value = {
			value = var:hmps_bot_exports_weekly
			multiply = 52
			multiply = global_var:hmps_bot_world_ratio
		}
	}
	set_variable = {
		name = hmps_bot_imports_weekly
		value = {
			every_country = {
				limit = { market = root.market }
				add = total_import_value
			}
		}
	}
	set_variable = {
		name = hmps_bot_imports
		value = {
			value = var:hmps_bot_imports_weekly
			multiply = 52
		}
	}
	set_variable = {
		name = hmps_bot_balance
		value = {
			value = var:hmps_bot_exports
			subtract = var:hmps_bot_imports
		}
	}
	set_variable = {
		name = hmps_bot_gdp
		value = {
			every_country = {
				limit = { market = root.market }
				add = gdp
			}
		}
	}
	if = {
		limit = { var:hmps_bot_gdp > 0 }
		set_variable = {
			name = hmps_bot_share
			value = {
				value = var:hmps_bot_balance
				divide = var:hmps_bot_gdp
			}
		}
	}
	else = {
		set_variable = { name = hmps_bot_share value = 0 }
	}

	# The trade-advantage penalties were replaced by the price level (bank rate phase 3b); this clears them from older saves
	if = {
		limit = { has_modifier = hmps_trade_surplus_penalty }
		remove_modifier = hmps_trade_surplus_penalty
	}
	if = {
		limit = { has_modifier = hmps_trade_deficit_penalty }
		remove_modifier = hmps_trade_deficit_penalty
	}
	if = {
		limit = { has_variable = hmps_bot_penalty }
		remove_variable = hmps_bot_penalty
		remove_variable = hmps_bot_buff
	}
}

# HMPS Bank Rate and Reserves, phase 1 (ledger). Runs once a week from HMPS_utility_je's on_weekly_pulse
# (common/journal_entries/z_HMPS_utility.txt), right after the Balance of Trade update, and once on day 0 from HMPS_balance_of_trade_start
# (common/on_actions/HMPS_on_action.txt) to fill the panel; the first weekly run then resets the starting reserves from real trade.
# Steps 1-2 and 4-5 keep each market's FX reserves (no effects yet); step 3 is the capital flow; step 6 is the bank rate.
# Money is stored on the market owner, in £:
#   hmps_fx_reserves     FX reserves (may go below zero in later phases)
#   hmps_fx_exports, hmps_fx_imports, hmps_fx_net   last week's market trade (weekly values), exports scaled by
#                        global_var:hmps_bot_world_ratio as in the Balance of Trade, so the world's net exports sum to zero
#   hmps_fx_gdp          market GDP; also the base for GDP shares when a country changes market
#   hmps_fx_import_bill  monthly imports (weekly x 52 / 12), at least 5% of market GDP a year / 12, so markets without trade
#                        never divide by zero
#   hmps_fx_gold         last week's gold credit: each gold mine and field's goods_output_gold_add (counts staffing) x 100
#   hmps_fx_topup        last week's world top-up
#   hmps_fx_months       reserves in months of imports
# Every country stores its market owner in hmps_fx_owner. When it changes, a former owner hands its whole pool to the new
# owner and a former member carries its GDP share of the old pool. A new owner with no reserves starts at 12 months.
# Globals: hmps_fx_world_months (import-weighted average), running totals hmps_fx_world_gold_total,
# hmps_fx_world_imports_total, hmps_fx_topup_count, hmps_fx_topup_total (read from saves by the campaign recorder).
# Whenever the world average falls below 12 months, every market gains the same number of months of its own imports.
# Markets owned by decentralized countries are skipped.
# Known gap: an annexed market owner's reserves are lost, because no hook fires for annexation.
# Phase 2 (bank rate), also on the market owner:
#   hmps_fx_rate         bank rate, 2 to 10 (4 = neutral, no effects); set by the market panel buttons (hmps_fx_rate_change_effect)
#   hmps_fx_rate_steps   steps from 4% (rate - 4); every rate effect follows it at once (no phase-in, so a hike can't be dropped before its costs arrive)
#   hmps_fx_rate_lock    timed (30 days) after a change, so the rate moves at most two points per month
# Step 6 gives new owners a rate; after step 7, hmps_bot.2 refreshes every country's rate modifiers (hmps_fx_rate_modifiers_effect).
# Known gap: a country that leaves a market starts at 4%.
# Phase 3 (capital flow), weekly, so every yearly figure is divided by 52. On the market owner, in £ (last week's values):
#   hmps_fx_trade        total trade (scaled exports + imports); picks the leader and sets the commission
#   hmps_fx_foreign      foreign money held: inflows add, outflows subtract (positive = owed abroad, negative = lent abroad);
#                        moves with the reserves when a country changes market
#   hmps_fx_interest     interest: debtors pay foreign money x own rate / 52 from reserves, creditors share it by what each lent
#   hmps_fx_flow         capital flow: (rate + pull - reference) x 1% of GDP a year / 52, no cap; sums to zero against the GDP-weighted
#                        reference. 0 for an isolationist market owner. Reserves and foreign money move by the full flow
#   hmps_fx_pull         safe-haven pull on the issuer (its share of 1 point) and a former issuer (its fading share); kept for the previews
#   hmps_fx_pool_moved   investment pool moved: senders owe 50% of their outflow, split among member countries by GDP share; a country
#                        pays only if its own investment_pool covers its full share (a negative pool is reset to 0, so money would be
#                        lost); receivers share only what was paid, by inflow, then by GDP share
#   hmps_fx_pool_share   on every member country: its own part of hmps_fx_pool_moved, for the Investment Pool tooltip
#   hmps_fx_commission   London's commission: every other market pays 0.5% of its total trade to the issuer (split with a fading
#                        former issuer). Safe-haven pull: in the capital flow the issuer's rate counts 1 point higher (same split)
# Globals: hmps_fx_issuer (reserve currency; Britain's market at the start), hmps_fx_leader (largest total trade), hmps_fx_challenger and
# hmps_fx_lead_weeks (the role passes after 104 weeks = 24 months in the lead), hmps_fx_old_issuer and hmps_fx_old_issuer_weeks (its
# commission and pull fade over 52 weeks; shares in hmps_fx_old_share and hmps_fx_issuer_share during the week), hmps_fx_ref_rate (rates +
# pull weighted by GDP, isolationist owners left out), hmps_fx_world_gdp (GDP of the markets in that average), and the test lines
# hmps_fx_left_flow, hmps_fx_left_pool and hmps_fx_left_interest (rounding remainders given to the largest receiver each week, near 0).
#   hmps_fx_change       change in reserves: net exports + gold + capital flow + interest + commission + top-up (not market moves)
# Phase 3b (price level), on the market owner:
#   hmps_fx_price_ratio   (reserves / market GDP) / the world's (global_var:hmps_fx_world_money_ratio); 1 if either is 0 or less
#   hmps_fx_price_money   money part, (ratio - 1) x 4; hmps_fx_price_rate: rate part, (4 - rate) / 2 (placeholder scales)
#   hmps_fx_price_target  money part + rate part, capped at -4 to 4, not rounded
#   hmps_fx_price_level   whole steps, -4 to 4 (positive = inflation); a new owner, or a country that leaves a market, starts at 0
#   hmps_fx_price_weeks   weeks in a row with the target at least 1 step away; at 26 the level steps toward it and the count restarts
# Step 7 sets these; hmps_bot.2 then pushes the level to every member as the Inflation or Deflation modifier (hmps_fx_price_modifiers_effect).
# Phases 4-5 add the Defend, Suspend and on-paper multipliers on inflows (marked in step 3d), and the crisis freeze in step 7.
hmps_fx_weekly_effect = {
	hmps_balance_of_trade_world_totals_effect = yes

	# 1. Market moves (uses last week's hmps_fx_gdp)
	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
			has_variable = hmps_fx_owner
		}
		save_scope_as = hmps_fx_mover
		market = {
			owner = { save_scope_as = hmps_fx_new_owner }
		}
		if = {
			limit = {
				NOT = { scope:hmps_fx_new_owner = var:hmps_fx_owner }
				scope:hmps_fx_new_owner = {
					NOT = { is_country_type = decentralized }
				}
			}
			scope:hmps_fx_new_owner = {
				if = {
					limit = { NOT = { has_variable = hmps_fx_reserves } }
					set_variable = { name = hmps_fx_reserves value = 0 }
				}
				if = {
					limit = { NOT = { has_variable = hmps_fx_foreign } }
					set_variable = { name = hmps_fx_foreign value = 0 }
				}
			}
			if = {
				# A former market owner: its whole pool moves, with its foreign money held
				limit = {
					scope:hmps_fx_mover = var:hmps_fx_owner
					has_variable = hmps_fx_reserves
				}
				scope:hmps_fx_new_owner = {
					change_variable = { name = hmps_fx_reserves add = scope:hmps_fx_mover.var:hmps_fx_reserves }
				}
				if = {
					limit = { has_variable = hmps_fx_foreign }
					scope:hmps_fx_new_owner = {
						change_variable = { name = hmps_fx_foreign add = scope:hmps_fx_mover.var:hmps_fx_foreign }
					}
				}
			}
			else_if = {
				# A former member: carries its GDP share of the old pool
				limit = {
					var:hmps_fx_owner = {
						is_country_alive = yes
						has_variable = hmps_fx_reserves
						has_variable = hmps_fx_gdp
						var:hmps_fx_gdp > 0
					}
				}
				set_variable = {
					name = hmps_fx_move
					value = {
						value = gdp
						divide = var:hmps_fx_owner.var:hmps_fx_gdp
						max = 1
						multiply = var:hmps_fx_owner.var:hmps_fx_reserves
					}
				}
				var:hmps_fx_owner = {
					change_variable = { name = hmps_fx_reserves subtract = scope:hmps_fx_mover.var:hmps_fx_move }
				}
				scope:hmps_fx_new_owner = {
					change_variable = { name = hmps_fx_reserves add = scope:hmps_fx_mover.var:hmps_fx_move }
				}
				remove_variable = hmps_fx_move
				# ...and the same share of the old market's foreign money held
				if = {
					limit = {
						var:hmps_fx_owner = { has_variable = hmps_fx_foreign }
					}
					set_variable = {
						name = hmps_fx_move
						value = {
							value = gdp
							divide = var:hmps_fx_owner.var:hmps_fx_gdp
							max = 1
							multiply = var:hmps_fx_owner.var:hmps_fx_foreign
						}
					}
					var:hmps_fx_owner = {
						change_variable = { name = hmps_fx_foreign subtract = scope:hmps_fx_mover.var:hmps_fx_move }
					}
					scope:hmps_fx_new_owner = {
						change_variable = { name = hmps_fx_foreign add = scope:hmps_fx_mover.var:hmps_fx_move }
					}
					remove_variable = hmps_fx_move
				}
			}
			# A former owner keeps nothing
			if = {
				limit = {
					NOT = { scope:hmps_fx_mover = scope:hmps_fx_new_owner }
					has_variable = hmps_fx_reserves
				}
				remove_variable = hmps_fx_reserves
			}
			if = {
				limit = {
					NOT = { scope:hmps_fx_mover = scope:hmps_fx_new_owner }
					has_variable = hmps_fx_foreign
				}
				remove_variable = hmps_fx_foreign
			}
		}
	}
	every_country = {
		limit = { NOT = { is_country_type = decentralized } }
		market = {
			owner = { save_scope_as = hmps_fx_new_owner }
		}
		set_variable = { name = hmps_fx_owner value = scope:hmps_fx_new_owner }
	}

	# 2. Trade, GDP and gold per market; new owners start at 12 months; reserves credited
	set_global_variable = { name = hmps_fx_world_reserves value = 0 }
	set_global_variable = { name = hmps_fx_world_bill value = 0 }
	# The first weekly run after day 0 (hmps_fx_start_pending) starts the running totals and reserves again from real trade
	if = {
		limit = {
			OR = {
				NOT = { has_global_variable = hmps_fx_world_gold_total }
				has_global_variable = hmps_fx_start_pending
			}
		}
		set_global_variable = { name = hmps_fx_world_gold_total value = 0 }
		set_global_variable = { name = hmps_fx_world_imports_total value = 0 }
		set_global_variable = { name = hmps_fx_topup_count value = 0 }
		set_global_variable = { name = hmps_fx_topup_total value = 0 }
	}
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		save_scope_as = hmps_fx_market
		owner = {
			set_variable = {
				name = hmps_fx_exports
				value = {
					every_country = {
						limit = { market = scope:hmps_fx_market }
						add = total_export_value
					}
				}
			}
			set_variable = {
				name = hmps_fx_exports
				value = {
					value = var:hmps_fx_exports
					multiply = global_var:hmps_bot_world_ratio
				}
			}
			set_variable = {
				name = hmps_fx_imports
				value = {
					every_country = {
						limit = { market = scope:hmps_fx_market }
						add = total_import_value
					}
				}
			}
			set_variable = {
				name = hmps_fx_net
				value = {
					value = var:hmps_fx_exports
					subtract = var:hmps_fx_imports
				}
			}
			set_variable = {
				name = hmps_fx_trade
				value = {
					value = var:hmps_fx_exports
					add = var:hmps_fx_imports
				}
			}
			set_variable = {
				name = hmps_fx_gdp
				value = {
					every_country = {
						limit = { market = scope:hmps_fx_market }
						add = gdp
					}
				}
			}
			# Weekly floor first (5% of GDP a year / 52), then converted to a monthly bill
			set_variable = {
				name = hmps_fx_import_bill
				value = {
					value = var:hmps_fx_gdp
					multiply = 0.05
					divide = 52
				}
			}
			if = {
				limit = { var:hmps_fx_imports > var:hmps_fx_import_bill }
				set_variable = { name = hmps_fx_import_bill value = var:hmps_fx_imports }
			}
			set_variable = {
				name = hmps_fx_import_bill
				value = {
					value = var:hmps_fx_import_bill
					multiply = 52
					divide = 12
				}
			}
			set_variable = {
				name = hmps_fx_gold
				value = {
					every_country = {
						limit = { market = scope:hmps_fx_market }
						every_scope_building = {
							limit = {
								OR = {
									is_building_type = building_gold_mine
									is_building_type = building_gold_field
								}
							}
							add = modifier:goods_output_gold_add
						}
					}
				}
			}
			set_variable = {
				name = hmps_fx_gold
				value = {
					value = var:hmps_fx_gold
					multiply = 100
				}
			}
			if = {
				limit = {
					OR = {
						NOT = { has_variable = hmps_fx_reserves }
						has_global_variable = hmps_fx_start_pending
					}
				}
				set_variable = {
					name = hmps_fx_reserves
					value = {
						value = var:hmps_fx_import_bill
						multiply = 12
					}
				}
			}
			change_variable = { name = hmps_fx_reserves add = var:hmps_fx_net }
			change_variable = { name = hmps_fx_reserves add = var:hmps_fx_gold }
			set_variable = { name = hmps_fx_topup value = 0 }
			change_global_variable = { name = hmps_fx_world_reserves add = var:hmps_fx_reserves }
			change_global_variable = { name = hmps_fx_world_bill add = var:hmps_fx_import_bill }
			change_global_variable = { name = hmps_fx_world_gold_total add = var:hmps_fx_gold }
			change_global_variable = { name = hmps_fx_world_imports_total add = var:hmps_fx_imports }
		}
	}
	if = {
		limit = { has_global_variable = hmps_fx_start_pending }
		remove_global_variable = hmps_fx_start_pending
	}

	# 3. Capital flow (phase 3). Every transfer moves money between markets, so the world total from step 2 is unchanged
	# a) Issuer: Britain's market holds the role at the start; the market with the largest total trade is the leader
	set_global_variable = { name = hmps_fx_top_trade value = -1 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			hmps_fx_rate_init_effect = yes
			if = {
				limit = { var:hmps_fx_trade > global_var:hmps_fx_top_trade }
				set_global_variable = { name = hmps_fx_top_trade value = var:hmps_fx_trade }
				save_scope_as = hmps_fx_leader_scope
				set_global_variable = { name = hmps_fx_leader value = scope:hmps_fx_leader_scope }
			}
		}
	}
	remove_global_variable = hmps_fx_top_trade
	if = {
		limit = { NOT = { has_global_variable = hmps_fx_issuer } }
		if = {
			limit = {
				exists = c:GBR
				c:GBR = {
					is_country_alive = yes
					NOT = { is_country_type = decentralized }
				}
			}
			c:GBR = {
				market = {
					owner = { save_scope_as = hmps_fx_issuer_scope }
				}
			}
		}
		else = {
			global_var:hmps_fx_leader = { save_scope_as = hmps_fx_issuer_scope }
		}
		set_global_variable = { name = hmps_fx_issuer value = scope:hmps_fx_issuer_scope }
	}
	# The issuer no longer owns a counted market: the role passes at once to its market's owner, else to the leader (no fade)
	if = {
		limit = {
			NOT = {
				global_var:hmps_fx_issuer = {
					is_country_alive = yes
					NOT = { is_country_type = decentralized }
					has_variable = hmps_fx_owner
					var:hmps_fx_owner = global_var:hmps_fx_issuer
				}
			}
		}
		if = {
			limit = {
				global_var:hmps_fx_issuer = {
					is_country_alive = yes
					NOT = { is_country_type = decentralized }
					has_variable = hmps_fx_owner
				}
			}
			global_var:hmps_fx_issuer = {
				var:hmps_fx_owner = { save_scope_as = hmps_fx_issuer_scope }
			}
		}
		else = {
			global_var:hmps_fx_leader = { save_scope_as = hmps_fx_issuer_scope }
		}
		set_global_variable = { name = hmps_fx_issuer value = scope:hmps_fx_issuer_scope }
	}
	# Challenger: weeks in a row in the lead (the notification comes with the final panel, phase 7)
	if = {
		limit = { global_var:hmps_fx_leader = global_var:hmps_fx_issuer }
		set_global_variable = { name = hmps_fx_lead_weeks value = 0 }
		if = {
			limit = { has_global_variable = hmps_fx_challenger }
			remove_global_variable = hmps_fx_challenger
		}
	}
	else_if = {
		limit = {
			has_global_variable = hmps_fx_challenger
			global_var:hmps_fx_leader = global_var:hmps_fx_challenger
		}
		change_global_variable = { name = hmps_fx_lead_weeks add = 1 }
	}
	else = {
		global_var:hmps_fx_leader = { save_scope_as = hmps_fx_challenger_scope }
		set_global_variable = { name = hmps_fx_challenger value = scope:hmps_fx_challenger_scope }
		set_global_variable = { name = hmps_fx_lead_weeks value = 1 }
	}
	# Handover after 104 weeks (24 months) in the lead; the old issuer's commission fades over 52 weeks (12 months)
	if = {
		limit = { global_var:hmps_fx_lead_weeks >= 104 }
		global_var:hmps_fx_issuer = { save_scope_as = hmps_fx_old_issuer_scope }
		global_var:hmps_fx_challenger = { save_scope_as = hmps_fx_issuer_scope }
		set_global_variable = { name = hmps_fx_old_issuer value = scope:hmps_fx_old_issuer_scope }
		set_global_variable = { name = hmps_fx_old_issuer_weeks value = 52 }
		set_global_variable = { name = hmps_fx_issuer value = scope:hmps_fx_issuer_scope }
		remove_global_variable = hmps_fx_challenger
		set_global_variable = { name = hmps_fx_lead_weeks value = 0 }
	}
	# A former issuer that wins the role back stops fading
	if = {
		limit = {
			has_global_variable = hmps_fx_old_issuer
			global_var:hmps_fx_old_issuer = global_var:hmps_fx_issuer
		}
		remove_global_variable = hmps_fx_old_issuer
		remove_global_variable = hmps_fx_old_issuer_weeks
	}
	# A former issuer that died, fell apart or no longer owns a market stops fading at once
	if = {
		limit = {
			has_global_variable = hmps_fx_old_issuer
			NOT = {
				global_var:hmps_fx_old_issuer = {
					is_country_alive = yes
					NOT = { is_country_type = decentralized }
					has_variable = hmps_fx_owner
					var:hmps_fx_owner = global_var:hmps_fx_old_issuer
				}
			}
		}
		remove_global_variable = hmps_fx_old_issuer
		remove_global_variable = hmps_fx_old_issuer_weeks
	}
	# Shares of London's commission (step f) and the safe-haven pull (step d): a former issuer keeps weeks left / 52, the issuer gets the rest
	if = {
		limit = { has_global_variable = hmps_fx_old_issuer }
		set_global_variable = {
			name = hmps_fx_old_share
			value = {
				value = global_var:hmps_fx_old_issuer_weeks
				divide = 52
			}
		}
		set_global_variable = {
			name = hmps_fx_issuer_share
			value = {
				value = 1
				subtract = global_var:hmps_fx_old_share
			}
		}
	}
	else = {
		set_global_variable = { name = hmps_fx_issuer_share value = 1 }
	}

	# b) Reference rate: the world average of effective rates, weighted by GDP. Effective rate = rate + safe-haven pull (hmps_fx_pull: the
	#    issuer's rate counts 1 point higher x its share; a former issuer's fades the same way). Isolationist market owners stay out of the
	#    average and have no flow. hmps_fx_world_gdp (the included markets) is kept for the coin previews (hmps_fx_flow_at_N script values)
	set_global_variable = { name = hmps_fx_ref_sum value = 0 }
	set_global_variable = { name = hmps_fx_world_gdp value = 0 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			if = {
				limit = { this = global_var:hmps_fx_issuer }
				set_variable = { name = hmps_fx_pull value = global_var:hmps_fx_issuer_share }
			}
			else_if = {
				limit = {
					has_global_variable = hmps_fx_old_issuer
					this = global_var:hmps_fx_old_issuer
				}
				set_variable = { name = hmps_fx_pull value = global_var:hmps_fx_old_share }
			}
			else_if = {
				limit = { has_variable = hmps_fx_pull }
				remove_variable = hmps_fx_pull
			}
			if = {
				limit = {
					NOT = { has_law_or_variant = law_type:law_isolationism }
				}
				set_variable = { name = hmps_fx_tmp value = var:hmps_fx_rate }
				if = {
					limit = { has_variable = hmps_fx_pull }
					change_variable = { name = hmps_fx_tmp add = var:hmps_fx_pull }
				}
				change_variable = { name = hmps_fx_tmp multiply = var:hmps_fx_gdp }
				change_global_variable = { name = hmps_fx_ref_sum add = var:hmps_fx_tmp }
				remove_variable = hmps_fx_tmp
				change_global_variable = { name = hmps_fx_world_gdp add = var:hmps_fx_gdp }
			}
		}
	}
	if = {
		limit = { global_var:hmps_fx_world_gdp > 0 }
		set_global_variable = {
			name = hmps_fx_ref_rate
			value = {
				value = global_var:hmps_fx_ref_sum
				divide = global_var:hmps_fx_world_gdp
			}
		}
	}
	else = {
		set_global_variable = { name = hmps_fx_ref_rate value = 4 }
	}
	remove_global_variable = hmps_fx_ref_sum

	# c) Interest on last week's foreign money held: debtors pay their own rate / 52 from reserves; creditors share it by what each lent.
	#    Rounding leaves a small remainder, which goes to the largest creditor
	set_global_variable = { name = hmps_fx_world_lent value = 0 }
	set_global_variable = { name = hmps_fx_interest_pot value = 0 }
	set_global_variable = { name = hmps_fx_left_interest value = 0 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			if = {
				limit = { NOT = { has_variable = hmps_fx_foreign } }
				set_variable = { name = hmps_fx_foreign value = 0 }
			}
			set_variable = { name = hmps_fx_interest value = 0 }
			set_variable = { name = hmps_fx_pool_moved value = 0 }
			set_variable = { name = hmps_fx_commission value = 0 }
			if = {
				limit = { var:hmps_fx_foreign < 0 }
				change_global_variable = { name = hmps_fx_world_lent subtract = var:hmps_fx_foreign }
			}
		}
	}
	if = {
		limit = { global_var:hmps_fx_world_lent > 0 }
		every_market = {
			limit = {
				owner = {
					NOT = { is_country_type = decentralized }
				}
			}
			owner = {
				if = {
					limit = { var:hmps_fx_foreign > 0 }
					set_variable = {
						name = hmps_fx_interest
						value = {
							value = var:hmps_fx_foreign
							multiply = var:hmps_fx_rate
							divide = 5200
							multiply = -1
						}
					}
					change_variable = { name = hmps_fx_reserves add = var:hmps_fx_interest }
					change_global_variable = { name = hmps_fx_interest_pot subtract = var:hmps_fx_interest }
				}
			}
		}
		set_global_variable = { name = hmps_fx_left_interest value = global_var:hmps_fx_interest_pot }
		set_global_variable = { name = hmps_fx_top_amount value = 0 }
		every_market = {
			limit = {
				owner = {
					NOT = { is_country_type = decentralized }
				}
			}
			owner = {
				if = {
					limit = { var:hmps_fx_foreign < 0 }
					set_variable = {
						name = hmps_fx_interest
						value = {
							value = var:hmps_fx_foreign
							divide = global_var:hmps_fx_world_lent
							multiply = global_var:hmps_fx_interest_pot
							multiply = -1
						}
					}
					change_variable = { name = hmps_fx_reserves add = var:hmps_fx_interest }
					change_global_variable = { name = hmps_fx_left_interest subtract = var:hmps_fx_interest }
					if = {
						limit = { var:hmps_fx_interest > global_var:hmps_fx_top_amount }
						save_scope_as = hmps_fx_top_scope
						set_global_variable = { name = hmps_fx_top value = scope:hmps_fx_top_scope }
						set_global_variable = { name = hmps_fx_top_amount value = var:hmps_fx_interest }
					}
				}
			}
		}
		if = {
			limit = { has_global_variable = hmps_fx_top }
			global_var:hmps_fx_top = {
				change_variable = { name = hmps_fx_interest add = global_var:hmps_fx_left_interest }
				change_variable = { name = hmps_fx_reserves add = global_var:hmps_fx_left_interest }
			}
			remove_global_variable = hmps_fx_top
		}
		remove_global_variable = hmps_fx_top_amount
	}

	# d) Capital flow: (effective rate - reference rate) x 1% of GDP a year, no cap. Against a GDP-weighted average the flows sum to zero
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			if = {
				limit = { has_law_or_variant = law_type:law_isolationism }
				set_variable = { name = hmps_fx_flow value = 0 }
			}
			else = {
				set_variable = { name = hmps_fx_tmp value = var:hmps_fx_rate }
				if = {
					limit = { has_variable = hmps_fx_pull }
					change_variable = { name = hmps_fx_tmp add = var:hmps_fx_pull }
				}
				set_variable = {
					name = hmps_fx_flow
					value = {
						value = var:hmps_fx_tmp
						subtract = global_var:hmps_fx_ref_rate
						multiply = var:hmps_fx_gdp
						divide = 5200
					}
				}
				remove_variable = hmps_fx_tmp
				# Phases 4-5: the Defend (x2), Suspend (x0.5) and on-paper (-25%) multipliers apply to positive flows here
			}
		}
	}
	# Reserves and foreign money held move by the full flow. Rounding leaves a small remainder, which goes to the largest inflow
	set_global_variable = { name = hmps_fx_world_inflow value = 0 }
	set_global_variable = { name = hmps_fx_left_flow value = 0 }
	set_global_variable = { name = hmps_fx_top_amount value = 0 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			change_variable = { name = hmps_fx_reserves add = var:hmps_fx_flow }
			change_variable = { name = hmps_fx_foreign add = var:hmps_fx_flow }
			change_global_variable = { name = hmps_fx_left_flow subtract = var:hmps_fx_flow }
			if = {
				limit = { var:hmps_fx_flow > 0 }
				change_global_variable = { name = hmps_fx_world_inflow add = var:hmps_fx_flow }
				if = {
					limit = { var:hmps_fx_flow > global_var:hmps_fx_top_amount }
					save_scope_as = hmps_fx_top_scope
					set_global_variable = { name = hmps_fx_top value = scope:hmps_fx_top_scope }
					set_global_variable = { name = hmps_fx_top_amount value = var:hmps_fx_flow }
				}
			}
		}
	}
	if = {
		limit = { has_global_variable = hmps_fx_top }
		global_var:hmps_fx_top = {
			change_variable = { name = hmps_fx_flow add = global_var:hmps_fx_left_flow }
			change_variable = { name = hmps_fx_reserves add = global_var:hmps_fx_left_flow }
			change_variable = { name = hmps_fx_foreign add = global_var:hmps_fx_left_flow }
		}
		change_global_variable = { name = hmps_fx_world_inflow add = global_var:hmps_fx_left_flow }
		remove_global_variable = hmps_fx_top
	}
	remove_global_variable = hmps_fx_top_amount

	# e) Investment pool: senders owe 50% of their outflow, split among member countries by GDP share. A country pays only if its
	#    pool covers its full share (a negative pool is reset to 0, so money would be lost); receivers share only what was paid
	set_global_variable = { name = hmps_fx_pool_paid value = 0 }
	every_country = {
		limit = { has_variable = hmps_fx_owner }
		set_variable = { name = hmps_fx_pool_share value = 0 }
	}
	every_country = {
		limit = {
			NOT = { is_country_type = decentralized }
			has_variable = hmps_fx_owner
			var:hmps_fx_owner = {
				NOT = { is_country_type = decentralized }
				has_variable = hmps_fx_flow
				has_variable = hmps_fx_gdp
				var:hmps_fx_flow < 0
				var:hmps_fx_gdp > 0
			}
		}
		save_scope_as = hmps_fx_pool_payer
		set_variable = {
			name = hmps_fx_pool_owed
			value = {
				value = gdp
				divide = var:hmps_fx_owner.var:hmps_fx_gdp
				max = 1
				multiply = var:hmps_fx_owner.var:hmps_fx_flow
				multiply = -0.5
			}
		}
		if = {
			limit = {
				var:hmps_fx_pool_owed > 0
				investment_pool >= var:hmps_fx_pool_owed
			}
			add_investment_pool = {
				value = var:hmps_fx_pool_owed
				multiply = -1
			}
			change_global_variable = { name = hmps_fx_pool_paid add = var:hmps_fx_pool_owed }
			change_variable = { name = hmps_fx_pool_share subtract = var:hmps_fx_pool_owed }
			var:hmps_fx_owner = {
				change_variable = { name = hmps_fx_pool_moved subtract = scope:hmps_fx_pool_payer.var:hmps_fx_pool_owed }
			}
		}
		remove_variable = hmps_fx_pool_owed
	}
	set_global_variable = { name = hmps_fx_left_pool value = global_var:hmps_fx_pool_paid }
	if = {
		limit = {
			global_var:hmps_fx_pool_paid > 0
			global_var:hmps_fx_world_inflow > 0
		}
		set_global_variable = { name = hmps_fx_top_amount value = 0 }
		every_country = {
			limit = {
				NOT = { is_country_type = decentralized }
				has_variable = hmps_fx_owner
				var:hmps_fx_owner = {
					NOT = { is_country_type = decentralized }
					has_variable = hmps_fx_flow
					has_variable = hmps_fx_gdp
					var:hmps_fx_flow > 0
					var:hmps_fx_gdp > 0
				}
			}
			save_scope_as = hmps_fx_pool_payee
			set_variable = {
				name = hmps_fx_pool_due
				value = {
					value = gdp
					divide = var:hmps_fx_owner.var:hmps_fx_gdp
					max = 1
					multiply = var:hmps_fx_owner.var:hmps_fx_flow
					divide = global_var:hmps_fx_world_inflow
					multiply = global_var:hmps_fx_pool_paid
				}
			}
			add_investment_pool = var:hmps_fx_pool_due
			change_variable = { name = hmps_fx_pool_share add = var:hmps_fx_pool_due }
			change_global_variable = { name = hmps_fx_left_pool subtract = var:hmps_fx_pool_due }
			var:hmps_fx_owner = {
				change_variable = { name = hmps_fx_pool_moved add = scope:hmps_fx_pool_payee.var:hmps_fx_pool_due }
			}
			if = {
				limit = { var:hmps_fx_pool_due > global_var:hmps_fx_top_amount }
				set_global_variable = { name = hmps_fx_top value = scope:hmps_fx_pool_payee }
				set_global_variable = { name = hmps_fx_top_amount value = var:hmps_fx_pool_due }
			}
			remove_variable = hmps_fx_pool_due
		}
		# Rounding leaves a small remainder, which goes to the largest receiver. It just gained far more, so a negative remainder
		# can't take its pool below zero (checked: its gain plus the remainder must not be negative)
		change_global_variable = { name = hmps_fx_top_amount add = global_var:hmps_fx_left_pool }
		if = {
			limit = {
				has_global_variable = hmps_fx_top
				global_var:hmps_fx_top_amount >= 0
			}
			global_var:hmps_fx_top = {
				add_investment_pool = global_var:hmps_fx_left_pool
				change_variable = { name = hmps_fx_pool_share add = global_var:hmps_fx_left_pool }
				var:hmps_fx_owner = {
					change_variable = { name = hmps_fx_pool_moved add = global_var:hmps_fx_left_pool }
				}
			}
		}
		if = {
			limit = { has_global_variable = hmps_fx_top }
			remove_global_variable = hmps_fx_top
		}
		remove_global_variable = hmps_fx_top_amount
	}

	# f) London's commission: every other market pays 0.5% of its weekly total trade, split between the issuer and a former issuer
	#    (weeks left / 52, fading over 52 weeks)
	hmps_fx_commission_effect = yes
	if = {
		limit = { has_global_variable = hmps_fx_old_issuer }
		change_global_variable = { name = hmps_fx_old_issuer_weeks subtract = 1 }
		if = {
			limit = { global_var:hmps_fx_old_issuer_weeks <= 0 }
			remove_global_variable = hmps_fx_old_issuer
			remove_global_variable = hmps_fx_old_issuer_weeks
		}
	}
	remove_global_variable = hmps_fx_issuer_share
	if = {
		limit = { has_global_variable = hmps_fx_old_share }
		remove_global_variable = hmps_fx_old_share
	}
	remove_global_variable = hmps_fx_world_lent
	remove_global_variable = hmps_fx_interest_pot
	remove_global_variable = hmps_fx_world_inflow
	remove_global_variable = hmps_fx_pool_paid

	# 4. World average and top-up
	if = {
		limit = { global_var:hmps_fx_world_bill > 0 }
		set_global_variable = {
			name = hmps_fx_world_months
			value = {
				value = global_var:hmps_fx_world_reserves
				divide = global_var:hmps_fx_world_bill
			}
		}
	}
	else = {
		set_global_variable = { name = hmps_fx_world_months value = 12 }
	}
	if = {
		limit = { global_var:hmps_fx_world_months < 12 }
		change_global_variable = { name = hmps_fx_topup_count add = 1 }
		every_market = {
			limit = {
				owner = {
					NOT = { is_country_type = decentralized }
				}
			}
			owner = {
				set_variable = {
					name = hmps_fx_topup
					value = {
						value = 12
						subtract = global_var:hmps_fx_world_months
						multiply = var:hmps_fx_import_bill
					}
				}
				change_variable = { name = hmps_fx_reserves add = var:hmps_fx_topup }
				change_global_variable = { name = hmps_fx_topup_total add = var:hmps_fx_topup }
			}
		}
		set_global_variable = { name = hmps_fx_world_months value = 12 }
	}

	# 5. Months of imports, and the week's change in reserves for the panel
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			set_variable = {
				name = hmps_fx_change
				value = {
					value = var:hmps_fx_net
					add = var:hmps_fx_gold
					add = var:hmps_fx_flow
					add = var:hmps_fx_interest
					add = var:hmps_fx_commission
					add = var:hmps_fx_topup
				}
			}
			if = {
				limit = { var:hmps_fx_import_bill > 0 }
				set_variable = {
					name = hmps_fx_months
					value = {
						value = var:hmps_fx_reserves
						divide = var:hmps_fx_import_bill
					}
				}
			}
			else = {
				set_variable = { name = hmps_fx_months value = 0 }
			}
		}
	}

	# 6. Bank rate: every market owner has a rate
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			hmps_fx_rate_init_effect = yes
		}
	}

	# 7. Price level: the target from reserves per GDP against the world's plus the rate; the level steps toward it after
	# 26 weeks in a row at least 1 step away. Phase 4 adds the crisis freeze here
	set_global_variable = { name = hmps_fx_tmp_reserves value = 0 }
	set_global_variable = { name = hmps_fx_tmp_gdp value = 0 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			change_global_variable = { name = hmps_fx_tmp_reserves add = var:hmps_fx_reserves }
			change_global_variable = { name = hmps_fx_tmp_gdp add = var:hmps_fx_gdp }
		}
	}
	set_global_variable = { name = hmps_fx_world_money_ratio value = 0 }
	if = {
		limit = {
			global_var:hmps_fx_tmp_reserves > 0
			global_var:hmps_fx_tmp_gdp > 0
		}
		set_global_variable = {
			name = hmps_fx_world_money_ratio
			value = {
				value = global_var:hmps_fx_tmp_reserves
				divide = global_var:hmps_fx_tmp_gdp
			}
		}
	}
	remove_global_variable = hmps_fx_tmp_reserves
	remove_global_variable = hmps_fx_tmp_gdp
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
			}
		}
		owner = {
			hmps_fx_price_level_effect = yes
		}
	}

	# Every country's rate and price-level modifiers, re-applied only where they changed
	every_country = {
		limit = { NOT = { is_country_type = decentralized } }
		trigger_event = { id = hmps_bot.2 }
	}
}

# HMPS Bank Rate and Reserves, phase 3b. Market owner scope, called by hmps_fx_weekly_effect (step 7) after the months and the rate.
# Money ratio = (reserves / market GDP) / the world's (global_var:hmps_fx_world_money_ratio). Target = money part (ratio - 1) x 4
# + rate part (4 - rate) / 2, capped at -4 to 4. Placeholder scales, not balanced yet. The level moves one whole step toward it once
# the target has been at least 1 step away for 26 weeks in a row; the count restarts when the gap drops below 1 or the level steps
hmps_fx_price_level_effect = {
	if = {
		limit = { NOT = { has_variable = hmps_fx_price_level } }
		set_variable = { name = hmps_fx_price_level value = 0 }
		set_variable = { name = hmps_fx_price_weeks value = 0 }
	}
	set_variable = { name = hmps_fx_price_ratio value = 1 }
	if = {
		limit = {
			global_var:hmps_fx_world_money_ratio > 0
			var:hmps_fx_gdp > 0
		}
		set_variable = {
			name = hmps_fx_price_ratio
			value = {
				value = var:hmps_fx_reserves
				divide = var:hmps_fx_gdp
				divide = global_var:hmps_fx_world_money_ratio
			}
		}
	}
	set_variable = {
		name = hmps_fx_price_money
		value = {
			value = var:hmps_fx_price_ratio
			subtract = 1
			multiply = 4
		}
	}
	set_variable = {
		name = hmps_fx_price_rate
		value = {
			value = 4
			subtract = var:hmps_fx_rate
			divide = 2
		}
	}
	set_variable = {
		name = hmps_fx_price_target
		value = {
			value = var:hmps_fx_price_money
			add = var:hmps_fx_price_rate
			max = 4
			min = -4
		}
	}
	set_variable = {
		name = hmps_fx_price_gap
		value = {
			value = var:hmps_fx_price_target
			subtract = var:hmps_fx_price_level
		}
	}
	if = {
		limit = {
			OR = {
				var:hmps_fx_price_gap >= 1
				var:hmps_fx_price_gap <= -1
			}
		}
		change_variable = { name = hmps_fx_price_weeks add = 1 }
		if = {
			limit = { var:hmps_fx_price_weeks >= 26 }
			if = {
				limit = { var:hmps_fx_price_gap > 0 }
				change_variable = { name = hmps_fx_price_level add = 1 }
			}
			else = {
				change_variable = { name = hmps_fx_price_level subtract = 1 }
			}
			set_variable = { name = hmps_fx_price_weeks value = 0 }
		}
	}
	else = {
		set_variable = { name = hmps_fx_price_weeks value = 0 }
	}
	remove_variable = hmps_fx_price_gap
}

# HMPS Bank Rate and Reserves, phase 3. Called by hmps_fx_weekly_effect (step 3f) after step a set global_var:hmps_fx_old_share.
# Every market except the issuer and a former issuer pays 0.5% of its weekly total trade. A former issuer gets old_share of the
# payments and the issuer gets the rest, so the world total is unchanged and each market pays once
hmps_fx_commission_effect = {
	if = {
		limit = { has_global_variable = hmps_fx_old_issuer }
		global_var:hmps_fx_old_issuer = { save_scope_as = hmps_fx_old_earner }
	}
	else = {
		global_var:hmps_fx_issuer = { save_scope_as = hmps_fx_old_earner }
	}
	set_global_variable = { name = hmps_fx_commission_paid value = 0 }
	every_market = {
		limit = {
			owner = {
				NOT = { is_country_type = decentralized }
				NOT = { this = global_var:hmps_fx_issuer }
				NOT = { this = scope:hmps_fx_old_earner }
			}
		}
		owner = {
			set_variable = {
				name = hmps_fx_tmp
				value = {
					value = var:hmps_fx_trade
					multiply = 0.005
				}
			}
			change_variable = { name = hmps_fx_reserves subtract = var:hmps_fx_tmp }
			change_variable = { name = hmps_fx_commission subtract = var:hmps_fx_tmp }
			change_global_variable = { name = hmps_fx_commission_paid add = var:hmps_fx_tmp }
			remove_variable = hmps_fx_tmp
		}
	}
	if = {
		limit = { has_global_variable = hmps_fx_old_issuer }
		set_global_variable = {
			name = hmps_fx_commission_old
			value = {
				value = global_var:hmps_fx_commission_paid
				multiply = global_var:hmps_fx_old_share
			}
		}
		scope:hmps_fx_old_earner = {
			change_variable = { name = hmps_fx_reserves add = global_var:hmps_fx_commission_old }
			change_variable = { name = hmps_fx_commission add = global_var:hmps_fx_commission_old }
		}
		change_global_variable = { name = hmps_fx_commission_paid subtract = global_var:hmps_fx_commission_old }
		remove_global_variable = hmps_fx_commission_old
	}
	global_var:hmps_fx_issuer = {
		change_variable = { name = hmps_fx_reserves add = global_var:hmps_fx_commission_paid }
		change_variable = { name = hmps_fx_commission add = global_var:hmps_fx_commission_paid }
	}
	remove_global_variable = hmps_fx_commission_paid
}

# HMPS Bank Rate and Reserves, phase 2. Market owner scope: a rate of 4%, if the owner has none yet
hmps_fx_rate_init_effect = {
	if = {
		limit = { NOT = { has_variable = hmps_fx_rate } }
		set_variable = { name = hmps_fx_rate value = 4 }
		set_variable = { name = hmps_fx_rate_steps value = 0 }
	}
}

# Market owner scope; RATE = the new rate (2 to 10, at most two points from the current one). Called by the bank rate coins
# (common/scripted_guis/HMPS_scripted_gui.txt).
# Locks the rate for 30 days and refreshes every member's modifiers at once (all rate effects change the same day)
hmps_fx_rate_change_effect = {
	hmps_fx_rate_init_effect = yes
	set_variable = { name = hmps_fx_rate value = $RATE$ }
	set_variable = {
		name = hmps_fx_rate_steps
		value = {
			value = var:hmps_fx_rate
			subtract = 4
		}
	}
	set_variable = { name = hmps_fx_rate_lock days = 30 }
	market = { save_scope_as = hmps_fx_rate_market }
	every_country = {
		limit = {
			market = scope:hmps_fx_rate_market
			NOT = { is_country_type = decentralized }
		}
		trigger_event = { id = hmps_bot.2 }
	}
}

# Country scope; root must be the country (add_modifier's multiplier reads var: from root), so it is only called from the
# hidden event hmps_bot.2 (events/00_HMPS_utility.txt). Copies the market owner's steps and re-applies the rate modifier
# (common/static_modifiers/z_HMPS_other_modifiers.txt) only when they changed, because re-applying makes the game recalculate.
hmps_fx_rate_modifiers_effect = {
	market = {
		owner = { save_scope_as = hmps_fx_rate_owner }
	}
	set_variable = { name = hmps_fx_rate_steps_new value = 0 }
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
			scope:hmps_fx_rate_owner = {
				NOT = { is_country_type = decentralized }
				has_variable = hmps_fx_rate
			}
		}
		set_variable = { name = hmps_fx_rate_steps_new value = scope:hmps_fx_rate_owner.var:hmps_fx_rate_steps }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = hmps_fx_rate_steps_here }
				var:hmps_fx_rate_steps_here != var:hmps_fx_rate_steps_new
			}
		}
		set_variable = { name = hmps_fx_rate_steps_here value = var:hmps_fx_rate_steps_new }
		if = {
			limit = { has_modifier = hmps_fx_rate_hike }
			remove_modifier = hmps_fx_rate_hike
		}
		if = {
			limit = { has_modifier = hmps_fx_rate_cut }
			remove_modifier = hmps_fx_rate_cut
		}
		# Multipliers must be positive, so cuts use their own modifiers
		if = {
			limit = { var:hmps_fx_rate_steps_here > 0 }
			set_variable = { name = hmps_fx_rate_mult value = var:hmps_fx_rate_steps_here }
			add_modifier = {
				name = hmps_fx_rate_hike
				multiplier = var:hmps_fx_rate_mult
			}
		}
		else_if = {
			limit = { var:hmps_fx_rate_steps_here < 0 }
			set_variable = {
				name = hmps_fx_rate_mult
				value = {
					value = var:hmps_fx_rate_steps_here
					multiply = -1
				}
			}
			add_modifier = {
				name = hmps_fx_rate_cut
				multiplier = var:hmps_fx_rate_mult
			}
		}
	}
	remove_variable = hmps_fx_rate_steps_new
}

# Country scope; root must be the country, so it is only called from hmps_bot.2 (same reason as hmps_fx_rate_modifiers_effect).
# Copies the market owner's price level and re-applies hmps_fx_price_inflation or hmps_fx_price_deflation
# (common/static_modifiers/z_HMPS_other_modifiers.txt), multiplied by the steps, only when the level changed
hmps_fx_price_modifiers_effect = {
	market = {
		owner = { save_scope_as = hmps_fx_price_owner }
	}
	set_variable = { name = hmps_fx_price_level_new value = 0 }
	if = {
		limit = {
			NOT = { is_country_type = decentralized }
			scope:hmps_fx_price_owner = {
				NOT = { is_country_type = decentralized }
				has_variable = hmps_fx_price_level
			}
		}
		set_variable = { name = hmps_fx_price_level_new value = scope:hmps_fx_price_owner.var:hmps_fx_price_level }
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = hmps_fx_price_level_here }
				var:hmps_fx_price_level_here != var:hmps_fx_price_level_new
			}
		}
		set_variable = { name = hmps_fx_price_level_here value = var:hmps_fx_price_level_new }
		if = {
			limit = { has_modifier = hmps_fx_price_inflation }
			remove_modifier = hmps_fx_price_inflation
		}
		if = {
			limit = { has_modifier = hmps_fx_price_deflation }
			remove_modifier = hmps_fx_price_deflation
		}
		# Multipliers must be positive, so deflation uses its own modifier
		if = {
			limit = { var:hmps_fx_price_level_here > 0 }
			set_variable = { name = hmps_fx_price_mult value = var:hmps_fx_price_level_here }
			add_modifier = {
				name = hmps_fx_price_inflation
				multiplier = var:hmps_fx_price_mult
			}
		}
		else_if = {
			limit = { var:hmps_fx_price_level_here < 0 }
			set_variable = {
				name = hmps_fx_price_mult
				value = {
					value = var:hmps_fx_price_level_here
					multiply = -1
				}
			}
			add_modifier = {
				name = hmps_fx_price_deflation
				multiplier = var:hmps_fx_price_mult
			}
		}
	}
	remove_variable = hmps_fx_price_level_new
}

# Oregon Trail (oregon.5): gives PROVINCE (ID without p:, e.g. xE5842F) to the USA only if a decentralized country owns it
hmps_oregon_trail_take_province = {
	if = {
		limit = {
			exists = c:USA
			exists = p:$PROVINCE$.state
			p:$PROVINCE$.state.owner = {
				is_country_type = decentralized
			}
		}
		p:$PROVINCE$.state.state_region = {
			set_owner_of_provinces = {
				country = c:USA
				provinces = { $PROVINCE$ }
			}
		}
	}
}

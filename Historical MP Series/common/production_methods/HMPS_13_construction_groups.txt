# Extra construction sector groups. The base group is Foundations (HMPS_13_construction.txt).
# Site Machinery is labour saving only (no construction points): Steam Plant (era 3) coal 5, laborers -1750; Motorised Plant (era 5) oil 5, laborers -2500.
# The -2500 cap leaves 0 laborers with Concrete Foundations or Piles and Caissons (2500 laborers).
# Fittings add construction points on top of Foundations: Bare Finish 0.5, Windows and Pipes 2.5, Electric Lifts 4.5.
# Fittings also give construction sector throughput per level: Windows and Pipes 0.5%, Electric Lifts 1%. This replaces the vanilla tech throughput (Reinforced Concrete, Pneumatic Tools).
# Each upgrade costs fewer goods per point than adding another construction sector level at every Foundations tier.

############ SITE MACHINERY

pm_steam_cranes_construction = {
	texture = "gfx/interface/icons/production_method_icons/pm_steam_cranes_construction.dds"

	unlocking_technologies = {
		steam_donkey # steam navvies, derricks and pile drivers on large works, about 1870-1910
	}

	state_modifiers = {
		workforce_scaled = {
			state_pollution_generation_add = 5
		}
	}

	building_modifiers = {
		workforce_scaled = {
			goods_input_coal_add = 5
		}

		level_scaled = {
			building_employment_laborers_add = -1750
		}
	}

	ai_value = 2000 # HMPS AI: upgrade construction methods quickly (tier 1 uses PRODUCTION_METHOD_BASE_VALUE 1000)
}

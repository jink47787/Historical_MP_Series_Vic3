# Builders' Yards (building_materials_works): two material lines (Timber, Masonry) plus the standard Automation group.
# 5000 jobs per level: Timber 3750 (it carries the yard's 500 shopkeepers and 2000 base laborers, and is never idle),
# Masonry 1250, No Masonry 0. Assembly Lines cuts 2500 laborers, so Laminated Timber with No Masonry ends at 0.
# Every non-default method is unlocked by its own technology; no two methods share one.

############ TIMBER LINE

	is_default = yes
	texture = "gfx/interface/icons/production_method_icons/pm_timber_yard_building_materials_works.dds"

	building_modifiers = {
		workforce_scaled = {
			goods_input_wood_add = 20

		}

}

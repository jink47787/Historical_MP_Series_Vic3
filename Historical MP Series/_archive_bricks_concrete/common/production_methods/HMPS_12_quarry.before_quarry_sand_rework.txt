# Quarries are bulky and dangerous: every method except hand quarrying hauls a lot of stone (transportation) and early methods
# carry high laborer mortality. They are not blocked in developed states, only capped per state.

	texture = "gfx/interface/icons/production_method_icons/pm_hand_quarrying_building_quarry.dds"
